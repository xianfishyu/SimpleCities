1using Godot;
2using System;
3
4/// <summary>
5/// 游戏场景内的周期自动存档调度器。Timer 继承场景暂停状态，因此暂停菜单打开时不会推进周期。
6/// </summary>
7public partial class AutosaveController : Node
8{
9    private const double MinimumIntervalSeconds = 0.001d;
10
11    private Timer _timer = null!;
12    private SaveManager? _saveManager;
13
14    [Export(PropertyHint.Range, "1,3600,1,or_greater")]
15    public double IntervalSeconds { get; set; } = 300d;
16
17    [Export]
18    public bool AutosaveEnabled { get; set; } = true;
19
20    public int AttemptCount { get; private set; }
21    public int SuccessfulSaveCount { get; private set; }
22    public int FailedSaveCount { get; private set; }
23    public int CanceledSaveCount { get; private set; }
24    public int SkippedBusyCount { get; private set; }
25    public bool LastAttemptSucceeded { get; private set; }
26    public string LastOperationToken { get; private set; } = string.Empty;
27
28    [Signal]
29    public delegate void AutosaveCompletedEventHandler(bool success);
30
31    public override void _Ready()
32    {
33        _timer = new Timer
34        {
35            Name = "AutosaveTimer",
36            OneShot = false,
37        };
38        AddChild(_timer);
39        _timer.Timeout += OnAutosaveTimeout;
40        _saveManager = GodotObject.IsInstanceValid(SaveManager.Instance)
41            ? SaveManager.Instance
42            : null;
43        if (_saveManager is not null)
44            _saveManager.OperationCompleted += OnSaveOperationCompleted;
45
46        if (AutosaveEnabled)
47            StartTimer();
48    }
49
50    public override void _ExitTree()
51    {
52        if (_timer != null && GodotObject.IsInstanceValid(_timer))
53        {
54            _timer.Timeout -= OnAutosaveTimeout;
55            _timer.Stop();
56        }
57        if (_saveManager is not null && GodotObject.IsInstanceValid(_saveManager))
58            _saveManager.OperationCompleted -= OnSaveOperationCompleted;
59        _saveManager = null;
60    }
61
62    /// <summary>启用或停止周期触发；重新启用时从完整周期开始计时。</summary>
63    public void SetAutosaveEnabled(bool enabled)
64    {
65        AutosaveEnabled = enabled;
66        if (_timer == null || !GodotObject.IsInstanceValid(_timer))
67            return;
68
69        if (enabled)
70            StartTimer();
71        else
72            _timer.Stop();
73    }
74
75    /// <summary>立即执行一次自动存档；不会改变玩家当前选中的手动槽。</summary>
76    public string RunAutosaveNow()
77    {
78        AttemptCount++;
79        SaveManager? saveManager = _saveManager is not null &&
80            GodotObject.IsInstanceValid(_saveManager)
81                ? _saveManager
82                : null;
83        if (saveManager is null)
84        {
85            LastOperationToken = string.Empty;
86            LastAttemptSucceeded = false;
87            FailedSaveCount++;
88            EmitSignal(SignalName.AutosaveCompleted, false);
89            return string.Empty;
90        }
91
92        LastOperationToken = saveManager.StartAutosave();
93        return LastOperationToken;
94    }
95
96    private void OnSaveOperationCompleted(SaveOperationResult result)
97    {
98        if (result.Kind != SaveOperationKind.Autosave)
99            return;
100
101        switch (result.ResultKind)
102        {
103            case SaveOperationResultKind.Succeeded:
104            case SaveOperationResultKind.SucceededWithWarnings:
105                LastAttemptSucceeded = true;
106                SuccessfulSaveCount++;
107                EmitSignal(SignalName.AutosaveCompleted, true);
108                break;
109            case SaveOperationResultKind.SkippedBusy:
110                LastAttemptSucceeded = false;
111                SkippedBusyCount++;
112                break;
113            case SaveOperationResultKind.Canceled:
114            case SaveOperationResultKind.RejectedSceneClosing:
115            case SaveOperationResultKind.RejectedShuttingDown:
116                LastAttemptSucceeded = false;
117                CanceledSaveCount++;
118                EmitSignal(SignalName.AutosaveCompleted, false);
119                break;
120            default:
121                LastAttemptSucceeded = false;
122                FailedSaveCount++;
123                EmitSignal(SignalName.AutosaveCompleted, false);
124                break;
125        }
126    }
127
128    private void StartTimer()
129    {
130        if (!double.IsFinite(IntervalSeconds) || IntervalSeconds <= 0d)
131        {
132            AutosaveEnabled = false;
133            _timer.Stop();
134            GD.PushError($"AutosaveController: interval must be finite and positive, got {IntervalSeconds}.");
135            return;
136        }
137
138        _timer.WaitTime = Math.Max(IntervalSeconds, MinimumIntervalSeconds);
139        _timer.Start();
140    }
141
142    private void OnAutosaveTimeout() => RunAutosaveNow();
143}

69    /// <summary>返回动作当前绑定的物理按键；动作不存在时返回 Key.None。</summary>
70    public Key GetBoundKey(string actionName)
71    {
72        if (!DefinitionsByAction.ContainsKey(actionName) || !InputMap.HasAction(actionName))
73            return Key.None;
74
75        foreach (InputEvent inputEvent in InputMap.ActionGetEvents(actionName))
76        {
77            if (inputEvent is InputEventKey keyEvent)
78                return NormalizeKey(keyEvent);
79        }
80
81        return Key.None;
82    }

97    /// <summary>用一次真实键盘事件匹配当前动作，供 HUD 和暂停菜单统一消费输入。</summary>
98    public bool EventMatchesAction(InputEvent inputEvent, string actionName)
99    {
100        if (inputEvent is InputEventAction actionEvent)
101            return actionEvent.Pressed && actionEvent.Action == actionName;
102
103        if (inputEvent is not InputEventKey keyEvent || !keyEvent.Pressed || keyEvent.Echo)
104            return false;
105        if (keyEvent.CtrlPressed || keyEvent.AltPressed || keyEvent.ShiftPressed || keyEvent.MetaPressed)
106            return false;
107
108        return NormalizeKey(keyEvent) == GetBoundKey(actionName);
109    }

111    public bool TryGetToolForEvent(InputEvent inputEvent, out ToolType tool)
112    {
113        foreach (BindingDefinition definition in BindingCatalog)
114        {
115            if (definition.Tool is not ToolType candidate || !EventMatchesAction(inputEvent, definition.ActionName))
116                continue;
117
118            tool = candidate;
119            return true;
120        }
121
122        tool = default;
123        return false;
124    }

141    public bool TryRebind(string actionName, Key key, out string error)
142    {
143        if (!DefinitionsByAction.TryGetValue(actionName, out BindingDefinition? definition))
144        {
145            error = $"未知输入动作：{actionName}";
146            return false;
147        }
148
149        if (!IsBindableKey(key))
150        {
151            error = "该按键不能用于绑定";
152            return false;
153        }
154
155        foreach (BindingDefinition candidate in BindingCatalog)
156        {
157            if (candidate.ActionName == actionName || GetBoundKey(candidate.ActionName) != key)
158                continue;
159
160            error = $"{OS.GetKeycodeString(key)} 已绑定到“{candidate.DisplayName}”";
161            return false;
162        }
163
164        Key previousKey = GetBoundKey(actionName);
165        if (previousKey == key)
166        {
167            error = string.Empty;
168            return true;
169        }
170
171        ApplyBinding(definition, key);
172        if (!SaveBindings(out error))
173        {
174            ApplyBinding(definition, previousKey);
175            return false;
176        }
177
178        BindingChanged?.Invoke(actionName);
179        return true;
180    }
181
182    public bool ResetToDefaults(out string error)
183    {
184        var previousKeys = BindingCatalog.ToDictionary(
185            definition => definition.ActionName,
186            definition => GetBoundKey(definition.ActionName),
187            StringComparer.Ordinal);
188
189        foreach (BindingDefinition definition in BindingCatalog)
190            ApplyBinding(definition, definition.DefaultKey);
191
192        if (!SaveBindings(out error))
193        {
194            foreach (BindingDefinition definition in BindingCatalog)
195                ApplyBinding(definition, previousKeys[definition.ActionName]);
196            return false;
197        }
198
199        foreach (BindingDefinition definition in BindingCatalog)
200            BindingChanged?.Invoke(definition.ActionName);
201        return true;
202    }

209    public static bool IsBindableKey(Key key)
210    {
211        return Enum.IsDefined(key) && key is not (Key.None or Key.Shift or Key.Ctrl or Key.Alt or Key.Meta);

214    private void InitializeBindings()
215    {
216        foreach (BindingDefinition definition in BindingCatalog)
217        {
218            if (!InputMap.HasAction(definition.ActionName))
219                InputMap.AddAction(definition.ActionName);
220            ApplyBinding(definition, definition.DefaultKey);
221        }
222
223        LoadBindings();
224    }

226    private static void ApplyBinding(BindingDefinition definition, Key key)
227    {
228        InputMap.ActionEraseEvents(definition.ActionName);
229        if (key == Key.None)
230            return;
231
232        InputMap.ActionAddEvent(definition.ActionName, new InputEventKey
233        {
234            PhysicalKeycode = key,
235        });
236    }

238    private void LoadBindings()
239    {
240        var config = new ConfigFile();
241        Error loadResult = config.Load(ConfigPath);
242        if (loadResult == Error.FileNotFound)
243            return;
244        if (loadResult != Error.Ok)
245        {
246            GD.PushWarning($"InputBindingManager: failed to load bindings ({loadResult}); defaults remain active.");
247            return;
248        }
249
250        var storedKeys = new Dictionary<string, Key>(StringComparer.Ordinal);
251        foreach (BindingDefinition definition in BindingCatalog)
252        {
253            if (config.HasSectionKey(ConfigSection, definition.ActionName))
254                storedKeys[definition.ActionName] =
255                    (Key)config.GetValue(ConfigSection, definition.ActionName).AsInt64();
256        }
257
258        if (!TryResolveStoredBindings(storedKeys, out Dictionary<string, Key> candidateKeys))
259        {
260            GD.PushWarning("InputBindingManager: saved bindings are invalid or duplicated; defaults remain active.");
261            return;
262        }
263
264        foreach (BindingDefinition definition in BindingCatalog)
265            ApplyBinding(definition, candidateKeys[definition.ActionName]);
266    }
267
268    internal static bool TryResolveStoredBindings(
269        IReadOnlyDictionary<string, Key> storedKeys,
270        out Dictionary<string, Key> candidateKeys)
271    {
272        candidateKeys = new Dictionary<string, Key>(StringComparer.Ordinal);
273        var occupiedKeys = new HashSet<Key>();
274
275        // Resolve persisted actions first so a newly added default never displaces a user's old binding.
276        foreach (BindingDefinition definition in BindingCatalog)
277        {
278            if (!storedKeys.TryGetValue(definition.ActionName, out Key key))
279                continue;
280
281            if (key != Key.None && (!IsBindableKey(key) || !occupiedKeys.Add(key)))
282                return false;
283
284            candidateKeys[definition.ActionName] = key;
285        }
286
287        foreach (BindingDefinition definition in BindingCatalog)
288        {
289            if (candidateKeys.ContainsKey(definition.ActionName))
290                continue;
291
292            candidateKeys[definition.ActionName] = occupiedKeys.Add(definition.DefaultKey)
293                ? definition.DefaultKey
294                : Key.None;
295        }
296
297        return true;
298    }
299
300    private bool SaveBindings(out string error)
301    {
302        var config = new ConfigFile();
303        foreach (BindingDefinition definition in BindingCatalog)
304            config.SetValue(ConfigSection, definition.ActionName, (long)GetBoundKey(definition.ActionName));
305
306        Error saveResult = config.Save(ConfigPath);
307        if (saveResult == Error.Ok)
308        {
309            error = string.Empty;
310            return true;
311        }
312
313        error = $"按键设置保存失败（{saveResult}）";
314        GD.PushWarning($"InputBindingManager: failed to save bindings ({saveResult}).");
315        return false;
316    }

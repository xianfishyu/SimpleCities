1using System;
2using System.Collections.Generic;
3using System.Diagnostics;
4using System.Linq;
5
6internal interface INonThrowingLoadCommitPlan : IDisposable
7{
8    string ParticipantID { get; }
9    bool IsGenerationCurrent { get; }
10    void CommitReferences();
11    IReadOnlyList<string> PublishNotifications();
12    void CompleteCommit();
13}
14
15internal sealed class PreparedAggregateLoad : IDisposable
16{
17    private readonly INonThrowingLoadCommitPlan[] _plans;
18    private TimeSpan? _referenceCommitDuration;
19    private bool _committed;
20    private bool _disposed;
21
22    internal PreparedAggregateLoad(IEnumerable<INonThrowingLoadCommitPlan> plans)
23    {
24        ArgumentNullException.ThrowIfNull(plans);
25        _plans = plans.ToArray();
26        if (_plans.Length == 0)
27            throw new ArgumentException("A prepared load aggregate cannot be empty.", nameof(plans));
28
29        var participantIDs = new HashSet<string>(StringComparer.Ordinal);
30        foreach (INonThrowingLoadCommitPlan plan in _plans)
31        {
32            ArgumentNullException.ThrowIfNull(plan);
33            if (string.IsNullOrWhiteSpace(plan.ParticipantID))
34                throw new ArgumentException("A load participant ID cannot be empty.", nameof(plans));
35            if (!participantIDs.Add(plan.ParticipantID))
36                throw new ArgumentException(
37                    $"Duplicate load participant ID '{plan.ParticipantID}'.",
38                    nameof(plans));
39        }
40    }
41
42    internal bool IsGenerationCurrent => _plans.All(plan => plan.IsGenerationCurrent);
43    internal TimeSpan? ReferenceCommitDuration => _referenceCommitDuration;
44
45    internal IReadOnlyList<string> Commit(IStorageOperationLease operationLease)
46    {
47        ObjectDisposedException.ThrowIf(_disposed, this);
48        ArgumentNullException.ThrowIfNull(operationLease);
49        if (operationLease.Kind != SaveOperationKind.Load)
50            throw new ArgumentException("An aggregate load requires a load operation lease.", nameof(operationLease));
51        if (_committed)
52            throw new InvalidOperationException("The prepared aggregate was already committed.");
53        if (!IsGenerationCurrent)
54            throw new LoadPreflightInvalidException("A load participant generation changed before commit.");
55
56        operationLease.ThrowIfCancellationRequested();
57        operationLease.AcquireCommitLease();
58        operationLease.CrossCommitBoundary(() =>
59        {
60            long referenceCommitStarted = Stopwatch.GetTimestamp();
61            try
62            {
63                if (!IsGenerationCurrent)
64                    throw new LoadPreflightInvalidException(
65                        "A load participant generation changed while entering commit.");
66                foreach (INonThrowingLoadCommitPlan plan in _plans)
67                    plan.CommitReferences();
68            }
69            finally
70            {
71                _referenceCommitDuration = Stopwatch.GetElapsedTime(referenceCommitStarted);
72            }
73        });
74        operationLease.MarkCommitted();
75        _committed = true;
76
77        var warnings = new List<string>();
78        foreach (INonThrowingLoadCommitPlan plan in _plans)
79        {
80            try
81            {
82                warnings.AddRange(plan.PublishNotifications().Where(warning =>
83                    !string.IsNullOrWhiteSpace(warning)));
84            }
85            catch (Exception exception)
86            {
87                warnings.Add($"Load observer '{plan.ParticipantID}' failed: {exception.Message}");
88            }
89        }
90        foreach (INonThrowingLoadCommitPlan plan in _plans)
91        {
92            try
93            {
94                plan.CompleteCommit();
95            }
96            catch (Exception exception)
97            {
98                warnings.Add($"Load participant '{plan.ParticipantID}' cleanup failed: {exception.Message}");
99            }
100        }
101        return warnings.AsReadOnly();
102    }
103
104    public void Dispose()
105    {
106        if (_disposed)
107            return;
108        _disposed = true;
109        foreach (INonThrowingLoadCommitPlan plan in _plans)
110        {
111            try
112            {
113                plan.Dispose();
114            }
115            catch
116            {
117                // Abandoning a preflight plan must continue releasing every participant.
118            }
119        }
120    }
121}
122
123internal sealed class LoadPreflightInvalidException(string message) : InvalidOperationException(message);

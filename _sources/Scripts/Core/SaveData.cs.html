84internal sealed class PreparedSaveSlot
85{
86    private readonly PreparedSaveParticipant[] _participants;
87
88    internal PreparedSaveSlot(
89        string slotID,
90        IEnumerable<PreparedSaveParticipant> participants)
91    {
92        if (string.IsNullOrWhiteSpace(slotID))
93            throw new ArgumentException("Prepared slot ID cannot be empty.", nameof(slotID));
94        ArgumentNullException.ThrowIfNull(participants);
95        SlotID = slotID;
96        _participants = participants.ToArray();
97    }
98
99    internal string SlotID { get; }
100    internal IReadOnlyList<PreparedSaveParticipant> Participants => _participants;
101
102    internal IPreparedSaveState GetPreparedState(IStreamingLoadTarget target)
103    {
104        ArgumentNullException.ThrowIfNull(target);
105        PreparedSaveParticipant? match = null;
106        foreach (PreparedSaveParticipant participant in _participants)
107        {
108            if (!ReferenceEquals(participant.Target, target))
109                continue;
110            if (match is not null)
111                throw new InvalidOperationException("A load target has multiple prepared states.");
112            match = participant;
113        }
114        return match?.State ?? throw new InvalidOperationException(
115            $"Load target '{target.SaveFileName}' has no prepared state.");
116    }
117
118    internal int CommitLegacy()
119    {
120        foreach (PreparedSaveParticipant participant in _participants)
121            participant.Target.CommitPreparedLoad(participant.State);
122        return _participants.Length;
123    }
124}

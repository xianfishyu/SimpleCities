1using System;
2using System.Collections.Generic;
3using System.IO;
4using System.Text.Json;
5
6internal sealed record SaveDeletionDescriptor(
7    string SlotID,
8    string OperationToken,
9    long UIGeneration,
10    SaveSlotOccupantKind OccupantKind,
11    string OccupantDigest,
12    string TombstonePath,
13    string ConfirmationSummary);
14
15internal static class SaveDeletionDescriptorCodec
16{
17    internal const int MaximumEncodedBytes = 4 * 1024;
18    internal const string TombstonePath = "tombstone";
19
20    internal static void Write(Stream destination, SaveDeletionDescriptor descriptor)
21    {
22        ArgumentNullException.ThrowIfNull(destination);
23        ArgumentNullException.ThrowIfNull(descriptor);
24        Validate(descriptor);
25
26        using var writer = new Utf8JsonWriter(destination, new JsonWriterOptions
27        {
28            Indented = false,
29            SkipValidation = false,
30        });
31        writer.WriteStartObject();
32        writer.WriteString("formatFamily", SaveSlotJson.FormatFamily);
33        writer.WriteNumber("schemaVersion", SaveSlotJson.SchemaVersion);
34        writer.WriteString("descriptorType", "delete");
35        writer.WriteString("slotId", descriptor.SlotID);
36        writer.WriteString("operationToken", descriptor.OperationToken);
37        writer.WriteNumber("uiGeneration", descriptor.UIGeneration);
38        writer.WriteString("occupantKind", ToToken(descriptor.OccupantKind));
39        writer.WriteString("occupantDigest", descriptor.OccupantDigest);
40        writer.WriteString("tombstonePath", descriptor.TombstonePath);
41        writer.WriteString("confirmationSummary", descriptor.ConfirmationSummary);
42        writer.WriteEndObject();
43        writer.Flush();
44    }
45
46    internal static SaveDeletionDescriptor Read(Stream source)
47    {
48        ArgumentNullException.ThrowIfNull(source);
49        if (!source.CanSeek)
50            throw new InvalidDataException("Deletion descriptor stream must be seekable.");
51        if (source.Length is < 2 or > MaximumEncodedBytes)
52            throw new InvalidDataException("Deletion descriptor exceeds its byte budget.");
53
54        using JsonDocument document = JsonDocument.Parse(source, new JsonDocumentOptions
55        {
56            AllowTrailingCommas = false,
57            CommentHandling = JsonCommentHandling.Disallow,
58            MaxDepth = 4,
59        });
60        IReadOnlyDictionary<string, JsonElement> fields = SaveSlotJson.ReadObject(
61            document.RootElement,
62            "V3 deletion descriptor",
63            "formatFamily",
64            "schemaVersion",
65            "descriptorType",
66            "slotId",
67            "operationToken",
68            "uiGeneration",
69            "occupantKind",
70            "occupantDigest",
71            "tombstonePath",
72            "confirmationSummary");
73        if (SaveSlotJson.ReadString(fields["formatFamily"], "deletion formatFamily") != SaveSlotJson.FormatFamily)
74            throw new JsonException("Deletion formatFamily is not simple-cities-v3.");
75        if (SaveSlotJson.ReadNonNegativeInt32(fields["schemaVersion"], "deletion schemaVersion") !=
76            SaveSlotJson.SchemaVersion)
77        {
78            throw new JsonException("Deletion schemaVersion is not supported.");
79        }
80        if (SaveSlotJson.ReadString(fields["descriptorType"], "deletion descriptorType") != "delete")
81            throw new JsonException("Deletion descriptorType is not delete.");
82
83        long generation = SaveSlotJson.ReadNonNegativeInt64(fields["uiGeneration"], "deletion uiGeneration");
84        if (generation == 0)
85            throw new JsonException("Deletion uiGeneration must be positive.");
86        var descriptor = new SaveDeletionDescriptor(
87            SaveSlotJson.ReadString(fields["slotId"], "deletion slotId"),
88            SaveSlotJson.ReadString(fields["operationToken"], "deletion operationToken"),
89            generation,
90            ParseOccupantKind(SaveSlotJson.ReadString(fields["occupantKind"], "deletion occupantKind")),
91            SaveSlotJson.ReadCanonicalSha256(fields["occupantDigest"], "deletion occupantDigest"),
92            SaveSlotJson.ReadString(fields["tombstonePath"], "deletion tombstonePath"),
93            SaveSlotJson.ReadString(fields["confirmationSummary"], "deletion confirmationSummary"));
94        Validate(descriptor);
95        return descriptor;
96    }
97
98    private static void Validate(SaveDeletionDescriptor descriptor)
99    {
100        SaveSlotStore.ValidateSlotID(descriptor.SlotID);
101        SavePublicationDescriptorCodec.ValidateOperationToken(descriptor.OperationToken);
102        if (descriptor.UIGeneration <= 0)
103            throw new JsonException("Deletion uiGeneration must be positive.");
104        _ = ToToken(descriptor.OccupantKind);
105        ValidateDigest(descriptor.OccupantDigest);
106        if (descriptor.TombstonePath != TombstonePath)
107            throw new JsonException("Deletion tombstonePath is not canonical.");
108        SaveSlotManifestCodec.ValidateText(descriptor.ConfirmationSummary, "deletion confirmationSummary");
109    }
110
111    private static string ToToken(SaveSlotOccupantKind kind) => kind switch
112    {
113        SaveSlotOccupantKind.CompleteV3 => "completeV3",
114        SaveSlotOccupantKind.CorruptV3 => "corruptV3",
115        _ => throw new JsonException("Deletion occupantKind must be completeV3 or corruptV3."),
116    };
117
118    private static SaveSlotOccupantKind ParseOccupantKind(string token) => token switch
119    {
120        "completeV3" => SaveSlotOccupantKind.CompleteV3,
121        "corruptV3" => SaveSlotOccupantKind.CorruptV3,
122        _ => throw new JsonException("Deletion occupantKind is not supported."),
123    };
124
125    private static void ValidateDigest(string digest)
126    {
127        if (digest.Length != 64)
128            throw new JsonException("Deletion occupantDigest must be a canonical SHA-256 digest.");
129        foreach (char character in digest)
130        {
131            if (!char.IsAsciiHexDigit(character) || char.IsAsciiLetterUpper(character))
132                throw new JsonException("Deletion occupantDigest must be a canonical SHA-256 digest.");
133        }
134    }

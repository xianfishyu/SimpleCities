1using System.Text.Json;
2
3/// <summary>
4/// JSON 序列化/反序列化静态工具。
5/// 统一序列化选项，供 SaveManager 和各子系统使用。
6/// </summary>
7public static class SaveJson
8{
9    private static readonly JsonSerializerOptions Options = new()
10    {
11        WriteIndented = true,
12        PropertyNameCaseInsensitive = true
13    };
14
15    /// <summary>将对象序列化为 JSON 字符串</summary>
16    public static string Serialize(object data) =>
17        JsonSerializer.Serialize(data, Options);
18
19    /// <summary>将 JSON 字符串反序列化为指定类型</summary>
20    public static T Deserialize<T>(string json) =>
21        JsonSerializer.Deserialize<T>(json, Options)!;
22}

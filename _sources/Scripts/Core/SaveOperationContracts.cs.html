45public sealed record SaveOperationResult
46{
47    public string OperationToken { get; }
48    public SaveOperationKind Kind { get; }
49    public string TargetSlotID { get; }
50    public SaveOperationResultKind ResultKind { get; }
51    public SaveOperationPhase FinalPhase { get; }
52    public bool Committed { get; }
53    public IReadOnlyList<string> Warnings { get; }
54    public string? Error { get; }
55
56    public bool IsSuccess => ResultKind is SaveOperationResultKind.Succeeded or
57        SaveOperationResultKind.SucceededWithWarnings;
58
59    public SaveOperationResult(
60        string operationToken,
61        SaveOperationKind kind,
62        string targetSlotID,
63        SaveOperationResultKind resultKind,
64        SaveOperationPhase finalPhase,
65        bool committed,
66        IEnumerable<string>? warnings = null,
67        string? error = null)
68    {
69        if (string.IsNullOrWhiteSpace(operationToken))
70            throw new ArgumentException("Operation token cannot be empty.", nameof(operationToken));
71        if (string.IsNullOrWhiteSpace(targetSlotID))
72            throw new ArgumentException("Target slot ID cannot be empty.", nameof(targetSlotID));
73        if (!Enum.IsDefined(kind))
74            throw new ArgumentOutOfRangeException(nameof(kind));
75        if (!Enum.IsDefined(resultKind))
76            throw new ArgumentOutOfRangeException(nameof(resultKind));
77        if (!Enum.IsDefined(finalPhase))
78            throw new ArgumentOutOfRangeException(nameof(finalPhase));
79        if (committed && resultKind is SaveOperationResultKind.Canceled or
80            SaveOperationResultKind.SkippedBusy or SaveOperationResultKind.RejectedShuttingDown or
81            SaveOperationResultKind.RejectedSceneClosing)
82        {
83            throw new ArgumentException("A canceled, skipped, or rejected operation cannot be committed.");
84        }
85
86        OperationToken = operationToken;
87        Kind = kind;
88        TargetSlotID = targetSlotID;
89        ResultKind = resultKind;
90        FinalPhase = finalPhase;
91        Committed = committed;
92        Warnings = Array.AsReadOnly((warnings ?? []).Where(warning =>
93            !string.IsNullOrWhiteSpace(warning)).ToArray());
94        Error = error;
95    }
96}

109internal sealed class UncoordinatedStorageOperationLease : IStorageOperationLease
110{
111    internal UncoordinatedStorageOperationLease(
112        SaveOperationKind kind,
113        string? operationToken = null)

116        OperationToken = operationToken ?? Guid.NewGuid().ToString("N");
117    }
118
119    public string OperationToken { get; }
120    public SaveOperationKind Kind { get; }
121    public void ThrowIfCancellationRequested() { }
122    public void AcquireCommitLease() { }
123    public void CrossCommitBoundary(Action boundaryAction)
124    {
125        ArgumentNullException.ThrowIfNull(boundaryAction);
126        boundaryAction();
127    }
128    public void MarkCommitted() { }
129    public void EnterCommitBoundary()
130    {
131        CrossCommitBoundary(static () => { });
132        MarkCommitted();
133    }

12    internal static string? Validate(string path)
13    {
14        try
15        {
16            ValidateCore(path);
17            return null;
18        }
19        catch (Exception exception) when (exception is InvalidDataException or IOException or UnauthorizedAccessException)
20        {
21            return exception.Message;
22        }
23    }

25    private static void ValidateCore(string path)
26    {
27        SaveSlotStore.EnsureOrdinaryFile(path, "thumbnail");
28        using var source = new FileStream(
29            path,
30            FileMode.Open,
31            FileAccess.Read,
32            FileShare.Read,
33            bufferSize: 16 * 1024,
34            FileOptions.SequentialScan);
35        long initialLength = source.Length;
36        if (initialLength is <= 0 or > SaveSlotStorageBudget.MaximumThumbnailEncodedBytes)
37            throw new InvalidDataException("Thumbnail exceeds its encoded byte budget.");
38
39        byte[] encoded = new byte[checked((int)initialLength)];
40        source.ReadExactly(encoded);
41        if (source.Length != initialLength || source.ReadByte() != -1)
42            throw new InvalidDataException("Thumbnail changed while it was being read.");
43        ValidateEncodedPng(encoded);
44    }

46    internal static void ValidateEncodedPng(ReadOnlySpan<byte> encoded)
47    {
48        if (encoded.Length > SaveSlotStorageBudget.MaximumThumbnailEncodedBytes)
49            throw new InvalidDataException("Thumbnail exceeds its encoded byte budget.");
50        if (encoded.Length < Signature.Length || !encoded[..Signature.Length].SequenceEqual(Signature))
51            throw new InvalidDataException("Thumbnail does not have a PNG signature.");
52
53        int offset = Signature.Length;
54        bool sawHeader = false;
55        bool sawPalette = false;
56        bool sawImageData = false;
57        bool imageDataEnded = false;
58        bool sawEnd = false;
59        int width = 0;
60        int height = 0;
61        int bitDepth = 0;
62        int colorType = 0;
63        int interlace = 0;
64        using var imageData = new MemoryStream();
65        while (offset < encoded.Length)
66        {
67            if (encoded.Length - offset < 12)
68                throw new InvalidDataException("Thumbnail contains a truncated PNG chunk.");
69            uint lengthValue = BinaryPrimitives.ReadUInt32BigEndian(encoded.Slice(offset, 4));
70            if (lengthValue > int.MaxValue)
71                throw new InvalidDataException("Thumbnail PNG chunk length is invalid.");
72            int length = (int)lengthValue;
73            offset += 4;
74            ReadOnlySpan<byte> type = encoded.Slice(offset, 4);
75            offset += 4;
76            ValidateChunkType(type);
77            if (length > encoded.Length - offset - 4)
78                throw new InvalidDataException("Thumbnail PNG chunk length is invalid.");
79            ReadOnlySpan<byte> data = encoded.Slice(offset, length);
80            uint expectedCrc = BinaryPrimitives.ReadUInt32BigEndian(encoded.Slice(offset + length, 4));
81            if (ComputeCrc32(type, data) != expectedCrc)
82                throw new InvalidDataException("Thumbnail PNG chunk CRC is invalid.");
83            offset += length + 4;
84
85            string chunkType = Encoding.ASCII.GetString(type);
86            switch (chunkType)
87            {
88                case "IHDR":
89                    if (sawHeader || offset - length - 12 != Signature.Length || length != 13)
90                        throw new InvalidDataException("Thumbnail PNG IHDR is invalid.");
91                    width = checked((int)BinaryPrimitives.ReadUInt32BigEndian(data[..4]));
92                    height = checked((int)BinaryPrimitives.ReadUInt32BigEndian(data.Slice(4, 4)));
93                    bitDepth = data[8];
94                    colorType = data[9];
95                    if (data[10] != 0 || data[11] != 0 || data[12] > 1)
96                        throw new InvalidDataException("Thumbnail PNG uses an unsupported compression, filter, or interlace method.");
97                    interlace = data[12];
98                    ValidateDimensionsAndFormat(width, height, bitDepth, colorType);
99                    sawHeader = true;
100                    break;
101                case "PLTE":
102                    if (!sawHeader || sawImageData || sawPalette || length is 0 or > 768 || length % 3 != 0)
103                        throw new InvalidDataException("Thumbnail PNG palette is invalid.");
104                    sawPalette = true;
105                    break;
106                case "IDAT":
107                    if (!sawHeader || imageDataEnded || length == 0)
108                        throw new InvalidDataException("Thumbnail PNG image data ordering is invalid.");
109                    if (colorType == 3 && !sawPalette)
110                        throw new InvalidDataException("Indexed thumbnail PNG is missing its palette.");
111                    if (imageData.Length > SaveSlotStorageBudget.MaximumThumbnailEncodedBytes - length)
112                        throw new InvalidDataException("Thumbnail PNG image data exceeds its byte budget.");
113                    imageData.Write(data);
114                    sawImageData = true;
115                    break;
116                case "IEND":
117                    if (!sawHeader || !sawImageData || sawEnd || length != 0)
118                        throw new InvalidDataException("Thumbnail PNG IEND is invalid.");
119                    sawEnd = true;
120                    imageDataEnded = true;
121                    if (offset != encoded.Length)
122                        throw new InvalidDataException("Thumbnail PNG contains bytes after IEND.");
123                    break;
124                default:
125                    if (!sawHeader || sawEnd)
126                        throw new InvalidDataException("Thumbnail PNG chunk ordering is invalid.");
127                    if (sawImageData)
128                        imageDataEnded = true;
129                    if (char.IsAsciiLetterUpper(chunkType[0]))
130                        throw new InvalidDataException($"Thumbnail PNG contains unsupported critical chunk '{chunkType}'.");
131                    break;
132            }
133        }
134
135        if (!sawEnd)
136            throw new InvalidDataException("Thumbnail PNG is missing IEND.");
137        ValidateDecodedScanlines(imageData.ToArray(), width, height, bitDepth, colorType, interlace);
138    }

140    private static void ValidateChunkType(ReadOnlySpan<byte> type)
141    {
142        foreach (byte value in type)
143        {
144            if (value is not (>= (byte)'A' and <= (byte)'Z') and
145                not (>= (byte)'a' and <= (byte)'z'))
146            {
147                throw new InvalidDataException("Thumbnail PNG chunk type is invalid.");
148            }
149        }
150        if (type[2] is >= (byte)'a' and <= (byte)'z')
151            throw new InvalidDataException("Thumbnail PNG chunk type uses an invalid reserved bit.");
152    }

154    private static void ValidateDimensionsAndFormat(int width, int height, int bitDepth, int colorType)
155    {
156        if (width <= 0 || height <= 0 ||
157            width > SaveSlotStorageBudget.MaximumThumbnailDimension ||
158            height > SaveSlotStorageBudget.MaximumThumbnailDimension ||
159            (long)width * height > SaveSlotStorageBudget.MaximumThumbnailPixels)
160        {
161            throw new InvalidDataException("Thumbnail PNG exceeds its dimension or pixel budget.");
162        }
163
164        bool valid = colorType switch
165        {
166            0 => bitDepth is 1 or 2 or 4 or 8 or 16,
167            2 => bitDepth is 8 or 16,
168            3 => bitDepth is 1 or 2 or 4 or 8,
169            4 => bitDepth is 8 or 16,
170            6 => bitDepth is 8 or 16,
171            _ => false,
172        };
173        if (!valid)
174            throw new InvalidDataException("Thumbnail PNG color type and bit depth are invalid.");
175    }

177    private static void ValidateDecodedScanlines(
178        byte[] compressed,
179        int width,
180        int height,
181        int bitDepth,
182        int colorType,
183        int interlace)
184    {
185        long[] rowLengths = interlace == 0
186            ? EnumerableRows(height, RowBytes(width, bitDepth, colorType))
187            : Adam7RowLengths(width, height, bitDepth, colorType);
188        long expected = 0;
189        foreach (long rowLength in rowLengths)
190            expected = checked(expected + rowLength + 1);
191        if (expected > SaveSlotStorageBudget.MaximumThumbnailDecodedBytes)
192            throw new InvalidDataException("Thumbnail PNG decoded data exceeds its byte budget.");
193
194        using var compressedStream = new MemoryStream(compressed, writable: false);
195        using var inflater = new ZLibStream(compressedStream, CompressionMode.Decompress, leaveOpen: false);
196        byte[] chunk = new byte[64 * 1024];
197        long decoded = 0;
198        int rowIndex = 0;
199        long rowOffset = 0;
200        int read;
201        while ((read = inflater.Read(chunk, 0, chunk.Length)) > 0)
202        {
203            decoded += read;
204            if (decoded > expected)
205                throw new InvalidDataException("Thumbnail PNG decodes beyond its expected scanline budget.");
206            for (int index = 0; index < read; index++)
207            {
208                if (rowIndex >= rowLengths.Length)
209                    throw new InvalidDataException("Thumbnail PNG contains excess decoded scanlines.");
210                if (rowOffset == 0 && chunk[index] > 4)
211                    throw new InvalidDataException("Thumbnail PNG uses an invalid scanline filter.");
212                rowOffset++;
213                if (rowOffset == rowLengths[rowIndex] + 1)
214                {
215                    rowOffset = 0;
216                    rowIndex++;
217                }
218            }
219        }
220        if (decoded != expected || rowOffset != 0 || rowIndex != rowLengths.Length)
221            throw new InvalidDataException("Thumbnail PNG decoded scanline length is invalid.");
222    }

224    private static long[] EnumerableRows(int height, long rowBytes)
225    {
226        var rows = new long[height];
227        Array.Fill(rows, rowBytes);
228        return rows;
229    }

231    private static long[] Adam7RowLengths(int width, int height, int bitDepth, int colorType)
232    {
233        int[] xStart = [0, 4, 0, 2, 0, 1, 0];
234        int[] yStart = [0, 0, 4, 0, 2, 0, 1];
235        int[] xStep = [8, 8, 4, 4, 2, 2, 1];
236        int[] yStep = [8, 8, 8, 4, 4, 2, 2];
237        var rows = new List<long>();
238        for (int pass = 0; pass < 7; pass++)
239        {
240            int passWidth = PassSize(width, xStart[pass], xStep[pass]);
241            int passHeight = PassSize(height, yStart[pass], yStep[pass]);
242            if (passWidth == 0 || passHeight == 0)
243                continue;
244            long rowBytes = RowBytes(passWidth, bitDepth, colorType);
245            for (int row = 0; row < passHeight; row++)
246                rows.Add(rowBytes);
247        }
248        return rows.ToArray();
249    }
250
251    private static int PassSize(int size, int start, int step) =>
252        size <= start ? 0 : (size - start + step - 1) / step;
253
254    private static long RowBytes(int width, int bitDepth, int colorType)
255    {
256        int channels = colorType switch
257        {
258            0 or 3 => 1,
259            2 => 3,
260            4 => 2,
261            6 => 4,
262            _ => throw new InvalidDataException("Thumbnail PNG color type is invalid."),
263        };
264        return checked(((long)width * channels * bitDepth + 7) / 8);
265    }

267    private static uint ComputeCrc32(ReadOnlySpan<byte> type, ReadOnlySpan<byte> data)
268    {
269        uint crc = uint.MaxValue;
270        foreach (byte value in type)
271            crc = UpdateCrc(crc, value);
272        foreach (byte value in data)
273            crc = UpdateCrc(crc, value);
274        return ~crc;
275    }
276
277    private static uint UpdateCrc(uint crc, byte value)
278    {
279        crc ^= value;
280        for (int bit = 0; bit < 8; bit++)
281            crc = (crc & 1) != 0 ? (crc >> 1) ^ 0xedb88320u : crc >> 1;
282        return crc;
283    }

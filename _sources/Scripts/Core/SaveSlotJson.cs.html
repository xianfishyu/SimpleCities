11    internal static IReadOnlyDictionary<string, JsonElement> ReadObject(
12        JsonElement element,
13        string context,
14        params string[] expectedProperties)
15    {
16        if (element.ValueKind != JsonValueKind.Object)
17            throw new JsonException($"{context} must be a JSON object.");
18
19        var expected = new HashSet<string>(expectedProperties, StringComparer.Ordinal);
20        var properties = new Dictionary<string, JsonElement>(StringComparer.Ordinal);
21        foreach (JsonProperty property in element.EnumerateObject())
22        {
23            if (!properties.TryAdd(property.Name, property.Value))
24                throw new JsonException($"{context} contains duplicate property '{property.Name}'.");
25            if (!expected.Contains(property.Name))
26                throw new JsonException($"{context} contains unknown property '{property.Name}'.");
27        }
28
29        foreach (string propertyName in expectedProperties)
30        {
31            if (!properties.ContainsKey(propertyName))
32                throw new JsonException($"{context} is missing property '{propertyName}'.");
33        }
34        return properties;
35    }

37    internal static JsonElement ReadArray(JsonElement element, string context)
38    {
39        if (element.ValueKind != JsonValueKind.Array)
40            throw new JsonException($"{context} must be a JSON array.");
41        return element;
42    }

44    internal static string ReadString(JsonElement element, string context)
45    {
46        if (element.ValueKind != JsonValueKind.String)
47            throw new JsonException($"{context} must be a JSON string.");
48        return element.GetString() ?? throw new JsonException($"{context} cannot be null.");
49    }
50
51    internal static string? ReadNullableString(JsonElement element, string context) =>
52        element.ValueKind == JsonValueKind.Null ? null : ReadString(element, context);

54    internal static int ReadNonNegativeInt32(JsonElement element, string context)
55    {
56        if (element.ValueKind != JsonValueKind.Number)
57            throw new JsonException($"{context} must be an integer number token.");
58        return ReadNonNegativeInt32(element.GetRawText(), context);
59    }
60
61    internal static int ReadNonNegativeInt32(string token, string context)
62    {
63        token = ReadCanonicalUnsignedIntegerToken(token, context, 10);
64        if (!int.TryParse(token, NumberStyles.None, CultureInfo.InvariantCulture, out int value))
65            throw new JsonException($"{context} exceeds Int32 range.");
66        return value;
67    }

76    internal static long ReadNonNegativeInt64(string token, string context)
77    {
78        token = ReadCanonicalUnsignedIntegerToken(token, context, 19);

80            throw new JsonException($"{context} exceeds Int64 range.");

84    internal static string ReadCanonicalSha256(JsonElement element, string context)
85    {

89    internal static string ReadCanonicalSha256(string value, string context)
90    {
91        if (value.Length != 64)
92            throw new JsonException($"{context} must contain 64 lowercase hexadecimal characters.");
93        foreach (char character in value)
94        {
95            if (!char.IsAsciiHexDigit(character) || char.IsAsciiLetterUpper(character))
96                throw new JsonException($"{context} must contain 64 lowercase hexadecimal characters.");
97        }
98        return value;
99    }
100
101    private static string ReadCanonicalUnsignedIntegerToken(
102        string token,
103        string context,
104        int maximumLength)
105    {
106        if (token.Length == 0 || token.Length > maximumLength ||
107            (token.Length > 1 && token[0] == '0'))
108        {
109            throw new JsonException($"{context} must use canonical unsigned decimal syntax.");
110        }
111        foreach (char character in token)
112        {
113            if (!char.IsAsciiDigit(character))
114                throw new JsonException($"{context} must use canonical unsigned decimal syntax.");
115        }
116        return token;
117    }

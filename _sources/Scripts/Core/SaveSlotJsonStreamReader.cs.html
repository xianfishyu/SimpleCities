15    internal void Validate()
16    {
17        if (MaximumEncodedBytes <= 0)
18            throw new ArgumentOutOfRangeException(nameof(MaximumEncodedBytes));
19        if (MaximumTokens <= 0)
20            throw new ArgumentOutOfRangeException(nameof(MaximumTokens));
21        if (MaximumDepth <= 0)
22            throw new ArgumentOutOfRangeException(nameof(MaximumDepth));
23        if (MaximumPropertyNameBytes <= 0)
24            throw new ArgumentOutOfRangeException(nameof(MaximumPropertyNameBytes));
25        if (MaximumStringTokenBytes <= 0)
26            throw new ArgumentOutOfRangeException(nameof(MaximumStringTokenBytes));
27        if (MaximumNumberTokenBytes <= 0)
28            throw new ArgumentOutOfRangeException(nameof(MaximumNumberTokenBytes));
29    }
30}

54internal sealed class SaveSlotJsonStreamReader : IDisposable
55{
56    private const int BufferSize = 8 * 1024;
57
58    private readonly Stream _source;
59    private readonly SaveSlotJsonReaderBudget _budget;
60    private readonly byte[] _buffer = ArrayPool<byte>.Shared.Rent(BufferSize);
61    private readonly byte[] _overflowProbe = new byte[1];
62    private JsonReaderState _state;
63    private int _buffered;
64    private int _consumed;
65    private long _discarded;
66    private long _bytesRead;
67    private long _tokenCount;
68    private bool _isFinalBlock;
69    private bool _finished;
70    private bool _returnedBuffer;
71    private string? _textValue;
72
73    internal SaveSlotJsonStreamReader(Stream source, SaveSlotJsonReaderBudget budget)
74    {
75        ArgumentNullException.ThrowIfNull(source);
76        if (!source.CanRead)
77            throw new ArgumentException("JSON source must be readable.", nameof(source));
78        budget.Validate();
79        _source = source;
80        _budget = budget;
81        _state = new JsonReaderState(new JsonReaderOptions
82        {
83            AllowTrailingCommas = false,
84            CommentHandling = JsonCommentHandling.Disallow,
85            MaxDepth = budget.MaximumDepth,
86        });
87    }
88
89    ~SaveSlotJsonStreamReader() => ReturnBuffer();
90
91    internal JsonTokenType TokenType { get; private set; }
92    internal long BytesRead => _bytesRead;
93    internal long BytesConsumed => _discarded + _consumed;
94    internal long TokenCount => _tokenCount;
95
96    internal bool Read()
97    {
98        if (_finished)
99            return false;
100        _textValue = null;
101
102        while (true)
103        {
104            CompactConsumedBytes();
105            FillAvailableBuffer();
106
107            var reader = new Utf8JsonReader(
108                _buffer.AsSpan(0, _buffered),
109                _isFinalBlock,
110                _state);
111            if (reader.Read())
112            {
113                _state = reader.CurrentState;
114                _consumed = checked((int)reader.BytesConsumed);
115                TokenType = reader.TokenType;
116                _tokenCount++;
117                if (_tokenCount > _budget.MaximumTokens)
118                    throw new JsonException("JSON token budget is exceeded.");
119                CaptureTokenValue(ref reader);
120                return true;
121            }
122
123            _state = reader.CurrentState;
124            _consumed = checked((int)reader.BytesConsumed);
125            if (_isFinalBlock)
126            {
127                _finished = true;
128                ReturnBuffer();
129                return false;
130            }
131            if (_consumed == 0 && _buffered == _buffer.Length)
132                throw new JsonException("JSON token exceeds its bounded reader buffer.");
133        }
134    }
135
136    internal void ReadRequired(string context)
137    {
138        if (!Read())
139            throw new JsonException($"{context} ended before its required JSON token.");
140    }
141
142    internal void Expect(JsonTokenType expected, string context)
143    {
144        if (TokenType != expected)
145            throw new JsonException($"{context} must be {expected}, not {TokenType}.");
146    }
147
148    internal string GetPropertyName(string context)
149    {
150        Expect(JsonTokenType.PropertyName, context);
151        return _textValue ?? throw new JsonException($"{context} property name is unavailable.");
152    }
153
154    internal string GetString(string context)
155    {
156        Expect(JsonTokenType.String, context);
157        return _textValue ?? throw new JsonException($"{context} string value cannot be null.");
158    }
159
160    internal string GetNumberToken(string context)
161    {
162        Expect(JsonTokenType.Number, context);
163        return _textValue ?? throw new JsonException($"{context} number token is unavailable.");
164    }
165
166    internal void RequireEndOfDocument(string context)
167    {
168        if (Read())
169            throw new JsonException($"{context} contains more than one JSON value.");
170        if (BytesConsumed != BytesRead)
171            throw new JsonException($"{context} was not consumed to EOF.");
172    }
173
174    public void Dispose()
175    {
176        ReturnBuffer();
177        GC.SuppressFinalize(this);
178    }
179
180    private void CaptureTokenValue(ref Utf8JsonReader reader)
181    {
182        ReadOnlySpan<byte> value = reader.HasValueSequence
183            ? reader.ValueSequence.ToArray()
184            : reader.ValueSpan;
185        switch (reader.TokenType)
186        {
187            case JsonTokenType.PropertyName:
188                if (reader.ValueIsEscaped || value.Length > _budget.MaximumPropertyNameBytes)
189                    throw new JsonException("JSON property name exceeds its raw UTF-8 budget or is escaped.");
190                _textValue = Encoding.UTF8.GetString(value);
191                break;
192            case JsonTokenType.String:
193                if (value.Length > _budget.MaximumStringTokenBytes)
194                    throw new JsonException("JSON string token exceeds its raw UTF-8 budget.");
195                _textValue = reader.GetString();
196                break;
197            case JsonTokenType.Number:
198                if (value.Length > _budget.MaximumNumberTokenBytes)
199                    throw new JsonException("JSON number token exceeds its lexeme budget.");
200                _textValue = Encoding.UTF8.GetString(value);
201                break;
202        }
203    }
204
205    private void CompactConsumedBytes()
206    {
207        if (_consumed == 0)
208            return;
209        int remaining = _buffered - _consumed;
210        if (remaining > 0)
211            Buffer.BlockCopy(_buffer, _consumed, _buffer, 0, remaining);
212        _discarded += _consumed;
213        _buffered = remaining;
214        _consumed = 0;
215    }
216
217    private void FillAvailableBuffer()
218    {
219        if (_isFinalBlock || _buffered == _buffer.Length)
220            return;
221
222        long remainingBudget = _budget.MaximumEncodedBytes - _bytesRead;
223        if (remainingBudget <= 0)
224        {
225            int overflow = _source.Read(_overflowProbe, 0, 1);
226            if (overflow == 0)
227            {
228                _isFinalBlock = true;
229                return;
230            }
231            throw new JsonException("JSON encoded byte budget is exceeded.");
232        }
233
234        int requested = (int)Math.Min(_buffer.Length - _buffered, remainingBudget);
235        int read = _source.Read(_buffer, _buffered, requested);
236        if (read == 0)
237        {
238            _isFinalBlock = true;
239            return;
240        }
241        _buffered += read;
242        _bytesRead += read;
243    }
244
245    private void ReturnBuffer()
246    {
247        if (_returnedBuffer)
248            return;
249        _returnedBuffer = true;
250        ArrayPool<byte>.Shared.Return(_buffer);
251    }

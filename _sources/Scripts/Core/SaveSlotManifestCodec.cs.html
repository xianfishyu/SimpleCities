15    internal static void Write(Stream destination, SaveSlotManifest manifest)
16    {
17        ArgumentNullException.ThrowIfNull(destination);
18        ArgumentNullException.ThrowIfNull(manifest);
19        ValidateModel(manifest);
20
21        using var writer = new Utf8JsonWriter(destination, new JsonWriterOptions
22        {
23            Indented = false,
24            SkipValidation = false,
25        });
26        writer.WriteStartObject();
27        writer.WriteString("formatFamily", SaveSlotJson.FormatFamily);
28        writer.WriteNumber("schemaVersion", SaveSlotJson.SchemaVersion);
29        writer.WriteString("slotId", manifest.SlotID);
30        writer.WriteString("displayName", manifest.DisplayName);
31        writer.WriteString("timestamp", manifest.Timestamp);
32        writer.WriteString("cityName", manifest.CityName);
33        if (manifest.Population.HasValue)
34            writer.WriteNumber("population", manifest.Population.Value);
35        else
36            writer.WriteNull("population");
37        if (manifest.Funds.HasValue)
38            writer.WriteNumber("funds", manifest.Funds.Value);
39        else
40            writer.WriteNull("funds");
41        if (manifest.ThumbnailFile is null)
42            writer.WriteNull("thumbnailFile");
43        else
44            writer.WriteString("thumbnailFile", manifest.ThumbnailFile);
45        writer.WriteStartArray("files");
46        foreach (SaveSlotManifestFile file in manifest.Files.OrderBy(file => file.Name, StringComparer.Ordinal))
47        {
48            writer.WriteStartObject();
49            writer.WriteString("name", file.Name);
50            writer.WriteNumber("encodedLength", file.EncodedLength);
51            writer.WriteString("sha256", file.Sha256);
52            writer.WriteEndObject();
53        }
54        writer.WriteEndArray();
55        writer.WriteEndObject();
56        writer.Flush();
57    }

65    internal static SaveSlotManifest Read(Stream source, ref bool declaresV3Family)
66    {
67        ArgumentNullException.ThrowIfNull(source);
68        declaresV3Family = false;
69        using var reader = new SaveSlotJsonStreamReader(source, SaveSlotStorageBudget.ManifestJson);
70        reader.ReadRequired("V3 manifest");
71        reader.Expect(JsonTokenType.StartObject, "V3 manifest");
72
73        ulong seen = 0;
74        string? formatFamily = null;
75        int schemaVersion = -1;
76        string? slotID = null;
77        string? displayName = null;
78        string? timestamp = null;
79        string? cityName = null;
80        long? population = null;
81        decimal? funds = null;
82        string? thumbnailFile = null;
83        List<SaveSlotManifestFile>? files = null;
84        while (ReadObjectProperty(reader, "V3 manifest", out string property))
85        {
86            reader.ReadRequired($"Manifest property '{property}'");
87            switch (property)
88            {
89                case "formatFamily":
90                    MarkField(ref seen, 1UL << 0, property, "V3 manifest");
91                    formatFamily = reader.GetString("manifest formatFamily");
92                    declaresV3Family = formatFamily == SaveSlotJson.FormatFamily;
93                    break;
94                case "schemaVersion":
95                    MarkField(ref seen, 1UL << 1, property, "V3 manifest");
96                    schemaVersion = SaveSlotJson.ReadNonNegativeInt32(
97                        reader.GetNumberToken("manifest schemaVersion"),
98                        "manifest schemaVersion");
99                    break;
100                case "slotId":
101                    MarkField(ref seen, 1UL << 2, property, "V3 manifest");
102                    slotID = reader.GetString("manifest slotId");
103                    break;
104                case "displayName":
105                    MarkField(ref seen, 1UL << 3, property, "V3 manifest");
106                    displayName = reader.GetString("manifest displayName");
107                    break;
108                case "timestamp":
109                    MarkField(ref seen, 1UL << 4, property, "V3 manifest");
110                    timestamp = reader.GetString("manifest timestamp");
111                    break;
112                case "cityName":
113                    MarkField(ref seen, 1UL << 5, property, "V3 manifest");
114                    cityName = reader.GetString("manifest cityName");
115                    break;
116                case "population":
117                    MarkField(ref seen, 1UL << 6, property, "V3 manifest");
118                    population = reader.TokenType == JsonTokenType.Null
119                        ? null
120                        : SaveSlotJson.ReadNonNegativeInt64(
121                            reader.GetNumberToken("manifest population"),
122                            "manifest population");
123                    break;
124                case "funds":
125                    MarkField(ref seen, 1UL << 7, property, "V3 manifest");
126                    funds = ReadNullableFunds(reader);
127                    break;
128                case "thumbnailFile":
129                    MarkField(ref seen, 1UL << 8, property, "V3 manifest");
130                    thumbnailFile = reader.TokenType == JsonTokenType.Null
131                        ? null
132                        : reader.GetString("manifest thumbnailFile");
133                    break;
134                case "files":
135                    MarkField(ref seen, 1UL << 9, property, "V3 manifest");
136                    files = ReadFiles(reader);
137                    break;
138                default:
139                    throw new JsonException($"V3 manifest contains unknown property '{property}'.");
140            }
141        }
142        RequireFields(seen, (1UL << 10) - 1, "V3 manifest");
143        reader.RequireEndOfDocument("V3 manifest");
144
145        if (formatFamily != SaveSlotJson.FormatFamily)
146            throw new JsonException("Manifest formatFamily is not simple-cities-v3.");
147        if (schemaVersion != SaveSlotJson.SchemaVersion)
148            throw new JsonException("Manifest schemaVersion is not supported.");
149
150        var manifest = new SaveSlotManifest(
151            slotID!,
152            displayName!,
153            timestamp!,
154            cityName!,
155            population,
156            funds,
157            thumbnailFile,
158            files!);
159        ValidateModel(manifest);
160        return manifest;
161    }

166    private static List<SaveSlotManifestFile> ReadFiles(SaveSlotJsonStreamReader reader)
167    {
168        reader.Expect(JsonTokenType.StartArray, "manifest files");
169        var files = new List<SaveSlotManifestFile>();
170        var caseInsensitiveNames = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
171        string? previousName = null;
172        long totalEncodedBytes = 0;
173        while (reader.Read())
174        {
175            if (reader.TokenType == JsonTokenType.EndArray)
176                return files;
177            if (files.Count >= SaveSlotStorageBudget.MaximumManifestFiles)
178                throw new JsonException("Manifest file item budget is exceeded.");
179            reader.Expect(JsonTokenType.StartObject, "manifest file");
180
181            ulong seen = 0;
182            string? name = null;
183            long encodedLength = -1;
184            string? sha256 = null;
185            while (ReadObjectProperty(reader, "manifest file", out string property))
186            {
187                reader.ReadRequired($"Manifest file property '{property}'");
188                switch (property)
189                {
190                    case "name":
191                        MarkField(ref seen, 1UL << 0, property, "manifest file");
192                        name = reader.GetString("manifest file name");
193                        break;
194                    case "encodedLength":
195                        MarkField(ref seen, 1UL << 1, property, "manifest file");
196                        encodedLength = SaveSlotJson.ReadNonNegativeInt64(
197                            reader.GetNumberToken("manifest file encodedLength"),
198                            "manifest file encodedLength");
199                        break;
200                    case "sha256":
201                        MarkField(ref seen, 1UL << 2, property, "manifest file");
202                        sha256 = SaveSlotJson.ReadCanonicalSha256(
203                            reader.GetString("manifest file sha256"),
204                            "manifest file sha256");
205                        break;
206                    default:
207                        throw new JsonException($"Manifest file contains unknown property '{property}'.");
208                }
209            }
210            RequireFields(seen, 0b111, "manifest file");
211            SaveSlotStore.ValidateManifestFileName(name!);
212            if (!caseInsensitiveNames.Add(name!))
213                throw new JsonException($"Manifest contains duplicate file '{name}'.");
214            if (previousName is not null && StringComparer.Ordinal.Compare(previousName, name) >= 0)
215                throw new JsonException("Manifest files must be sorted by ordinal file name.");
216            ValidatePayloadLength(encodedLength, name!);
217            if (totalEncodedBytes > SaveSlotStorageBudget.MaximumSlotEncodedBytes - encodedLength)
218                throw new JsonException("Manifest payloads exceed the slot encoded byte budget.");
219            totalEncodedBytes += encodedLength;
220            previousName = name;
221            files.Add(new SaveSlotManifestFile(name!, encodedLength, sha256!));
222        }
223        throw new JsonException("Manifest files array is incomplete.");
224    }

226    private static void ValidateModel(SaveSlotManifest manifest)
227    {
228        SaveSlotStore.ValidateSlotID(manifest.SlotID);
229        ValidateText(manifest.DisplayName, "displayName");
230        ValidateText(manifest.CityName, "cityName");
231        if (!DateTimeOffset.TryParseExact(
232                manifest.Timestamp,
233                TimestampFormat,
234                CultureInfo.InvariantCulture,
235                DateTimeStyles.AssumeUniversal | DateTimeStyles.AdjustToUniversal,
236                out DateTimeOffset timestamp) || timestamp.Offset != TimeSpan.Zero)
237        {
238            throw new JsonException("Manifest timestamp must use exact seven-digit UTC format.");
239        }
240        if (manifest.Population < 0)
241            throw new JsonException("Manifest population cannot be negative.");
242        if (manifest.Funds.HasValue && decimal.Round(manifest.Funds.Value, 2) != manifest.Funds.Value)
243            throw new JsonException("Manifest funds cannot contain more than two decimal places.");
244        if (manifest.ThumbnailFile is not null)
245            ValidateThumbnailFileName(manifest.ThumbnailFile);
246        ArgumentNullException.ThrowIfNull(manifest.Files);
247        if (manifest.Files.Count is < 1 or > SaveSlotStorageBudget.MaximumManifestFiles)
248            throw new JsonException("Manifest must contain between 1 and 64 business files.");
249        var names = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
250        long totalEncodedBytes = 0;
251        foreach (SaveSlotManifestFile file in manifest.Files)
252        {
253            ArgumentNullException.ThrowIfNull(file);
254            SaveSlotStore.ValidateManifestFileName(file.Name);
255            if (!names.Add(file.Name))
256                throw new JsonException($"Manifest contains duplicate file '{file.Name}'.");
257            ValidatePayloadLength(file.EncodedLength, file.Name);
258            _ = SaveSlotJson.ReadCanonicalSha256(file.Sha256, $"manifest file '{file.Name}' sha256");
259            if (totalEncodedBytes > SaveSlotStorageBudget.MaximumSlotEncodedBytes - file.EncodedLength)
260                throw new JsonException("Manifest payloads exceed the slot encoded byte budget.");
261            totalEncodedBytes += file.EncodedLength;
262        }
263    }

265    private static decimal? ReadNullableFunds(SaveSlotJsonStreamReader reader)
266    {
267        if (reader.TokenType == JsonTokenType.Null)
268            return null;
269        string token = reader.GetNumberToken("manifest funds");
270        if (!IsCanonicalFundsToken(token) || !decimal.TryParse(
271                token,
272                NumberStyles.AllowLeadingSign | NumberStyles.AllowDecimalPoint,
273                CultureInfo.InvariantCulture,
274                out decimal value))
275        {
276            throw new JsonException("Manifest funds uses an invalid decimal token.");
277        }
278        if (value == 0m && token.StartsWith("-", StringComparison.Ordinal))
279            throw new JsonException("Manifest funds cannot use negative zero.");
280        return value;
281    }

290        SaveSlotJsonStreamReader reader,
291        string context,
292        out string property)
293    {
294        reader.ReadRequired(context);
295        if (reader.TokenType == JsonTokenType.EndObject)
296        {
297            property = string.Empty;
298            return false;

301        return true;

304    private static void MarkField(ref ulong seen, ulong field, string property, string context)
305    {
306        if ((seen & field) != 0)
307            throw new JsonException($"{context} contains duplicate property '{property}'.");
308        seen |= field;
309    }

317    private static bool IsCanonicalFundsToken(string token)
318    {
319        if (token.Length is 0 or > 64 || token[0] == '+')
320            return false;
321        int index = token[0] == '-' ? 1 : 0;
322        if (index == token.Length)
323            return false;
324        int integerStart = index;
325        while (index < token.Length && char.IsAsciiDigit(token[index]))
326            index++;
327        int integerLength = index - integerStart;
328        if (integerLength == 0 || (integerLength > 1 && token[integerStart] == '0'))
329            return false;
330        if (index == token.Length)
331            return true;
332        if (token[index++] != '.')
333            return false;
334        int decimalLength = token.Length - index;
335        if (decimalLength is < 1 or > 2)
336            return false;
337        while (index < token.Length)
338        {
339            if (!char.IsAsciiDigit(token[index++]))
340                return false;
341        }
342        return true;
343    }

345    internal static void ValidateText(string value, string fieldName)
346    {
347        if (string.IsNullOrWhiteSpace(value) || !string.Equals(value, value.Trim(), StringComparison.Ordinal))
348            throw new JsonException($"Manifest {fieldName} cannot be empty or contain surrounding whitespace.");
349        int scalars = 0;
350        foreach (System.Text.Rune rune in value.EnumerateRunes())
351        {
352            if (System.Text.Rune.IsControl(rune))
353                throw new JsonException($"Manifest {fieldName} cannot contain control characters.");
354            scalars++;
355        }
356        if (scalars > MaximumTextScalars || Encoding.UTF8.GetByteCount(value) > MaximumTextUtf8Bytes)
357            throw new JsonException($"Manifest {fieldName} exceeds its length budget.");
358    }

360    private static void ValidateThumbnailFileName(string fileName)
361    {
362        if (!fileName.EndsWith(".png", StringComparison.Ordinal) ||
363            !string.Equals(Path.GetFileName(fileName), fileName, StringComparison.Ordinal))
364        {
365            throw new JsonException("Manifest thumbnailFile must be a safe lowercase .png file name.");
366        }
367        foreach (char character in fileName[..^4])
368        {
369            if (!char.IsAsciiLetterOrDigit(character) && character is not '_' and not '-')
370                throw new JsonException("Manifest thumbnailFile contains an unsafe character.");
371        }
372    }

114    private SavePublishResult SaveCore(
115        string slotID,
116        string? displayName,
117        IReadOnlyList<IStreamingSaveable> saveables,
118        IReadOnlyList<CapturedSaveParticipant>? capturedParticipants,
119        bool requireExisting,
120        IStorageOperationLease operationLease)
121    {
122        ValidateSlotID(slotID);
123        if (displayName is not null)
124            ValidateDisplayName(displayName);
125        else if (!requireExisting)
126            throw new ArgumentNullException(nameof(displayName));
127        ArgumentNullException.ThrowIfNull(saveables);
128        ArgumentNullException.ThrowIfNull(operationLease);
129        if (operationLease.Kind is not SaveOperationKind.Publish and not SaveOperationKind.Autosave)
130            throw new ArgumentException("A publish operation requires a publish or autosave lease.", nameof(operationLease));
131        SavePublicationDescriptorCodec.ValidateOperationToken(operationLease.OperationToken);
132
133        IReadOnlyList<SaveParticipantDefinition> participantDefinitions =
134            ValidateParticipants(saveables);
135        string slotDir = GetSlotDir(slotID);
136        string operationToken = operationLease.OperationToken;
137        string operationDir = GetOperationDir(slotID, operationToken);
138        string stagingDir = Path.Combine(operationDir, SavePublicationDescriptorCodec.StagingPath);
139        string backupDir = Path.Combine(operationDir, SavePublicationDescriptorCodec.BackupPath);
140
141        EnsureRootDirectory();
142        using FileStream rootLock = AcquireRootLock();
143        RecoverInterruptedTransactions();
144        operationLease.ThrowIfCancellationRequested();
145        SaveSlotClassification occupant = ClassifySlotCore(slotID, slotDir);
146        if (occupant.Kind is not SaveSlotOccupantKind.Absent and not SaveSlotOccupantKind.CompleteV3)
147        {
148            throw new InvalidDataException(
149                $"Save slot '{slotID}' is occupied by {occupant.Kind} and cannot be overwritten.");
150        }
151        if (requireExisting && occupant.Kind != SaveSlotOccupantKind.CompleteV3)
152            throw new InvalidDataException($"Save slot '{slotID}' does not exist as a complete V3 slot.");
153        string effectiveDisplayName = displayName ?? occupant.Manifest!.DisplayName;
154
155        IReadOnlyList<SaveParticipant> participants = capturedParticipants is null
156            ? participantDefinitions.Select(definition => new SaveParticipant(
157                definition.FileName,
158                definition.Saveable,
159                definition.Saveable.CaptureSnapshot())).ToArray()
160            : MatchCapturedParticipants(participantDefinitions, capturedParticipants);
161        operationLease.ThrowIfCancellationRequested();
162
163        Directory.CreateDirectory(operationDir);
164        Directory.CreateDirectory(stagingDir);
165        try
166        {
167            var files = new List<SaveSlotManifestFile>(participants.Count);
168            foreach (SaveParticipant participant in participants)
169            {
170                operationLease.ThrowIfCancellationRequested();
171                string path = Path.Combine(stagingDir, participant.FileName);
172                using var stream = new FileStream(
173                    path,
174                    FileMode.CreateNew,
175                    FileAccess.ReadWrite,
176                    FileShare.None,
177                    bufferSize: 64 * 1024,
178                    FileOptions.SequentialScan);
179                using (var cancellable = new SaveOperationWriteStream(stream, operationLease))
180                    participant.Saveable.WriteSnapshot(cancellable, participant.Snapshot);
181                stream.Flush(flushToDisk: false);
182                long length = stream.Length;
183                stream.Position = 0;
184                string sha256 = ComputeSha256(stream, operationLease);
185                files.Add(new SaveSlotManifestFile(participant.FileName, length, sha256));
186            }
187
188            operationLease.ThrowIfCancellationRequested();
189
190            var manifest = new SaveSlotManifest(
191                slotID,
192                effectiveDisplayName,
193                SaveSlotManifestCodec.CreateTimestamp(DateTime.UtcNow),
194                "Unknown City",
195                null,
196                null,
197                null,
198                files.OrderBy(file => file.Name, StringComparer.Ordinal).ToArray());
199            using (var manifestStream = new FileStream(
200                       Path.Combine(stagingDir, ManifestFile),
201                       FileMode.CreateNew,
202                       FileAccess.Write,
203                       FileShare.None))
204            {
205                SaveSlotManifestCodec.Write(manifestStream, manifest);
206                manifestStream.Flush(flushToDisk: false);
207            }
208
209            SaveSlotClassification staged = ClassifySlotCore(slotID, stagingDir);
210            if (staged.Kind != SaveSlotOccupantKind.CompleteV3)
211                throw new InvalidDataException($"Staged V3 slot is invalid: {staged.Error}");
212
213            string newDigest = ComputeAggregateDigest(stagingDir, staged.Manifest!);
214            string? oldDigest = occupant.Kind == SaveSlotOccupantKind.CompleteV3
215                ? ComputeAggregateDigest(slotDir, occupant.Manifest!)
216                : null;
217            WritePublicationDescriptor(operationDir, new SavePublicationDescriptor(
218                slotID,
219                operationToken,
220                oldDigest,
221                newDigest,
222                SavePublicationDescriptorCodec.StagingPath,
223                SavePublicationDescriptorCodec.BackupPath));
224
225            _publicationObserver?.Invoke(SavePublicationPhase.Staged);
226            operationLease.ThrowIfCancellationRequested();
227            operationLease.AcquireCommitLease();
228            _publicationObserver?.Invoke(SavePublicationPhase.CommitLeaseAcquired);
229        }
230        catch
231        {
232            TryDeleteTransactionDirectory(stagingDir);
233            TryDeleteTransactionDirectory(operationDir);
234            TryDeleteEmptyTransactionParents(slotID);
235            throw;
236        }
237
238        bool previousSlotMoved = false;
239        try
240        {
241            if (Directory.Exists(slotDir))
242            {
243                operationLease.CrossCommitBoundary(() => Directory.Move(slotDir, backupDir));
244                previousSlotMoved = true;
245                _publicationObserver?.Invoke(SavePublicationPhase.PreviousSlotMoved);
246                Directory.Move(stagingDir, slotDir);
247                operationLease.MarkCommitted();
248            }
249            else
250            {
251                operationLease.CrossCommitBoundary(() => Directory.Move(stagingDir, slotDir));
252                operationLease.MarkCommitted();
253            }
254        }
255        catch
256        {
257            if (previousSlotMoved)
258                TryRestoreBackup(slotDir, backupDir);
259            if (!Directory.Exists(slotDir) || previousSlotMoved)
260            {
261                TryDeleteTransactionDirectory(stagingDir);
262                TryDeleteTransactionDirectory(operationDir);
263                TryDeleteEmptyTransactionParents(slotID);
264            }
265            throw;
266        }
267
268        SavePublicationDescriptor descriptor = ReadPublicationDescriptor(operationDir);
269        PathDigestState published = ReadPathDigestState(slotID, slotDir);
270        if (!published.Matches(descriptor.NewAggregateDigest))
271        {
272            throw new SavePublicationRecoveryException(
273                $"Published slot '{slotID}' does not match its descriptor digest; recovery evidence was preserved.");
274        }
275
276        try
277        {
278            _publicationObserver?.Invoke(SavePublicationPhase.CanonicalPublished);
279            CleanupPublishedOperation(slotID, operationDir, stagingDir, backupDir);
280            return new SavePublishResult(
281                SavePublishResultKind.Published,
282                slotID,
283                operationToken,
284                participants.Count);
285        }
286        catch (Exception exception) when (exception is IOException or UnauthorizedAccessException)
287        {
288            return new SavePublishResult(
289                SavePublishResultKind.PublishedWithCleanupPending,
290                slotID,
291                operationToken,
292                participants.Count,
293                exception.Message);
294        }
295    }

318    internal PreparedSaveSlot PrepareLoad(
319        string slotID,
320        IReadOnlyList<CapturedLoadParticipant> loadParticipants,
321        IStorageOperationLease operationLease)
322    {
323        ArgumentNullException.ThrowIfNull(loadParticipants);
324        ArgumentNullException.ThrowIfNull(operationLease);
325        if (operationLease.Kind != SaveOperationKind.Load)
326            throw new ArgumentException("Preparing a load requires a load operation lease.", nameof(operationLease));
327        string slotDir = GetSlotDir(slotID);
328        EnsureRootDirectory();
329        using FileStream rootLock = AcquireRootLock();
330        RecoverInterruptedTransactions();
331        operationLease.ThrowIfCancellationRequested();
332        EnsureOrdinaryDirectory(slotDir, "slot");
333
334        SlotEntrySnapshot firstEntries = CaptureSlotEntries(slotDir);
335        string manifestPath = Path.Combine(slotDir, ManifestFile);
336        if (!firstEntries.Files.Contains(ManifestFile, StringComparer.Ordinal))
337            throw new InvalidDataException($"Save slot '{slotID}' does not contain a V3 manifest.");
338        EnsureOrdinaryFile(manifestPath, "manifest");
339
340        SaveSlotManifest manifest;
341        var payloadHandles = new List<(SaveSlotManifestFile File, FileStream Handle)>();
342        var prepared = new List<(IStreamingLoadTarget Target, IPreparedSaveState State)>();
343        using (var manifestHandle = new FileStream(
344                   manifestPath,
345                   FileMode.Open,
346                   FileAccess.Read,
347                   FileShare.Read,
348                   bufferSize: 16 * 1024,
349                   FileOptions.SequentialScan))
350        {
351            long manifestInitialLength = manifestHandle.Length;
352            if (manifestInitialLength is <= 0 or > SaveSlotStorageBudget.MaximumManifestEncodedBytes)
353                throw new InvalidDataException("V3 manifest exceeds its encoded byte budget.");
354            bool declaresV3 = false;
355            try
356            {
357                manifest = SaveSlotManifestCodec.Read(manifestHandle, ref declaresV3);
358            }
359            catch (JsonException exception)
360            {
361                string kind = declaresV3 ? "corrupt V3" : "foreign";
362                throw new InvalidDataException(
363                    $"Save slot '{slotID}' has a {kind} manifest and cannot be loaded.",
364                    exception);
365            }
366            if (manifestHandle.Position != manifestInitialLength ||
367                manifestHandle.Length != manifestInitialLength)
368            {
369                throw new InvalidDataException("V3 manifest changed or was not consumed to EOF.");
370            }
371            if (!string.Equals(manifest.SlotID, slotID, StringComparison.Ordinal))
372                throw new InvalidDataException(
373                    $"Manifest slotId '{manifest.SlotID}' does not match directory '{slotID}'.");
374
375            ValidateSlotEntries(firstEntries, manifest);
376            Dictionary<string, CapturedLoadParticipant> loadParticipantMap =
377                MatchLoadParticipants(loadParticipants, manifest);
378            prepared.Capacity = loadParticipantMap.Count;
379            try
380            {
381                foreach (SaveSlotManifestFile file in manifest.Files)
382                {
383                    operationLease.ThrowIfCancellationRequested();
384                    string path = Path.Combine(slotDir, file.Name);
385                    EnsureOrdinaryFile(path, $"payload '{file.Name}'");
386                    var handle = new FileStream(
387                        path,
388                        FileMode.Open,
389                        FileAccess.Read,
390                        FileShare.Read,
391                        bufferSize: 64 * 1024,
392                        FileOptions.SequentialScan);
393                    if (handle.Length != file.EncodedLength)
394                    {
395                        handle.Dispose();
396                        throw new InvalidDataException(
397                            $"Payload '{file.Name}' initial length does not match manifest.");
398                    }
399                    payloadHandles.Add((file, handle));
400                }
401
402                SlotEntrySnapshot secondEntries = CaptureSlotEntries(slotDir);
403                ValidateSlotEntries(secondEntries, manifest);
404                if (!firstEntries.Equals(secondEntries))
405                    throw new InvalidDataException("Save slot file set changed while payload handles were opened.");
406
407                foreach ((SaveSlotManifestFile file, FileStream handle) in payloadHandles)
408                {
409                    operationLease.ThrowIfCancellationRequested();
410                    using var verified = new SaveVerifiedReadStream(handle, operationLease);
411                    CapturedLoadParticipant participant = loadParticipantMap[file.Name];
412                    IPreparedSaveState state = participant.Reader.PrepareLoad(verified);
413                    verified.VerifyComplete(file);
414                    prepared.Add((participant.Target, state));
415                }
416
417                SlotEntrySnapshot thirdEntries = CaptureSlotEntries(slotDir);
418                ValidateSlotEntries(thirdEntries, manifest);
419                if (!firstEntries.Equals(thirdEntries))
420                    throw new InvalidDataException("Save slot file set changed during payload preparation.");
421            }
422            finally
423            {
424                foreach ((_, FileStream handle) in payloadHandles)
425                    handle.Dispose();
426            }
427        }
428
429        operationLease.ThrowIfCancellationRequested();
430        return new PreparedSaveSlot(
431            slotID,
432            prepared.Select(item => new PreparedSaveParticipant(item.Target, item.State)));
433    }

446    private static Dictionary<string, CapturedLoadParticipant> MatchLoadParticipants(
447        IReadOnlyList<CapturedLoadParticipant> loadParticipants,
448        SaveSlotManifest manifest)
449    {
450        var manifestFiles = manifest.Files.ToDictionary(file => file.Name, StringComparer.Ordinal);
451        var participantMap = new Dictionary<string, CapturedLoadParticipant>(StringComparer.Ordinal);
452        foreach (CapturedLoadParticipant participant in loadParticipants)
453        {
454            ArgumentNullException.ThrowIfNull(participant);
455            ArgumentNullException.ThrowIfNull(participant.Target);
456            ArgumentNullException.ThrowIfNull(participant.Reader);
457            string fileName = GetDataFileName(participant.Target.SaveFileName);
458            if (!string.Equals(fileName, participant.FileName, StringComparison.Ordinal))
459                throw new InvalidDataException($"Captured load reader name '{participant.FileName}' is stale.");
460            if (!participantMap.TryAdd(fileName, participant))
461                throw new InvalidDataException($"Multiple saveables require file '{fileName}'.");
462            if (!manifestFiles.ContainsKey(fileName))
463                throw new InvalidDataException($"Manifest does not contain required file '{fileName}'.");
464        }
465        if (participantMap.Count != manifestFiles.Count)
466            throw new InvalidDataException("Manifest declares an unsupported business payload.");
467        return participantMap;
468    }

470    private static SlotEntrySnapshot CaptureSlotEntries(string slotDir)
471    {
472        string[] directories = Directory.EnumerateDirectories(slotDir, "*", SearchOption.TopDirectoryOnly)
473            .Select(Path.GetFileName)
474            .Order(StringComparer.Ordinal)
475            .ToArray()!;
476        string[] files = Directory.EnumerateFiles(slotDir, "*", SearchOption.TopDirectoryOnly)
477            .Select(Path.GetFileName)
478            .Order(StringComparer.Ordinal)
479            .ToArray()!;
480        return new SlotEntrySnapshot(directories, files);

483    private static void ValidateSlotEntries(SlotEntrySnapshot entries, SaveSlotManifest manifest)
484    {
485        if (entries.Directories.Length != 0)
486            throw new InvalidDataException("Save slot contains a subdirectory.");
487        var expected = new HashSet<string>(StringComparer.Ordinal) { ManifestFile };
488        foreach (SaveSlotManifestFile file in manifest.Files)
489            expected.Add(file.Name);
490        var actual = new HashSet<string>(entries.Files, StringComparer.Ordinal);
491        if (manifest.ThumbnailFile is not null)
492        {
493            string thumbnail = manifest.ThumbnailFile;
494            if (actual.Contains(thumbnail))
495                expected.Add(thumbnail);
496        }
497        if (!expected.SetEquals(actual))
498            throw new InvalidDataException("Save slot file set does not match manifest.");
499    }

524    public IReadOnlyList<SaveSlotSummary> ListSlots()
525    {
526        if (!Directory.Exists(_saveBaseDir))
527            return Array.Empty<SaveSlotSummary>();
528        EnsureRootDirectory();
529        using FileStream rootLock = AcquireRootLock();
530        RecoverInterruptedTransactions();
531
532        var summaries = new List<SaveSlotSummary>();
533        foreach (string slotDir in Directory.EnumerateDirectories(_saveBaseDir, "*", SearchOption.TopDirectoryOnly))
534        {
535            string slotID = Path.GetFileName(slotDir);
536            if (string.Equals(slotID, TransactionRootName, StringComparison.Ordinal))
537                continue;
538
539            SaveSlotClassification classification;
540            try
541            {
542                classification = ClassifySlotCore(slotID, slotDir);
543            }
544            catch (ArgumentException)
545            {
546                continue;
547            }
548            if (classification.Kind is SaveSlotOccupantKind.Foreign or
549                SaveSlotOccupantKind.Unsafe or SaveSlotOccupantKind.Absent)
550            {
551                continue;
552            }
553
554            if (classification.Kind == SaveSlotOccupantKind.CompleteV3 &&
555                classification.Manifest is SaveSlotManifest manifest)
556            {
557                DateTimeOffset savedAtUtc = DateTimeOffset.ParseExact(
558                    manifest.Timestamp,
559                    "yyyy-MM-dd'T'HH:mm:ss.fffffff'Z'",
560                    CultureInfo.InvariantCulture,
561                    DateTimeStyles.AssumeUniversal | DateTimeStyles.AdjustToUniversal);
562                summaries.Add(new SaveSlotSummary
563                {
564                    SlotID = slotID,
565                    DisplayName = manifest.DisplayName,
566                    SavedAtUtc = savedAtUtc,
567                    CityName = manifest.CityName,
568                    Population = manifest.Population,
569                    Funds = manifest.Funds,
570                    ThumbnailPath = ResolveThumbnailPath(
571                        slotDir,
572                        manifest.ThumbnailFile,
573                        out string? thumbnailWarning),
574                    Warning = thumbnailWarning,
575                    Files = manifest.Files.Select(file => file.Name).ToArray(),
576                    IsValid = true,
577                    OccupantKind = SaveSlotOccupantKind.CompleteV3,
578                    OccupantDigest = ComputeAggregateDigest(slotDir, manifest),
579                });
580            }
581            else
582            {
583                summaries.Add(new SaveSlotSummary
584                {
585                    SlotID = slotID,
586                    DisplayName = slotID,
587                    IsValid = false,
588                    Error = classification.Error ?? "V3 slot is corrupt.",
589                    OccupantKind = SaveSlotOccupantKind.CorruptV3,
590                    OccupantDigest = ComputeRawDirectoryDigest(slotDir),
591                });
592            }
593        }
594
595        return summaries
596            .OrderByDescending(summary => summary.IsValid)
597            .ThenByDescending(summary => summary.SavedAtUtc)
598            .ThenBy(summary => summary.SlotID, StringComparer.Ordinal)
599            .ToArray();
600    }

626    internal SaveDeleteResult Delete(
627        SaveDeletionAuthorization authorization,
628        IStorageOperationLease operationLease)
629    {
630        ArgumentNullException.ThrowIfNull(authorization);
631        ArgumentNullException.ThrowIfNull(operationLease);
632        if (operationLease.Kind != SaveOperationKind.Delete)
633            throw new ArgumentException("Deleting a slot requires a delete operation lease.", nameof(operationLease));
634        if (!string.Equals(
635                operationLease.OperationToken,
636                authorization.OperationToken,
637                StringComparison.Ordinal))
638        {
639            throw new ArgumentException("Delete lease token does not match its authorization.", nameof(operationLease));
640        }
641        ValidateSlotID(authorization.SlotID);
642        SavePublicationDescriptorCodec.ValidateOperationToken(authorization.OperationToken);
643        if (authorization.UIGeneration <= 0)
644            throw new ArgumentException("Deletion UI generation must be positive.", nameof(authorization));
645        string slotID = authorization.SlotID;
646        string slotDir = GetSlotDir(slotID);
647        if (!Directory.Exists(_saveBaseDir))
648            throw new InvalidDataException($"Save slot '{slotID}' does not exist.");
649        EnsureRootDirectory();
650        using FileStream rootLock = AcquireRootLock();
651        RecoverInterruptedTransactions();
652        operationLease.ThrowIfCancellationRequested();
653        SaveSlotClassification classification = ClassifySlotCore(slotID, slotDir);
654        if (classification.Kind is not SaveSlotOccupantKind.CompleteV3 and
655            not SaveSlotOccupantKind.CorruptV3)
656        {
657            throw new InvalidDataException(
658                $"Save slot '{slotID}' is {classification.Kind} and cannot be deleted by DeleteV3.");
659        }
660        if (classification.Kind != authorization.OccupantKind)
661            throw new InvalidDataException("Deletion authority occupant kind is stale.");
662        string occupantDigest = ComputeOccupantDigest(slotDir, classification);
663        if (!string.Equals(occupantDigest, authorization.OccupantDigest, StringComparison.Ordinal))
664            throw new InvalidDataException("Deletion authority occupant digest is stale.");
665
666        string operationDir = GetOperationDir(slotID, authorization.OperationToken);
667        string tombstoneDir = Path.Combine(operationDir, SaveDeletionDescriptorCodec.TombstonePath);
668        Directory.CreateDirectory(operationDir);
669        var descriptor = new SaveDeletionDescriptor(
670            slotID,
671            authorization.OperationToken,
672            authorization.UIGeneration,
673            authorization.OccupantKind,
674            authorization.OccupantDigest,
675            SaveDeletionDescriptorCodec.TombstonePath,
676            authorization.ConfirmationSummary);
677        WriteDeletionDescriptor(operationDir, descriptor);
678        _publicationObserver?.Invoke(SavePublicationPhase.DeletionDescriptorPublished);
679        operationLease.ThrowIfCancellationRequested();
680        operationLease.AcquireCommitLease();
681
682        operationLease.CrossCommitBoundary(() => Directory.Move(slotDir, tombstoneDir));
683        operationLease.MarkCommitted();
684        try
685        {
686            _publicationObserver?.Invoke(SavePublicationPhase.DeletionTombstoned);
687            TryDeleteTransactionDirectory(tombstoneDir);
688            TryDeleteTransactionDirectory(operationDir);
689            TryDeleteEmptyTransactionParents(slotID);
690            return new SaveDeleteResult(
691                SaveDeleteResultKind.Deleted,
692                slotID,
693                authorization.OperationToken);
694        }
695        catch (Exception exception) when (exception is IOException or UnauthorizedAccessException)
696        {
697            return new SaveDeleteResult(
698                SaveDeleteResultKind.DeletedWithCleanupPending,
699                slotID,
700                authorization.OperationToken,
701                exception.Message);
702        }
703    }

705    private static IReadOnlyList<SaveParticipantDefinition> ValidateParticipants(
706        IReadOnlyList<IStreamingSaveable> saveables)
707    {
708        var participants = new List<SaveParticipantDefinition>(saveables.Count);
709        var names = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
710        foreach (IStreamingSaveable saveable in saveables)
711        {
712            ArgumentNullException.ThrowIfNull(saveable);
713            string fileName = GetDataFileName(saveable.SaveFileName);
714            if (string.Equals(fileName, ManifestFile, StringComparison.OrdinalIgnoreCase))
715                throw new ArgumentException("Save file name 'manifest' is reserved.", nameof(saveables));
716            if (!names.Add(fileName))
717                throw new ArgumentException($"Duplicate save file name '{fileName}'.", nameof(saveables));
718            participants.Add(new SaveParticipantDefinition(fileName, saveable));
719        }
720        return participants.OrderBy(participant => participant.FileName, StringComparer.Ordinal).ToArray();
721    }

723    private static IReadOnlyList<SaveParticipant> MatchCapturedParticipants(
724        IReadOnlyList<SaveParticipantDefinition> definitions,
725        IReadOnlyList<CapturedSaveParticipant> capturedParticipants)
726    {
727        if (definitions.Count != capturedParticipants.Count)
728            throw new ArgumentException("Captured participant count does not match saveables.", nameof(capturedParticipants));
729
730        var capturedByName = new Dictionary<string, CapturedSaveParticipant>(StringComparer.Ordinal);
731        foreach (CapturedSaveParticipant captured in capturedParticipants)
732        {
733            ArgumentNullException.ThrowIfNull(captured);
734            ArgumentNullException.ThrowIfNull(captured.Saveable);
735            ArgumentNullException.ThrowIfNull(captured.Snapshot);
736            if (!capturedByName.TryAdd(captured.FileName, captured))
737                throw new ArgumentException($"Duplicate captured file name '{captured.FileName}'.", nameof(capturedParticipants));
738        }
739
740        var participants = new List<SaveParticipant>(definitions.Count);
741        foreach (SaveParticipantDefinition definition in definitions)
742        {
743            if (!capturedByName.TryGetValue(definition.FileName, out CapturedSaveParticipant? captured) ||
744                !ReferenceEquals(definition.Saveable, captured.Saveable))
745            {
746                throw new ArgumentException(
747                    $"Captured participant '{definition.FileName}' does not match its active saveable.",
748                    nameof(capturedParticipants));
749            }
750            participants.Add(new SaveParticipant(
751                definition.FileName,
752                definition.Saveable,
753                captured.Snapshot));
754        }
755        return participants;
756    }

758    private SaveSlotClassification ClassifySlotCore(string slotID, string slotDir)
759    {
760        if (!Directory.Exists(slotDir))
761        {
762            return File.Exists(slotDir)
763                ? new SaveSlotClassification(
764                    SaveSlotOccupantKind.Unsafe,
765                    Error: "Slot path is occupied by a file.")
766                : new SaveSlotClassification(SaveSlotOccupantKind.Absent);
767        }
768        try
769        {
770            if (File.GetAttributes(slotDir).HasFlag(FileAttributes.ReparsePoint))
771                return new SaveSlotClassification(SaveSlotOccupantKind.Unsafe, Error: "Slot is a filesystem link.");
772
773            string manifestPath = Path.Combine(slotDir, ManifestFile);
774            if (!File.Exists(manifestPath))
775                return new SaveSlotClassification(SaveSlotOccupantKind.Foreign, Error: "V3 manifest is missing.");
776            if (File.GetAttributes(manifestPath).HasFlag(FileAttributes.ReparsePoint))
777                return new SaveSlotClassification(SaveSlotOccupantKind.Unsafe, Error: "Manifest is a filesystem link.");
778
779            SaveSlotManifest manifest;
780            using (var stream = new FileStream(
781                       manifestPath,
782                       FileMode.Open,
783                       FileAccess.Read,
784                       FileShare.Read,
785                       bufferSize: 16 * 1024,
786                       FileOptions.SequentialScan))
787            {
788                bool declaresV3 = false;
789                try
790                {
791                    manifest = SaveSlotManifestCodec.Read(stream, ref declaresV3);
792                }
793                catch (Exception exception) when (exception is JsonException or InvalidDataException)
794                {
795                    return new SaveSlotClassification(
796                        declaresV3 ? SaveSlotOccupantKind.CorruptV3 : SaveSlotOccupantKind.Foreign,
797                        Error: exception.Message);
798                }
799            }
800
801            if (!string.Equals(manifest.SlotID, slotID, StringComparison.Ordinal))
802            {
803                return new SaveSlotClassification(
804                    SaveSlotOccupantKind.CorruptV3,
805                    Error: $"Manifest slotId '{manifest.SlotID}' does not match directory '{slotID}'.");
806            }
807
808            var allowedFiles = new HashSet<string>(StringComparer.Ordinal) { ManifestFile };
809            foreach (SaveSlotManifestFile file in manifest.Files)
810                allowedFiles.Add(file.Name);
811            if (manifest.ThumbnailFile is not null)
812                allowedFiles.Add(manifest.ThumbnailFile);
813
814            string[] directories = Directory.EnumerateDirectories(slotDir, "*", SearchOption.TopDirectoryOnly)
815                .ToArray();
816            if (directories.Length != 0)
817                return new SaveSlotClassification(SaveSlotOccupantKind.CorruptV3, Error: "Slot contains a subdirectory.");
818            string[] actualFiles = Directory.EnumerateFiles(slotDir, "*", SearchOption.TopDirectoryOnly)
819                .Select(Path.GetFileName)
820                .ToArray()!;
821            var actualFileSet = new HashSet<string>(actualFiles, StringComparer.Ordinal);
822            bool fileSetMatches = manifest.ThumbnailFile is null
823                ? allowedFiles.SetEquals(actualFileSet)
824                : actualFileSet.IsSubsetOf(allowedFiles) &&
825                  manifest.Files.All(file => actualFileSet.Contains(file.Name)) &&
826                  actualFileSet.Contains(ManifestFile);
827            if (!fileSetMatches)
828                return new SaveSlotClassification(SaveSlotOccupantKind.CorruptV3, Error: "Slot file set does not match manifest.");
829
830            foreach (SaveSlotManifestFile file in manifest.Files)
831            {
832                string path = Path.Combine(slotDir, file.Name);
833                if (File.GetAttributes(path).HasFlag(FileAttributes.ReparsePoint))
834                    return new SaveSlotClassification(SaveSlotOccupantKind.Unsafe, Error: $"Payload '{file.Name}' is a filesystem link.");
835                using var stream = new FileStream(
836                    path,
837                    FileMode.Open,
838                    FileAccess.Read,
839                    FileShare.Read,
840                    bufferSize: 64 * 1024,
841                    FileOptions.SequentialScan);
842                ValidatePayloadStream(stream, file);
843            }
844
845            return new SaveSlotClassification(SaveSlotOccupantKind.CompleteV3, manifest);
846        }
847        catch (InvalidDataException exception)
848        {
849            return new SaveSlotClassification(SaveSlotOccupantKind.CorruptV3, Error: exception.Message);
850        }
851        catch (JsonException exception)
852        {
853            return new SaveSlotClassification(SaveSlotOccupantKind.CorruptV3, Error: exception.Message);
854        }
855        catch (Exception exception) when (exception is IOException or UnauthorizedAccessException)
856        {
857            return new SaveSlotClassification(SaveSlotOccupantKind.Unsafe, Error: exception.Message);
858        }
859    }

861    private static void ValidatePayloadStream(Stream stream, SaveSlotManifestFile file)
862    {
863        if (stream.Length != file.EncodedLength)
864            throw new InvalidDataException($"Payload '{file.Name}' length does not match manifest.");
865        stream.Position = 0;
866        string actualHash = ComputeSha256(stream);
867        if (!string.Equals(actualHash, file.Sha256, StringComparison.Ordinal))
868            throw new InvalidDataException($"Payload '{file.Name}' SHA-256 does not match manifest.");
869    }
870
871    private static string ComputeSha256(Stream stream) =>
872        Convert.ToHexString(SHA256.HashData(stream)).ToLowerInvariant();
873
874    private static string ComputeSha256(
875        Stream stream,
876        IStorageOperationLease operationLease)
877    {
878        ArgumentNullException.ThrowIfNull(stream);
879        ArgumentNullException.ThrowIfNull(operationLease);
880        using IncrementalHash hash = IncrementalHash.CreateHash(HashAlgorithmName.SHA256);
881        byte[] buffer = new byte[64 * 1024];
882        while (true)
883        {
884            operationLease.ThrowIfCancellationRequested();
885            int read = stream.Read(buffer, 0, buffer.Length);
886            if (read == 0)
887                break;
888            hash.AppendData(buffer, 0, read);
889        }
890        return Convert.ToHexString(hash.GetHashAndReset()).ToLowerInvariant();
891    }

893    internal static string ComputeAggregateDigest(string slotDir, SaveSlotManifest manifest)
894    {
895        var fileNames = new List<string>(manifest.Files.Count + 2) { ManifestFile };
896        fileNames.AddRange(manifest.Files.Select(file => file.Name));
897        fileNames.Sort(StringComparer.Ordinal);
898
899        using IncrementalHash aggregate = IncrementalHash.CreateHash(HashAlgorithmName.SHA256);
900        Span<byte> lengthBytes = stackalloc byte[sizeof(long)];
901        foreach (string fileName in fileNames)
902        {
903            byte[] nameBytes = Encoding.UTF8.GetBytes(fileName);
904            BinaryPrimitives.WriteInt64BigEndian(lengthBytes, nameBytes.LongLength);
905            aggregate.AppendData(lengthBytes);
906            aggregate.AppendData(nameBytes);
907
908            string path = Path.Combine(slotDir, fileName);
909            using var stream = new FileStream(
910                path,
911                FileMode.Open,
912                FileAccess.Read,
913                FileShare.Read,
914                bufferSize: 64 * 1024,
915                FileOptions.SequentialScan);
916            BinaryPrimitives.WriteInt64BigEndian(lengthBytes, stream.Length);
917            aggregate.AppendData(lengthBytes);
918            byte[] fileDigest = SHA256.HashData(stream);
919            aggregate.AppendData(fileDigest);
920        }
921        return Convert.ToHexString(aggregate.GetHashAndReset()).ToLowerInvariant();
922    }

924    private static string ComputeRawDirectoryDigest(string directory)
925    {
926        EnsureOrdinaryDirectory(directory, "occupant");
927        string root = Path.GetFullPath(directory);
928        string[] entries = Directory.EnumerateFileSystemEntries(root, "*", SearchOption.AllDirectories)
929            .Take(MaximumOccupantDigestEntries + 1)
930            .ToArray();
931        if (entries.Length > MaximumOccupantDigestEntries)
932            throw new InvalidDataException("Save occupant exceeds the deletion digest entry budget.");
933
934        using IncrementalHash aggregate = IncrementalHash.CreateHash(HashAlgorithmName.SHA256);
935        Span<byte> lengthBytes = stackalloc byte[sizeof(long)];
936        foreach (string entry in entries.OrderBy(
937                     path => Path.GetRelativePath(root, path),
938                     StringComparer.Ordinal))
939        {
940            string relative = Path.GetRelativePath(root, entry);
941            if (relative == "." || relative.StartsWith("..", StringComparison.Ordinal))
942                throw new IOException("Save occupant entry resolves outside its root.");
943            FileAttributes attributes = File.GetAttributes(entry);
944            if (attributes.HasFlag(FileAttributes.ReparsePoint))
945                throw new IOException("Save occupant contains a filesystem link.");
946
947            bool isDirectory = attributes.HasFlag(FileAttributes.Directory);
948            aggregate.AppendData([isDirectory ? (byte)'D' : (byte)'F']);
949            byte[] pathBytes = Encoding.UTF8.GetBytes(relative.Replace('\\', '/'));
950            BinaryPrimitives.WriteInt64BigEndian(lengthBytes, pathBytes.LongLength);
951            aggregate.AppendData(lengthBytes);
952            aggregate.AppendData(pathBytes);
953            if (isDirectory)
954                continue;
955
956            using var stream = new FileStream(
957                entry,
958                FileMode.Open,
959                FileAccess.Read,
960                FileShare.Read,
961                bufferSize: 64 * 1024,
962                FileOptions.SequentialScan);
963            BinaryPrimitives.WriteInt64BigEndian(lengthBytes, stream.Length);
964            aggregate.AppendData(lengthBytes);
965            aggregate.AppendData(SHA256.HashData(stream));
966        }
967        return Convert.ToHexString(aggregate.GetHashAndReset()).ToLowerInvariant();
968    }

1027    {
1028        string transactionRoot = Path.Combine(_saveBaseDir, TransactionRootName);
1029        if (!Directory.Exists(transactionRoot))
1030        {

1033            return;
1034        }
1035        EnsureOrdinaryDirectory(transactionRoot, "transaction root");
1036
1037        string[] slotDirectories = Directory.GetDirectories(transactionRoot)
1038            .Order(StringComparer.Ordinal)
1039            .ToArray();

1055            if (Directory.GetFiles(slotTransactions).Length != 0)
1056                throw new SavePublicationRecoveryException(
1057                    $"Publication transaction slot '{slotID}' contains an unexpected file.");

1060                         .Order(StringComparer.Ordinal))

1069    private void RecoverPublicationOperation(string slotID, string operationDir)
1070    {
1071        EnsureOrdinaryDirectory(operationDir, "publication operation");
1072        string operationToken = Path.GetFileName(operationDir);
1073        try
1074        {
1075            SavePublicationDescriptorCodec.ValidateOperationToken(operationToken);
1076        }
1077        catch (JsonException exception)
1078        {
1079            throw new SavePublicationRecoveryException(
1080                $"Publication operation directory '{operationToken}' is invalid.", exception);
1081        }
1082
1083        string descriptorPath = Path.Combine(operationDir, PublicationDescriptorFile);
1084        string deletionDescriptorPath = Path.Combine(operationDir, DeletionDescriptorFile);
1085        bool hasPublishDescriptor = File.Exists(descriptorPath);
1086        bool hasDeleteDescriptor = File.Exists(deletionDescriptorPath);
1087        if (hasPublishDescriptor && hasDeleteDescriptor)
1088            throw new SavePublicationRecoveryException("Transaction contains both publish and delete descriptors.");
1089        if (hasDeleteDescriptor)
1090        {
1091            RecoverDeletionOperation(slotID, operationToken, operationDir, deletionDescriptorPath);
1092            return;
1093        }
1094        if (!hasPublishDescriptor)
1095        {
1096            QuarantineOperation(slotID, operationToken, operationDir);
1097            return;
1098        }
1099        if (File.GetAttributes(descriptorPath).HasFlag(FileAttributes.ReparsePoint))
1100            throw new SavePublicationRecoveryException("Publication descriptor is a filesystem link.");
1101
1102        SavePublicationDescriptor descriptor;
1103        try
1104        {
1105            using var descriptorStream = new FileStream(
1106                descriptorPath,
1107                FileMode.Open,
1108                FileAccess.Read,
1109                FileShare.Read,
1110                bufferSize: 4 * 1024,
1111                FileOptions.SequentialScan);
1112            descriptor = SavePublicationDescriptorCodec.Read(descriptorStream);
1113        }
1114        catch (Exception exception) when (exception is JsonException or InvalidDataException or IOException)
1115        {
1116            throw new SavePublicationRecoveryException("Publication descriptor is invalid.", exception);
1117        }
1118        if (descriptor.SlotID != slotID || descriptor.OperationToken != operationToken)
1119            throw new SavePublicationRecoveryException("Publication descriptor identity does not match its path.");
1120
1121        string slotDir = GetSlotDir(slotID);
1122        string stagingDir = Path.Combine(operationDir, descriptor.StagingPath);
1123        string backupDir = Path.Combine(operationDir, descriptor.BackupPath);
1124        PathDigestState canonical = ReadPathDigestState(slotID, slotDir);
1125        PathDigestState staging = ReadPathDigestState(slotID, stagingDir);
1126        PathDigestState backup = ReadPathDigestState(slotID, backupDir);
1127        bool canonicalNew = canonical.Matches(descriptor.NewAggregateDigest);
1128        bool canonicalOld = descriptor.OldAggregateDigest is not null &&
1129                            canonical.Matches(descriptor.OldAggregateDigest);
1130        bool stagingNew = staging.Matches(descriptor.NewAggregateDigest);
1131        bool backupOld = descriptor.OldAggregateDigest is not null &&
1132                         backup.Matches(descriptor.OldAggregateDigest);
1133
1134        bool stagingConsistentWithNew = staging.Kind == PathDigestKind.Absent || stagingNew;
1135        bool backupConsistentWithOld = backup.Kind == PathDigestKind.Absent || backupOld;
1136        if (canonicalNew && stagingConsistentWithNew && backupConsistentWithOld)
1137        {
1138            CleanupPublishedOperation(slotID, operationDir, stagingDir, backupDir);
1139            return;
1140        }
1141        if (canonicalOld && stagingNew && backup.Kind == PathDigestKind.Absent)
1142        {
1143            QuarantineOperation(slotID, operationToken, operationDir);
1144            return;
1145        }
1146        if (canonical.Kind == PathDigestKind.Absent && backupOld && stagingNew)
1147        {
1148            Directory.Move(stagingDir, slotDir);
1149            PathDigestState published = ReadPathDigestState(slotID, slotDir);
1150            if (!published.Matches(descriptor.NewAggregateDigest))
1151                throw new SavePublicationRecoveryException("Recovered publication does not match its new digest.");
1152            CleanupPublishedOperation(slotID, operationDir, stagingDir, backupDir);
1153            return;
1154        }
1155        if (canonical.Kind == PathDigestKind.Absent && backupOld &&
1156            staging.Kind == PathDigestKind.Absent)
1157        {
1158            Directory.Move(backupDir, slotDir);
1159            TryDeleteTransactionDirectory(operationDir);
1160            TryDeleteEmptyTransactionParents(slotID);
1161            return;
1162        }
1163        if (descriptor.OldAggregateDigest is null &&
1164            canonical.Kind == PathDigestKind.Absent && stagingNew &&
1165            backup.Kind == PathDigestKind.Absent)
1166        {
1167            QuarantineOperation(slotID, operationToken, operationDir);
1168            return;
1169        }
1170
1171        throw new SavePublicationRecoveryException(
1172            $"Publication recovery is ambiguous for slot '{slotID}' operation '{operationToken}'.");
1173    }

1175    private void RecoverDeletionOperation(
1176        string slotID,
1177        string operationToken,
1178        string operationDir,
1179        string descriptorPath)
1180    {
1181        if (File.GetAttributes(descriptorPath).HasFlag(FileAttributes.ReparsePoint))
1182            throw new SaveDeletionRecoveryException("Deletion descriptor is a filesystem link.");
1183        SaveDeletionDescriptor descriptor;
1184        try
1185        {
1186            using var stream = new FileStream(
1187                descriptorPath,
1188                FileMode.Open,
1189                FileAccess.Read,
1190                FileShare.Read,
1191                bufferSize: 4 * 1024,
1192                FileOptions.SequentialScan);
1193            descriptor = SaveDeletionDescriptorCodec.Read(stream);
1194        }
1195        catch (Exception exception) when (exception is JsonException or InvalidDataException or IOException)
1196        {
1197            throw new SaveDeletionRecoveryException("Deletion descriptor is invalid.", exception);
1198        }
1199        if (descriptor.SlotID != slotID || descriptor.OperationToken != operationToken)
1200            throw new SaveDeletionRecoveryException("Deletion descriptor identity does not match its path.");
1201
1202        string slotDir = GetSlotDir(slotID);
1203        string tombstoneDir = Path.Combine(operationDir, descriptor.TombstonePath);
1204        OccupantDigestState canonical = ReadDeletionDigestState(
1205            slotID, slotDir, descriptor.OccupantKind);
1206        OccupantDigestState tombstone = ReadDeletionDigestState(
1207            slotID, tombstoneDir, descriptor.OccupantKind);
1208        bool canonicalMatches = canonical.Matches(descriptor.OccupantDigest);
1209        bool tombstoneMatches = tombstone.Matches(descriptor.OccupantDigest);
1210
1211        if (canonicalMatches && tombstone.Kind == OccupantDigestKind.Absent)
1212        {
1213            Directory.Move(slotDir, tombstoneDir);
1214            tombstone = ReadDeletionDigestState(slotID, tombstoneDir, descriptor.OccupantKind);
1215            if (!tombstone.Matches(descriptor.OccupantDigest))
1216                throw new SaveDeletionRecoveryException("Deletion tombstone does not match its descriptor digest.");
1217            TryDeleteTransactionDirectory(tombstoneDir);
1218            TryDeleteTransactionDirectory(operationDir);
1219            TryDeleteEmptyTransactionParents(slotID);
1220            return;
1221        }
1222        if (canonical.Kind == OccupantDigestKind.Absent && tombstoneMatches)
1223        {
1224            TryDeleteTransactionDirectory(tombstoneDir);
1225            TryDeleteTransactionDirectory(operationDir);
1226            TryDeleteEmptyTransactionParents(slotID);
1227            return;
1228        }
1229
1230        throw new SaveDeletionRecoveryException(
1231            $"Deletion recovery is ambiguous for slot '{slotID}' operation '{operationToken}'.");
1232    }

1327    private void QuarantineOperation(string slotID, string operationToken, string operationDir)
1328    {
1329        string quarantineRoot = Path.Combine(_saveBaseDir, QuarantineRootName);
1330        if (File.Exists(quarantineRoot))
1331            throw new SavePublicationRecoveryException("Publication quarantine root is occupied by a file.");
1332        Directory.CreateDirectory(quarantineRoot);
1333        EnsureOrdinaryDirectory(quarantineRoot, "quarantine root");
1334        string quarantineSlot = Path.Combine(quarantineRoot, slotID);
1335        if (File.Exists(quarantineSlot))
1336            throw new SavePublicationRecoveryException("Publication quarantine slot is occupied by a file.");
1337        Directory.CreateDirectory(quarantineSlot);
1338        EnsureOrdinaryDirectory(quarantineSlot, "quarantine slot");
1339
1340        string destination = Path.Combine(quarantineSlot, operationToken);
1341        if (Directory.Exists(destination) || File.Exists(destination))
1342        {
1343            destination = Path.Combine(
1344                quarantineSlot,
1345                $"{operationToken}-{Guid.NewGuid():N}");
1346        }
1347        Directory.Move(operationDir, destination);
1348        TryDeleteEmptyTransactionParents(slotID);
1349    }

1389    private static string GetDataFileName(string saveFileName)
1390    {
1391        if (string.IsNullOrWhiteSpace(saveFileName))
1392            throw new ArgumentException("Save file name cannot be empty.", nameof(saveFileName));
1393        foreach (char character in saveFileName)
1394        {
1395            if (!char.IsAsciiLetterOrDigit(character) && character is not '_' and not '-')
1396            {
1397                throw new ArgumentException(
1398                    "Save file name may only contain ASCII letters, digits, '_' or '-'.",
1399                    nameof(saveFileName));
1400            }
1401        }
1402        return saveFileName + ".json";
1403    }

1405    internal static void ValidateManifestFileName(string fileName)
1406    {
1407        const string extension = ".json";
1408        if (string.Equals(fileName, ManifestFile, StringComparison.OrdinalIgnoreCase))
1409            throw new ArgumentException("Manifest file name is reserved.", nameof(fileName));
1410        if (string.IsNullOrWhiteSpace(fileName) || !fileName.EndsWith(extension, StringComparison.Ordinal))
1411            throw new ArgumentException("Manifest file name must use the .json extension.", nameof(fileName));
1412        string saveFileName = fileName[..^extension.Length];
1413        if (!string.Equals(GetDataFileName(saveFileName), fileName, StringComparison.Ordinal))
1414            throw new ArgumentException("Manifest file name is not a safe system JSON name.", nameof(fileName));
1415    }

1417    internal static void ValidateSlotID(string slotID)
1418    {
1419        if (string.IsNullOrWhiteSpace(slotID) || slotID.Length > MaxSlotIDLength)
1420            throw new ArgumentException("Save slot ID must contain 1 to 128 characters.", nameof(slotID));
1421        foreach (char character in slotID)
1422        {
1423            if (!char.IsAsciiLetterOrDigit(character) && character is not '_' and not '-')
1424                throw new ArgumentException("Save slot ID contains an unsafe character.", nameof(slotID));
1425        }
1426    }

1440    private static string? ResolveThumbnailPath(
1441        string slotDir,
1442        string? thumbnailFile,
1443        out string? warning)
1444    {
1445        warning = null;
1446        if (thumbnailFile is null)
1447            return null;
1448        string thumbnailPath = Path.GetFullPath(Path.Combine(slotDir, thumbnailFile));
1449        string relative = Path.GetRelativePath(slotDir, thumbnailPath);
1450        if (!string.Equals(relative, thumbnailFile, StringComparison.Ordinal))
1451        {
1452            warning = "Thumbnail path resolves outside its slot.";
1453            return null;
1454        }
1455        if (!File.Exists(thumbnailPath))
1456        {
1457            warning = "Thumbnail file is missing; using the placeholder.";
1458            return null;
1459        }
1460        warning = SavePreviewPngValidator.Validate(thumbnailPath);
1461        if (warning is not null)
1462            return null;
1463        return thumbnailPath;
1464    }

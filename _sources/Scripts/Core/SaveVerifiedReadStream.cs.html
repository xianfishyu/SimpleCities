1using System;
2using System.IO;
3using System.Security.Cryptography;
4
5internal sealed class SaveVerifiedReadStream : Stream
6{
7    private readonly FileStream _source;
8    private readonly IStorageOperationLease? _operationLease;
9    private readonly IncrementalHash _hash = IncrementalHash.CreateHash(HashAlgorithmName.SHA256);
10    private bool _finished;
11    private bool _disposed;
12
13    internal SaveVerifiedReadStream(
14        FileStream source,
15        IStorageOperationLease? operationLease = null)
16    {
17        ArgumentNullException.ThrowIfNull(source);
18        if (!source.CanRead || !source.CanSeek)
19            throw new ArgumentException("Verified payload source must be a readable file handle.", nameof(source));
20        if (source.Position != 0)
21            throw new ArgumentException("Verified payload source must start at byte zero.", nameof(source));
22        _source = source;
23        _operationLease = operationLease;
24        InitialLength = source.Length;
25    }
26
27    internal long InitialLength { get; }
28    internal long BytesRead { get; private set; }
29    internal bool ReachedEof { get; private set; }
30
31    public override bool CanRead => !_disposed;
32    public override bool CanSeek => false;
33    public override bool CanWrite => false;
34    public override long Length => throw new NotSupportedException();
35    public override long Position
36    {
37        get => BytesRead;
38        set => throw new NotSupportedException();
39    }
40
41    public override int Read(byte[] buffer, int offset, int count)
42    {
43        ObjectDisposedException.ThrowIf(_disposed, this);
44        _operationLease?.ThrowIfCancellationRequested();
45        int read = _source.Read(buffer, offset, count);
46        Record(buffer.AsSpan(offset, read), read);
47        return read;
48    }
49
50    public override int Read(Span<byte> buffer)
51    {
52        ObjectDisposedException.ThrowIf(_disposed, this);
53        _operationLease?.ThrowIfCancellationRequested();
54        int read = _source.Read(buffer);
55        Record(buffer[..read], read);
56        return read;
57    }
58
59    internal void VerifyComplete(SaveSlotManifestFile file)
60    {
61        ArgumentNullException.ThrowIfNull(file);
62        ObjectDisposedException.ThrowIf(_disposed, this);
63        if (_finished)
64            throw new InvalidOperationException("Payload verification has already finished.");
65        _finished = true;
66
67        if (!ReachedEof)
68        {
69            Span<byte> probe = stackalloc byte[1];
70            if (Read(probe) != 0)
71                throw new InvalidDataException($"Payload '{file.Name}' was not consumed to EOF.");
72        }
73        if (InitialLength != file.EncodedLength ||
74            _source.Length != InitialLength ||
75            BytesRead != InitialLength)
76        {
77            throw new InvalidDataException($"Payload '{file.Name}' length does not match its declared and consumed lengths.");
78        }
79        string actualHash = Convert.ToHexString(_hash.GetHashAndReset()).ToLowerInvariant();
80        if (!string.Equals(actualHash, file.Sha256, StringComparison.Ordinal))
81            throw new InvalidDataException($"Payload '{file.Name}' SHA-256 does not match manifest.");
82    }
83
84    private void Record(ReadOnlySpan<byte> bytes, int read)
85    {
86        if (read == 0)
87        {
88            ReachedEof = true;
89            return;
90        }
91        if (ReachedEof)
92            throw new IOException("Payload source returned bytes after EOF.");
93        BytesRead = checked(BytesRead + read);
94        _hash.AppendData(bytes);
95    }
96
97    protected override void Dispose(bool disposing)
98    {
99        if (_disposed)
100            return;
101        _disposed = true;
102        if (disposing)
103            _hash.Dispose();
104        base.Dispose(disposing);
105    }
106
107    public override void Flush() { }
108    public override long Seek(long offset, SeekOrigin origin) => throw new NotSupportedException();
109    public override void SetLength(long value) => throw new NotSupportedException();
110    public override void Write(byte[] buffer, int offset, int count) => throw new NotSupportedException();
111}

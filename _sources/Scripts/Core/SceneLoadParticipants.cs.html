41/// <summary>同一活动场景中的网络、工具和表现参与者。</summary>
42internal sealed class SceneLoadParticipants
43{
44    internal SceneLoadParticipants(
45        ISceneNetworkLoadParticipant network,
46        ISceneToolLoadParticipant tools,
47        IScenePresentationLoadParticipant presentation,
48        SceneStoragePolicy storage)
49    {
50        ArgumentNullException.ThrowIfNull(network);
51        ArgumentNullException.ThrowIfNull(tools);
52        ArgumentNullException.ThrowIfNull(presentation);
53        ArgumentNullException.ThrowIfNull(storage);
54        if (!storage.RequiredSaveFileNames.Contains(network.SaveFileName, StringComparer.Ordinal))
55            throw new ArgumentException("Scene storage must include its network payload.", nameof(storage));
56        Network = network;
57        Tools = tools;
58        Presentation = presentation;
59        Storage = storage;
60    }
61
62    internal ISceneNetworkLoadParticipant Network { get; }
63    internal ISceneToolLoadParticipant Tools { get; }
64    internal IScenePresentationLoadParticipant Presentation { get; }
65    internal SceneStoragePolicy Storage { get; }
66}

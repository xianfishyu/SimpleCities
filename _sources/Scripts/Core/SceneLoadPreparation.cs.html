1using System;
2
3/// <summary>可跨准备线程传递的纯表现数据，不拥有已创建的引擎资源。</summary>
4internal interface IPreparedScenePresentation;
5
6/// <summary>从已验证的网络载荷准备纯表现数据。</summary>
7internal interface IScenePresentationPreparer
8{
9    IPreparedScenePresentation Prepare(IPreparedSaveState networkState);
10}
11
12/// <summary>在活动场景捕获、供加载准备阶段使用的目标与准备器。</summary>
13internal sealed class SceneLoadPreparationContext
14{
15    internal SceneLoadPreparationContext(
16        long generation,
17        IStreamingLoadTarget networkTarget,
18        IScenePresentationPreparer presentationPreparer)
19    {
20        ArgumentNullException.ThrowIfNull(networkTarget);
21        ArgumentNullException.ThrowIfNull(presentationPreparer);
22        Generation = generation;
23        NetworkTarget = networkTarget;
24        PresentationPreparer = presentationPreparer;
25    }
26
27    internal long Generation { get; }
28    internal IStreamingLoadTarget NetworkTarget { get; }
29    internal IScenePresentationPreparer PresentationPreparer { get; }
30}
31
32/// <summary>完成准备但尚未发布的场景加载载荷。</summary>
33internal sealed class PreparedSceneLoad
34{
35    internal PreparedSceneLoad(
36        SceneLoadPreparationContext context,
37        PreparedSaveSlot slot,
38        IPreparedSaveState networkState,
39        IPreparedScenePresentation presentation,
40        TimeSpan workerPrepareDuration)
41    {
42        ArgumentNullException.ThrowIfNull(context);
43        ArgumentNullException.ThrowIfNull(slot);
44        ArgumentNullException.ThrowIfNull(networkState);
45        ArgumentNullException.ThrowIfNull(presentation);
46        if (!ReferenceEquals(slot.GetPreparedState(context.NetworkTarget), networkState))
47            throw new InvalidOperationException("Prepared network state does not belong to this scene load target.");
48
49        Context = context;
50        Slot = slot;
51        NetworkState = networkState;
52        Presentation = presentation;
53        WorkerPrepareDuration = workerPrepareDuration;
54    }
55
56    internal SceneLoadPreparationContext Context { get; }
57    internal PreparedSaveSlot Slot { get; }
58    internal IPreparedSaveState NetworkState { get; }
59    internal IPreparedScenePresentation Presentation { get; }
60    internal TimeSpan WorkerPrepareDuration { get; }
61}

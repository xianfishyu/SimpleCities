1using System;
2using System.Collections.Generic;
3using System.Linq;
4
5/// <summary>由场景装配提供的保存根和必需载荷，创建后保持不变。</summary>
6internal sealed class SceneStoragePolicy
7{
8    internal SceneStoragePolicy(string saveBasePath, IEnumerable<string> requiredSaveFileNames)
9    {
10        if (string.IsNullOrWhiteSpace(saveBasePath))
11            throw new ArgumentException("A scene save root is required.", nameof(saveBasePath));
12        ArgumentNullException.ThrowIfNull(requiredSaveFileNames);
13        string[] names = requiredSaveFileNames.ToArray();
14        if (names.Length == 0 || names.Any(string.IsNullOrWhiteSpace) ||
15            names.Distinct(StringComparer.OrdinalIgnoreCase).Count() != names.Length)
16        {
17            throw new ArgumentException("Scene payload names must be nonempty and unique.", nameof(requiredSaveFileNames));
18        }
19
20        SaveBasePath = saveBasePath;
21        RequiredSaveFileNames = Array.AsReadOnly(names);
22    }
23
24    internal string SaveBasePath { get; }
25    internal IReadOnlyList<string> RequiredSaveFileNames { get; }
26
27    internal string ResolveSaveBaseDir(Func<string, string> globalizePath)
28    {
29        ArgumentNullException.ThrowIfNull(globalizePath);
30        string resolved = globalizePath(SaveBasePath);
31        if (string.IsNullOrWhiteSpace(resolved))
32            throw new InvalidOperationException("The scene save root could not be resolved.");
33        return resolved;
34    }
35
36    internal IReadOnlyList<IStreamingSaveable> SelectParticipants(IReadOnlyList<IStreamingSaveable> saveables)
37    {
38        ArgumentNullException.ThrowIfNull(saveables);
39        var selected = new List<IStreamingSaveable>(RequiredSaveFileNames.Count);
40        foreach (string fileName in RequiredSaveFileNames)
41        {
42            IStreamingSaveable? match = null;
43            foreach (IStreamingSaveable candidate in saveables)
44            {
45                if (!string.Equals(candidate.SaveFileName, fileName, StringComparison.Ordinal))
46                    continue;
47                if (match is not null)
48                    throw new InvalidOperationException($"Multiple saveables provide '{fileName}'.");
49                match = candidate;
50            }
51            selected.Add(match ?? throw new InvalidOperationException(
52                $"Required saveable '{fileName}' is not registered."));
53        }
54        return selected;
55    }
56}

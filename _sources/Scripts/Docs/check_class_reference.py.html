22def source_paths(roots):
23    output = subprocess.check_output(
24        ["git", "ls-files", "-z", "--cached", "--others", "--exclude-standard", "--", *roots],
25        cwd=ROOT,
26    ).decode("utf-8")
27    return sorted({
28        name for name in output.split("\0")
29        if name and (ROOT / name).is_file()
30        and not {"obj", "bin"}.intersection(Path(name).parts)
31    })

34def source_digest(path):
35    data = path.read_bytes().removeprefix(b"\xef\xbb\xbf").replace(b"\r\n", b"\n")
36    return hashlib.sha256(data).hexdigest()

39def named_script_types(paths):
40    """Only explicit named classes; anonymous GDScript script types are excluded."""
41    found = set()
42    for name in paths:
43        path = ROOT / name
44        if path.suffix == ".py":
45            tree = ast.parse(path.read_text(encoding="utf-8-sig"), filename=name)
46            found.update((name, node.name) for node in ast.walk(tree) if isinstance(node, ast.ClassDef))
47        elif path.suffix == ".gd":
48            text = path.read_text(encoding="utf-8-sig")
49            for match in re.finditer(r"^\s*(?:class_name|class)\s+(\w+)", text, re.M):
50                found.add((name, match[1]))
51    return found

54def scan():
55    baseline = json.loads(MANIFEST.read_text(encoding="utf-8"))
56    issues = []
57    if baseline.get("version") != 2:
58        raise ValueError("Expected version=2 class-reference inventory.")
59    paths = source_paths(baseline["sourceRoots"])
60    actual_cs = {name for name in paths if name.endswith(".cs")}
61    recorded_cs = {item["path"] for item in baseline["sources"] if item["language"] == "C#"}
62    for name in sorted(actual_cs - recorded_cs):
63        issues.append(f"Unrecorded C# source: {name}")
64    for name in sorted(recorded_cs - actual_cs):
65        issues.append(f"Missing C# source: {name}")
66
67    cache = {}
68
69    def check_declaration(page, anchor, signature):
70        target = ROOT / page
71        if not target.is_file():
72            issues.append(f"Missing reference page: {page}")
73            return
74        if page not in cache:
75            text = target.read_text(encoding="utf-8")
76            cache[page] = (anchors(text), html.unescape(text).replace("\\|", "|"))
77        page_anchors, text = cache[page]
78        if anchor not in page_anchors:
79            issues.append(f"Missing declaration anchor: {page}#{anchor}")
80        # Check at the matching declaration, not just somewhere in this page.
81        start = text.find(f'<a id="{anchor}"></a>')
82        if start >= 0:
83            stop = text.find('<a id="', start + 1)
84            section = text[start:stop] if stop >= 0 else text[start:]
85            if signature not in section:
86                issues.append(f"Missing declaration signature: {page}#{anchor}: {signature}")
87
88    for source in baseline["sources"]:
89        path = ROOT / source["path"]
90        if not path.is_file():
91            issues.append(f"Missing baseline source: {source['path']}")
92        elif source_digest(path) != source["sha256"]:
93            issues.append(f"Source changed; review reference and baseline: {source['path']}")
94
95    type_keys = set()
96    type_pages = set()
97    member_count = 0
98    for item in baseline["types"]:
99        if item["key"] in type_keys:
100            issues.append(f"Duplicate type: {item['key']}")
101        type_keys.add(item["key"])
102        if item["page"].casefold() in type_pages:
103            issues.append(f"Multiple types share a canonical page: {item['page']}")
104        type_pages.add(item["page"].casefold())
105        for part in item["parts"]:
106            check_declaration(item["page"], item["anchor"], part["signature"])
107        for member in item["members"]:
108            check_declaration(item["page"], member["anchor"], member["signature"])
109            member_count += 1
110
111    scripts = named_script_types(paths)
112    recorded_scripts = {(item["source"], item["name"]) for item in baseline["otherTypes"]}
113    for name, symbol in sorted(scripts - recorded_scripts):
114        issues.append(f"Unrecorded script class: {name}: {symbol}")
115    for name, symbol in sorted(recorded_scripts - scripts):
116        issues.append(f"Missing script class: {name}: {symbol}")
117    for item in baseline["otherTypes"]:
118        if item["page"].casefold() in type_pages:
119            issues.append(f"Multiple types share a canonical page: {item['page']}")
120        type_pages.add(item["page"].casefold())
121        check_declaration(item["page"], item["anchor"], item["signature"])
122        for member in item["members"]:
123            check_declaration(item["page"], member["anchor"], member["signature"])
124
125    generated = ClassReferenceRenderer(baseline).pages()
126    for page, expected in generated.items():
127        target = ROOT / page
128        if not target.is_file() or target.read_text(encoding="utf-8") != expected:
129            issues.append(f"Generated reference page differs from reviewed inventory: {page}")
130
131    return {
132        "csharpFiles": len(recorded_cs),
133        "csharpTypes": len(type_keys),
134        "csharpTypeDeclarations": sum(item["typeDeclarations"] for item in baseline["sources"] if item["language"] == "C#"),
135        "csharpMemberDeclarations": member_count,
136        "otherLanguageTypes": len(recorded_scripts),
137        "referencePages": len(cache),
138        "generatedReferencePages": len(generated),
139        "issues": issues,
140    }

143def main():
144    parser = argparse.ArgumentParser(description=__doc__)
145    parser.add_argument("--json", action="store_true")
146    args = parser.parse_args()
147    try:
148        result = scan()
149    except (OSError, ValueError, KeyError) as error:
150        print(f"无法检查类参考：{error}")
151        return True
152    if args.json:
153        print(json.dumps(result, ensure_ascii=False, indent=2))
154    else:
155        for issue in result["issues"]:
156            print(issue)
157        print(
158            f"C#: {result['csharpFiles']} files, {result['csharpTypes']} types, "
159            f"{result['csharpMemberDeclarations']} member declarations; "
160            f"other languages: {result['otherLanguageTypes']} types; errors: {len(result['issues'])}"
161        )
162    return bool(result["issues"])

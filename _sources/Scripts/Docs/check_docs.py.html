1"""Check local Markdown links and GitHub-style heading fragments; no dependencies."""
2
3import argparse
4import json
5import re
6import subprocess
7import unicodedata
8from pathlib import Path
9from urllib.parse import unquote, urlsplit
10
11ROOT = Path(__file__).resolve().parents[2]
12
13
14def markdown_files():
15    paths = subprocess.check_output(
16        ["git", "ls-files", "-z", "--cached", "--others", "--exclude-standard", "--", "*.md"],
17        cwd=ROOT,
18    ).decode("utf-8").split("\0")
19    return sorted({p for p in paths if p and not p.startswith("addons/") and (ROOT / p).is_file()})
20

22def prose(text):
23    """Keep line numbers while excluding fenced examples and HTML comments."""
24    fence = None
25    lines = []
26    for line in text.splitlines():
27        match = re.match(r"^\s{0,3}(`{3,}|~{3,})", line)
28        if match and (fence is None or (match[1][0] == fence[0] and len(match[1]) >= len(fence))):
29            fence = match[1] if fence is None else None
30            lines.append("")
31        else:
32            lines.append(line if fence is None else "")
33    return re.sub(r"<!--[\s\S]*?-->", lambda m: "\n" * m[0].count("\n"), "\n".join(lines))

36def anchors(text):
37    content = prose(text)
38    result = set(re.findall(r"\b(?:id|name)=[\"']([^\"']+)[\"']", content))
39    counts = {}
40    for line in content.splitlines():
41        match = re.match(r"^\s{0,3}#{1,6}\s+(.+?)\s*#*\s*$", line)
42        if not match:
43            continue
44        heading = re.sub(r"<[^>]*>", "", match[1])
45        heading = re.sub(r"!?\[([^\]]*)\]\([^)]*\)", r"\1", heading)
46        slug = "".join(c for c in heading.lower() if c.isspace() or c in "_-" or unicodedata.category(c)[0] in "LN")
47        slug = re.sub(r"\s", "-", slug)
48        number = counts.get(slug, 0)
49        counts[slug] = number + 1
50        result.add(f"{slug}-{number}" if number else slug)
51    return result

54def links(text):
55    content = prose(text)
56    # Inline links, image links, and reference definitions. A destination may
57    # contain balanced parentheses; optional link titles are not destinations.
58    pattern = r"!?\[[^\]\n]*\]\(\s*(<[^>\n]+>|(?:[^\s()]+|\([^()]*\))+)(?:\s+[^\n]*?)?\s*\)"
59    for match in re.finditer(pattern, content):
60        yield content.count("\n", 0, match.start()) + 1, match[1].strip("<>")
61    for match in re.finditer(r"^\s{0,3}\[[^\]]+\]:\s*(<[^>\n]+>|\S+)", content, re.M):
62        yield content.count("\n", 0, match.start()) + 1, match[1].strip("<>")

65def scan():
66    issues = []
67    checked = 0
68    cache = {}
69    files = markdown_files()
70    for name in files:
71        source = ROOT / name
72        text = source.read_text(encoding="utf-8-sig")
73        for line, href in links(text):
74            parsed = urlsplit(href)
75            if parsed.scheme or parsed.netloc:
76                continue
77            target = (ROOT / unquote(parsed.path).lstrip("/")) if parsed.path.startswith("/") else (source.parent / unquote(parsed.path))
78            target = target.resolve() if parsed.path else source
79            checked += 1
80            reason = None
81            if not target.exists():
82                reason = "missing target"
83            elif parsed.fragment and target.suffix.lower() == ".md":
84                if target not in cache:
85                    cache[target] = anchors(target.read_text(encoding="utf-8-sig"))
86                if unquote(parsed.fragment) not in cache[target]:
87                    reason = "missing anchor"
88            if reason:
89                issues.append({"file": name, "line": line, "href": href, "reason": reason})
90    return {"files": len(files), "links": checked, "issues": issues}

93def main():
94    parser = argparse.ArgumentParser(description=__doc__)
95    parser.add_argument("--json", action="store_true", help="Print machine-readable results")
96    args = parser.parse_args()
97    result = scan()
98    if args.json:
99        print(json.dumps(result, ensure_ascii=False, indent=2))
100    else:
101        for issue in result["issues"]:
102            print(f"{issue['file']}:{issue['line']}: {issue['reason']}: {issue['href']}")
103        print(f"Markdown files: {result['files']}; local links: {result['links']}; errors: {len(result['issues'])}")
104    return bool(result["issues"])

Scripts/Docs/publish_docs.ps1

1param(
2    [switch]$SkipBuild,
3    [switch]$CheckConnection
4)
5
6$ErrorActionPreference = 'Stop'
7$docsPublishRoot = [System.IO.Path]::GetFullPath((Join-Path $PSScriptRoot '../..'))
8$docsPublishWork = [System.IO.Path]::GetFullPath((Join-Path $docsPublishRoot '.scratch/docs-site'))
9$docsPublishRepository = 'xianfishyu/SimpleCities'
10$docsPublishBranch = 'codex/gh-pages'
11$docsPublishRemote = "https://github.com/$docsPublishRepository.git"
12$docsGitAuth = @('-c', 'core.autocrlf=false', '-c', 'credential.helper=', '-c', 'credential.helper=!gh auth git-credential')
13
14function Invoke-DocsGit {
15    param([string[]]$Arguments)
16    & git @docsGitAuth @Arguments
17    if ($LASTEXITCODE -ne 0) { throw "文档发布 Git 操作失败：$($Arguments[0])" }
18}
19
20Push-Location -LiteralPath $docsPublishRoot
21try {
22    if (-not $SkipBuild) {
23        & (Join-Path $PSScriptRoot 'serve_docs.ps1') -BuildOnly
24        if ($LASTEXITCODE -ne 0) { throw '文档站构建失败。' }
25    }
26    $docsPublishPython = Join-Path $docsPublishWork 'venv/Scripts/python.exe'
27    $docsPublishProxy = & $docsPublishPython -c "import urllib.request; proxies = urllib.request.getproxies(); print(proxies.get('https') or proxies.get('http') or '')"
28    if ($LASTEXITCODE -ne 0) { throw '无法读取当前系统代理。' }
29    if ($docsPublishProxy) {
30        $docsGitAuth += @('-c', "http.proxy=$($docsPublishProxy.Trim())")
31    }
32    if ($CheckConnection) {
33        Invoke-DocsGit -Arguments @('ls-remote', $docsPublishRemote, 'HEAD')
34        return
35    }
36    $docsPublishBuild = Get-Content -LiteralPath (Join-Path $docsPublishWork 'current.json') -Raw | ConvertFrom-Json
37    $docsPublishSite = [System.IO.Path]::GetFullPath($docsPublishBuild.site)
38    if (-not $docsPublishSite.StartsWith($docsPublishWork + [System.IO.Path]::DirectorySeparatorChar, [System.StringComparison]::OrdinalIgnoreCase)) {
39        throw '发布目录必须是本项目独立构建产物。'
40    }
41    if (-not (Test-Path -LiteralPath (Join-Path $docsPublishSite 'index.html') -PathType Leaf)) {
42        throw '缺少站点首页。'
43    }
44    & (Join-Path $docsPublishWork 'venv/Scripts/python.exe') -B (Join-Path $PSScriptRoot 'docs_site.py') check
45    if ($LASTEXITCODE -ne 0) { throw '站点链接检查失败。' }
46
47    $docsPagesRaw = & gh api "repos/$docsPublishRepository/pages" 2>$null
48    $docsPagesCode = $LASTEXITCODE
49    $docsPages = $docsPagesRaw | ConvertFrom-Json
50    if ($docsPagesCode -eq 0) {
51        if ($docsPages.source.branch -ne $docsPublishBranch -or $docsPages.source.path -ne '/') {
52            throw '现有 Pages 使用其他发布来源，保留其配置并停止。'
53        }
54    }
55    elseif ($docsPages.status -ne 404) { throw '无法读取 Pages 配置。' }
56
57    $docsPublishCheckout = Join-Path $docsPublishWork ('publish-' + [guid]::NewGuid().ToString('N'))
58    New-Item -ItemType Directory -Path $docsPublishCheckout | Out-Null
59    Invoke-DocsGit -Arguments @('init', '--quiet', '--initial-branch', $docsPublishBranch, $docsPublishCheckout)
60    Invoke-DocsGit -Arguments @('-C', $docsPublishCheckout, 'remote', 'add', 'origin', $docsPublishRemote)
61    $docsBranchRaw = & gh api "repos/$docsPublishRepository/git/ref/heads/$docsPublishBranch" 2>$null
62    $docsBranchCode = $LASTEXITCODE
63    $docsBranch = $docsBranchRaw | ConvertFrom-Json
64    if ($docsBranchCode -eq 0) {
65        Invoke-DocsGit -Arguments @('-C', $docsPublishCheckout, 'fetch', '--quiet', '--depth=1', 'origin', $docsPublishBranch)
66        Invoke-DocsGit -Arguments @('-C', $docsPublishCheckout, 'checkout', '--quiet', '-B', $docsPublishBranch, 'FETCH_HEAD')
67        $docsOwnerPath = Join-Path $docsPublishCheckout 'simplecities-docs.json'
68        if (-not (Test-Path -LiteralPath $docsOwnerPath)) { throw "现有 $docsPublishBranch 未标记为本工具管理的站点，保留内容并停止。" }
69        $docsOwner = Get-Content -LiteralPath $docsOwnerPath -Raw | ConvertFrom-Json
70        if ($docsOwner.generator -ne 'SimpleCities docs_site.py') { throw '现有发布分支属于其他工具，保留内容并停止。' }
71        $docsResolvedCheckout = [System.IO.Path]::GetFullPath($docsPublishCheckout)
72        if (-not $docsResolvedCheckout.StartsWith($docsPublishWork + [System.IO.Path]::DirectorySeparatorChar, [System.StringComparison]::OrdinalIgnoreCase)) { throw '发布工作区边界无效。' }
73        Invoke-DocsGit -Arguments @('-C', $docsResolvedCheckout, 'rm', '-r', '--quiet', '--ignore-unmatch', '--', '.')
74    }
75    elseif ($docsBranch.status -ne 404) { throw "无法读取 $docsPublishBranch 发布分支。" }
76
77    Get-ChildItem -LiteralPath $docsPublishSite -Force | Copy-Item -Destination $docsPublishCheckout -Recurse -Force
78    if ($docsPagesCode -eq 0 -and $docsPages.cname) {
79        Set-Content -LiteralPath (Join-Path $docsPublishCheckout 'CNAME') -Value $docsPages.cname -Encoding utf8
80    }
81    $docsOwner = @{
82        generator = 'SimpleCities docs_site.py'
83        sourceCommit = $docsPublishBuild.sourceCommit
84        sourceDirty = $docsPublishBuild.sourceDirty
85        builtAtUtc = $docsPublishBuild.builtAtUtc
86        htmlPages = $docsPublishBuild.audit.htmlPages
87    }
88    $docsOwner | ConvertTo-Json | Set-Content -LiteralPath (Join-Path $docsPublishCheckout 'simplecities-docs.json') -Encoding utf8
89    Invoke-DocsGit -Arguments @('-C', $docsPublishCheckout, 'add', '--all')
90    $docsAuthorName = git config user.name
91    $docsAuthorEmail = git config user.email
92    if (-not $docsAuthorName -or -not $docsAuthorEmail) { throw '当前项目缺少提交身份配置。' }
93    Invoke-DocsGit -Arguments @('-C', $docsPublishCheckout, '-c', "user.name=$docsAuthorName", '-c', "user.email=$docsAuthorEmail", 'commit', '--quiet', '-m', '发布：部署 SimpleCities 类参考站点')
94    Invoke-DocsGit -Arguments @('-C', $docsPublishCheckout, 'push', 'origin', "HEAD:refs/heads/$docsPublishBranch")
95
96    if ($docsPagesCode -ne 0) {
97        $docsPagesRequest = Join-Path $docsPublishWork 'pages-request.json'
98        @{ source = @{ branch = $docsPublishBranch; path = '/' } } | ConvertTo-Json | Set-Content -LiteralPath $docsPagesRequest -Encoding utf8
99        $docsPagesResponse = gh api --method POST "repos/$docsPublishRepository/pages" --input $docsPagesRequest
100        if ($LASTEXITCODE -ne 0) { throw '发布分支已推送，但启用 Pages 失败；保留分支以便继续配置。' }
101        $docsPagesResponse | Set-Content -LiteralPath (Join-Path $docsPublishWork 'pages-response.json') -Encoding utf8
102    }
103    gh api "repos/$docsPublishRepository/pages" --jq '{html_url,status,source}'
104    if ($LASTEXITCODE -ne 0) { throw '发布后无法读取 Pages 状态。' }
105}
106finally {
107    Pop-Location
108}

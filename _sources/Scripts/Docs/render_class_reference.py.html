44def link(label, href):
45    return "[" + html.escape(label).replace("[", "&#91;").replace("]", "&#93;") + "](" + href + ")"

48def source_link(page, source, line, label=None):
49    return link(label or f"{source}:{line}", relative(page, source) + f"#L{line}")

52def is_exported(member):
53    return any(a.rsplit(".", 1)[-1] in {"Export", "ExportAttribute"} for a in member.get("attributes", []))

56def qualifiers(member):
57    values = [member.get("access", "")]
58    values += [m for m in member.get("modifiers", []) if m not in {"public", "protected", "internal", "private"}]
59    values += member.get("accessors", [])
60    if is_exported(member):
61        values.append("Export")
62    if member.get("configurations") and set(member["configurations"]) != {"Debug", "Release"}:
63        values.append(" / ".join(member["configurations"]))
64    if member.get("source", "").startswith("tests/godot/"):
65        values.append("Debug 探针")
66    return " · ".join(code(v) for v in values if v)

69def initial_value(member):
70    values = [x["value"] for x in member.get("initializers", []) if x.get("value") is not None]
71    if not values:
72        return "由构造或运行时决定"
73    result = ", ".join(values)
74    return code(result) if len(result) <= 100 and "\n" not in result else "见源码初始化"

81def table(rows, headers):
82    return ["| " + " | ".join(headers) + " |", "| " + " | ".join("---" for _ in headers) + " |"] + ["| " + " | ".join(row) + " |" for row in rows] + [""]

85class ClassReferenceRenderer:
86    def __init__(self, manifest):
87        self.manifest = manifest
88        self.types = {t["key"]: t for t in manifest["types"]}
89        self.names = defaultdict(list)
90        for key, typ in self.types.items():
91            self.names[typ["name"].rsplit(".", 1)[-1]].append(key)
92        self.children = defaultdict(list)
93        self.parents = {}
94        self.derived = defaultdict(list)
95        for key, typ in self.types.items():
96            parent = key.rpartition(".")[0]
97            if parent in self.types:
98                self.parents[key] = parent
99                self.children[parent].append(key)
100            for base in typ.get("bases", []):
101                resolved = self.resolve(base.split("<", 1)[0], typ)
102                if resolved:
103                    self.derived[resolved].append(key)
104
105    def resolve(self, name, context):
106        name = name.removeprefix("global::")
107        for aliases in context.get("aliases", {}).values():
108            if name in aliases:
109                name = aliases[name]
110                break
111        if name in self.types:
112            return name
113        scope = context["key"].split(".")[:-1]
114        while scope:
115            candidate = ".".join(scope + [name])
116            if candidate in self.types:
117                return candidate
118            scope.pop()
119        matches = self.names.get(name, [])
120        return matches[0] if len(matches) == 1 else None
121
122    def type_link(self, key, page, label=None):
123        typ = self.types[key]
124        target = "" if page == typ["page"] else relative(page, typ["page"])
125        return link(label or typ["name"] + typ.get("genericParameters", ""), target + "#" + typ["anchor"])
126
127    def type_text(self, name, context):
128        result = []
129        previous = 0
130        for token in re.finditer(r"(?:global::)?[A-Za-z_][A-Za-z0-9_.]*", name):
131            result.append(html.escape(name[previous:token.start()]))
132            identifier = token[0]
133            resolved = self.resolve(identifier, context)
134            if resolved:
135                result.append(self.type_link(resolved, context["page"], identifier))
136            elif identifier in ENGINE_BASES:
137                result.append(link(identifier, "https://docs.godotengine.org/en/stable/classes/class_" + identifier.lower() + ".html"))
138            else:
139                result.append(html.escape(identifier))
140            previous = token.end()
141        result.append(html.escape(name[previous:]))
142        return "".join(result)

144    def nav(self, typ):
145        page = typ["page"]
146        group = page.split("/")[2]
147        return link("类参考首页", relative(page, "docs/class-reference/README.md")) + " · " + link(GROUPS[group][0], relative(page, f"docs/class-reference/{group}/README.md")) + " · " + link("类型索引", relative(page, "docs/class-reference/type-index.md"))

149    def member_name(self, member):
150        name = member["name"]
151        if member["kind"] in {"method", "constructor", "operator", "destructor"}:
152            return name + "(" + ", ".join(p["type"] for p in member.get("parameters", [])) + ")"
153        return name

155    def callable(self, member, typ):
156        params = ", ".join(code(p["declaration"]) for p in member.get("parameters", []))
157        return link(member["name"], "#" + member["anchor"]) + "(" + params + ")"

159    def overview(self, kind, members, typ):
160        if not members:
161            return []
162        lines = ["## " + TITLES[kind][0], ""]
163        rows = []
164        for member in sorted(members, key=lambda m: (m["name"].lower(), m["signature"])):
165            result = self.type_text(member.get("returnType") or "—", typ)
166            if kind in {"method", "constructor", "operator", "destructor"}:
167                rows.append([result if kind != "constructor" else "—", self.callable(member, typ), qualifiers(member)])
168            elif kind in {"constant", "enumValue"}:
169                rows.append([result, link(member["name"], "#" + member["anchor"]), initial_value(member)])
170            elif kind == "event":
171                rows.append([result, link(member["name"], "#" + member["anchor"]), qualifiers(member)])
172            else:
173                rows.append([result, link(member["name"], "#" + member["anchor"]), initial_value(member) if kind in {"property", "field"} else qualifiers(member), qualifiers(member) if kind in {"property", "field"} else ""])
174        if kind in {"method", "constructor", "operator", "destructor"}:
175            return lines + table(rows, ["返回类型", "签名", "限定符"])
176        if kind in {"constant", "enumValue"}:
177            return lines + table(rows, ["类型", "名称", "值"])
178        if kind == "event":
179            return lines + table(rows, ["类型", "名称", "访问与限定"])
180        return lines + table(rows, ["类型", "名称", "源码初始值" if kind in {"property", "field"} else "限定符", "访问与限定"])
181
182    def details(self, kind, members, typ):
183        if not members:
184            return []
185        lines = ["## " + TITLES[kind][1], ""]
186        for member in sorted(members, key=lambda m: (m["name"].lower(), m["signature"])):
187            lines += [f'<a id="{member["anchor"]}"></a>', "", "### " + code(self.member_name(member)), "", "```csharp", member["signature"], "```", ""]
188            text = member.get("description", "—")
189            lines += [text if text != "—" else "本成员尚无独立的行为说明；调用边界以本类型描述和链接源码为准。", ""]
190            if kind in {"property", "field", "constant", "enumValue"}:
191                lines += ["源码初始值：" + initial_value(member) + "。", ""]
192            lines += ["访问与限定：" + qualifiers(member) + "。", ""]
193            if member.get("parameters"):
194                rows = [[code(p["name"]), self.type_text(p["type"], typ), code(p["defaultValue"]) if p.get("defaultValue") is not None else "必填"] for p in member["parameters"]]
195                lines += table(rows, ["参数", "类型", "默认值"])
196            lines += ["源码：" + source_link(typ["page"], member["source"], member["line"]) + "。", ""]
197        return lines

199    def render_type(self, typ):
200        page = typ["page"]
201        name = typ["name"] + typ.get("genericParameters", "")
202        lines = ["# " + html.escape(name), "", self.nav(typ), "", f'<a id="{typ["anchor"]}"></a>', ""]
203        bases = typ.get("bases", [])
204        if bases:
205            lines += ["**继承 / 实现：** " + "、".join(self.type_text(b, typ) for b in bases), ""]
206        elif typ["kind"] != "interface":
207            base = "System.ValueType" if "struct" in typ["kind"] else "System.Enum" if typ["kind"] == "enum" else "System.MulticastDelegate" if typ["kind"] == "delegate" else "System.Object"
208            lines += ["**基础类型：** " + code(base), ""]
209        if typ["key"] in self.parents:
210            lines += ["**外层类型：** " + self.type_link(self.parents[typ["key"]], page), ""]
211        derived = self.derived[typ["key"]]
212        if derived:
213            lines += ["**派生 / 实现类型：** " + "、".join(self.type_link(k, page) for k in sorted(derived)), ""]
214        lines += [brief(typ["description"]), "", "**种类：** " + code(typ["kind"]) + "　**声明可见性：** " + code(typ["access"]) + "　**命名空间：** " + code(typ["namespace"] or "全局"), "", "## 描述", "", typ["description"], ""]
215        for paragraph in typ.get("remarks", []):
216            lines += [paragraph, ""]
217        if any(p["source"].startswith("tests/godot/") for p in typ["parts"]):
218            lines += ["> **注意：** `tests/godot/` 中的声明由主项目 Debug 配置纳入。它们补充测试和故障注入接口，表中保留相应标记。", ""]
219        lines += ["<details>", "<summary>C# 声明与源码</summary>", ""]
220        for signature in dict.fromkeys(p["signature"] for p in typ["parts"]):
221            lines += ["```csharp", signature, "```", ""]
222        for part in typ["parts"]:
223            lines += ["- " + source_link(page, part["source"], part["line"])]
224        lines += ["", "</details>", "", "## 教程与示例", "", "- " + link("调用约束与纯核心示例", relative(page, "docs/class-reference/usage.md")), "- " + link("所属系统手册", relative(page, GROUPS[page.split('/')[2]][2])), ""]
225        members = typ["members"]
226        public = [m for m in members if m["access"] != "private" or is_exported(m)]
227        private = [m for m in members if m["access"] == "private" and not is_exported(m)]
228        grouped = {kind: [m for m in public if m["kind"] == kind] for kind in TITLES}
229        params = typ.get("parameters", []) if typ["kind"] != "delegate" else []
230        if params and typ["kind"].startswith("record"):
231            lines += ["## 位置属性", "", "以下属性由位置 record 参数生成，单独说明，不计入显式成员数量。", ""]
232            lines += table([[self.type_text(p["type"], typ), link(p["name"], "#position-" + p["name"].lower()), code(p["defaultValue"]) if p.get("defaultValue") is not None else "由构造参数提供"] for p in params], ["类型", "名称", "默认参数"])
233        for kind in ["property", "field", "constructor", "method", "operator", "event"]:
234            lines += self.overview(kind, grouped[kind], typ)
235        children = self.children[typ["key"]]
236        signals = [self.types[k] for k in children if "Signal" in self.types[k].get("attributes", [])]
237        enums = [self.types[k] for k in children if self.types[k]["kind"] == "enum"]
238        other = [self.types[k] for k in children if k not in {t["key"] for t in signals + enums}]
239        for title, items in [("Godot 信号", signals), ("枚举", enums), ("嵌套类型", other)]:
240            if not items:
241                continue
242            lines += ["## " + title, ""]
243            rows = []
244            for child in items:
245                label = child["name"].rsplit(".", 1)[-1]
246                if title == "Godot 信号":
247                    label = label.removesuffix("EventHandler") + "(" + ", ".join(p["declaration"] for p in child.get("parameters", [])) + ")"
248                rows.append([self.type_link(child["key"], page, label), brief(child["description"])])
249            lines += table(rows, ["名称", "说明"])
250        for kind in ["constant", "enumValue", "destructor", "other"]:
251            lines += self.overview(kind, grouped[kind], typ)
252        if params:
253            lines += ['<a id="primary-constructor"></a>', "", "## 主构造函数", "", code(typ["name"].rsplit(".", 1)[-1] + "(" + ", ".join(p["declaration"] for p in params) + ")"), "", "位置 record 的同名属性由编译器生成；普通类/struct 的主构造参数是否成为成员，以显式声明为准。", ""]
254            lines += table([[code(p["name"]), self.type_text(p["type"], typ), p.get("description") or "构造输入；类型和默认值见声明。"] for p in params], ["参数", "类型", "含义"])
255        if params and typ["kind"].startswith("record"):
256            lines += ["## 位置属性说明", ""]
257            for param in params:
258                lines += [f'<a id="position-{param["name"].lower()}"></a>', "", "### " + param["name"], "", self.type_text(param["type"], typ) + " " + code(param["name"]), "", param.get("description") or "由同名位置构造参数初始化的属性。", ""]
259        for kind in TITLES:
260            lines += self.details(kind, grouped[kind], typ)
261        if private:
262            lines += ["## 内部实现参考", "", "<details>", f"<summary>私有成员（{len(private)} 处显式声明）</summary>", "", "这些成员用于维护和定位实现；私有的真正 `[Export]` 配置已列入上方属性/字段。", ""]
263            rows = []
264            for member in private:
265                note = member.get("description", "—")
266                rows.append([f'<a id="{member["anchor"]}"></a>' + code(member["signature"]), cell(note), source_link(page, member["source"], member["line"], str(member["line"]))])
267            lines += table(rows, ["声明", "说明", "源码行"])
268            lines += ["</details>", ""]
269        if not members and not params:
270            lines += ["本类型没有额外的显式成员；使用其声明、继承接口或类型说明中的契约。", ""]
271        if typ.get("aliases"):
272            lines += ["## 源码别名", ""]
273            aliases = {k: v for values in typ["aliases"].values() for k, v in values.items()}
274            lines += table([[code(k), self.type_text(v, typ)] for k, v in sorted(aliases.items())], ["别名", "实际类型"])
275        return "\n".join(lines).rstrip() + "\n"

277    def render_other(self, typ):
278        page = typ["page"]
279        language = "python" if typ["source"].endswith(".py") else "gdscript"
280        lines = ["# " + typ["name"], "", link("类参考首页", relative(page, "docs/class-reference/README.md")) + " · " + link("其他语言类型", relative(page, "docs/class-reference/other-languages.md")), "", f'<a id="{typ["anchor"]}"></a>', "", brief(typ["description"]), "", "## 描述", "", typ["description"], "", "```" + language, typ["signature"], "```", "", "源码：" + source_link(page, typ["source"], typ["line"]) + "。", "", "## 成员", ""]
281        lines += table([[link(m["signature"], "#" + m["anchor"]), cell(m["description"])] for m in typ["members"]], ["声明", "说明"])
282        lines += ["## 成员说明", ""]
283        for member in typ["members"]:
284            lines += [f'<a id="{member["anchor"]}"></a>', "", "### " + code(member["signature"].split("(", 1)[0]), "", "```" + language, member["signature"], "```", "", member["description"], "", "源码：" + source_link(page, typ["source"], member["line"]) + "。", ""]
285        return "\n".join(lines).rstrip() + "\n"

287    def render_index(self, page, title, keys, description):
288        lines = ["# " + title, "", link("类参考首页", relative(page, "docs/class-reference/README.md")) + " · " + link("类型总索引", relative(page, "docs/class-reference/type-index.md")), "", description, ""]
289        rows = [[self.type_link(k, page), code(self.types[k]["kind"]), code(self.types[k]["access"]), brief(self.types[k]["description"])] for k in sorted(keys, key=str.lower)]
290        lines += table(rows, ["类型", "种类", "声明可见性", "用途"])
291        return "\n".join(lines).rstrip() + "\n"

293    def pages(self):
294        result = {t["page"]: self.render_type(t) for t in self.types.values()}
295        if len(result) != len(self.types):
296            raise ValueError("类型页面路径重复。")
297        for typ in self.manifest["otherTypes"]:
298            result[typ["page"]] = self.render_other(typ)
299        for group, (title, description, _) in GROUPS.items():
300            page = f"docs/class-reference/{group}/README.md"
301            keys = [k for k, t in self.types.items() if t["page"].split("/")[2] == group]
302            result[page] = self.render_index(page, title, keys, description)
303        for legacy in self.manifest["legacyGroups"]:
304            page = legacy["page"]
305            if page not in result:
306                result[page] = self.render_index(page, Path(page).stem + " 类型导航", legacy["types"], "按原源码文件或测试集合保留的导航入口。每种类型的正文已放入独立页面。")
307        page = "docs/class-reference/type-index.md"
308        result[page] = self.render_index(page, "类型总索引", self.types, "按完整类型名查找。生产、内部和测试类型均保留，类型正文各自独立成页。")
309        other = ["", "## GDScript 与 Python", ""] + table([[link(t["name"], relative(page, t["page"]) + "#" + t["anchor"]), brief(t["description"])] for t in self.manifest["otherTypes"]], ["类型", "用途"])
310        result[page] += "\n".join(other)
311        page = "docs/class-reference/other-languages.md"
312        result[page] = "\n".join(["# GDScript 与 Python 命名类", "", link("类参考首页", "README.md"), "", "各命名类独立成页；未声明名称的脚本及模块函数不计入命名类型覆盖。", ""] + table([[link(t["name"], relative(page, t["page"]) + "#" + t["anchor"]), brief(t["description"])] for t in self.manifest["otherTypes"]], ["类型", "用途"]))
313        return result

316def main():
317    parser = argparse.ArgumentParser(description=__doc__)
318    parser.add_argument("--check", action="store_true", help="只检查，不修改页面")
319    args = parser.parse_args()
320    manifest = json.loads(MANIFEST.read_text(encoding="utf-8"))
321    if manifest.get("version") != 2:
322        raise SystemExit("需要 version=2 的结构化类参考清单。")
323    pages = ClassReferenceRenderer(manifest).pages()
324    different = []
325    for name, content in sorted(pages.items()):
326        path = ROOT / name
327        if not path.is_file() or path.read_text(encoding="utf-8") != content:
328            different.append(name)
329            if not args.check:
330                path.parent.mkdir(parents=True, exist_ok=True)
331                path.write_text(content, encoding="utf-8")
332    if args.check:
333        for name in different:
334            print("页面未同步：" + name)
335    print(f"类型参考页面：{len(pages)}；{'未同步' if args.check else '写入'}：{len(different)}")
336    return bool(different) if args.check else False

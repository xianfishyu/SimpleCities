3/// <summary>
4/// 地图背景渲染器 — 暗色底 + 网格线（对齐原点）
5/// </summary>
6public partial class MapBackground : CanvasLayer
7{
8    public static MapBackground Instance { get; private set; } = null!;
9
10    // ═══════════════════════════════════════════
11    // RoadConfig 引用
12    // ═══════════════════════════════════════════
13
14    [ExportGroup("配置引用")]
15    /// <summary>道路配置资源，网格大小和偏移从 CellSize 自动推导</summary>
16    [Export] public RoadConfig Config { get; set; } = null!;
17
18    // ═══════════════════════════════════════════
19    // 背景设置
20    // ═══════════════════════════════════════════
21
22    [ExportGroup("背景设置")]
23    [Export] public Color BackgroundColor = new(0.118f, 0.118f, 0.118f);
24
25    // ═══════════════════════════════════════════
26    // 网格设置
27    // ═══════════════════════════════════════════
28
29    [ExportGroup("网格设置")]
30    /// <summary>主网格 = CellSize × 此倍数</summary>
31    [Export(PropertyHint.Range, "1,20,1")] public int MajorGridCells = 5;
32    [Export] public float MainLineWidth = 1.5f;
33    [Export] public Color MajorGridColor = new(0.25f, 0.25f, 0.25f);
34
35    /// <summary>次网格 = CellSize × 此倍数</summary>
36    [Export(PropertyHint.Range, "1,10,1")] public int MinorGridCells = 1;
37    [Export] public float LineWidth = 0.5f;
38    [Export] public Color MinorGridColor = new(0.18f, 0.18f, 0.18f);
39
40    [Export] public float DotGridSize = 10f;
41    [Export] public float DotRadius = 0.5f;
42    [Export] public Color DotColor = new(0.20f, 0.20f, 0.20f);
43
44    // ═══════════════════════════════════════════
45    // 显示设置
46    // ═══════════════════════════════════════════
47
48    [ExportGroup("显示设置")]
49    [Export] public bool ShowGrid = true;
50    [Export] public bool ShowMainGrid = true;
51    [Export] public bool ShowMinorGrid = true;
52    [Export] public bool ShowDotGrid = true;
53
54    // ═══════════════════════════════════════════
55    // 节点引用
56    // ═══════════════════════════════════════════
57
58    [ExportGroup("节点引用")]
59    [Export] public ColorRect Display = null!;
60
61    private ShaderMaterial? _shaderMaterial;
62    private Vector2 _gridOffset;
63
64    public override void _Ready()
65    {
66		Instance = this;
67
68        // 回退到默认 RoadConfig
69        if (Config == null)
70        {
71            GD.PushError("MapBackground: Config (RoadConfig resource) is not assigned in the scene.");
72            Config = new RoadConfig();
73        }
74
75        _gridOffset = Vector2.Zero;
76
77        Display.Visible = true;
78        Display.AnchorLeft = 0;
79        Display.AnchorTop = 0;
80        Display.AnchorRight = 1;
81        Display.AnchorBottom = 1;
82        Display.OffsetLeft = 0;
83        Display.OffsetTop = 0;
84        Display.OffsetRight = 0;
85        Display.OffsetBottom = 0;
86
87        _shaderMaterial = Display.Material as ShaderMaterial;
88    }
89
90	public override void _ExitTree()
91	{
92		if (ReferenceEquals(Instance, this))
93			Instance = null!;
94		_shaderMaterial = null;
95	}
96
97    public override void _Process(double delta)
98    {
99        if (_shaderMaterial == null)
100            return;
101
102        MainCamera camera = MainCamera.Instance;
103        Vector2 cameraPos = camera?.GlobalPosition ?? Vector2.Zero;
104        float cameraZoom = camera?.Zoom.X ?? 1.0f;
105        Vector2 viewportSize = GetViewport().GetVisibleRect().Size;
106
107        // 背景
108        _shaderMaterial.SetShaderParameter("background_color",
109            new Vector3(BackgroundColor.R, BackgroundColor.G, BackgroundColor.B));
110
111        // 网格偏移（从 Config.CellSize 推导）
112        _shaderMaterial.SetShaderParameter("grid_offset", _gridOffset);
113
114        // 网格（尺寸 = CellSize × 倍数，Inspector 设倍数不会被覆盖）
115        float cellSize = Config.CellSize;
116        _shaderMaterial.SetShaderParameter("major_grid_size", cellSize * MajorGridCells);
117        _shaderMaterial.SetShaderParameter("major_line_width", MainLineWidth);
118        _shaderMaterial.SetShaderParameter("major_grid_color",
119            new Vector3(MajorGridColor.R, MajorGridColor.G, MajorGridColor.B));
120        _shaderMaterial.SetShaderParameter("minor_grid_size", cellSize * MinorGridCells);
121        _shaderMaterial.SetShaderParameter("minor_line_width", LineWidth);
122        _shaderMaterial.SetShaderParameter("minor_grid_color",
123            new Vector3(MinorGridColor.R, MinorGridColor.G, MinorGridColor.B));
124        _shaderMaterial.SetShaderParameter("show_major_grid", ShowMainGrid && ShowGrid);
125        _shaderMaterial.SetShaderParameter("show_minor_grid", ShowMinorGrid && ShowGrid);
126        _shaderMaterial.SetShaderParameter("show_dot_grid", ShowDotGrid && ShowGrid);
127        _shaderMaterial.SetShaderParameter("dot_grid_size", DotGridSize);
128        _shaderMaterial.SetShaderParameter("dot_radius", DotRadius);
129        _shaderMaterial.SetShaderParameter("dot_color",
130            new Vector3(DotColor.R, DotColor.G, DotColor.B));
131
132        // 相机
133        _shaderMaterial.SetShaderParameter("camera_pos", cameraPos);
134        _shaderMaterial.SetShaderParameter("camera_zoom", cameraZoom);
135        _shaderMaterial.SetShaderParameter("viewport_size", viewportSize);
136    }
137
138    public void ToggleGrid()
139    {
140        ShowGrid = !ShowGrid;
141    }
142}

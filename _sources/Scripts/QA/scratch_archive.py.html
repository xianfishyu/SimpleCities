20def read_manifest(repo):

25def digest(path):
26    with path.open('rb') as stream:
27        return hashlib.file_digest(stream, 'sha256').hexdigest()

30def archive_rows(manifest, group):
31    rows = list(manifest['archives'])
32    existing = {row['directory'] for row in rows}
33    for directory in sorted({file_directory(row['path']) for row in manifest['files']} - existing):
34        rows.append({'directory': directory, 'path': None, 'files': 0})
35    if group:
36        rows = [row for row in rows if row['directory'] == group]
37        if not rows:
38            raise ValueError('Unknown directory: ' + group)
39    rows = [dict(row, recoverableFiles=sum(file_directory(item['path']) == row['directory']
40                                         for item in manifest['files'])) for row in rows]
41    return rows

44def file_directory(name):
45    parts = PurePosixPath(name).parts
46    if not parts or parts[0] != '.scratch' or '..' in parts or '\\' in name or ':' in name:
47        raise ValueError('Invalid scratch path: ' + name)
48    return parts[1] if len(parts) > 2 else '.'

51class GitBlobs:
52    """Read historical blobs through one Git process, including binary files."""
53
54    def __init__(self, repo):
55        self.process = subprocess.Popen(['git', 'cat-file', '--batch'], cwd=repo,
56                                        stdin=subprocess.PIPE, stdout=subprocess.PIPE)
57
58    def __enter__(self):
59        return self
60
61    def __exit__(self, *_):
62        self.process.stdin.close()
63        self.process.stdout.close()
64        self.process.wait()
65
66    def read(self, row):
67        self.process.stdin.write((row['gitBlob'] + '\n').encode('ascii'))
68        self.process.stdin.flush()
69        header = self.process.stdout.readline().split()
70        if len(header) != 3 or header[1] != b'blob':
71            raise ValueError('Source Git blob unavailable; fetch full history or restore the local archive: ' + row['path'])
72        size = int(header[2])
73        data = self.process.stdout.read(size)
74        if len(data) != size or self.process.stdout.read(1) != b'\n':
75            raise ValueError('Incomplete Git blob: ' + row['path'])
76        return data

79def stored_git_file(blobs, row):
80    data = blobs.read(row)
81    transform = row.get('gitToOriginal', 'identity')
82    if transform == 'lf-to-crlf':
83        data = data.replace(b'\n', b'\r\n')
84    elif transform == 'lf-to-mixed-eol':
85        mask = zlib.decompress(base64.b64decode(row['crlfMask'], validate=True))
86        parts = data.split(b'\n')
87        if len(mask) != (len(parts) - 1 + 7) // 8:
88            raise ValueError('Invalid newline mask: ' + row['path'])
89        restored = bytearray()
90        for index, part in enumerate(parts):
91            if index:
92                line = index - 1
93                restored.extend(b'\r\n' if mask[line // 8] & (1 << (line % 8)) else b'\n')
94            restored.extend(part)
95        data = restored
96    elif transform != 'identity':
97        raise ValueError('Unknown Git transform: ' + transform)
98    if len(data) != row['bytes'] or hashlib.sha256(data).hexdigest() != row['sha256']:
99        raise ValueError('Source Git identity differs: ' + row['path'])
100    return data

103def write_restored(destination, name, data, row):
104    target = (destination / name).resolve()
105    if not target.is_relative_to(destination.resolve()):
106        raise ValueError('Member escapes destination: ' + name)
107    if target.exists():
108        if not target.is_file() or digest(target) != row['sha256']:
109            raise ValueError('Refusing to overwrite different file: ' + str(target))
110    else:
111        target.parent.mkdir(parents=True, exist_ok=True)
112        target.write_bytes(data)

115def restore_or_verify(repo, manifest, group, destination, source='auto'):
116    with ExitStack() as stack:
117        return process_archives(repo, manifest, group, destination, source, stack)

120def process_archives(repo, manifest, group, destination, source, stack):
121    results = []
122    files = {row['path']: row for row in manifest['files']}
123    blobs = None
124    for archive in archive_rows(manifest, group):
125        path = repo / archive['path'] if archive['path'] else None
126        readable = [row for row in manifest['files']
127                    if not row.get('archive') and file_directory(row['path']) == archive['directory']]
128        if source == 'git' or path is None or (source == 'auto' and not path.exists()):
129            if blobs is None:
130                blobs = stack.enter_context(GitBlobs(repo))
131            expected = [row for row in manifest['files'] if row.get('archive') and row['archive'] == archive['path']]
132            expected += readable
133            for row in expected:
134                data = stored_git_file(blobs, row)
135                if destination:
136                    write_restored(destination, row['path'], data, row)
137            results.append({'directory': archive['directory'], 'files': len(expected), 'verified': True, 'source': 'git-history'})
138            continue
139        if digest(path) != archive['sha256']:
140            raise ValueError('Archive hash differs: ' + archive['path'])
141        expected = {row['path'] for row in manifest['files'] if row.get('archive') == archive['path']}
142        seen = set()
143        with tarfile.open(path, 'r:xz') as stream:
144            for member in stream:
145                if not member.isfile() or member.name not in expected or member.name in seen:
146                    raise ValueError('Unexpected member: ' + member.name)
147                seen.add(member.name)
148                member_stream = stream.extractfile(member)
149                if member_stream is None:
150                    raise ValueError('Unreadable member: ' + member.name)
151                data = member_stream.read()
152                if len(data) != files[member.name]['bytes'] or hashlib.sha256(data).hexdigest() != files[member.name]['sha256']:
153                    raise ValueError('Member identity differs: ' + member.name)
154                if destination:
155                    write_restored(destination, member.name, data, files[member.name])
156        if seen != expected:
157            raise ValueError('Missing archive members: ' + archive['directory'])
158        if readable:
159            if blobs is None:
160                blobs = stack.enter_context(GitBlobs(repo))
161            for row in readable:
162                data = stored_git_file(blobs, row)
163                if destination:
164                    write_restored(destination, row['path'], data, row)
165        results.append({'directory': archive['directory'], 'files': len(seen) + len(readable), 'verified': True,
166                        'source': 'local-archive+git-history' if readable else 'local-archive'})
167    return results

170def read_bundle_manifest(archive):
171    path = archive.with_name(archive.name + '.manifest.json.gz')
172    with gzip.open(path, 'rt', encoding='utf-8') as stream:
173        manifest = json.load(stream)
174    if manifest.get('format') != 'scratch-bundle-v1':
175        raise ValueError('Unknown bundle format: ' + str(path))
176    return manifest

179def local_bundle(repo, group):
180    for name in LOCAL_BUNDLES:
181        path = repo / EVIDENCE_DIRECTORY / name
182        if path.exists():
183            manifest = read_bundle_manifest(path)
184            if not group or group in {file_directory(row['path']) for row in manifest['files']}:
185                return path, manifest
186    return None, None

189def process_bundle(archive, manifest, group, destination):
190    identity = manifest['archive']
191    if archive.stat().st_size != identity['bytes'] or digest(archive) != identity['sha256']:
192        raise ValueError('Bundle hash differs: ' + str(archive))
193    files = {row['path']: row for row in manifest['files']}
194    if len(files) != len(manifest['files']):
195        raise ValueError('Duplicate bundle paths')
196    groups = {file_directory(name) for name in files}
197    if group and group not in groups:
198        raise ValueError('Unknown directory: ' + group)
199    directories = set(manifest['directories'])
200    for name in directories:
201        file_directory(name)
202    seen, seen_dirs = set(), set()
203    with tarfile.open(archive, 'r:xz') as stream:
204        for member in stream:
205            if member.isdir():
206                if member.name not in directories or member.name in seen_dirs:
207                    raise ValueError('Unexpected bundle directory: ' + member.name)
208                seen_dirs.add(member.name)
209                continue
210            if not member.isfile() or member.name not in files or member.name in seen:
211                raise ValueError('Unexpected bundle member: ' + member.name)
212            seen.add(member.name)
213            row = files[member.name]
214            with stream.extractfile(member) as source:
215                data = source.read()
216            if len(data) != row['bytes'] or hashlib.sha256(data).hexdigest() != row['sha256']:
217                raise ValueError('Bundle member identity differs: ' + member.name)
218            if destination and (not group or file_directory(member.name) == group):
219                write_restored(destination, member.name, data, row)
220    if seen != set(files) or seen_dirs != directories:
221        raise ValueError('Missing bundle members')
222    if destination:
223        for name in directories:
224            if not group or name == '.scratch' or name.startswith('.scratch/' + group + '/') or name == '.scratch/' + group:
225                target = (destination / name).resolve()
226                if not target.is_relative_to(destination.resolve()):
227                    raise ValueError('Directory escapes destination: ' + name)
228                target.mkdir(parents=True, exist_ok=True)
229    return {'files': len(seen), 'directories': len(seen_dirs), 'verified': True, 'source': 'whole-bundle'}

232def check_working_tree(repo, manifest):
233    root = repo / '.scratch'
234    if not root.is_dir() or is_link(root):
235        raise ValueError('Expected ordinary scratch directory: ' + str(root))
236    expected = {row['path']: row for row in manifest['files']}
237    files, directories = set(), {'.scratch'}
238    for current, dirs, names in os.walk(root, followlinks=False):
239        for name in dirs:
240            path = Path(current) / name
241            if is_link(path):
242                raise ValueError('Refusing linked directory: ' + str(path))
243            directories.add(path.relative_to(repo).as_posix())
244        for name in names:
245            path = Path(current) / name
246            relative = path.relative_to(repo).as_posix()
247            if is_link(path) or not path.is_file():
248                raise ValueError('Refusing nonregular file: ' + str(path))
249            files.add(relative)
250            row = expected.get(relative)
251            if not row or path.stat().st_size != row['bytes'] or digest(path) != row['sha256']:
252                raise ValueError('Working file missing from backup or changed: ' + relative)
253    if files != set(expected) or directories != set(manifest['directories']):
254        raise ValueError('Working scratch tree differs from backup')
255    return {'workingFiles': len(files), 'workingTreeMatchesBundle': True}

258def is_link(path):
259    # Windows reparse attributes cover junctions on Python 3.11 as well.
260    attributes = getattr(path.lstat(), 'st_file_attributes', 0)
261    return path.is_symlink() or bool(attributes & getattr(stat, 'FILE_ATTRIBUTE_REPARSE_POINT', 0))

264def main():
265    parser = argparse.ArgumentParser(description=__doc__)
266    parser.add_argument('action', choices=['list', 'verify', 'extract'])
267    parser.add_argument('--directory', help='One original .scratch directory; default all')
268    parser.add_argument('--destination', type=Path, help='Restore root; default repository root')
269    parser.add_argument('--bundle', type=Path, help='Complete local tar.xz bundle with adjacent manifest.json.gz')
270    parser.add_argument('--check-working-tree', action='store_true',
271                        help='With bundle verify, also require every local scratch file and directory to match the backup')
272    parser.add_argument('--source', choices=['auto', 'archive', 'git'], default='auto',
273                        help='Prefer local archive, require archive, or use historical Git blobs')
274    args = parser.parse_args()
275    if args.check_working_tree and (not args.bundle or args.action != 'verify'):
276        parser.error('--check-working-tree requires verify --bundle')
277    repo = Path(__file__).resolve().parents[2]
278    bundle = args.bundle
279    if bundle:
280        manifest = read_bundle_manifest(bundle)
281    elif args.source != 'git' and not args.check_working_tree:
282        bundle, manifest = local_bundle(repo, args.directory)
283        if not bundle:
284            manifest = read_manifest(repo)
285    else:
286        manifest = read_manifest(repo)
287    if args.action == 'list':
288        if bundle:
289            listing = sorted({file_directory(row['path']) for row in manifest['files']})
290            if args.directory:
291                if args.directory not in listing:
292                    raise ValueError('Unknown directory: ' + args.directory)
293                listing = [args.directory]
294        else:
295            listing = archive_rows(manifest, args.directory)
296        print(json.dumps(listing, ensure_ascii=True, indent=2))
297    else:
298        destination = (args.destination or repo).resolve() if args.action == 'extract' else None
299        if bundle:
300            result = process_bundle(bundle, manifest, args.directory, destination)
301            if args.check_working_tree:
302                result.update(check_working_tree(repo, manifest))
303        else:
304            result = restore_or_verify(repo, manifest, args.directory, destination, args.source)
305        if destination:
306            # A restored scratch directory must remain outside Godot's resource import scan.
307            marker = destination / '.scratch/.gdignore'
308            if not marker.exists():
309                write_restored(destination, '.scratch/.gdignore', b'', {'sha256': hashlib.sha256(b'').hexdigest()})
310        print(json.dumps(result, ensure_ascii=True))
311    return 0

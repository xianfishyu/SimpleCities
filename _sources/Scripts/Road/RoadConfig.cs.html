13        RoadType.Dirt,

27    public bool TryValidateRoadTypeStyles(out string error)
28    {
29        if (!TryCreateRoadTypeStyleDefinitions(out List<RoadTypeStyleDefinition> styles, out error))
30            return false;
31
32        return TryValidateRoadTypeStyles(styles, out error);
33    }
34
35    public Godot.Collections.Dictionary GetRoadTypeStylesValidationResult()
36    {
37        bool valid = TryValidateRoadTypeStyles(out string error);
38        return new Godot.Collections.Dictionary
39        {
40            ["valid"] = valid,
41            ["error"] = error,
42        };
43    }
44
45    public RoadTypeStyle GetRoadTypeStyle(RoadType roadType)
46    {
47        if (!TryCreateRoadTypeStyleDefinitions(out List<RoadTypeStyleDefinition> styles, out string error) ||
48            !TryValidateRoadTypeStyles(styles, out error))
49        {
50            throw new InvalidOperationException($"RoadConfig RoadTypeStyles are invalid: {error}");
51        }
52
53        RoadTypeStyleDefinition resolved = ResolveRoadTypeStyle(styles, roadType);
54        foreach (RoadTypeStyle style in RoadTypeStyles!)
55        {
56            if (style.RoadType == resolved.RoadType)
57                return style;
58        }
59
60        throw new InvalidOperationException($"RoadConfig does not contain RoadTypeStyle '{roadType}'.");
61    }
62
63    internal static bool TryValidateRoadTypeStyles(
64        IReadOnlyList<RoadTypeStyleDefinition>? styles,
65        out string error)
66    {
67        if (styles is null)
68        {
69            error = "RoadTypeStyles array cannot be null.";
70            return false;
71        }
72
73        if (styles.Count != RequiredRoadTypes.Length)
74        {
75            error = $"RoadTypeStyles must contain exactly {RequiredRoadTypes.Length} entries; found {styles.Count}.";
76            return false;
77        }
78
79        var seen = new HashSet<RoadType>();
80        for (int index = 0; index < styles.Count; index++)
81        {
82            RoadTypeStyleDefinition style = styles[index];
83            if (!RoadTypeStyle.TryValidate(style, out error))
84            {
85                error = $"RoadTypeStyles[{index}] is invalid: {error}";
86                return false;
87            }
88
89            if (!seen.Add(style.RoadType))
90            {
91                error = $"RoadTypeStyles contains duplicate RoadType '{style.RoadType}'.";
92                return false;
93            }
94        }
95
96        foreach (RoadType roadType in RequiredRoadTypes)
97        {
98            if (!seen.Contains(roadType))
99            {
100                error = $"RoadTypeStyles is missing RoadType '{roadType}'.";
101                return false;
102            }
103        }
104
105        error = string.Empty;
106        return true;
107    }
108
109    internal static RoadTypeStyleDefinition ResolveRoadTypeStyle(
110        IReadOnlyList<RoadTypeStyleDefinition>? styles,
111        RoadType roadType)
112    {
113        if (!RoadTypeContract.IsDefined(roadType))
114            throw new ArgumentOutOfRangeException(nameof(roadType), roadType, "RoadType is not defined.");
115        if (!TryValidateRoadTypeStyles(styles, out string error))
116            throw new InvalidOperationException($"RoadConfig RoadTypeStyles are invalid: {error}");
117
118        foreach (RoadTypeStyleDefinition style in styles!)
119        {
120            if (style.RoadType == roadType)
121                return style;
122        }
123
124        throw new InvalidOperationException($"RoadConfig does not contain RoadTypeStyle '{roadType}'.");
125    }
126
127    private bool TryCreateRoadTypeStyleDefinitions(
128        out List<RoadTypeStyleDefinition> styles,
129        out string error)
130    {
131        styles = [];
132        if (RoadTypeStyles is null)
133        {
134            error = "RoadTypeStyles array cannot be null.";
135            return false;
136        }
137
138        styles.Capacity = RoadTypeStyles.Count;
139        for (int index = 0; index < RoadTypeStyles.Count; index++)
140        {
141            RoadTypeStyle? style = RoadTypeStyles[index];
142            if (style is null)
143            {
144                error = $"RoadTypeStyles[{index}] cannot be null.";
145                return false;
146            }
147            styles.Add(style.ToDefinition());
148        }
149
150        error = string.Empty;
151        return true;
152    }

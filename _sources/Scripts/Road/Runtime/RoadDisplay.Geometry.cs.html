42        {

51    private sealed class GeometryGroup
52    {
53        internal RoadSurfaceGroup Source { get; }
54        internal Piece[] Pieces { get; }
55        internal Fragment[] Fragments { get; }
56        internal GeometryGroup(RoadSurfaceGroup source, CancellationToken token)
57        {
58            Source = source;
59            Pieces = new Piece[source.Pieces.Count];
60            int fragmentCount = 0;
61            for (int i = 0; i < Pieces.Length; i++)
62            {
63                token.ThrowIfCancellationRequested();
64                RoadSurfacePiece item = source.Pieces[i];
65                var polygon = new Vector2[item.Corners.Count];
66                for (int c = 0; c < polygon.Length; c++) polygon[c] = ToVector(item.Corners[c]);
67                ValidatePolygon(polygon);
68                var piece = new Piece(this, i, item.Edge.Id, polygon, ToVector(item.Start), ToVector(item.End),
69                    item.StartParameter, item.EndParameter, item.JunctionNode);
70                Pieces[i] = piece;
71                fragmentCount = checked(fragmentCount + FragmentCount(piece));
72            }
73            Fragments = new Fragment[fragmentCount];
74            int offset = 0;
75            foreach (Piece piece in Pieces)
76            {
77                int count = FragmentCount(piece);
78                for (int part = 0; part < count; part++)
79                {
80                    token.ThrowIfCancellationRequested();
81                    Fragments[offset++] = new(piece, FragmentBounds(piece, part, count));
82                }
83            }
84        }
85    }

159        internal RoadDisplay Upload()
160        {
161            if (_uploaded) throw new InvalidOperationException("V4 display geometry was already uploaded.");
162            _uploaded = true;
163            if (_basis is not null) ObjectDisposedException.ThrowIf(_basis._disposed, _basis);
164            if (_basis is not null && ReferenceEquals(_target._pages, _basis._pages))
165            {
166                _target._resources = (_basis._resources ?? throw new InvalidOperationException("V4 source resources were not uploaded.")).Acquire();
167                _uploads = null;
168                return _target;
169            }
170            var owned = new Dictionary<PageKey, PageMeshes>(_target._pages.Count);
171            var meshes = new List<ArrayMesh>();
172            int ordinal = 0;
173            try
174            {
175                foreach (GeometryPage content in _target._pages)
176                {
177                    PageMeshes page;
178                    if (_basis is not null && _basis._resources!.Pages.TryGetValue(content.Key, out PageMeshes? previous) && ReferenceEquals(previous.Content, content))
179                        page = previous.Acquire();
180                    else
181                    {
182                        MeshArrays[] arrays = _uploads![content.Key];
183                        var created = new ArrayMesh[arrays.Length];
184                        try
185                        {
186                            for (int i = 0; i < created.Length; i++)
187                            {
188                                created[i] = UploadMesh(arrays[i]);
189#if DEBUG
190                                AfterMeshUpload?.Invoke(created[i], ordinal);
191#endif
192                                ordinal++;
193                            }
194                            page = new(content, created);
195                        }
196                        catch { foreach (ArrayMesh? mesh in created) mesh?.Dispose(); throw; }
197                    }
198                    try { owned.Add(content.Key, page); }
199                    catch { page.Release(); throw; }
200                    meshes.AddRange(page.Meshes);
201                }
202                _target._resources = new(owned, meshes.ToArray());
203                return _target;
204            }
205            catch { foreach (PageMeshes page in owned.Values) page.Release(); throw; }
206            finally { _uploads = null; }
207        }

209        private static ArrayMesh UploadMesh(MeshArrays data)
210        {
211            var mesh = new ArrayMesh();
212            try
213            {
214                using var arrays = new Godot.Collections.Array();
215                arrays.Resize((int)Godot.Mesh.ArrayType.Max);
216                arrays[(int)Godot.Mesh.ArrayType.Vertex] = data.Vertices;
217                arrays[(int)Godot.Mesh.ArrayType.Color] = data.Colors;
218                arrays[(int)Godot.Mesh.ArrayType.Index] = data.Indices;
219                mesh.AddSurfaceFromArrays(Godot.Mesh.PrimitiveType.Triangles, arrays);
220                if (mesh.GetSurfaceCount() != 1 || mesh.SurfaceGetArrayLen(0) != data.Vertices.Length || mesh.SurfaceGetArrayIndexLen(0) != data.Indices.Length)
221                    throw new InvalidOperationException("V4 road mesh preflight failed.");
222                return mesh;
223            }
224            catch { mesh.Dispose(); throw; }
225        }

244    private static PreparedGeometry PrepareRevision(RoadSurfaceRevision revision, CancellationToken cancellationToken, double surfaceMs)
245    {
246        var pages = ImmutableList.CreateBuilder<GeometryPage>();
247        var groups = new List<GeometryGroup>(OwnersPerPage);
248        RoadSurfaceLayer? active = null;
249        foreach (RoadSurfaceGroup source in revision.Groups.Values)
250        {
251            cancellationToken.ThrowIfCancellationRequested();
252            if (active is RoadSurfaceLayer previous && previous != source.Key.Layer) Flush();
253            active = source.Key.Layer;
254            groups.Add(new(source, cancellationToken));
255            if (groups.Count == OwnersPerPage) Flush();
256        }
257        Flush();
258        ImmutableList<GeometryPage> immutable = pages.ToImmutable();
259        PieceIndex index = PieceIndex.Create(immutable.SelectMany(page => page.Groups), cancellationToken);
260        var uploads = new Dictionary<PageKey, MeshArrays[]>(immutable.Count);
261        foreach (GeometryPage page in immutable) uploads.Add(page.Key, PrepareMeshArrays(page, cancellationToken));
262        return new PreparedGeometry(new(revision, immutable, index), null, uploads, surfaceMs);
263
264        void Flush()
265        {
266            if (groups.Count == 0) return;
267            pages.Add(MakePage(groups, 0, groups.Count));
268            groups.Clear();
269            BalancePage(pages, pages.Count - 1);
270        }
271    }

273    internal static PreparedGeometry PrepareChange(RoadDisplay basis, RoadPlan plan, CancellationToken token)
274    {
275        ObjectDisposedException.ThrowIf(basis._disposed, basis);
276        long start = Stopwatch.GetTimestamp();
277        RoadSurfaceUpdate update = basis._surface.Apply(plan, token);
278        double surfaceMs = Stopwatch.GetElapsedTime(start).TotalMilliseconds;
279        var pages = basis._pages.ToBuilder();
280        var removed = new List<GeometryGroup>();
281        var added = new List<GeometryGroup>();
282        foreach (RoadSurfaceGroupKey key in update.ChangedGroups.Order())
283        {
284            token.ThrowIfCancellationRequested();
285            int pageIndex = FindPageIndex(pages, key);
286            RoadSurfaceGroup? source = update.Target.FindGroup(key);
287            if (pageIndex < 0)
288            {
289                if (source is null) throw new InvalidOperationException("V4 source surface group is missing.");
290                var fresh = new GeometryGroup(source, token);
291                added.Add(fresh);
292                pages.Insert(~pageIndex, MakePage(new[] { fresh }, 0, 1));
293                continue;
294            }
295            GeometryPage previous = pages[pageIndex];
296            int groupIndex = FindGroupIndex(previous.Groups, key.Owner);
297            GeometryGroup? old = groupIndex >= 0 ? previous.Groups[groupIndex] : null;
298            if (old is not null && ReferenceEquals(old.Source, source)) continue;
299            var groups = previous.Groups.ToList();
300            if (old is not null) { removed.Add(old); groups.RemoveAt(groupIndex); }
301            if (source is not null)
302            {
303                var fresh = new GeometryGroup(source, token);
304                added.Add(fresh);
305                groups.Insert(groupIndex >= 0 ? groupIndex : ~groupIndex, fresh);
306            }
307            else if (old is null) throw new InvalidOperationException("V4 source surface group is missing.");
308            if (groups.Count == 0) pages.RemoveAt(pageIndex);
309            else if (groups.Count > OwnersPerPage)
310            {
311                int middle = groups.Count / 2;
312                pages[pageIndex] = MakePage(groups, 0, middle);
313                pages.Insert(pageIndex + 1, MakePage(groups, middle, groups.Count - middle));
314            }
315            else
316            {
317                pages[pageIndex] = MakePage(groups, 0, groups.Count);
318                BalancePage(pages, pageIndex);
319            }
320        }
321        ImmutableList<GeometryPage> targetPages = pages.ToImmutable();
322        var uploads = new Dictionary<PageKey, MeshArrays[]>();
323        foreach (GeometryPage page in targetPages)
324        {
325            token.ThrowIfCancellationRequested();
326            int previous = FindPageIndex(basis._pages, new(page.Key.Layer, page.Key.FirstOwner));
327            if (previous < 0 || !ReferenceEquals(basis._pages[previous], page))
328                uploads.Add(page.Key, PrepareMeshArrays(page, token));
329        }
330        PieceIndex index = basis._index.Replace(removed, added, token);
331        token.ThrowIfCancellationRequested();
332        return new PreparedGeometry(new(update.Target, targetPages, index), basis, uploads, surfaceMs);
333    }

343    // Pages contain actual owners, not numeric ID slots. Local redistribution bounds
344    // draw calls even when a legal archive has sparse IDs or editing leaves ID gaps.
345    private static void BalancePage(ImmutableList<GeometryPage>.Builder pages, int index)
346    {
347        GeometryPage page = pages[index];
348        if (page.Groups.Length >= MinimumOwnersPerPage) return;
349        int left;
350        if (index > 0 && pages[index - 1].Key.Layer == page.Key.Layer) left = index - 1;
351        else if (index + 1 < pages.Count && pages[index + 1].Key.Layer == page.Key.Layer) left = index;
352        else return;
353        var combined = new List<GeometryGroup>(pages[left].Groups.Length + pages[left + 1].Groups.Length);
354        combined.AddRange(pages[left].Groups);
355        combined.AddRange(pages[left + 1].Groups);
356        if (combined.Count <= OwnersPerPage)
357        {
358            pages[left] = MakePage(combined, 0, combined.Count);
359            pages.RemoveAt(left + 1);
360        }
361        else
362        {
363            int middle = combined.Count / 2;
364            pages[left] = MakePage(combined, 0, middle);
365            pages[left + 1] = MakePage(combined, middle, combined.Count - middle);
366        }
367    }

369    // A negative result is the insertion point for a layer that has no page yet.
370    // Within a layer, IDs in a gap select the predecessor page, then group lookup
371    // separately proves whether that owner exists.
372    private static int FindPageIndex(IReadOnlyList<GeometryPage> pages, RoadSurfaceGroupKey group)
373    {
374        var key = new PageKey(group.Layer, group.Owner);
375        int low = 0, high = pages.Count;
376        while (low < high)
377        {
378            int middle = low + (high - low) / 2;
379            if (pages[middle].Key.CompareTo(key) <= 0) low = middle + 1;
380            else high = middle;
381        }
382        int previous = low - 1;
383        if (previous >= 0 && pages[previous].Key.Layer == group.Layer) return previous;
384        if (low < pages.Count && pages[low].Key.Layer == group.Layer) return low;
385        return ~low;
386    }

406    private static int FindGroupIndex(IReadOnlyList<GeometryGroup> groups, long owner)
407    {
408        int low = 0, high = groups.Count - 1;
409        while (low <= high)
410        {
411            int middle = low + (high - low) / 2;
412            if (groups[middle].Source.Key.Owner == owner) return middle;
413            if (groups[middle].Source.Key.Owner < owner) low = middle + 1;
414            else high = middle - 1;
415        }
416        return ~low;
417    }

419    private static MeshArrays[] PrepareMeshArrays(GeometryPage page, CancellationToken token)
420    {
421        var chunks = new List<MeshArrays>();
422        int capacity = Math.Min(VerticesPerMesh / 3, page.Groups.Sum(group => group.Pieces.Length));
423        var pieces = new List<Piece>(capacity);
424        int vertices = 0, indices = 0;
425        foreach (GeometryGroup group in page.Groups)
426            foreach (Piece piece in group.Pieces)
427            {
428                token.ThrowIfCancellationRequested();
429                if (vertices + piece.Polygon.Length > VerticesPerMesh) Flush();
430                pieces.Add(piece);
431                vertices += piece.Polygon.Length;
432                indices += 3 * (piece.Polygon.Length - 2);
433            }
434        Flush();
435        return chunks.ToArray();
436
437        void Flush()
438        {
439            if (pieces.Count == 0) return;
440            var data = new MeshArrays(new Vector3[vertices], new Color[vertices], new int[indices]);
441            int vertex = 0, index = 0;
442            foreach (Piece piece in pieces)
443            {
444                token.ThrowIfCancellationRequested();
445                RoadProfile profile = RoadProfiles.Get(piece.Group.Source.Pieces[piece.Order].Edge.Profile);
446                var color = new Color(profile.Red / 255f, profile.Green / 255f, profile.Blue / 255f);
447                int offset = vertex;
448                foreach (Vector2 point in piece.Polygon) { data.Vertices[vertex] = new(point.X, point.Y, 0); data.Colors[vertex++] = color; }
449                for (int corner = 1; corner < piece.Polygon.Length - 1; corner++)
450                { data.Indices[index++] = offset; data.Indices[index++] = offset + corner; data.Indices[index++] = offset + corner + 1; }
451            }
452            chunks.Add(data);
453            pieces.Clear();
454            vertices = indices = 0;
455        }
456    }

461    private static SpatialBounds FragmentBounds(Piece piece, int part, int count)
462    {
463        Vector2[] polygon = piece.Polygon;
464        Vector2 a = polygon[0], b = polygon[1], c = polygon[2], d = polygon[^1];
465        if (count > 1)
466        {
467            a = polygon[0].Lerp(polygon[1], (float)part / count);
468            b = polygon[0].Lerp(polygon[1], (float)(part + 1) / count);
469            c = polygon[3].Lerp(polygon[2], (float)(part + 1) / count);
470            d = polygon[3].Lerp(polygon[2], (float)part / count);
471        }
472        return new(Math.Min(Math.Min(a.X, b.X), Math.Min(c.X, d.X)) - 0.001,
473            Math.Min(Math.Min(a.Y, b.Y), Math.Min(c.Y, d.Y)) - 0.001,
474            Math.Max(Math.Max(a.X, b.X), Math.Max(c.X, d.X)) + 0.001,
475            Math.Max(Math.Max(a.Y, b.Y), Math.Max(c.Y, d.Y)) + 0.001);
476    }

10    {
11        internal Piece Piece { get; } = piece;
12        internal SpatialBounds Bounds { get; } = bounds;
13        internal bool Intersects(SpatialBounds other) => Bounds.MinX <= other.MaxX && Bounds.MaxX >= other.MinX &&
14            Bounds.MinY <= other.MaxY && Bounds.MaxY >= other.MinY;
15    }

17    /// <summary>Immutable spatial buckets with stable fragment identities; only touched buckets are replaced.</summary>
18    private sealed class PieceIndex(ImmutableDictionary<(int X, int Y), Fragment[]> buckets)
19    {
20        private readonly ImmutableDictionary<(int X, int Y), Fragment[]> _buckets = buckets;
21        private const double BucketSize = 100;
22        private static int Bucket(double coordinate) => (int)Math.Floor(coordinate / BucketSize);
23        private static bool Valid(SpatialBounds bounds) => bounds.IsValid &&
24            Math.Abs(bounds.MinX / BucketSize) < int.MaxValue / 2d && Math.Abs(bounds.MaxX / BucketSize) < int.MaxValue / 2d &&
25            Math.Abs(bounds.MinY / BucketSize) < int.MaxValue / 2d && Math.Abs(bounds.MaxY / BucketSize) < int.MaxValue / 2d;
26
27        internal static PieceIndex Create(IEnumerable<GeometryGroup> groups, CancellationToken token)
28        {
29            var counts = new Dictionary<(int, int), int>();
30            foreach (GeometryGroup group in groups)
31                foreach (Fragment fragment in group.Fragments)
32                {
33                    token.ThrowIfCancellationRequested();
34                    SpatialBounds b = fragment.Bounds;
35                    for (int x = Bucket(b.MinX); x <= Bucket(b.MaxX); x++)
36                        for (int y = Bucket(b.MinY); y <= Bucket(b.MaxY); y++)
37                            counts[(x, y)] = counts.GetValueOrDefault((x, y)) + 1;
38                }
39            var builder = ImmutableDictionary.CreateBuilder<(int, int), Fragment[]>();
40            foreach (var pair in counts) builder.Add(pair.Key, new Fragment[pair.Value]);
41            var positions = new Dictionary<(int, int), int>(counts.Count);
42            foreach (GeometryGroup group in groups)
43                foreach (Fragment fragment in group.Fragments)
44                {
45                    token.ThrowIfCancellationRequested();
46                    SpatialBounds b = fragment.Bounds;
47                    for (int x = Bucket(b.MinX); x <= Bucket(b.MaxX); x++)
48                        for (int y = Bucket(b.MinY); y <= Bucket(b.MaxY); y++)
49                        {
50                            int position = positions.GetValueOrDefault((x, y));
51                            builder[(x, y)][position] = fragment;
52                            positions[(x, y)] = position + 1;
53                        }
54                }
55            return new(builder.ToImmutable());
56        }

58        internal PieceIndex Replace(IReadOnlyList<GeometryGroup> removed, IReadOnlyList<GeometryGroup> added, CancellationToken token)
59        {
60            if (removed.Count == 0 && added.Count == 0) return this;
61            var removedSet = new HashSet<GeometryGroup>(removed);
62            var touched = new Dictionary<(int, int), List<Fragment>>();
63            foreach (GeometryGroup group in removed)
64                foreach (Fragment fragment in group.Fragments) Visit(fragment, append: false);
65            foreach (GeometryGroup group in added)
66                foreach (Fragment fragment in group.Fragments) Visit(fragment, append: true);
67            var builder = _buckets.ToBuilder();
68            foreach (var pair in touched)
69            {
70                token.ThrowIfCancellationRequested();
71                if (pair.Value.Count == 0) builder.Remove(pair.Key);
72                else builder[pair.Key] = pair.Value.ToArray();
73            }
74            return new(builder.ToImmutable());
75
76            void Visit(Fragment fragment, bool append)
77            {
78                token.ThrowIfCancellationRequested();
79                SpatialBounds b = fragment.Bounds;
80                for (int x = Bucket(b.MinX); x <= Bucket(b.MaxX); x++)
81                    for (int y = Bucket(b.MinY); y <= Bucket(b.MaxY); y++)
82                    {
83                        if (!touched.TryGetValue((x, y), out List<Fragment>? values))
84                        {
85                            values = new();
86                            if (_buckets.TryGetValue((x, y), out Fragment[]? previous))
87                                foreach (Fragment old in previous)
88                                    if (!removedSet.Contains(old.Piece.Group)) values.Add(old);
89                            touched.Add((x, y), values);
90                        }
91                        if (append) values.Add(fragment);
92                    }
93            }
94        }

96        internal SpatialQueryResult<Piece> QueryBounds(SpatialBounds bounds, RoadStateToken source, SpatialQueryBudget limits)
97        {
98            if (!limits.IsValid || !Valid(bounds)) return new(source, SpatialQueryStatus.InvalidParameters, [], default, "空间查询参数非法");
99            int left = Bucket(bounds.MinX), right = Bucket(bounds.MaxX), top = Bucket(bounds.MinY), bottom = Bucket(bounds.MaxY);
100            if (left == right && top == bottom)
101            {
102                List<Piece>? results = null;
103                int candidates = 0;
104                if (_buckets.TryGetValue((left, top), out Fragment[]? entries))
105                    foreach (Fragment fragment in entries)
106                    {
107                        if (candidates >= limits.MaxCandidates)
108                            return new(source, SpatialQueryStatus.BudgetExceeded, [], new(1, candidates, 0, 0, 0), "空间查询候选或分桶预算已耗尽");
109                        candidates++;
110                        if (fragment.Intersects(bounds)) (results ??= []).Add(fragment.Piece);
111                    }
112                return new(source, SpatialQueryStatus.Ready, results is null ? [] : results.AsReadOnly(), new(1, candidates, 0, 0, 0), "");
113            }
114            var query = new Query(this, source, limits, bounds);
115            for (int x = left; x <= right; x++)
116                for (int y = top; y <= bottom; y++)
117                    if (!query.Visit(x, y)) return query.Result();
118            return query.Result();
119        }

121        internal SpatialQueryResult<Piece> QuerySegment(RoadPoint from, RoadPoint to, RoadStateToken source, SpatialQueryBudget limits)
122        {
123            var query = new Query(this, source, limits, SpatialBounds.Between(from, to));
124            if (!from.IsFinite || !to.IsFinite || !query.IsValid) return query.Invalid();
125            int x = Bucket(from.X), y = Bucket(from.Y), endX = Bucket(to.X), endY = Bucket(to.Y);
126            int stepX = Math.Sign(to.X - from.X), stepY = Math.Sign(to.Y - from.Y);
127            double dx = to.X - from.X, dy = to.Y - from.Y;
128            double nextX = stepX == 0 ? double.PositiveInfinity : ((x + (stepX > 0 ? 1d : 0d)) * BucketSize - from.X) / dx;
129            double nextY = stepY == 0 ? double.PositiveInfinity : ((y + (stepY > 0 ? 1d : 0d)) * BucketSize - from.Y) / dy;
130            double deltaX = stepX == 0 ? double.PositiveInfinity : BucketSize / Math.Abs(dx);
131            double deltaY = stepY == 0 ? double.PositiveInfinity : BucketSize / Math.Abs(dy);
132            while (true)
133            {
134                if (!query.Visit(x, y) || (x == endX && y == endY)) return query.Result();
135                // Match floor endpoint ownership and both side buckets at a corner touch.
136                if (x == endX) nextX = double.PositiveInfinity;
137                if (y == endY) nextY = double.PositiveInfinity;
138                if (nextX < nextY) { x += stepX; nextX += deltaX; }
139                else if (nextY < nextX) { y += stepY; nextY += deltaY; }
140                else
141                {
142                    if (!query.Visit(x + stepX, y) || !query.Visit(x, y + stepY)) return query.Result();
143                    x += stepX; y += stepY; nextX += deltaX; nextY += deltaY;
144                }
145            }
146        }

148        private sealed class Query(PieceIndex owner, RoadStateToken source, SpatialQueryBudget budget, SpatialBounds bounds)
149        {
150            private readonly HashSet<(int, int)> _visited = [];
151            private readonly HashSet<Fragment> _seen = [];
152            private readonly List<Piece> _results = [];
153            private SpatialQueryStatus _status = SpatialQueryStatus.Ready;
154            internal bool IsValid => budget.IsValid && Valid(bounds);
155            internal bool Visit(int x, int y)
156            {
157                if (_visited.Contains((x, y))) return true;
158                if (_visited.Count >= budget.MaxBuckets) { _status = SpatialQueryStatus.BudgetExceeded; return false; }
159                _visited.Add((x, y));
160                if (!owner._buckets.TryGetValue((x, y), out Fragment[]? entries)) return true;
161                foreach (Fragment fragment in entries)
162                {
163                    if (_seen.Contains(fragment)) continue;
164                    if (_seen.Count >= budget.MaxCandidates) { _status = SpatialQueryStatus.BudgetExceeded; return false; }
165                    _seen.Add(fragment);
166                    if (fragment.Intersects(bounds)) _results.Add(fragment.Piece);
167                }
168                return true;
169            }
170            internal SpatialQueryResult<Piece> Invalid() => new(source, SpatialQueryStatus.InvalidParameters, [], default, "空间查询参数非法");
171            internal SpatialQueryResult<Piece> Result() => !IsValid ? Invalid() :
172                new(source, _status, _status == SpatialQueryStatus.Ready ? _results.AsReadOnly() : [],
173                    new(_visited.Count, _seen.Count, 0, 0, 0), _status == SpatialQueryStatus.Ready ? "" : "空间查询候选或分桶预算已耗尽");
174        }
175    }

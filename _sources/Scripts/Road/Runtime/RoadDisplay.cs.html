13    private static void ValidatePolygon(Vector2[] polygon)
14    {
15        double winding = 0;
16        for (int i = 0; i < polygon.Length; i++)
17        {
18            Vector2 a = polygon[i], b = polygon[(i + 1) % polygon.Length], c = polygon[(i + 2) % polygon.Length];
19            // Compute from the shared binary32 vertices, widening before arithmetic so large
20            // world offsets do not hide a collapsed or concave triangle through cancellation.
21            double cross = ((double)b.X - a.X) * ((double)c.Y - b.Y) - ((double)b.Y - a.Y) * ((double)c.X - b.X);
22            if (cross == 0 || (winding != 0 && Math.Sign(cross) != winding))
23                throw new InvalidOperationException("V4 road surface is degenerate or non-convex after display conversion.");
24            winding = Math.Sign(cross);
25        }
26    }

35    internal SpatialQueryResult<HitResult> QueryHit(Vector2 world, SpatialQueryBudget? budget = null)
36    {
37        SpatialQueryBudget limits = budget ?? SpatialQueryBudget.Default;
38        SpatialQueryResult<Piece> candidates = _index.QueryBounds(new(world.X, world.Y, world.X, world.Y), Snapshot.Token, limits);
39        if (candidates.Status != SpatialQueryStatus.Ready)
40            return new(Snapshot.Token, candidates.Status, [], candidates.Metrics, candidates.Reason);
41        int exact = 0;
42        // Last drawn polygon owns the visible surface where differently coloured pieces overlap.
43        foreach (Piece piece in candidates.Results.Distinct().OrderByDescending(piece => piece, PaintComparer.Instance))
44        {
45            if (exact >= limits.MaxExactTests)
46                return new(Snapshot.Token, SpatialQueryStatus.BudgetExceeded, [], candidates.Metrics with { ExactGeometryTests = exact }, "空间查询精确检查预算已耗尽");
47            exact++;
48            if (!Geometry2D.IsPointInPolygon(world, piece.Polygon)) continue;
49            Vector2 direction = piece.End - piece.Start;
50            double fraction = direction.LengthSquared() == 0 ? 0 : Mathf.Clamp((world - piece.Start).Dot(direction) / direction.LengthSquared(), 0, 1);
51            double parameter = piece.StartParameter + fraction * (piece.EndParameter - piece.StartParameter);
52            return new(Snapshot.Token, SpatialQueryStatus.Ready,
53                [new HitResult(new CoreRoadLocation(Snapshot.Token, piece.Edge, parameter), piece.JunctionNode)],
54                candidates.Metrics with { ExactGeometryTests = exact, Hits = 1 }, "");
55        }
56        return new(Snapshot.Token, SpatialQueryStatus.Ready, [], candidates.Metrics with { ExactGeometryTests = exact }, "");
57    }

59    internal SpatialQueryResult<RoadGridSpan> QuerySpan(Vector2 world, SpatialQueryBudget? budget = null)
60    {
61        SpatialQueryResult<HitResult> result = QueryHit(world, budget);
62        HitResult? hit = result.Results.Cast<HitResult?>().FirstOrDefault();
63        // A junction patch has several incidences. Only the visible branch outside it selects a span.
64        RoadGridSpan? span = hit is HitResult picked && picked.JunctionNode is null
65            ? RoadSpanQuery.Pick(Snapshot, picked.Location) : null;
66        return new(Snapshot.Token, result.Status, span is null ? [] : [span],
67            result.Metrics with { Hits = span is null ? 0 : 1 }, result.Reason);
68    }

70    internal SpatialQueryResult<RoadGridSpan> QueryTraceSpans(Vector2 from, Vector2 to, SpatialQueryBudget? budget = null)
71    {
72        if (from == to) return QuerySpan(from, budget);
73        SpatialQueryBudget limits = budget ?? SpatialQueryBudget.Default;
74        SpatialQueryResult<Piece> candidates = _index.QuerySegment(new(from.X, from.Y), new(to.X, to.Y), Snapshot.Token, limits);
75        if (candidates.Status != SpatialQueryStatus.Ready)
76            return new(Snapshot.Token, candidates.Status, [], candidates.Metrics, candidates.Reason);
77        SpatialQueryMetrics metrics = candidates.Metrics;
78        var result = new List<RoadGridSpan>();
79        var seen = new HashSet<RoadSpanKey>();
80
81        // Partition the pointer segment at every visible polygon boundary and grid-span boundary.
82        // This captures every crossed span even when an input event traverses the entire map.
83        var breaks = new List<double> { 0, 1 };
84        Vector2 travel = to - from;
85        foreach (Piece piece in candidates.Results.Distinct())
86        {
87            if (metrics.ExactGeometryTests >= limits.MaxExactTests) return Exhausted();
88            metrics = metrics with { ExactGeometryTests = metrics.ExactGeometryTests + 1 };
89            if (!Clip(piece.Polygon, from, travel, out double entry, out double exit)) continue;
90            breaks.Add(entry);
91            breaks.Add(exit);
92            Vector2 axis = piece.End - piece.Start;
93            double lengthSquared = axis.LengthSquared();
94            if (lengthSquared == 0 || piece.JunctionNode is not null) continue;
95            double fractionStart = (from - piece.Start).Dot(axis) / lengthSquared;
96            double fractionDelta = travel.Dot(axis) / lengthSquared;
97            if (fractionDelta == 0) continue;
98            // All eight supported directions cross one main-grid line per span. For diagonals
99            // either coordinate gives the same crossing, so use the dominant coordinate only.
100            double origin = Math.Abs(axis.X) >= Math.Abs(axis.Y) ? piece.Start.X : piece.Start.Y;
101            double extent = Math.Abs(axis.X) >= Math.Abs(axis.Y) ? axis.X : axis.Y;
102            double startCoordinate = origin + extent * fractionStart;
103            double coordinateDelta = extent * fractionDelta;
104            double low = Math.Max(Math.Min(origin, origin + extent), Math.Min(startCoordinate + coordinateDelta * entry, startCoordinate + coordinateDelta * exit));
105            double high = Math.Min(Math.Max(origin, origin + extent), Math.Max(startCoordinate + coordinateDelta * entry, startCoordinate + coordinateDelta * exit));
106            int cell = Snapshot.Map.CellSizeMetres;
107            for (int grid = (int)Math.Ceiling(low / cell); grid <= Math.Floor(high / cell); grid++)
108            {
109                double t = (grid * cell - startCoordinate) / coordinateDelta;
110                if (t > entry && t < exit) breaks.Add(t);
111            }
112        }
113        breaks.Sort();
114        if (!Add(from)) return Exhausted();
115        for (int i = 1; i < breaks.Count; i++)
116        {
117            double left = breaks[i - 1], right = breaks[i];
118            if (right > left && !Add(from + travel * (float)((left + right) / 2))) return Exhausted();
119        }
120        if (!Add(to)) return Exhausted();
121        return new(Snapshot.Token, SpatialQueryStatus.Ready, result.AsReadOnly(),
122            metrics with { Hits = result.Select(span => span.Edge).Distinct().Count() }, "");
123
124        bool Add(Vector2 point)
125        {
126            var remaining = new SpatialQueryBudget(limits.MaxBuckets - metrics.BucketsVisited,
127                limits.MaxCandidates - metrics.FragmentCandidates, limits.MaxExactTests - metrics.ExactGeometryTests);
128            if (!remaining.IsValid) return false;
129            SpatialQueryResult<RoadGridSpan> picked = QuerySpan(point, remaining);
130            metrics = new(metrics.BucketsVisited + picked.Metrics.BucketsVisited,
131                metrics.FragmentCandidates + picked.Metrics.FragmentCandidates,
132                metrics.ExactGeometryTests + picked.Metrics.ExactGeometryTests, 0, 0);
133            if (picked.Status != SpatialQueryStatus.Ready) return false;
134            RoadGridSpan? span = picked.Results.FirstOrDefault();
135            if (span is not null && seen.Add(span.Key)) result.Add(span);
136            return true;
137        }
138
139        SpatialQueryResult<RoadGridSpan> Exhausted() => new(Snapshot.Token, SpatialQueryStatus.BudgetExceeded,
140            [], metrics with { Hits = 0 }, "拖选查询预算已耗尽，本次手势不可提交");
141    }

145    private static bool Clip(Vector2[] polygon, Vector2 from, Vector2 travel, out double entry, out double exit)
146    {
147        entry = 0;
148        exit = 1;
149        double area = 0;
150        for (int i = 0; i < polygon.Length; i++) area += polygon[i].Cross(polygon[(i + 1) % polygon.Length]);
151        double winding = Math.Sign(area);
152        if (winding == 0) return false;
153        for (int i = 0; i < polygon.Length; i++)
154        {
155            Vector2 start = polygon[i], side = polygon[(i + 1) % polygon.Length] - start;
156            double value = winding * side.Cross(from - start);
157            double slope = winding * side.Cross(travel);
158            if (slope == 0)
159            {
160                if (value < 0) return false;
161                continue;
162            }
163            double crossing = -value / slope;
164            if (slope > 0) entry = Math.Max(entry, crossing);
165            else exit = Math.Min(exit, crossing);
166            if (entry > exit) return false;
167        }
168        return true;
169    }

171    internal Vector2 SurfaceCenter(CoreRoadLocation location)
172    {
173        if (location.Source != Snapshot.Token) throw new InvalidOperationException("V4 surface location is stale.");
174        GeometryGroup? group = FindGeometryGroup(new(RoadSurfaceLayer.Ribbon, location.Edge.Value));
175        if (group is null)
176            throw new InvalidOperationException("V4 surface location does not belong to the displayed road.");
177        if (!double.IsFinite(location.Parameter) || location.Parameter < 0 || location.Parameter > 1)
178            throw new InvalidOperationException("V4 surface parameter is invalid.");
179        Piece[] edgePieces = group.Pieces;
180        int low = 0, high = edgePieces.Length;
181        while (low + 1 < high)
182        {
183            int middle = (low + high) / 2;
184            if (edgePieces[middle].StartParameter <= location.Parameter) low = middle;
185            else high = middle;
186        }
187        Piece piece = edgePieces[low];
188        double range = piece.EndParameter - piece.StartParameter;
189        return range == 0 ? piece.Start : piece.Start.Lerp(piece.End, (float)((location.Parameter - piece.StartParameter) / range));
190    }

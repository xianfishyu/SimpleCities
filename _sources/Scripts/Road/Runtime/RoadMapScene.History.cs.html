1using Godot;
2using System.Diagnostics;
3using SimpleCities.RoadCore;
4
5public partial class RoadMapScene
6{
7    private Button _undoRoad = null!;
8    private Button _redoRoad = null!;
9    private bool CanUseHistory => CanEdit && !_draftStart.HasValue && !_selectionSession.IsSelecting;
10
11    private void InitializeHistory()
12    {
13        _undoRoad = GetNode<Button>(Controls + "History/Undo");
14        _redoRoad = GetNode<Button>(Controls + "History/Redo");
15        _undoRoad.Pressed += () => UndoRoadEdit();
16        _redoRoad.Pressed += () => RedoRoadEdit();
17    }
18
19    private void UpdateHistoryControls()
20    {
21        _undoRoad.Disabled = !CanUseHistory || _roads!.Network.History.UndoCount == 0;
22        _redoRoad.Disabled = !CanUseHistory || _roads!.Network.History.RedoCount == 0;
23    }
24
25    public bool UndoRoadEdit() => SubmitHistory(redo: false);
26    public bool RedoRoadEdit() => SubmitHistory(redo: true);
27
28    private bool SubmitHistory(bool redo)
29    {
30        if (!CanUseHistory || (redo ? _roads!.Network.History.RedoCount : _roads!.Network.History.UndoCount) == 0)
31            return false;
32        ClearRoadSelection();
33        SubmitRoadOperation((network, token) =>
34        {
35            RoadEditResult result = redo ? network.PlanRedo(token) : network.PlanUndo(token);
36            return new PlannedRoadOperation(result.Plan, result.Status == RoadEditStatus.NoChange, result.Reason);
37        }, Stopwatch.GetTimestamp(), redo ? "道路操作已重做" : "道路操作已撤销", "");
38        return true;
39    }
40
41    private bool HandleHistoryInput(InputEvent input)
42    {
43        if (input is not InputEventKey { Pressed: true, Echo: false, CtrlPressed: true } key ||
44            key.Keycode is not (Key.Z or Key.Y)) return false;
45        if (key.Keycode == Key.Y || key.ShiftPressed) RedoRoadEdit();
46        else UndoRoadEdit();
47        GetViewport().SetInputAsHandled();
48        return true;
49    }
50
51    public Godot.Collections.Dictionary GetHistoryState()
52    {
53        if (_roads is null) return new();
54        var history = _roads.Network.History;
55        var snapshot = _roads.Network.Snapshot;
56        return new()
57        {
58            ["undoCount"] = history.UndoCount, ["redoCount"] = history.RedoCount,
59            ["retainedCount"] = history.RetainedCount, ["estimatedRetainedBytes"] = history.EstimatedBytes,
60            ["sourceToken"] = snapshot.Token.ToString(), ["contentRevision"] = snapshot.Token.ContentRevision,
61            ["changeSequence"] = snapshot.Token.ChangeSequence,
62            ["nextNodeId"] = snapshot.NextNodeId, ["nextEdgeId"] = snapshot.NextEdgeId,
63        };
64    }
65}

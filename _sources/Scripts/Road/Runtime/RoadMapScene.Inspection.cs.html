1using System;
2using System.IO;
3using System.Security.Cryptography;
4using SimpleCities.RoadCore;
5
6public partial class RoadMapScene
7{
8    /// <summary>Exact geometry/entity digest for large-map inspection without exporting native Variant trees.
9    /// Includes map, nodes, edge identities, profiles and binary64 points; excludes tokens and allocation watermarks.</summary>
10    public string GetRoadContentFingerprint()
11    {
12        if (_roads is null) return "";
13        RoadSnapshot snapshot = _roads.Network.Snapshot;
14        using var bytes = new MemoryStream();
15        using var writer = new BinaryWriter(bytes);
16        writer.Write("v4-core-content-v1");
17        writer.Write(snapshot.Map.CellSizeMetres);
18        writer.Write(snapshot.NodeCount);
19        foreach (RoadNode node in snapshot.Nodes)
20        {
21            writer.Write(node.Id.Value);
22            Point(node.Position);
23        }
24        writer.Write(snapshot.EdgeCount);
25        foreach (RoadEdge edge in snapshot.Edges)
26        {
27            writer.Write(edge.Id.Value);
28            writer.Write(edge.Start.Value);
29            writer.Write(edge.End.Value);
30            writer.Write(edge.Profile.Value);
31            writer.Write(edge.Points.Count);
32            foreach (RoadPoint point in edge.Points) Point(point);
33        }
34        writer.Flush();
35        return Convert.ToHexStringLower(SHA256.HashData(bytes.GetBuffer().AsSpan(0, checked((int)bytes.Length))));
36
37        void Point(RoadPoint point)
38        {
39            writer.Write(point.X);
40            writer.Write(point.Y);
41        }
42    }
43}

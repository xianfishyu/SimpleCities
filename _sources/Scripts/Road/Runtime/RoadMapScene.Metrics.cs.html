24    private void ResetOperationMetrics()
25    {
26        _domainMilliseconds = _surfaceMilliseconds = _geometryMilliseconds = _preflightMilliseconds = -1;
27        _referenceMilliseconds = _publicationMilliseconds = _workerQueueMilliseconds = _resumeMilliseconds = -1;
28        _cancelWorkerMilliseconds = _cancelToolMilliseconds = -1;

30        _reusedDisplayContent = false;
31        _changedDisplayPages = 0;

35    /// <summary>Observed stages of the last accepted operation, in milliseconds; -1 means unobserved.
36    /// Reading allocates; baseline collectors read once after completion, never on every render frame.</summary>
37    public Godot.Collections.Dictionary GetOperationMetrics()
38    {
39        long workerExitedAt = Volatile.Read(ref _workerExitedAt);
40        string cancelWorkerState = _cancelRequestedAt == 0 ? "NotRequested" : workerExitedAt == 0 ? "Pending" :
41            workerExitedAt <= _cancelRequestedAt ? "ExitedBeforeRequest" : "ExitedAfterRequest";
42        return new()
43        {
44            ["phase"] = BuildPhase, ["sourceToken"] = BuildSourceToken, ["presentedToken"] = BuildPresentedToken,
45            ["domainMs"] = _domainMilliseconds, ["surfaceMs"] = _surfaceMilliseconds,
46            ["geometryMs"] = _geometryMilliseconds,
47            ["preflightMs"] = _preflightMilliseconds, ["referenceCommitMs"] = _referenceMilliseconds,
48            ["presentationPublishMs"] = _publicationMilliseconds, ["inputToWorkerMs"] = _workerQueueMilliseconds,
49            ["workerToMainMs"] = _resumeMilliseconds, ["inputToDrawMs"] = BuildDrawnElapsedMilliseconds,
50            ["cancelToWorkerExitMs"] = _cancelWorkerMilliseconds, ["cancelToToolReadyMs"] = _cancelToolMilliseconds,
51            ["reusedDisplayContent"] = _reusedDisplayContent,
52            ["changedDisplayPages"] = _changedDisplayPages, ["displayUploadBytes"] = _displayUploadBytes,
53            ["cancelWorkerState"] = cancelWorkerState,
54            ["cancelWorkerWarningMs"] = 10, ["cancelToolWarningMs"] = 100,
55            ["cancelWorkerWarningExceeded"] = _cancelWorkerMilliseconds > 10,
56            ["cancelToolWarningExceeded"] = _cancelToolMilliseconds > 100,
57            ["cancelToolReadyObserved"] = _cancelToolMilliseconds >= 0,
58        };
59    }
60
61    private void CompleteCancellationMetrics()
62    {
63        if (_cancelRequestedAt == 0) return;
64        // A worker may already have exited while its continuation waited on the main thread.
65        _cancelWorkerMilliseconds = _workerExitedAt == 0 ? -1 :
66            System.Math.Max(0, Stopwatch.GetElapsedTime(_cancelRequestedAt, _workerExitedAt).TotalMilliseconds);
67        // The continuation refreshes the actual controls before observing this boundary.
68        // Exiting the scene is cleanup, not a claim that its tools became usable again.
69        _cancelToolMilliseconds = IsSceneAlive && CanEdit && !_toolMode.Disabled && !_profileChoice.Disabled
70            ? Stopwatch.GetElapsedTime(_cancelInputAt).TotalMilliseconds : -1;
71    }

75    public bool RetryRoadDisplay()
76    {
77        if (!CanRetryDisplay) return false;
78        _displayRetryRunning = true;
79        _displayPhase = DisplayPhase.Preparing;
80        StatusText = "正在重新准备道路显示…";

86    private async void PrepareDisplayRetry()
87    {
88        RoadNetwork network = _roads!.Network;
89        RoadSnapshot snapshot = network.Snapshot;
90        long generation = _displayGeneration;
91        using var cancellation = new CancellationTokenSource();
92        _displayRetryCancellation = cancellation;
93        CancellationToken token = cancellation.Token;
94        RoadDisplay? display = null;
95#if DEBUG
96        Action? beforeWork = BeforeDisplayRetryWork;
97#endif
98        bool IsCurrent() => IsSceneAlive && generation == _displayGeneration &&
99            ReferenceEquals(_roads?.Network, network) && ReferenceEquals(network.Snapshot, snapshot);
100        try
101        {
102            RoadDisplay.PreparedGeometry geometry = await Task.Run(() =>
103            {
104#if DEBUG
105                beforeWork?.Invoke();
106#endif
107                token.ThrowIfCancellationRequested();
108                return RoadDisplay.PrepareFull(snapshot, token);
109            });
110            // A load in admission has priority. If it fails, leave the explicit retry available.
111            if (!IsCurrent() || _toolAdmission is not null || token.IsCancellationRequested) return;
112            display = _view.PrepareDisplay(geometry);
113            if (!IsCurrent() || _toolAdmission is not null || token.IsCancellationRequested) return;
114            _view.PublishDisplay(display);
115            display = null;
116            _displayPhase = DisplayPhase.AwaitingDraw;
117        }
118        catch (OperationCanceledException) when (token.IsCancellationRequested) { }
119        catch (Exception exception)
120        {
121            if (IsCurrent()) FailRoadDisplay(exception.Message);
122        }
123        finally
124        {
125            display?.Dispose();
126            _displayRetryRunning = false;
127            if (ReferenceEquals(_displayRetryCancellation, cancellation)) _displayRetryCancellation = null;
128            if (IsCurrent() && _displayPhase == DisplayPhase.Preparing) _displayPhase = DisplayPhase.Failed;
129            if (IsSceneAlive) UpdateDisplayRecoveryControls();
130        }
131    }
132
133    private bool CompleteRoadDisplayFrame()
134    {
135        if (_view.DrawError.Length != 0)
136        {
137            if (_displayPhase is DisplayPhase.Current or DisplayPhase.AwaitingDraw) FailRoadDisplay(_view.DrawError);
138            return false;
139        }
140        if (_displayPhase != DisplayPhase.AwaitingDraw || !IsPresentationCurrent ||
141            _view.DrawSubmittedToken != StateToken) return false;
142        _view.CompleteDisplayDraw();
143        bool recovered = _displayError.Length != 0;
144        _displayPhase = DisplayPhase.Current;
145        _displayError = "";
146        if (recovered) StatusText = "道路显示已恢复";
147        UpdateDisplayRecoveryControls();
148        UpdateHistoryControls();
149        return true;
150    }

37    internal bool CancelRoadInteraction()
38    {
39        // Load admission owns the frozen gesture until it commits or rolls back.
40        // Consume the tool cancel intent without changing that captured state.
41        if (_toolAdmission is not null) return true;
42        long inputAt = Stopwatch.GetTimestamp();
43        bool active = _draftStart.HasValue || _selectionSession.IsSelecting ||
44            _selectionSession.Selected.Count != 0 || _buildOperation.IsBusy;
45        CancelDraft();
46        ClearRoadSelection();
47        if (_buildOperation.TryCancel())
48        {
49            _cancelInputAt = inputAt;
50            _cancelRequestedAt = Stopwatch.GetTimestamp();
51            _buildCancellation?.Cancel();
52            StatusText = "正在取消…";
53        }
54        return active;
55    }

61    internal RoadStatistics? CaptureStatistics()
62    {
63        RoadMapView? view = _view;
64        if (view is null || view.Presented is not RoadSnapshot snapshot) return null;
65        if (_statistics?.Source == snapshot.Token) return _statistics;
66        int segments = 0, loops = 0;
67        foreach (RoadEdge edge in snapshot.Edges)
68        {
69            segments = checked(segments + edge.Points.Count - 1);
70            if (edge.Start == edge.End) loops++;
71        }
72        _presentedNodePositions.Clear();
73        foreach (RoadNode node in snapshot.Nodes) _presentedNodePositions.Add(node.Position);
74        _statistics = new(snapshot.Token, snapshot.NodeCount, snapshot.EdgeCount, segments,
75            view.CurrentDisplay?.QueryFragmentCount ?? 0, loops);
76        return _statistics;
77    }
78
79    internal (Vector2 Point, bool HasNode)? DescribeCursor(Vector2 world)
80    {
81        if (!world.IsFinite() || CaptureStatistics() is null || _view.Presented is not RoadSnapshot snapshot)
82            return null;
83        RoadPoint point = snapshot.Map.SnapBuildPoint(new(world.X, world.Y));
84        return (new((float)point.X, (float)point.Y), _presentedNodePositions.Contains(point));
85    }

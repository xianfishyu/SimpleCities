68    private void UpdateSelectionPointer(Vector2 screen)
69    {
70        RoadSnapshot snapshot = _roads!.Network.Snapshot;
71        Vector2 world = SelectionWorld(screen);
72        var hoverQuery = _view.QuerySpan(world);
73        if (hoverQuery.Status != SpatialQueryStatus.Ready)
74        {
75            RejectSelectionQuery(hoverQuery.Reason);
76            return;
77        }
78        RoadGridSpan? hover = hoverQuery.Results.FirstOrDefault();
79        if (_toolMode.Selected == 3 && hover is not null)
80        {
81            RoadProfileId target = _selectionSession.IsSelecting ? _selectionProfile : RoadProfiles.All[_profileChoice.Selected].Id;
82            if (snapshot.FindEdge(hover.Edge)?.Profile == target) hover = null;
83        }
84        if (!_selectionSession.Hover(snapshot, hover))
85        {
86            _selectionLastWorld = null;
87            _view.ClearSelection();
88            return;
89        }
90        if (_selectionSession.IsSelecting)
91        {
92            var trace = _view.QuerySpans(_selectionLastWorld ?? world, world);
93            if (trace.Status != SpatialQueryStatus.Ready)
94            {
95                RejectSelectionQuery(trace.Reason);
96                return;
97            }
98            var crossed = trace.Results;
99            _selectionSession.Accumulate(snapshot, _toolMode.Selected == 3
100                ? crossed.Where(span => snapshot.FindEdge(span.Edge)?.Profile != _selectionProfile)
101                : crossed);
102            _selectionLastWorld = world;
103            StatusText = $"已选 {_selectionSession.Selected.Count} 个道路格段";
104        }
105        _view.SetSelection(_selectionSession.Hovered, _selectionSession.Selected);
106    }

114    private bool HandleSelectionInput(InputEvent input)
115    {
116        if (input is InputEventMouseMotion motion && IsPointerOverUi(motion.Position))
117        {
118            _selectionLastWorld = null;
119            if (_roads is not null) _selectionSession.Hover(_roads.Network.Snapshot, null);
120            _view.SetSelection(null, _selectionSession.Selected);
121        }
122        if (!IsSelectionTool || input is not InputEventMouseButton { ButtonIndex: MouseButton.Left, Pressed: false } release)
123            return false;
124        if (_selectionSession.IsSelecting)
125        {
126            long inputTimestamp = Stopwatch.GetTimestamp();
127            bool canSubmit = CanEdit && !IsPointerOverUi(release.Position);
128            if (canSubmit)
129                UpdateSelectionPointer(release.Position);
130            if (!_selectionSession.IsSelecting) return true;
131            _selectionSession.End();
132            _selectionLastWorld = null;
133            if (IsSpanEditTool)
134            {
135                RoadGridSpan[] spans = canSubmit ? _selectionSession.Selected.ToArray() : [];
136                int mode = _toolMode.Selected;
137                if (spans.Length != 0) SubmitSpanEdit(spans, mode, _selectionProfile, inputTimestamp);
138                else ClearRoadSelection();
139                return true;
140            }
141            StatusText = $"已选 {_selectionSession.Selected.Count} 个道路格段 · Esc 清除";
142        }
143        return true;
144    }
145
146    private bool HandleSelectionPointerEvent(InputEvent input)
147    {
148        if (!IsSelectionTool || !CanEdit) return false;
149        if (input is InputEventMouseButton { ButtonIndex: MouseButton.Left, Pressed: true } press)
150        {
151            _selectionSession.Begin(_roads!.Network.Snapshot);
152            _selectionProfile = RoadProfiles.All[_profileChoice.Selected].Id;
153            _selectionLastWorld = null;
154            UpdateSelectionPointer(press.Position);
155            return true;
156        }
157        if (input is InputEventMouseMotion motion) UpdateSelectionPointer(motion.Position);
158        return false; // Camera wheel/pan still flows through the shared input handler.
159    }

161    private void SubmitSpanEdit(RoadGridSpan[] spans, int mode, RoadProfileId profile, long inputTimestamp) =>
162        SubmitRoadOperation((network, token) =>
163        {
164            RoadEditResult result = mode == 2 ? network.PlanRemove(spans, token) : network.PlanChangeProfile(spans, profile, token);
165            return new PlannedRoadOperation(result.Plan, result.Status == RoadEditStatus.NoChange, result.Reason);
166        }, inputTimestamp, mode == 2 ? "格段已删除" : "格段已改造", "选择道路格段");

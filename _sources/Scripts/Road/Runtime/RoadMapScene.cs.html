122    public bool CreateMap(int cellSizeMetres)
123    {
124        if (_saveManager.IsOperationBusy || _pendingOperation.Length != 0 || _toolAdmission is not null || _buildOperation.IsBusy ||
125            _displayPhase == DisplayPhase.AwaitingDraw)
126            return false;
127        // Validate before unregistering the current scene or replacing its immutable map.
128        var replacement = new RoadSaveParticipant(new RoadNetwork(new MapDefinition(cellSizeMetres)));
129        CancelDraft();
130        ClearRoadSelection();
131        if (_roads is not null)
132        {
133            _saveManager.UnregisterSceneLoad(this);
134            _saveManager.Unregister(_roads);
135        }
136        _roads = replacement;
137        if (!_saveManager.Register(_roads) || !_saveManager.RegisterSceneLoad(
138                new SceneLoadParticipants(_roads, this, _view, RoadSaveParticipant.Storage)))
139            throw new InvalidOperationException("Cannot register the V4 scene.");
140        _view.ShowNewMap(_roads.Network.Snapshot);
141        ResetDisplayRecoveryReferences();
142        _displayRetryCancellation?.Cancel();
143        StatusText = "已创建空地图";
144        UpdateMapInfo();
145        UpdateHistoryControls();
146        return true;
147    }

259    // At most one preview worker plus one replaceable latest request. Superseded work cannot publish.
260    private async void RunPreviewWorker()
261    {
262        if (_previewWorkerRunning) return;
263        _previewWorkerRunning = true;
264        try
265        {
266            while (IsSceneAlive && _queuedPreview is CoreRoadBuildRequest request)
267            {
268                _queuedPreview = null;
269                RoadNetwork network = _roads!.Network;
270                using var cancellation = new CancellationTokenSource();
271                _previewCancellation = cancellation;
272                CancellationToken token = cancellation.Token;
273#if DEBUG
274                Action? beforeWork = BeforePreviewWork;
275#endif
276                try
277                {
278                    CoreRoadBuildResult result = await Task.Run(() =>
279                    {
280#if DEBUG
281                        beforeWork?.Invoke();
282#endif
283                        return network.PlanBuild(request, token);
284                    });
285                    if (token.IsCancellationRequested) continue;
286                    CompletePreview(request, result, result.Status switch
287                    {
288                        RoadBuildStatus.Ready => "松开以建造道路",
289                        RoadBuildStatus.NoChange => "拖动以预览道路",
290                        _ => result.Reason,
291                    });
292                }
293                catch (OperationCanceledException) when (token.IsCancellationRequested) { }
294                catch (Exception exception)
295                {
296                    CompletePreview(request, new CoreRoadBuildResult(RoadBuildStatus.Rejected, null, "建造预览检查失败"),
297                        $"建造预览检查失败：{exception.Message}");
298                }
299                finally
300                {
301                    if (ReferenceEquals(_previewCancellation, cancellation)) _previewCancellation = null;
302                }
303            }
304        }
305        finally { _previewWorkerRunning = false; }
306    }

309    {
310        if (!IsSceneAlive || request != _previewRequest || !_draftStart.HasValue ||

328    public override void _Input(InputEvent @event)
329    {
330        // Load admission preserves the captured tool state until the aggregate commits or abandons it.
331        // Camera input still flows through _UnhandledInput while these road gestures are suspended.
332        if (_toolAdmission is not null) return;
333        if (!UseProductControls && @event is InputEventKey { Pressed: true, Keycode: Key.Escape })
334        {
335            CancelRoadInteraction();
336        }
337        if (HandleSelectionInput(@event)) return;
338        if (@event is InputEventMouseButton { ButtonIndex: MouseButton.Left, Pressed: false } release && _draftStart is RoadPoint start)
339        {
340            long inputTimestamp = Stopwatch.GetTimestamp();
341            RoadPoint end = _roads!.Network.Snapshot.Map.SnapDragEnd(start, WorldPoint(release.Position));
342            var request = new CoreRoadBuildRequest(_draftSource, start, end, _draftProfile);
343            CancelDraft();
344            // Release over UI ends the captured gesture without constructing behind a control.
345            if (CanEdit && !IsPointerOverUi(release.Position))
346                SubmitBuild(request, inputTimestamp);
347        }
348    }

350    public override void _UnhandledInput(InputEvent @event)
351    {
352        if (!UseProductControls && HandleHistoryInput(@event)) return;
353        if (HandleSelectionPointerEvent(@event)) return;
354        if (@event is InputEventMouseButton { ButtonIndex: MouseButton.Left, Pressed: true } press && CanEdit && !IsSelectionTool)
355        {
356            MapDefinition map = _roads!.Network.Snapshot.Map;
357            RoadPoint cursor = WorldPoint(press.Position);
358            if (cursor.X < -4000 || cursor.X > 4000 || cursor.Y < -4000 || cursor.Y > 4000) return;
359            _draftStart = map.SnapBuildPoint(cursor);
360            _draftSource = _roads.Network.Snapshot.Token;
361            _draftProfile = RoadProfiles.All[_profileChoice.Selected].Id;
362            ClearRoadSelection();
363            UpdateDraftPreview(_draftStart.Value, _draftStart.Value);
364        }
365        if (@event is InputEventMouseMotion move && _draftStart is RoadPoint draft && CanEdit)
366        {
367            RoadPoint end = _roads!.Network.Snapshot.Map.SnapDragEnd(draft, WorldPoint(move.Position));
368            UpdateDraftPreview(draft, end);
369        }
370        else if (@event is InputEventMouseMotion hover && CanEdit && !IsSelectionTool)
371        {
372            RoadPoint point = WorldPoint(hover.Position);
373            var hits = _view.QueryHit(new Vector2((float)point.X, (float)point.Y), null);
374            UpdateSelectionPointer(hover.Position);
375            if (hits.Status == SpatialQueryStatus.Ready && hits.Results.Count != 0)
376            {
377                RoadDisplay.HitResult hit = hits.Results[0];
378                if (hit.JunctionNode is NodeId junctionId && junctionId.Value > 0 &&
379                    RoadJunctionQuery.Read(_roads!.Network.Snapshot, junctionId) is RoadJunctionReadModel junction)
380                    StatusText = $"{(_roads.Network.Snapshot.Map.IsCellCenter(junction.Node.Position) ? "格心路口" : "路口")} · {junction.Incidences.Count} 个方向";
381                else StatusText = $"道路位置：{hit.Location.Parameter:P0}";
382            }
383        }
384        if (UseProductControls) return; // MainCamera owns the product's camera input.
385        if (@event is InputEventMouseButton { Pressed: true } button &&
386            button.ButtonIndex is MouseButton.WheelUp or MouseButton.WheelDown)
387        {
388            float zoom = Mathf.Clamp(_camera.Zoom.X * (button.ButtonIndex == MouseButton.WheelUp ? 1.15f : 1 / 1.15f), 0.04f, 2f);
389            _camera.Zoom = new Vector2(zoom, zoom);
390        }
391        if (@event is InputEventMouseMotion motion && motion.ButtonMask.HasFlag(MouseButtonMask.Middle))
392            _camera.Position -= motion.Relative / _camera.Zoom;
393    }

461    private static Godot.Collections.Array<Godot.Collections.Dictionary> DescribeJunctions(RoadSnapshot snapshot)
462    {
463        var junctions = new Godot.Collections.Array<Godot.Collections.Dictionary>();
464        foreach (RoadNode node in snapshot.Nodes)
465        {
466            RoadJunctionReadModel? junction = RoadJunctionQuery.Read(snapshot, node.Id);
467            if (junction is not null && junction.Incidences.Count >= 3) junctions.Add(DescribeJunction(junction));
468        }
469        return junctions;
470    }

472    private static Godot.Collections.Dictionary DescribeJunction(RoadJunctionReadModel junction)
473    {
474        var incidences = new Godot.Collections.Array<Godot.Collections.Dictionary>();
475        foreach (RoadIncidence incidence in junction.Incidences)
476            incidences.Add(new()
477            {
478                ["edgeId"] = incidence.Key.Edge.Value, ["role"] = incidence.Key.Role.ToString(),
479                ["profile"] = incidence.Profile.Value,
480                ["outward"] = new Vector2((float)incidence.Outward.X, (float)incidence.Outward.Y),
481                ["bearingRadians"] = incidence.BearingRadians,
482            });
483        var turns = new Godot.Collections.Array<Godot.Collections.Dictionary>();
484        foreach (RoadTurnMovement turn in junction.Turns)
485            turns.Add(new()
486            {
487                ["fromEdgeId"] = turn.From.Edge.Value, ["fromRole"] = turn.From.Role.ToString(),
488                ["toEdgeId"] = turn.To.Edge.Value, ["toRole"] = turn.To.Role.ToString(),
489                ["signedAngleRadians"] = turn.SignedAngleRadians,
490            });
491        return new()
492        {
493            ["nodeId"] = junction.Node.Id.Value,
494            ["position"] = new Vector2((float)junction.Node.Position.X, (float)junction.Node.Position.Y),
495            ["sourceToken"] = junction.Source.ToString(), ["incidences"] = incidences, ["turns"] = turns,
496        };
497    }

499    private static Godot.Collections.Array<Godot.Collections.Dictionary> DescribeEdges(RoadSnapshot snapshot)
500    {
501        var edges = new Godot.Collections.Array<Godot.Collections.Dictionary>();
502        foreach (RoadEdge edge in snapshot.Edges)
503        {
504            var points = new Godot.Collections.Array<Vector2>();
505            foreach (RoadPoint point in edge.Points)
506                points.Add(new Vector2((float)point.X, (float)point.Y));
507            edges.Add(new()
508            {
509                ["edgeId"] = edge.Id.Value,
510                ["startNodeId"] = edge.Start.Value,
511                ["endNodeId"] = edge.End.Value,
512                ["profile"] = edge.Profile.Value,
513                ["points"] = points,
514            });
515        }
516        return edges;
517    }

528    private async void SubmitRoadOperation(Func<RoadNetwork, CancellationToken, PlannedRoadOperation> prepare,
529        long inputTimestamp, string completedText, string noChangeText)
530    {
531        RoadNetwork network = _roads!.Network;
532        if (!_buildOperation.TryBegin(inputTimestamp)) return;
533        ResetOperationMetrics();
534        using var cancellation = new CancellationTokenSource();
535        _buildCancellation = cancellation;
536        CancellationToken token = cancellation.Token;
537        _buildSourceToken = network.Snapshot.Token.ToString();
538        _buildPresentedToken = "";
539        _operationCompletedText = completedText;
540        StatusText = "正在准备道路… Esc 取消";
541        bool committed = false;
542        RoadDisplay? display = null;
543        RoadDisplay? reusable = _view.ReusableDisplay;
544        RoadDisplay? basis = _view.CurrentDisplay;
545#if DEBUG
546        Action? beforeWork = BeforeBuildWork;
547#endif
548        try
549        {
550            (PlannedRoadOperation Result, RoadDisplay.PreparedGeometry? Geometry) prepared = await Task.Run(() =>
551            {
552                long workerStarted = Stopwatch.GetTimestamp();
553                _workerQueueMilliseconds = Stopwatch.GetElapsedTime(inputTimestamp, workerStarted).TotalMilliseconds;
554                try
555                {
556#if DEBUG
557                    beforeWork?.Invoke();
558#endif
559                    token.ThrowIfCancellationRequested();
560                    long domainStarted = Stopwatch.GetTimestamp();
561                    PlannedRoadOperation result = prepare(network, token);
562                    _domainMilliseconds = Stopwatch.GetElapsedTime(domainStarted).TotalMilliseconds;
563                    long geometryStarted = Stopwatch.GetTimestamp();
564                    RoadDisplay.PreparedGeometry? geometry = result.Plan is RoadPlan reusePlan
565                        ? reusable?.TryPrepareReuse(reusePlan.Target, token) : null;
566                    _reusedDisplayContent = geometry is not null;
567                    geometry ??= result.Plan is RoadPlan displayPlan
568                        ? RoadDisplay.PrepareChange(basis ?? throw new InvalidOperationException("Missing V4 source display."), displayPlan, token) : null;
569                    _surfaceMilliseconds = geometry?.SurfacePreparationMilliseconds ?? 0;
570                    _geometryMilliseconds = Stopwatch.GetElapsedTime(geometryStarted).TotalMilliseconds - _surfaceMilliseconds;
571                    _changedDisplayPages = geometry?.ChangedPageCount ?? 0;
572                    _displayUploadBytes = geometry?.UploadBytes ?? 0;
573                    token.ThrowIfCancellationRequested();
574                    return (result, geometry);
575                }
576                finally { Volatile.Write(ref _workerExitedAt, Stopwatch.GetTimestamp()); }
577            });
578            _resumeMilliseconds = Stopwatch.GetElapsedTime(_workerExitedAt).TotalMilliseconds;
579            if (!IsSceneAlive || !_buildOperation.CanPublish)
580            {
581                if (IsSceneAlive) StatusText = "已取消";
582                return;
583            }
584            if (prepared.Result.Plan is not RoadPlan plan)
585            {
586                StatusText = prepared.Result.NoChange ? noChangeText : prepared.Result.Reason;
587                return;
588            }
589            long preflightStarted = Stopwatch.GetTimestamp();
590            try { display = _view.PrepareDisplay(prepared.Geometry ?? throw new InvalidOperationException("Missing prepared display geometry.")); }
591            finally { _preflightMilliseconds = Stopwatch.GetElapsedTime(preflightStarted).TotalMilliseconds; }
592            if (!network.CanCommit(plan) || !_buildOperation.CanPublish) return;
593            // No await or callbacks between the validated reference publications.
594            long referenceStarted = Stopwatch.GetTimestamp();
595            committed = _buildOperation.TryCommit(() => network.TryCommit(plan));
596            _referenceMilliseconds = Stopwatch.GetElapsedTime(referenceStarted).TotalMilliseconds;
597            if (!committed) return;
598            long publishStarted = Stopwatch.GetTimestamp();
599            try { _view.PublishDisplay(display); }
600            finally { _publicationMilliseconds = Stopwatch.GetElapsedTime(publishStarted).TotalMilliseconds; }
601            display = null;
602            _buildPresentedToken = plan.Target.Token.ToString();
603            _displayPhase = DisplayPhase.AwaitingDraw;
604            StatusText = "道路已提交，正在更新显示…";
605            UpdateMapInfo();
606        }
607        catch (OperationCanceledException) when (token.IsCancellationRequested)
608        {
609            if (IsSceneAlive) StatusText = "已取消";
610        }
611        catch (Exception exception)
612        {
613            if (IsSceneAlive)
614            {
615                if (committed) FailRoadDisplay(exception.Message);
616                else StatusText = $"道路操作未完成：{exception.Message}";
617            }
618        }
619        finally
620        {
621            display?.Dispose();
622            if (IsSceneAlive) ClearRoadSelection();
623            if (ReferenceEquals(_buildCancellation, cancellation)) _buildCancellation = null;
624            if (!committed || !IsSceneAlive || _buildPresentedToken.Length == 0)
625                _buildOperation.Finish(committed ? "DisplayFailed" : "Finished");
626            if (IsSceneAlive) UpdateOperationControls();
627            CompleteCancellationMetrics();
628        }
629    }

631    private void OnFramePostDraw()
632    {
633        if (!CompleteRoadDisplayFrame()) return;
634        // This is the first completed render after the committed display was published.
635        // Input remains gated until this point; no new write can replace the measured target.
636        if (_buildOperation.Phase != "Committed" || !IsPresentationCurrent ||
637            _view.Presented?.Token.ToString() != _buildPresentedToken ||
638            _view.DrawSubmittedToken != _buildPresentedToken)
639            return;
640        if (_buildOperation.RecordFirstDraw())
641        {
642            StatusText = $"{_operationCompletedText} · {_buildOperation.DrawnElapsedMilliseconds:F1} ms";
643            UpdateHistoryControls();
644        }
645    }

649    private sealed class ToolAdmission(RoadMapScene owner, long generation) : ISceneToolLoadAdmission, INonThrowingLoadCommitPlan
650    {
651        private readonly RoadSpanSelectionSession _emptySelection = new();
652        private CancellationTokenSource? _previewToCancel;
653        private CancellationTokenSource? _retryToCancel;
654        public string ParticipantID => "v4-tools";
655        public bool IsGenerationCurrent => ReferenceEquals(owner._toolAdmission, this) &&
656            owner._saveManager.SceneGeneration == generation;
657        public INonThrowingLoadCommitPlan PreflightFullReset()
658        {
659            if (!IsGenerationCurrent)
660                throw new LoadPreflightInvalidException("V4 tool admission is stale.");
661            return this;
662        }
663        public void CommitReferences()
664        {
665            owner.ResetDisplayRecoveryReferences();
666            owner._displayPhase = DisplayPhase.AwaitingDraw;
667            _retryToCancel = owner._displayRetryCancellation;
668            owner._draftStart = null;
669            owner._draftSource = default;
670            owner._draftProfile = default;
671            owner._previewRequest = null;
672            owner._queuedPreview = null;
673            owner._previewResult = null;
674            owner._completedPreview = null;
675            _previewToCancel = owner._previewCancellation;
676            owner._previewCancellation = null;
677            owner._selectionSession = _emptySelection;
678            owner._selectionLastWorld = null;
679            owner._selectionProfile = default;
680        }
681        public IReadOnlyList<string> PublishNotifications()
682        {
683            owner._view.QueueRedraw();
684            return Array.Empty<string>();
685        }
686        public void CompleteCommit()
687        {
688            // Callback-capable cancellation runs after all reference swaps; stale results are already detached.
689            try
690            {
691                _previewToCancel?.Cancel();
692                _retryToCancel?.Cancel();
693            }
694            finally
695            {
696                _previewToCancel = null;
697                _retryToCancel = null;
698                Dispose();
699            }
700        }
701        public void Dispose()
702        {
703            if (ReferenceEquals(owner._toolAdmission, this))
704                owner._toolAdmission = null;
705        }
706    }

35    private string FormatDrawToken(RoadSnapshot? snapshot)
36    {
37        if (snapshot is null) return "";
38        if (_formattedDrawSource != snapshot.Token)
39        {

67    internal void PublishDisplay(RoadDisplay target)
68    {
69        if (_awaitingDraw) throw new InvalidOperationException("A V4 display is already awaiting its first draw.");
70#if DEBUG
71        BeforeDisplayPublish?.Invoke();
72#endif
73        // Schedule before taking ownership; everything after this is a prepared reference swap.
74        QueueRedraw();
75        _previousDisplay = CommitDisplay(target);
76        _awaitingDraw = true;
77    }

79    internal void CompleteDisplayDraw()
80    {
81        if (!_awaitingDraw || DrawSubmittedToken != FormatDrawToken(Presented) || DrawError.Length != 0) return;
82        _awaitingDraw = false;
83        RoadDisplay? previous = _previousDisplay;
84        _previousDisplay = null;
85        _cachedDisplay?.Dispose();
86        _cachedDisplay = null;
87        // Retain just the preceding successful picture, not a history of derived data.
88        // A load/new lineage cannot use it and must release the old map's resources.
89        if (previous is not null && previous.Snapshot.Token.NetworkInstance == Presented?.Token.NetworkInstance &&
90            previous.Snapshot.Token.Lineage == Presented?.Token.Lineage)
91            _cachedDisplay = previous;
92        else previous?.Dispose();
93    }
94    internal RoadDisplay? CommitDisplay(RoadDisplay target)
95    {
96        RoadDisplay? previous = _display;

99        _drawingUnavailable = false;
100        _preview = [];
101        _selection.Clear();
102        return previous;
103    }

120    internal void ShowPreview(RoadPoint? start, RoadPoint? end, SimpleCities.RoadCore.RoadBuildResult? result, RoadProfileId profile)
121    {
122        var segments = new List<PreviewSegment>();
123        if (start is RoadPoint a && end is RoadPoint b)
124        {
125            var neutral = new Color("9aa5ad");
126            if (result is null || result.Conflicts.Count == 0)
127                Add(0, 1, result?.Status == RoadBuildStatus.Ready ? new Color("75dfcb") : neutral, false);
128            else
129            {
130                double cursor = 0;
131                foreach (RoadConflictSpan conflict in result.Conflicts)
132                {
133                    if (cursor < conflict.StartParameter) Add(cursor, conflict.StartParameter, neutral, false);
134                    Add(conflict.StartParameter, conflict.EndParameter, new Color("ef6f76"), true);
135                    cursor = conflict.EndParameter;
136                }
137                if (cursor < 1) Add(cursor, 1, neutral, false);
138            }
139
140            void Add(double from, double to, Color color, bool conflict)
141            {
142                Vector2 Point(double t) => new((float)(a.X + (b.X - a.X) * t), (float)(a.Y + (b.Y - a.Y) * t));
143                segments.Add(new PreviewSegment(Point(from), Point(to), color, conflict, (float)RoadProfiles.Get(profile).DisplayWidth));
144            }
145        }
146        _preview = segments.ToArray();
147        QueueRedraw();
148    }

158    internal Godot.Collections.Dictionary PickRoad(Vector2 world)
159    {
160        var query = QueryHit(world, null);
161        if (query.Status != SpatialQueryStatus.Ready)
162            return new() { ["queryStatus"] = query.Status.ToString(), ["reason"] = query.Reason };
163        return query.Results.Count == 0 ? new() : DescribeHit(query.Results[0]);

191    public override void _Draw()
192    {
193        if (Presented is null)
194            return;
195        var bounds = new Rect2(-4000, -4000, 8000, 8000);
196        DrawRect(bounds, new Color(0.35f, 0.35f, 0.35f), filled: false, width: 3);
197        if (_drawingUnavailable) return;
198        try
199        {
200#if DEBUG
201            BeforeDisplayDraw?.Invoke();
202#endif
203            _display?.Draw(this);
204            _selection.Draw(this);
205            foreach (PreviewSegment segment in _preview)
206                DrawLine(segment.Start, segment.End, segment.Color, segment.Width);
207            DrawSubmittedToken = FormatDrawToken(Presented);
208        }
209        catch (Exception exception)
210        {
211            DrawError = exception.Message;
212            DrawSubmittedToken = "";
213            _preview = [];
214            _selection.Clear();
215            if (_awaitingDraw)
216            {
217                RoadDisplay? failed = _display;
218                _display = _previousDisplay;
219                _previousDisplay = null;
220                _awaitingDraw = false;
221                failed?.Dispose();
222                // Replace any commands already emitted by the failed draw in this same frame.
223                RenderingServer.CanvasItemClear(GetCanvasItem());
224                try
225                {
226                    DrawRect(bounds, new Color(0.35f, 0.35f, 0.35f), filled: false, width: 3);
227                    _display?.Draw(this);
228                    DrawSubmittedToken = FormatDrawToken(Presented);
229                }
230                catch (Exception fallbackException)
231                {
232                    // A lost device may also make the retained resources unusable. Do not retry every frame.
233                    DrawError = fallbackException.Message;
234                    _drawingUnavailable = true;
235                    RenderingServer.CanvasItemClear(GetCanvasItem());
236                }
237            }
238            else _drawingUnavailable = true;
239        }

250    private sealed class Admission(RoadMapView owner, long generation) : IScenePresentationLoadAdmission
251    {
252        internal bool IsCurrent => ReferenceEquals(owner._admission, this) && owner._sceneGeneration == generation;
253        public IScenePresentationPreparer Preparer { get; } = new RoadLoadPresentationPreparer();
254        public INonThrowingLoadCommitPlan PreflightPreparedLoad(IPreparedScenePresentation presentation, IPreparedSaveState targetState)
255        {
256            if (!IsCurrent || presentation is not PreparedRoadPresentation prepared ||
257                targetState is not RoadSaveParticipant.Target target || !ReferenceEquals(prepared.Target, target.Value))
258                throw new LoadPreflightInvalidException("V4 presentation does not match the target map.");
259            return new CommitPlan(owner, this, prepared.Geometry.Upload());
260        }
261
262        public void Dispose()
263        {
264            if (ReferenceEquals(owner._admission, this))
265                owner._admission = null;
266        }
267    }

269    private sealed class CommitPlan(RoadMapView owner, Admission admission, RoadDisplay target) : INonThrowingLoadCommitPlan
270    {
271        private RoadDisplay? _previous;
272        private bool _committed;
273        private bool _disposed;
274        public string ParticipantID => "v4-presentation";
275        public bool IsGenerationCurrent => admission.IsCurrent;
276        public void CommitReferences()
277        {
278            _previous = owner.CommitDisplay(target);
279            owner._previousDisplay = _previous;
280            owner._awaitingDraw = true;
281            _committed = true;
282        }
283        public IReadOnlyList<string> PublishNotifications()
284        {
285            owner.QueueRedraw();
286            return Array.Empty<string>();
287        }
288        public void CompleteCommit()
289        {
290            // The view retains the old complete picture until the new one has actually drawn.
291            _previous = null;
292            admission.Dispose();
293        }
294        public void Dispose()
295        {
296            if (_disposed) return;
297            _disposed = true;
298            if (!_committed) target.Dispose();
299            if (!_committed) _previous?.Dispose();
300            _previous = null;
301            admission.Dispose();
302        }
303    }

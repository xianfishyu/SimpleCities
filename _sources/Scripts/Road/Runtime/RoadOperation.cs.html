1using System;
2using System.Diagnostics;
3
4/// <summary>
5/// Main-thread arbitration for one road write. Cancellation revokes publication immediately;
6/// ownership remains held until the worker and prepared resources have been cleaned up.
7/// </summary>
8public sealed class RoadOperation
9{
10    private long _startedAt;
11    private double _finishedElapsed;
12    public string Phase { get; private set; } = "Idle";
13    public bool IsBusy { get; private set; }
14    public bool CanPublish => IsBusy && Phase == "Preparing";
15    public bool IsWaiting => CanPublish && ElapsedMilliseconds > 300;
16    public double ElapsedMilliseconds => IsBusy ? Stopwatch.GetElapsedTime(_startedAt).TotalMilliseconds : _finishedElapsed;
17    public double DrawnElapsedMilliseconds { get; private set; } = -1;
18
19    public bool TryBegin(long? inputTimestamp = null)
20    {
21        if (IsBusy) return false;
22        IsBusy = true;
23        Phase = "Preparing";
24        DrawnElapsedMilliseconds = -1;
25        _startedAt = inputTimestamp ?? Stopwatch.GetTimestamp();
26        return true;
27    }
28
29    public bool TryCancel()
30    {
31        if (!CanPublish) return false;
32        Phase = "Cancelling";
33        return true;
34    }
35
36    public bool TryCommit(Func<bool> publish)
37    {
38        if (!CanPublish) return false;
39        if (!publish()) return false;
40        Phase = "Committed";
41        return true;
42    }
43
44    public bool RecordFirstDraw()
45    {
46        if (!IsBusy || Phase != "Committed") return false;
47        DrawnElapsedMilliseconds = ElapsedMilliseconds;
48        Finish("Drawn");
49        return true;
50    }
51
52    public void Finish(string outcome = "Finished")
53    {
54        if (!IsBusy) return;
55        _finishedElapsed = ElapsedMilliseconds;
56        Phase = Phase == "Cancelling" ? "Cancelled" : outcome;
57        IsBusy = false;
58    }
59}

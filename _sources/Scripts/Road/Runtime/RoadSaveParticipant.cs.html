1using System;
2using System.Collections.Generic;
3using System.IO;
4using SimpleCities.RoadCore;
5
6/// <summary>将 V4 核心内容接到通用槽位事务及联合加载；核心不引用应用契约。</summary>
7internal sealed class RoadSaveParticipant(RoadNetwork network) : ISceneNetworkLoadParticipant
8{
9    internal static SceneStoragePolicy Storage { get; } = new("user://saves-v4", ["road_network_v4"]);
10    private Admission? _admission;
11    internal RoadNetwork Network { get; } = network;
12    public string SaveFileName => "road_network_v4";
13    public ISaveSnapshot CaptureSnapshot() => new Snapshot(Network.Snapshot);
14    public void WriteSnapshot(Stream destination, ISaveSnapshot snapshot) =>
15        RoadCodec.Write(destination, ((Snapshot)snapshot).Value);
16    public IStreamingLoadReader CaptureLoadReader() => new Reader(Network);
17
18    public void CommitPreparedLoad(IPreparedSaveState preparedState)
19    {
20        using ISceneNetworkLoadAdmission admission = BeginSceneLoadAdmission();
21        using var aggregate = new PreparedAggregateLoad([admission.PreflightPreparedLoad(preparedState, out _)]);
22        aggregate.Commit(new UncoordinatedStorageOperationLease(SaveOperationKind.Load));
23    }
24
25    public ISceneNetworkLoadAdmission BeginSceneLoadAdmission()
26    {
27        if (_admission is not null)
28            throw new InvalidOperationException("V4 network is already loading.");
29        return _admission = new Admission(this);
30    }
31
32    internal sealed record Prepared(PreparedRoadState Value, RoadPlan Plan) : IPreparedSaveState;
33    internal sealed record Target(RoadSnapshot Value) : IPreparedSaveState;
34    private sealed record Snapshot(RoadSnapshot Value) : ISaveSnapshot;
35    private sealed class Reader(RoadNetwork network) : IStreamingLoadReader
36    {
37        public IPreparedSaveState PrepareLoad(Stream source)
38        {
39            PreparedRoadState content = RoadCodec.Read(source);
40            // Plan against the immutable published state on the load worker. Publication stays on the main thread.
41            return new Prepared(content, network.PlanLoad(content));
42        }
43    }
44
45    private sealed class Admission(RoadSaveParticipant owner) : ISceneNetworkLoadAdmission
46    {
47        internal bool IsCurrent => ReferenceEquals(owner._admission, this);
48        public INonThrowingLoadCommitPlan PreflightPreparedLoad(IPreparedSaveState state, out IPreparedSaveState targetState)
49        {
50            if (!IsCurrent || state is not Prepared prepared || !owner.Network.CanCommit(prepared.Plan))
51                throw new LoadPreflightInvalidException("Invalid V4 network load admission or payload.");
52            RoadPlan plan = prepared.Plan;
53            targetState = new Target(plan.Target);
54            return new CommitPlan(owner, this, plan);
55        }
56
57        public void Dispose()
58        {
59            if (IsCurrent)
60                owner._admission = null;
61        }
62    }
63
64    private sealed class CommitPlan(RoadSaveParticipant owner, Admission admission, RoadPlan plan)
65        : INonThrowingLoadCommitPlan
66    {
67        public string ParticipantID => "v4-network";
68        public bool IsGenerationCurrent => admission.IsCurrent && owner.Network.CanCommit(plan);
69        // Aggregate validates all plans on this same thread immediately before these reference swaps.
70        public void CommitReferences() => owner.Network.TryCommit(plan);
71        public IReadOnlyList<string> PublishNotifications() => Array.Empty<string>();
72        public void CompleteCommit() => admission.Dispose();
73        public void Dispose() => admission.Dispose();
74    }
75}

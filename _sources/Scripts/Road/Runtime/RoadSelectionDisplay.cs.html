1using Godot;
2using System;
3using System.Collections.Generic;
4using System.Linq;
5using SimpleCities.RoadCore;
6
7/// <summary>按实际道路格段缓存显示坐标；选择改变时转换，绘制时复用。</summary>
8internal sealed class RoadSelectionDisplay
9{
10    private sealed record Stroke(RoadGridSpan Span, Vector2[] Points, Color Color, float Width, bool Selected);
11    private Stroke[] _strokes = [];
12    private RoadStateToken? _source;
13    private RoadGridSpan? _hover;
14    private RoadGridSpan[] _selected = [];
15
16    internal void Set(RoadSnapshot snapshot, RoadGridSpan? hover, IReadOnlyList<RoadGridSpan> selected)
17    {
18        if (_source == snapshot.Token && SameOptional(_hover, hover) && _selected.Length == selected.Count &&
19            _selected.Where((span, index) => !RoadDisplay.SameSpan(span, selected[index])).Any() == false)
20            return;
21        var strokes = new List<Stroke>();
22        foreach (RoadGridSpan span in selected) Add(span, true);
23        if (hover is not null && !selected.Any(span => RoadDisplay.SameSpan(span, hover))) Add(hover, false);
24        _strokes = strokes.ToArray();
25        _source = snapshot.Token;
26        _hover = hover;
27        _selected = selected.ToArray();
28
29        void Add(RoadGridSpan span, bool isSelected)
30        {
31            if (span.Source != snapshot.Token) return;
32            RoadEdge? edge = snapshot.FindEdge(span.Edge);
33            if (edge is null) return;
34            Vector2[] points = span.Points.Select(point => new Vector2((float)point.X, (float)point.Y)).ToArray();
35            if (points.Length < 2 || points.Any(point => !point.IsFinite()))
36                throw new InvalidOperationException("V4 selection display coordinates are invalid.");
37            float width = (float)RoadProfiles.Get(edge.Profile).DisplayWidth;
38            strokes.Add(new Stroke(span, points,
39                isSelected ? new Color(1f, 0.78f, 0.25f, 0.68f) : new Color(0.35f, 0.93f, 1f, 0.58f),
40                width * (isSelected ? 0.8f : 0.45f), isSelected));
41        }
42    }
43
44    private static bool SameOptional(RoadGridSpan? left, RoadGridSpan? right) =>
45        left is null ? right is null : right is not null && RoadDisplay.SameSpan(left, right);
46
47    internal void Clear()
48    {
49        _strokes = [];
50        _source = null;
51        _hover = null;
52        _selected = [];
53    }
54
55    internal void Draw(Node2D canvas)
56    {
57        foreach (Stroke stroke in _strokes)
58            canvas.DrawPolyline(stroke.Points, stroke.Color, stroke.Width, antialiased: true);
59    }
60
61    internal Godot.Collections.Array<Godot.Collections.Dictionary> Describe()
62    {
63        var strokes = new Godot.Collections.Array<Godot.Collections.Dictionary>();
64        foreach (Stroke stroke in _strokes)
65        {
66            var ranges = new Godot.Collections.Array<Godot.Collections.Dictionary>();
67            foreach (var range in stroke.Span.Ranges)
68                ranges.Add(new() { ["startParameter"] = range.StartParameter, ["endParameter"] = range.EndParameter });
69            strokes.Add(new()
70            {
71                ["sourceToken"] = stroke.Span.Source.ToString(),
72                ["edgeId"] = stroke.Span.Edge.Value,
73                ["ranges"] = ranges,
74                ["points"] = stroke.Points,
75                ["color"] = stroke.Color,
76                ["width"] = stroke.Width,
77                ["selected"] = stroke.Selected,
78            });
79        }
80        return strokes;
81    }
82}

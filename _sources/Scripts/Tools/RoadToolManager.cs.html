1using Godot;

3/// <summary>正式 HUD 的 V4 工具入口；路网、历史和加载准入均由同一个 RoadMapScene 持有。</summary>
4public partial class RoadToolManager : Node2D
5{
6    [Export] public RoadConfig Config { get; set; } = null!;
7    public static RoadToolManager Instance { get; private set; } = null!;
8    private RoadMapScene _runtime = null!;
9
10    public override void _EnterTree()
11    {
12        Instance = this;
13        _runtime = GetParent<RoadMapScene>();
14    }
15
16    public override void _Ready()
17    {
18        Callable.From(() => CurrentTool = ToolType.Select).CallDeferred();
19    }
20
21    public override void _Process(double delta)
22    {
23        if (_runtime.PresentedCellSizeMetres > 0) Config.CellSize = _runtime.PresentedCellSizeMetres;
24    }
25
26    public override void _UnhandledInput(InputEvent input)
27    {
28        if (input is InputEventKey { Pressed: true, Echo: false, Keycode: Key.Escape } && CancelRoadSessions())
29            GetViewport().SetInputAsHandled();
30    }
31
32    public override void _ExitTree()
33    {
34        if (ReferenceEquals(Instance, this)) Instance = null!;
35    }
36
37    public ToolType CurrentTool
38    {
39        get => _runtime.ActiveToolMode switch { 0 => ToolType.Road, 2 => ToolType.RoadRemove, 3 => ToolType.RoadUpgrade, _ => ToolType.Select };
40        set
41        {
42            if (CurrentTool == value) return;
43            int mode = value switch { ToolType.Select => 1, ToolType.Road => 0, ToolType.RoadRemove => 2, ToolType.RoadUpgrade => 3, _ => -1 };
44            if (mode >= 0) _runtime.SetToolMode(mode);
45        }
46    }
47
48    public bool UndoRoadEdit() => _runtime.UndoRoadEdit();
49    public bool RedoRoadEdit() => _runtime.RedoRoadEdit();
50    public bool CanUndoRoadEdit() => _runtime.CanUndoRoadEdit;
51    public bool CanRedoRoadEdit() => _runtime.CanRedoRoadEdit;
52    public RoadType GetSelectedRoadType() => _runtime.SelectedRoadType;
53    public bool SetSelectedRoadType(RoadType type) => _runtime.SetSelectedRoadType(type);
54    public bool CancelRoadSessions() => _runtime.CancelRoadInteraction();
55    public bool RetryRoadDisplay() => _runtime.RetryRoadDisplay();
56    public string RoadStatus => _runtime.RoadOperationStatus;
57    public bool CanRetryRoadDisplay => _runtime.CanRetryRoadDisplay;
58    public bool RoadOperationPending => _runtime.RoadOperationPending;
59}

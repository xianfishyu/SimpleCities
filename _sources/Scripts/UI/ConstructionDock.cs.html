219    /// <summary>为场景中预置的分类按钮写入显示数据，并登记各自的点击回调。</summary>
220    private void BuildCategoryBar()
221    {
222        foreach (CategoryDescriptor category in Categories)
223        {
224            Button button = GetNode<Button>($"DockPanel/DockStack/CategoryScroll/CategoryBar/{category.NodeName}");
225            if (button is ConstructionDockButton dockButton)
226            {
227                dockButton.DisplayText = category.DisplayName;
228                dockButton.VisualRole = ConstructionDockButtonVisualRole.PrimaryCategory;
229            }
230            button.ToggleMode = true;
231            button.FocusMode = FocusModeEnum.All;
232            button.Disabled = false;
233            button.TooltipText = string.Empty;
234            string categoryId = category.Id;
235            Action handler = () => OnCategoryPressed(categoryId);
236            button.Pressed += handler;
237            _disconnectActions.Add(() => button.Pressed -= handler);
238            _categoryButtons[categoryId] = button;
239        }
240    }

242    /// <summary>清空旧工具列表后，根据当前分类生成道路工具或未实现功能的占位项。</summary>

256    private void RenderRoadsMenu()
257    {
258        if (Category?.Tools == null) return;
259
260        var tools = new List<ConstructionToolDefinition>();
261        foreach (ConstructionToolDefinition? tool in Category.Tools)
262        {
263            if (tool == null)
264            {
265                GD.PushWarning("ConstructionDock: Category contains an empty tool reference.");
266                continue;
267            }
268            if (tool.ToolType is ToolType.Road or ToolType.RoadUpgrade)
269                tools.Add(tool);
270        }
271        tools.Sort(static (left, right) => left.SortOrder.CompareTo(right.SortOrder));
272
273        foreach (ConstructionToolDefinition tool in tools)
274            AddToolButton(tool);
275    }

296    /// <summary>从资源定义创建可交互工具按钮，并将点击操作映射到对应 ToolType。</summary>
297    private void AddToolButton(ConstructionToolDefinition tool)
298    {
299        var button = new ConstructionDockButton
300        {
301            Name = ToolNodeName(tool.ToolType),
302            DisplayText = tool.DisplayName,
303            VisualRole = ConstructionDockButtonVisualRole.SecondaryTool,
304            IconTexture = tool.Icon,
305            ToggleMode = true,
306            FocusMode = FocusModeEnum.All,
307            CustomMinimumSize = new Vector2(DockButtonWidth, ToolTrayHeight),
308            TooltipText = tool.Description,
309        };
310        BuildDockButtonPresentation(button);
311
312        ToolType toolType = tool.ToolType;
313        Action handler = () => OnToolPressed(toolType);
314        button.Pressed += handler;
315        _toolList.AddChild(button);
316        _toolButtons[toolType] = button;
317        _toolDefinitions[toolType] = tool;
318        _toolOrder.Add(toolType);
319    }

393    /// <summary>同类再次点击只切换托盘开关；切换分类则重建工具列表并展开托盘。</summary>
394    private void OnCategoryPressed(string categoryId)
395    {
396        if (!_categoryValid)
397        {
398            SetTrayVisible(false);
399            return;
400        }
401
402        if (categoryId == _activeCategoryId)
403        {
404            SetTrayVisible(!_toolTray.Visible);
405            NotifyContextDisplay();
406            return;
407        }
408
409        _activeCategoryId = categoryId;
410        RenderActiveMenu();
411        SetTrayVisible(true);
412        _categoryScroll.EnsureControlVisible(_categoryButtons[categoryId]);
413        NotifyContextDisplay();
414    }

441    /// <summary>轮询外部工具切换，使底栏按钮与实际工具状态保持一致。</summary>
442    private void SyncFromToolManager(bool force = false)
443    {
444        if (!_categoryValid) return;
445
446        if (_toolManager == null)
447        {
448            _toolManager = GodotObject.IsInstanceValid(RoadToolManager.Instance) ? RoadToolManager.Instance : null;
449            if (_toolManager == null)
450            {
451                LogMissingToolManager();
452                return;
453            }
454        }
455
456        ToolType currentTool = _toolManager.CurrentTool;
457        if (!force && _hasSyncedTool && currentTool == _lastSyncedTool) return;
458
459        _lastSyncedTool = currentTool;
460        _hasSyncedTool = true;
461        SyncSelectedTool(currentTool);
462    }

464    private void SyncSelectedTool(ToolType? toolType)
465    {
466        foreach ((ToolType candidate, Button button) in _toolButtons)
467        {
468            button.ButtonPressed = toolType == candidate;
469            if (button is ConstructionDockButton dockButton)
470                dockButton.Selected = button.ButtonPressed;
471        }
472    }

474    private void SyncCategoryButtons()
475    {
476        foreach ((string categoryId, Button button) in _categoryButtons)
477        {
478            button.ButtonPressed = categoryId == _activeCategoryId;
479            if (button is ConstructionDockButton dockButton)
480                dockButton.Selected = button.ButtonPressed;
481        }

484    /// <summary>将底栏固定到视口底部，并根据展开状态及可用宽度计算可滚动分类栏的尺寸。</summary>
485    private void ApplyDockLayout()
486    {
487        if (_dockPanel == null || _toolTray == null || _toolScroll == null || _categoryScroll == null || _categoryBar == null)
488            return;
489
490        float height = _toolTray.Visible ? ExpandedHeight : CollapsedHeight;
491
492        AnchorLeft = 0f;
493        AnchorRight = 1f;
494        AnchorTop = 1f;
495        AnchorBottom = 1f;
496        OffsetLeft = 0f;
497        OffsetRight = 0f;
498        OffsetBottom = 0f;
499        OffsetTop = OffsetBottom - height;
500
501        _dockPanel.SetAnchorsPreset(LayoutPreset.FullRect);
502        _dockPanel.OffsetLeft = 0f;
503        _dockPanel.OffsetTop = 0f;
504        _dockPanel.OffsetRight = 0f;
505        _dockPanel.OffsetBottom = 0f;
506        _toolTray.CustomMinimumSize = new Vector2(0f, ToolTrayHeight);
507        _toolScroll.CustomMinimumSize = new Vector2(0f, ToolTrayHeight);
508        _categoryScroll.CustomMinimumSize = new Vector2(0f, CategoryBarHeight);
509        float categoryButtonWidth = Mathf.Min(
510            DockButtonWidth,
511            Mathf.Max(72f, Mathf.Floor((Size.X - (Categories.Length - 1) * DockButtonSeparation) / Categories.Length)));
512        foreach (Button categoryButton in _categoryButtons.Values)
513            categoryButton.CustomMinimumSize = new Vector2(categoryButtonWidth, CategoryBarHeight);
514        _categoryBar.CustomMinimumSize = new Vector2(
515            Categories.Length * categoryButtonWidth + (Categories.Length - 1) * DockButtonSeparation,
516            CategoryBarHeight);
517    }

519    /// <summary>把分类、展开后的当前工具和 HUD 其他面板串成可循环的键盘焦点顺序。</summary>
520    private void UpdateFocusChain()
521    {
522        if (!IsInsideTree()) return;
523        if (!_categoryValid) return;
524        if (_categoryButtons.Count == 0) return;
525
526        Button? previousCategory = null;
527        for (int index = 0; index < Categories.Length; index++)
528        {
529            CategoryDescriptor category = Categories[index];
530            if (!_categoryButtons.TryGetValue(category.Id, out Button? button)) continue;
531            if (!button.IsInsideTree()) return;
532
533            button.FocusPrevious = previousCategory?.GetPath() ?? _debugFocusPath;
534            if (previousCategory != null)
535                previousCategory.FocusNext = button.GetPath();
536            previousCategory = button;
537        }
538
539        if (previousCategory == null) return;
540
541        if (_toolTray.Visible && _toolOrder.Count > 0)
542        {
543            Button? previousToolButton = null;
544            foreach (ToolType toolType in _toolOrder)
545            {
546                if (!_toolButtons.TryGetValue(toolType, out Button? toolButton) || !toolButton.IsInsideTree())
547                    return;
548
549                toolButton.FocusPrevious = previousToolButton?.GetPath() ?? previousCategory.GetPath();
550                if (previousToolButton != null)
551                {
552                    previousToolButton.FocusNext = toolButton.GetPath();
553                    previousToolButton.FocusNeighborRight = toolButton.GetPath();
554                    toolButton.FocusNeighborLeft = previousToolButton.GetPath();
555                }
556
557                previousToolButton = toolButton;
558            }
559
560            if (previousToolButton == null) return;
561            previousCategory.FocusNext = _toolButtons[_toolOrder[0]].GetPath();
562            previousToolButton.FocusNext = _contextFocusPath;
563        }
564        else
565        {
566            previousCategory.FocusNext = _contextFocusPath;
567        }
568    }

578    private bool TryGetCategory(string categoryId, out CategoryDescriptor? category)
579    {
580        foreach (CategoryDescriptor candidate in Categories)
581        {
582            if (candidate.Id != categoryId) continue;
583            category = candidate;
584            return true;
585        }
586
587        category = null;
588        return false;
589    }

3/// <summary>区分分类按钮和工具按钮，决定选中指示器的呈现方式。</summary>
4public enum ConstructionDockButtonVisualRole

10/// <summary>
11/// 建造栏按钮的通用呈现组件。场景中预置的分类按钮和运行时生成的工具按钮都复用它。
12/// </summary>
13public partial class ConstructionDockButton : Button
14{
15    private TextureRect? _icon;
16    private Label? _label;

18    private bool _isReady;
19
20    private Texture2D? _iconTexture;
21    private string _displayText = string.Empty;
22    private bool _selected;

69    /// <summary>取得由场景或 ConstructionDock 动态创建的视觉子节点后完成首次同步。</summary>
70    public override void _Ready()
71    {
72        Text = string.Empty;
73        _icon = GetNodeOrNull<TextureRect>("Presentation/Icon");
74        _label = GetNodeOrNull<Label>("Presentation/Label");
75        _primarySelectionIndicator = GetNodeOrNull<ColorRect>("PrimarySelectionIndicator");
76        _isReady = true;
77
78        SynchronizePresentation();
79    }
80
81    public override void _Notification(int what)
82    {
83        if (what == NotificationThemeChanged)
84            SynchronizePresentation();
85    }
86
87    public override void _Draw() => SynchronizePresentation();
88
89    public override void _ExitTree()
90    {
91        _isReady = false;
92        _icon = null;
93        _label = null;
94        _primarySelectionIndicator = null;
95    }

97    /// <summary>将导出的图标、文字、选中态和主题色同步到内部视觉节点。</summary>
98    private void SynchronizePresentation()
99    {
100        Text = string.Empty;
101        if (!_isReady)
102            return;
103
104        PresentationColors colors = ResolvePresentationColors();
105
106        if (_icon != null)
107        {
108            _icon.Texture = _iconTexture;
109            _icon.Modulate = colors.Icon;
110        }
111
112        if (_label != null)
113        {
114            _label.Text = _displayText;
115            _label.AddThemeColorOverride("font_color", colors.Label);
116        }
117
118        if (_primarySelectionIndicator != null)
119        {
120            _primarySelectionIndicator.Visible = _selected
121                && _visualRole == ConstructionDockButtonVisualRole.PrimaryCategory;
122            _primarySelectionIndicator.Color = colors.Indicator;
123        }
124    }
125
126    /// <summary>依次按禁用、选中、默认状态解析语义主题色，并保留 Godot 默认主题作为后备。</summary>
127    private PresentationColors ResolvePresentationColors()
128    {
129        if (Disabled)
130        {
131            Color disabledColor = ResolveThemeColor("disabled_color", "font_disabled_color");
132            return new PresentationColors(disabledColor, disabledColor, disabledColor);
133        }
134
135        if (_selected)
136        {
137            Color selectedIconColor = ResolveThemeColor("selected_color", "font_pressed_color");
138            Color selectedLabelColor = ResolveThemeColor("selected_label_color", "font_pressed_color");
139            return new PresentationColors(selectedIconColor, selectedLabelColor, selectedIconColor);
140        }
141
142        Color primaryColor = ResolveThemeColor("primary_color", "font_color");
143        return new PresentationColors(primaryColor, primaryColor, primaryColor);
144    }
145
146    private Color ResolveThemeColor(string semanticName, string fallbackName)
147    {
148        string themeType = ThemeTypeVariation.ToString();
149        if (Theme != null && themeType.Length > 0 && Theme.HasColor(semanticName, themeType))
150            return Theme.GetColor(semanticName, themeType);
151
152        if (themeType.Length > 0 && HasThemeColor(semanticName, themeType))
153            return GetThemeColor(semanticName, themeType);
154
155        if (HasThemeColor(semanticName))
156            return GetThemeColor(semanticName);
157
158        if (Theme != null && themeType.Length > 0 && Theme.HasColor(fallbackName, themeType))
159            return Theme.GetColor(fallbackName, themeType);
160
161        if (themeType.Length > 0 && HasThemeColor(fallbackName, themeType))
162            return GetThemeColor(fallbackName, themeType);
163
164        if (HasThemeColor(fallbackName))
165            return GetThemeColor(fallbackName);
166
167        return Colors.White;
168    }
169
170    private readonly record struct PresentationColors(Color Icon, Color Label, Color Indicator);

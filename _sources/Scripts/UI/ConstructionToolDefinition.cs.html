1using Godot;
2
3[GlobalClass]
4/// <summary>
5/// 单个建造工具的资源定义。ConstructionDock 用它生成按钮，ToolContextPanel 用它显示说明。
6/// </summary>
7public partial class ConstructionToolDefinition : Resource
8{
9    /// <summary>用于资源校验和去重的稳定工具 ID。</summary>
10    [Export] public string Id { get; set; } = string.Empty;
11    /// <summary>显示在建造栏和工具上下文面板中的名称。</summary>
12    [Export] public string DisplayName { get; set; } = string.Empty;
13    [Export] public string ShortcutHint { get; set; } = string.Empty;
14    [Export] public ToolType ToolType { get; set; }
15    [Export] public Texture2D? Icon { get; set; }
16    [Export] public int SortOrder { get; set; }
17    [Export(PropertyHint.MultilineText)] public string Description { get; set; } = string.Empty;
18
19    public bool TryValidate(out string error) => TryValidate(Id, DisplayName, out error);
20
21    public static bool TryValidate(string id, string displayName, out string error)
22    {
23        if (string.IsNullOrWhiteSpace(id))
24        {
25            error = "Tool ID cannot be empty.";
26            return false;
27        }
28
29        if (string.IsNullOrWhiteSpace(displayName))
30        {
31            error = $"Tool '{id}' display name cannot be empty.";
32            return false;
33        }
34
35        error = string.Empty;
36        return true;
37    }
38}

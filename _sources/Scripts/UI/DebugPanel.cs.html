1using Godot;
2using SimpleCities.RoadCore;
3
4/// <summary>
5/// 左上角的运行时诊断面板，展示帧率、鼠标网格位置和当前路网的规模统计。
6/// </summary>
7public partial class DebugPanel : PanelContainer
8{
9    [Export] public RoadConfig? Config { get; set; }
10
11    private Button _toggleButton = null!;
12    private VBoxContainer _debugContent = null!;
13    private Label _fpsValue = null!;
14    private Label _gridValue = null!;
15    private Label _nodeValue = null!;
16    private Label _canonicalEdgeValue = null!;
17    private Label _geometrySegmentValue = null!;
18    private Label _queryFragmentValue = null!;
19    private Label _selfLoopValue = null!;
20
21    private RoadMapScene? _network;
22    private RoadStateToken? _lastDiagnosticsSource;
23    private ulong _nextFpsUpdate;
24    private (Vector2 Point, bool HasNode)? _lastCursor;
25
26    public NodePath ToggleFocusPath => _toggleButton.GetPath();
27
28    public override void _Ready()
29    {
30        _toggleButton = GetNode<Button>("PanelMargin/Rows/DebugToggleButton");
31        _debugContent = GetNode<VBoxContainer>("PanelMargin/Rows/DebugContent");
32        _fpsValue = GetNode<Label>("PanelMargin/Rows/DebugContent/FpsRow/FpsValue");
33        _gridValue = GetNode<Label>("PanelMargin/Rows/DebugContent/GridRow/GridValue");
34        _nodeValue = GetNode<Label>("PanelMargin/Rows/DebugContent/NodeRow/NodeValue");
35        _canonicalEdgeValue = GetNode<Label>("PanelMargin/Rows/DebugContent/CanonicalEdgeRow/CanonicalEdgeValue");
36        _geometrySegmentValue = GetNode<Label>("PanelMargin/Rows/DebugContent/GeometrySegmentRow/GeometrySegmentValue");
37        _queryFragmentValue = GetNode<Label>("PanelMargin/Rows/DebugContent/QueryFragmentRow/QueryFragmentValue");
38        _selfLoopValue = GetNode<Label>("PanelMargin/Rows/DebugContent/SelfLoopRow/SelfLoopValue");
39
40        _debugContent.Visible = false;
41        _toggleButton.FocusMode = FocusModeEnum.All;
42        _toggleButton.Pressed += ToggleDebugContent;
43    }
44
45    public override void _ExitTree()
46    {
47        if (_toggleButton != null)
48            _toggleButton.Pressed -= ToggleDebugContent;
49    }
50
51    /// <summary>注入 HUD 已解析的路网和配置，避免面板自行查找场景节点。</summary>
52    public void SetDependencies(RoadMapScene? network, RoadConfig? config)
53    {
54        _network = network;
55        Config = config;
56        _lastDiagnosticsSource = null;
57    }
58
59    public void ConfigureFocus(NodePath previousPath, NodePath nextPath)
60    {
61        _toggleButton.FocusPrevious = previousPath;
62        _toggleButton.FocusNext = nextPath;
63    }
64
65    /// <summary>由 GameHUD 每帧调用；图指标只读取已提交的不可变快照。</summary>
66    public void UpdateMetrics()
67    {
68        ulong now = Time.GetTicksMsec();
69        if (now >= _nextFpsUpdate)
70        {
71            _fpsValue.Text = Engine.GetFramesPerSecond().ToString();
72            _nextFpsUpdate = now + 250;
73        }
74
75        if (!_debugContent.Visible)
76            return;
77
78        UpdateGridText();
79        RefreshDiagnostics();
80    }
81
82    private void ToggleDebugContent()
83    {
84        _debugContent.Visible = !_debugContent.Visible;
85        _toggleButton.Text = _debugContent.Visible ? "Debug ▲" : "Debug ▼";
86        if (_debugContent.Visible)
87        {
88            _lastDiagnosticsSource = null;
89            RefreshDiagnostics();
90        }
91    }
92
93    private void RefreshDiagnostics()
94    {
95        RoadStatistics? captured = _network?.CaptureStatistics();
96        if (captured is not RoadStatistics snapshot)
97        {
98            SetDiagnosticsUnavailable();
99            return;
100        }
101
102        if (snapshot.Source == _lastDiagnosticsSource)
103            return;
104
105        _lastDiagnosticsSource = snapshot.Source;
106        _nodeValue.Text = snapshot.NodeCount.ToString();
107        _canonicalEdgeValue.Text = snapshot.CanonicalEdgeCount.ToString();
108        _geometrySegmentValue.Text = snapshot.GeometrySegmentCount.ToString();
109        _queryFragmentValue.Text = snapshot.QueryFragmentCount.ToString();
110        _selfLoopValue.Text = snapshot.SelfLoopCount.ToString();
111    }
112
113    private void SetDiagnosticsUnavailable()
114    {
115        _lastDiagnosticsSource = null;
116        _nodeValue.Text = "--";
117        _canonicalEdgeValue.Text = "--";
118        _geometrySegmentValue.Text = "--";
119        _queryFragmentValue.Text = "--";
120        _selfLoopValue.Text = "--";
121    }
122
123    /// <summary>将鼠标世界坐标吸附到网格，并标记该位置是否已有路口节点。</summary>
124    private void UpdateGridText()
125    {
126        var cursor = Config is not null && GodotObject.IsInstanceValid(MainCamera.Instance)
127            ? _network?.DescribeCursor(MainCamera.Instance.GetGlobalMousePosition()) : null;
128        if (cursor == _lastCursor) return;
129        _lastCursor = cursor;
130        if (cursor is null) { _gridValue.Text = "--"; return; }
131        Vector2 snapped = cursor.Value.Point;
132        _gridValue.Text = $"({snapped.X:0.#}, {snapped.Y:0.#}){(cursor.Value.HasNode ? " [结构节点]" : string.Empty)}";
133    }
134}

36    /// <summary>解析依赖并完成一次 HUD 生命周期内的组件配置、信号连接和首帧布局。</summary>
37    public override void _Ready()
38    {
39        _toolManager = GodotObject.IsInstanceValid(RoadToolManager.Instance) ? RoadToolManager.Instance : null;
40        _network = GetParent() as RoadMapScene;
41        if (_toolManager == null)
42            GD.PushWarning("GameHUD: ToolManager.Instance is missing; tool display is degraded.");
43        if (_network == null)
44            GD.PushWarning("GameHUD: V4 map runtime is missing; debug metrics are degraded.");
45
46        if (Config == null)
47        {
48            GD.PushWarning("GameHUD: Config (RoadConfig resource) is not assigned; using fallback RoadConfig for UI display.");
49            Config = new RoadConfig();
50        }
51
52        EnsureUIManager();
53        ResolveChildNodes();
54        _panelResizedCallable = Callable.From(OnPanelResized);
55        ConfigureComponents();
56        _bindingManager = GodotObject.IsInstanceValid(InputBindingManager.Instance) ? InputBindingManager.Instance : null;
57        if (_bindingManager is not null) _bindingManager.BindingChanged += OnBindingChanged;
58        _contextDirty = true;
59        _toolContextPanel.RoadDisplayRetryRequested += RetryRoadDisplay;
60        if (_network is not null) _network.ProductUiContains = IsPointerOverHud;
61        RegisterManagedPanels();
62        WireConstructionDock();
63        WirePauseMenu();
64        WireLayoutSignals();
65        ConfigureFocusChain();
66        ConnectViewportResize();
67        QueueResponsiveLayoutRefresh();
68    }

70    /// <summary>断开所有跨节点事件和面板注册，保证 HUD 可安全地再次进入场景树。</summary>
71    public override void _ExitTree()
72    {
73        if (_pauseMenu != null && _pauseMenu.IsOpen)
74        {
75            _pauseMenu.SetExitConvergencePending(false);
76            _pauseMenu.Close();
77        }
78        DisconnectViewportResize();
79        if (_constructionDock != null)
80        {
81            _constructionDock.TrayVisibilityChanged -= OnDockTrayVisibilityChanged;
82            _constructionDock.ContextDisplayChanged -= OnDockContextDisplayChanged;
83        }
84        if (_pauseMenu != null)
85        {
86            _pauseMenu.ContinueRequested -= ClosePauseMenu;
87            _pauseMenu.ReturnToMainMenuRequested -= ReturnToMainMenu;
88            _pauseMenu.QuitToDesktopRequested -= QuitToDesktop;
89            _pauseMenu.ConfigureSaveManager(null);
90        }
91        DisconnectLayoutSignals();
92        if (_bindingManager is not null) _bindingManager.BindingChanged -= OnBindingChanged;
93        _bindingManager = null;
94
95        _uiManager?.Unregister("ContextPanel");
96        _uiManager?.Unregister("DebugPanel");
97        _uiManager?.Unregister(PauseMenuPanelName);
98        _layoutRefreshQueued = false;
99        if (_toolContextPanel != null) _toolContextPanel.RoadDisplayRetryRequested -= RetryRoadDisplay;
100        if (_network is not null) _network.ProductUiContains = null;
101        RequestReadyOnReentry();
102    }

104    /// <summary>统一消费可配置的暂停和工具动作；暂停时由 PauseMenu 自身接管输入。</summary>
105    public override void _Input(InputEvent @event)
106    {
107        if (!GodotObject.IsInstanceValid(InputBindingManager.Instance))
108            return;
109
110        if (_pauseMenu.IsOpen)
111            return;
112
113        if (InputBindingManager.Instance.EventMatchesAction(@event, InputBindingManager.PauseMenuAction))
114        {
115            if (_toolManager?.CancelRoadSessions() != true) OpenPauseMenu();
116            GetViewport().SetInputAsHandled();
117            return;
118        }
119
120        if (_uiManager.IsModalActive || _toolManager == null)
121            return;
122
123        if (InputBindingManager.Instance.EventMatchesAction(@event, InputBindingManager.EditUndoAction))
124        {
125            _toolManager.UndoRoadEdit();
126            GetViewport().SetInputAsHandled();
127            return;
128        }
129
130        if (InputBindingManager.Instance.EventMatchesAction(@event, InputBindingManager.EditRedoAction))
131        {
132            _toolManager.RedoRoadEdit();
133            GetViewport().SetInputAsHandled();
134            return;
135        }
136
137        if (!InputBindingManager.Instance.TryGetToolForEvent(@event, out ToolType tool))
138            return;
139
140        _toolManager.CurrentTool = tool;
141        GetViewport().SetInputAsHandled();
142    }

144    /// <summary>持续同步工具上下文和轻量调试指标，不在此执行任何游戏规则。</summary>
145    public override void _Process(double delta)
146    {
147        ToolType currentTool = _toolManager?.CurrentTool ?? ToolType.Select;
148        RoadType? currentRoadType = _toolManager?.GetSelectedRoadType();
149        if (_constructionDock?.UsesCatalogContext != false &&
150            (_contextDirty || currentTool != _lastContextTool || currentRoadType != _lastContextRoadType || !ReferenceEquals(Config, _lastContextConfig) ||
151             Config.CellSize != _lastContextCellSize))
152        {
153            _toolContextPanel.UpdateContext(currentTool, Config);
154            _lastContextTool = currentTool;
155            _lastContextRoadType = currentRoadType;
156            _lastContextConfig = Config;
157            _lastContextCellSize = Config.CellSize;
158            _contextDirty = false;
159        }
160        _debugPanel.UpdateMetrics();
161        _toolContextPanel.UpdateRoadState(_toolManager?.RoadStatus ?? "", _toolManager?.CanRetryRoadDisplay == true,
162            _toolManager?.RoadOperationPending == true);
163    }

238    private void WireLayoutSignals()
239    {
240        if (_layoutSignalsConnected) return;
241
242        ConnectResizeSignal(_toolContextPanel);
243        ConnectResizeSignal(_debugPanel);
244        if (!_debugPanel.IsConnected(Control.SignalName.MinimumSizeChanged, _panelResizedCallable))
245            _debugPanel.Connect(Control.SignalName.MinimumSizeChanged, _panelResizedCallable);
246        if (_constructionDock != null)
247            ConnectResizeSignal(_constructionDock);
248        _layoutSignalsConnected = true;
249    }
250
251    private void DisconnectLayoutSignals()
252    {
253        if (!_layoutSignalsConnected) return;
254
255        if (_toolContextPanel != null)
256            DisconnectResizeSignal(_toolContextPanel);
257        if (_debugPanel != null)
258        {
259            DisconnectResizeSignal(_debugPanel);
260            if (_debugPanel.IsConnected(Control.SignalName.MinimumSizeChanged, _panelResizedCallable))
261                _debugPanel.Disconnect(Control.SignalName.MinimumSizeChanged, _panelResizedCallable);
262        }
263        if (_constructionDock != null)
264            DisconnectResizeSignal(_constructionDock);
265        _layoutSignalsConnected = false;
266    }

329    /// <summary>将布局延后到容器完成尺寸计算后执行，并合并同一帧的多次尺寸变化。</summary>
330    private void QueueResponsiveLayoutRefresh()
331    {
332        if (_layoutRefreshQueued || !IsInsideTree()) return;
333
334        _layoutRefreshQueued = true;
335        CallDeferred(MethodName.ApplyResponsiveLayoutAfterContainersSettled);
336    }

338    private void ApplyResponsiveLayoutAfterContainersSettled()
339    {
340        _layoutRefreshQueued = false;

344    }

346    /// <summary>协调右侧上下文、右上系统操作和左上调试面板，避免彼此或底边栏重叠。</summary>
347    private void ApplyResponsiveLayout()
348    {
349        Vector2 viewportSize = GetViewport().GetVisibleRect().Size;
350        if (_constructionDock == null)
351        {
352            _toolContextPanel.ApplyResponsiveLayout();
353            PlaceTopLeftDebugPanel(viewportSize);
354            return;
355        }
356
357        _toolContextPanel.ReservedBottomTop = _constructionDock.Position.Y;
358        _toolContextPanel.ApplyResponsiveLayoutForViewport(viewportSize);
359        PlaceTopLeftDebugPanel(viewportSize);
360    }
361
362    private void PlaceTopLeftDebugPanel(Vector2 viewportSize)
363    {
364        float reservedBottomTop = _constructionDock?.Position.Y ?? viewportSize.Y;
365        Vector2 debugSize = EffectiveSize(_debugPanel);
366        debugSize.Y = _debugPanel.GetCombinedMinimumSize().Y;
367        debugSize.X = Mathf.Min(debugSize.X, Mathf.Max(0f, viewportSize.X - (PanelMargin * 2f)));
368        debugSize.Y = Mathf.Min(debugSize.Y, Mathf.Max(0f, reservedBottomTop - PanelMargin - PanelGap));
369        _debugPanel.Size = debugSize;
370        _debugPanel.Position = new Vector2(PanelMargin, PanelMargin);
371    }

373    private static Vector2 EffectiveSize(Control panel)
374    {
375        Vector2 minimumSize = panel.GetCombinedMinimumSize();
376        return new Vector2(Mathf.Max(panel.Size.X, minimumSize.X), Mathf.Max(panel.Size.Y, minimumSize.Y));
377    }

403    private async void ReturnToMainMenu()
404    {
405        if (_exitTransitionPending)
406            return;
407        _exitTransitionPending = true;
408        _pauseMenu.SetExitConvergencePending(true);
409
410        SaveManager? saveManager = GodotObject.IsInstanceValid(SaveManager.Instance)
411            ? SaveManager.Instance
412            : null;
413        try
414        {
415            if (saveManager is not null)
416                await saveManager.DrainCurrentSceneOperationsAsync();
417            if (!IsInsideTree())
418                return;
419            if (saveManager?.IsApplicationExitPending == true)
420                return;
421
422            _pauseMenu.SetExitConvergencePending(false);
423            ClosePauseMenu();
424            Error result = GetTree().ChangeSceneToFile(MainMenuScenePath);
425            if (result == Error.Ok)
426                return;
427
428            bool resumed = saveManager?.ResumeCurrentSceneOperations() != false;
429            GD.PushError(
430                $"GameHUD: failed to change to main menu ({result}); save admission resumed={resumed}.");
431            OpenPauseMenu();
432            _exitTransitionPending = false;
433        }
434        catch (Exception exception)
435        {
436            if (IsInsideTree())
437            {
438                saveManager?.ResumeCurrentSceneOperations();
439                _pauseMenu.SetExitConvergencePending(false);
440                _exitTransitionPending = false;
441            }
442            GD.PushError($"GameHUD: failed to converge scene exit: {exception.Message}");
443        }
444    }
445
446    private void QuitToDesktop()
447    {
448        if (_exitTransitionPending)
449            return;
450        _exitTransitionPending = true;
451        _pauseMenu.SetExitConvergencePending(true);
452        SaveManager? saveManager = GodotObject.IsInstanceValid(SaveManager.Instance)
453            ? SaveManager.Instance
454            : null;
455        if (saveManager is not null)
456            saveManager.RequestApplicationQuit();
457        else
458            GetTree().Quit();
459    }

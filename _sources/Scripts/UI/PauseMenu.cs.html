137    public override void _Process(double delta)
138    {
139        if (!_refreshSaveSlotsWhenIdle || !IsOpen || _view != MenuView.SaveManagement ||
140            IsSaveOperationBusy)
141        {
142            return;
143        }
144
145        SaveManager? saveManager = ActiveSaveManager();
146        if (saveManager?.IsOperationBusy != false)
147        {
148            _saveManagerIdleFrames = 0;
149            return;
150        }
151        if (++_saveManagerIdleFrames < 3)
152            return;
153
154        _refreshSaveSlotsWhenIdle = false;
155        _saveManagerIdleFrames = 0;
156        SetSaveOperationControlsDisabled(false);
157        RefreshSaveSlots(PreferredSaveSlotID());
158    }

160    public override void _Input(InputEvent @event)
161    {
162        if (!IsOpen || @event is not InputEventKey keyEvent || !keyEvent.Pressed || keyEvent.Echo)
163            return;
164
165        if (_exitConvergencePending)
166        {
167            GetViewport().SetInputAsHandled();
168            return;
169        }
170
171        if (_capturingAction != null)
172        {
173            HandleBindingInput(keyEvent);
174            GetViewport().SetInputAsHandled();
175            return;
176        }
177
178        if (!GodotObject.IsInstanceValid(InputBindingManager.Instance) ||
179            !InputBindingManager.Instance.EventMatchesAction(@event, InputBindingManager.PauseMenuAction))
180            return;
181
182        if (IsSaveOperationBusy)
183        {
184            if (!_activeOperationCrossedBoundary && !_activeOperationCancelRequested)
185            {
186                _activeOperationCancelRequested = true;
187                _saveManager?.CancelOperation(_activeOperationToken);
188                ShowSaveStatus("正在取消操作...", success: false);
189            }
190        }
191        else if (_view == MenuView.Confirmation)
192            CancelConfirmation();
193        else if (_view == MenuView.SaveManagement)
194            ShowMainView();
195        else if (_view == MenuView.Bindings)
196            ShowSettingsView();
197        else if (_view == MenuView.Settings)
198            ShowMainView();
199        else
200            ContinueRequested?.Invoke();
201
202        GetViewport().SetInputAsHandled();
203    }

500    private void TryGrabDeferredFocus(Control control)
501    {

504            control.FocusMode == FocusModeEnum.None ||

558    private void ConfirmRequestedAction()
559    {
560        if (IsSaveOperationBusy || _exitConvergencePending)
561            return;
562        switch (_confirmationAction)
563        {
564            case ConfirmationAction.ReturnToMainMenu:
565                ReturnToMainMenuRequested?.Invoke();
566                break;
567            case ConfirmationAction.QuitToDesktop:
568                QuitToDesktopRequested?.Invoke();
569                break;
570            case ConfirmationAction.OverwriteSave:
571                OverwriteConfirmedSave();
572                break;
573            case ConfirmationAction.LoadSave:
574                LoadConfirmedSave();
575                break;
576            case ConfirmationAction.DeleteSave:
577                DeleteConfirmedSave();
578                break;
579            default:
580                throw new ArgumentOutOfRangeException();
581        }
582    }

658    private void RequestDeleteSave()
659    {
660        if (IsSaveOperationBusy)
661            return;
662        SaveSlotSummary? summary = SelectedSaveSlot();
663        if (summary == null)
664            return;
665        SaveManager? saveManager = ActiveSaveManager();
666        string operationToken = saveManager?.ArmDeletion(summary) ?? string.Empty;
667        if (operationToken.Length == 0)
668        {
669            RefreshSaveSlots(summary.SlotID);
670            ShowSaveStatus("存档列表已变化，请重新确认", success: false);
671            return;
672        }
673
674        string displayName = summary.IsValid ? summary.DisplayName : summary.SlotID;

802    private void OnSaveOperationCompleted(SaveOperationResult result)
803    {
804        SaveManager? saveManager = ActiveSaveManager();
805        if (saveManager is null || !IsOpen ||
806            !IsMatchingOperation(result.OperationToken, result.Kind) ||
807            _operationMenuGeneration != _menuOpenGeneration ||
808            _operationSceneGeneration != saveManager.SceneGeneration)
809        {
810            return;
811        }
812
813        MenuOperationAction menuOperation = _activeMenuOperation;
814        string displayName = _activeOperationDisplayName;
815        string preferredSlotID = result.TargetSlotID;
816        ClearSaveOperation();
817
818        if (!result.IsSuccess)
819        {
820            RefreshSaveSlots(PreferredSaveSlotID());
821            string message = result.ResultKind == SaveOperationResultKind.Canceled
822                ? "操作已取消"
823                : result.Error?.Length > 0
824                    ? $"{OperationFailureLabel(menuOperation)}：{result.Error}"
825                    : OperationFailureLabel(menuOperation);
826            ShowSaveStatus(message, success: false);
827            return;
828        }
829
830        string warningSuffix = result.Warnings.Count == 0
831            ? string.Empty
832            : $"（{string.Join("；", result.Warnings)}）";
833        switch (menuOperation)
834        {
835            case MenuOperationAction.SaveAs:
836                _saveNameInput.Text = string.Empty;
837                RefreshSaveSlots(preferredSlotID);
838                ShowSaveStatus($"已创建“{displayName}”{warningSuffix}", success: true);
839                break;
840            case MenuOperationAction.Overwrite:
841                RefreshSaveSlots(preferredSlotID);
842                ShowSaveStatus($"已覆盖“{displayName}”{warningSuffix}", success: true);
843                break;
844            case MenuOperationAction.Delete:
845                RefreshSaveSlots(saveManager.CurrentSlotID);
846                ShowSaveStatus($"已删除“{displayName}”{warningSuffix}", success: true);
847                break;
848            case MenuOperationAction.Load:
849                if (result.Warnings.Count != 0)
850                    GD.PushWarning($"Load completed with observer warnings: {string.Join("; ", result.Warnings)}");
851                ShowSaveStatus($"已加载“{displayName}”{warningSuffix}", success: true);
852                ContinueRequested?.Invoke();
853                break;
854        }
855    }

935    private void RefreshSaveSlots(string? preferredSlotID)
936    {
937        _saveSlots.Clear();
938        _saveSlotList.Clear();
939        _selectedSaveSlotIndex = -1;
940
941        SaveManager? saveManager = ActiveSaveManager();
942        if (saveManager == null)
943        {
944            _saveSlotSummaryLabel.Text = "存档管理不可用";
945            _saveSlotSummaryLabel.TooltipText = string.Empty;
946            UpdateSaveActionAvailability();
947            return;
948        }
949
950        _saveSlots.AddRange(saveManager.ListSlots());
951        int preferredIndex = -1;
952        for (int index = 0; index < _saveSlots.Count; index++)
953        {
954            SaveSlotSummary summary = _saveSlots[index];
955            string slotKind = summary.IsAutosave ? "自动" : "手动";
956            string itemText = summary.IsValid
957                ? $"{slotKind}  ·  {summary.DisplayName}  ·  {FormatSaveTime(summary.SavedAtUtc)}"
958                : $"损坏{slotKind}存档  ·  {summary.SlotID}";
959            _saveSlotList.AddItem(itemText);
960            _saveSlotList.SetItemMetadata(index, summary.SlotID);
961            if (!summary.IsValid)
962                _saveSlotList.SetItemCustomFgColor(index, new Color("#FF6B6B"));
963            if (string.Equals(summary.SlotID, preferredSlotID, StringComparison.Ordinal))
964                preferredIndex = index;
965        }
966
967        if (_saveSlots.Count == 0)
968        {
969            _saveSlotSummaryLabel.Text = "暂无存档";
970            _saveSlotSummaryLabel.TooltipText = string.Empty;
971            UpdateSaveActionAvailability();
972            return;
973        }
974
975        _selectedSaveSlotIndex = preferredIndex >= 0 ? preferredIndex : 0;
976        _saveSlotList.Select(_selectedSaveSlotIndex);
977        UpdateSelectedSaveSummary();
978    }

1167    private void HandleBindingInput(InputEventKey keyEvent)
1168    {
1169        if (_capturingAction == null)
1170            return;
1171
1172        if (keyEvent.CtrlPressed || keyEvent.AltPressed || keyEvent.ShiftPressed || keyEvent.MetaPressed)
1173        {
1174            ShowBindingStatus("暂不支持组合键", success: false);
1175            return;
1176        }
1177
1178        if (!GodotObject.IsInstanceValid(InputBindingManager.Instance))
1179        {
1180            CancelBindingCapture(showStatus: false);
1181            ShowBindingStatus("输入设置不可用", success: false);
1182            return;
1183        }
1184
1185        string actionName = _capturingAction;
1186        Key key = InputBindingManager.NormalizeKey(keyEvent);
1187        if (!InputBindingManager.Instance.TryRebind(actionName, key, out string error))
1188        {
1189            ShowBindingStatus(error, success: false);
1190            return;
1191        }
1192
1193        _capturingAction = null;
1194        RefreshBindingButtons();
1195        ShowBindingStatus(
1196            $"“{InputBindingManager.Instance.GetDisplayName(actionName)}”已绑定为 {InputBindingManager.Instance.GetBindingText(actionName)}",
1197            success: true);
1198    }

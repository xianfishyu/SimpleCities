172    /// <summary>
173    /// 同步工具说明。优先使用资源化工具定义；选择和拆除等内置工具则使用底栏提供的后备文案。
174    /// </summary>
175    public void UpdateContext(ToolType currentTool, RoadConfig? config)
176    {
177        Config = config;
178        RefreshRoadTypeStylesIfNeeded();
179        _shortcutRow.Visible = true;
180        _cellSizeRow.Visible = true;
181        ConstructionToolDefinition? definition = FindTool(currentTool);
182        _categoryValue.Text = Category?.DisplayName ?? "道路 unavailable";
183        if (definition != null)
184        {
185            _toolValue.Text = definition.DisplayName;
186            _operationValue.Text = definition.Description;
187            string shortcutHint = ResolveShortcutHint(currentTool, definition.ShortcutHint);
188            _shortcutValue.Text = shortcutHint;
189            _shortcutRow.Visible = !string.IsNullOrWhiteSpace(shortcutHint);
190        }
191        else if (ConstructionDock.TryGetBuiltInToolPresentation(currentTool, out string displayName, out string description, out string shortcutHint))
192        {
193            _toolValue.Text = displayName;
194            _operationValue.Text = description;
195            shortcutHint = ResolveShortcutHint(currentTool, shortcutHint);
196            _shortcutValue.Text = shortcutHint;
197            _shortcutRow.Visible = !string.IsNullOrWhiteSpace(shortcutHint);
198        }
199        else
200        {
201            _toolValue.Text = currentTool.ToString();
202            _operationValue.Text = "工具定义不可用。";
203            _shortcutValue.Text = "--";
204        }
205        _cellSizeValue.Text = Config == null ? "CellSize: unavailable" : $"CellSize: {Config.CellSize:F0}";
206        _roadTypeRow.Visible = currentTool is ToolType.Road or ToolType.RoadUpgrade;
207        SyncSelectedRoadTypeButtons();
208    }

228    /// <summary>
229    /// 按视口和底边栏顶部位置重新计算面板边界，避免覆盖底栏；760px 以下使用紧凑模式。
230    /// </summary>
231    public void ApplyResponsiveLayoutForViewport(Vector2 viewportSize)
232    {
233        if (_contentScroll == null || _content == null || _focusEntryButton == null) return;
234        if (viewportSize.X <= 0f) return;
235        float reservedBottomTop = ReservedBottomTop > 0f ? ReservedBottomTop : viewportSize.Y - 16f;
236
237        bool shouldCompact = viewportSize.X < 760f;
238        _usesCompactViewport = shouldCompact;
239        if (!shouldCompact)
240            _compactExpanded = false;
241
242        _compact = shouldCompact && !_compactExpanded;
243        float width = _compact ? CompactWidth : WideWidth;
244        OffsetLeft = -width - RightMargin;
245        OffsetRight = -RightMargin;
246        float top = shouldCompact ? CompactTop : WideTop;
247        float bottomLimit = Mathf.Max(top + 44f, Mathf.Min(viewportSize.Y - 16f, reservedBottomTop - 16f));
248        OffsetTop = top;
249        OffsetBottom = _compact ? top + 44f : bottomLimit;
250        ApplyMinimumContributionState();
251    }

253    private void ApplyMinimumContributionState()
254    {
255        bool contentVisible = !_compact;
256        _contentScroll.Visible = contentVisible;
257        _roadRuntimeStatus.Visible = contentVisible && _lastRoadStatus.Length > 0;
258        _contentScroll.VerticalScrollMode = ScrollContainer.ScrollMode.Auto;
259        _contentScroll.HorizontalScrollMode = ScrollContainer.ScrollMode.Disabled;
260        _focusEntryButton.Text = _compact ? "工" : "工具上下文";
261        if (_compact)
262        {
263            CustomMinimumSize = new Vector2(CompactWidth, 44f);
264            _focusEntryButton.CustomMinimumSize = new Vector2(CompactWidth, 36f);
265            _contentScroll.CustomMinimumSize = Vector2.Zero;
266            ApplyPanelMargins(0);
267            AddThemeStyleboxOverride("panel", _compactPanelStyle);
268            Size = new Vector2(CompactWidth, 44f);
269            return;
270        }
271
272        CustomMinimumSize = new Vector2(WideWidth, 44f);
273        _focusEntryButton.CustomMinimumSize = new Vector2(44f, 36f);
274        _contentScroll.CustomMinimumSize = new Vector2(0f, 40f);
275        ApplyPanelMargins(_usesCompactViewport ? CompactExpandedMargin : ExpandedMargin);
276        if (_usesCompactViewport)
277            AddThemeStyleboxOverride("panel", _compactPanelStyle);
278        else
279            RemoveThemeStyleboxOverride("panel");
280    }

306    private ConstructionToolDefinition? FindTool(ToolType currentTool)
307    {
308        if (Category?.Tools == null) return null;
309
310        foreach (ConstructionToolDefinition? tool in Category.Tools)
311            if (tool?.ToolType == currentTool)
312                return tool;
313
314        return null;
315    }

317    private void WireRoadTypeButtons()
318    {
319        DisconnectRoadTypeButtons();
320        _roadTypeButtons.Clear();
321        _roadTypeButtonGroup = new ButtonGroup { AllowUnpress = false };
322
323        foreach (RoadType roadType in RoadTypeOrder)
324        {
325            Button button = _roadTypeSelector.GetNode<Button>(GetRoadTypeButtonName(roadType));
326            button.ButtonGroup = _roadTypeButtonGroup;
327            button.ToggleMode = true;
328            button.FocusMode = FocusModeEnum.All;
329            button.MouseDefaultCursorShape = CursorShape.PointingHand;
330            RoadType capturedRoadType = roadType;
331            Action handler = () => OnRoadTypePressed(capturedRoadType);
332            button.Pressed += handler;
333            _roadTypeDisconnectActions.Add(() => button.Pressed -= handler);
334            _roadTypeButtons[roadType] = button;
335        }
336
337        ConfigureRoadTypeFocusNeighbors();
338    }
339
340    private void DisconnectRoadTypeButtons()
341    {
342        foreach (Action disconnect in _roadTypeDisconnectActions)
343            disconnect();
344        _roadTypeDisconnectActions.Clear();
345    }

347    private void ConfigureRoadTypeFocusNeighbors()
348    {
349        for (int index = 0; index < RoadTypeOrder.Length; index++)
350        {
351            Button button = _roadTypeButtons[RoadTypeOrder[index]];
352            if (index > 0)
353            {
354                button.FocusNeighborLeft = _roadTypeButtons[RoadTypeOrder[index - 1]].GetPath();
355                button.FocusPrevious = _roadTypeButtons[RoadTypeOrder[index - 1]].GetPath();
356            }
357            if (index + 1 < RoadTypeOrder.Length)
358            {
359                button.FocusNeighborRight = _roadTypeButtons[RoadTypeOrder[index + 1]].GetPath();
360                button.FocusNext = _roadTypeButtons[RoadTypeOrder[index + 1]].GetPath();
361            }
362        }
363    }

365    private void RefreshRoadTypeStylesIfNeeded()
366    {
367        if (ReferenceEquals(_roadTypeStyleConfig, Config))
368            return;
369
370        _roadTypeStyleConfig = Config;
371        _roadTypeStylesValid = Config != null && Config.TryValidateRoadTypeStyles(out _);
372        _roadTypeSelectorAvailable = _roadTypeStylesValid &&
373            _selectedRoadTypeGetter is not null &&
374            _selectedRoadTypeSetter is not null;
375
376        foreach (RoadType roadType in RoadTypeOrder)
377        {
378            if (!_roadTypeButtons.TryGetValue(roadType, out Button? button))
379                continue;
380
381            RoadTypePresentation presentation = ResolveRoadTypePresentation(roadType);
382            button.TooltipText = presentation.DisplayName;
383            ColorRect swatch = button.GetNode<ColorRect>("Swatch");
384            swatch.Color = presentation.Color;
385            Label label = button.GetNode<Label>("Label");
386            label.Text = presentation.DisplayName;
387        }
388
389        ApplyRoadTypeAvailability();
390    }

392    private void ApplyRoadTypeAvailability()
393    {
394        if (_roadTypeSelector == null || _roadTypeStatus == null)
395            return;
396
397        foreach (Button button in _roadTypeButtons.Values)
398            button.Disabled = !_roadTypeSelectorAvailable;
399
400        _roadTypeStatus.Visible = !_roadTypeStylesValid || !_roadTypeSelectorAvailable;
401        _roadTypeStatus.Text = !_roadTypeStylesValid
402            ? "道路类型样式不可用，选择已禁用。"
403            : "道路类型控制器不可用，选择已禁用。";
404    }

406    private RoadTypePresentation ResolveRoadTypePresentation(RoadType roadType)
407    {
408        if (_roadTypeStylesValid && Config?.RoadTypeStyles is not null)
409        {
410            foreach (RoadTypeStyle? style in Config.RoadTypeStyles)
411            {
412                if (style?.RoadType != roadType)
413                    continue;
414
415                string displayName = string.IsNullOrWhiteSpace(style.DisplayName)
416                    ? FallbackRoadTypePresentations[roadType].DisplayName
417                    : style.DisplayName;
418                return new RoadTypePresentation(displayName, style.Color);
419            }
420        }
421
422        return FallbackRoadTypePresentations[roadType];
423    }
424
425    private void OnRoadTypePressed(RoadType roadType)
426    {
427        if (!_roadTypeSelectorAvailable ||
428            _selectedRoadTypeSetter is null ||
429            !RoadTypeContract.IsDefined(roadType) ||
430            !TrySetSelectedRoadType(roadType))
431        {
432            SyncSelectedRoadTypeButtons();
433            return;
434        }
435
436        SyncSelectedRoadTypeButtons();
437    }
438
439    private bool TrySetSelectedRoadType(RoadType roadType)
440    {
441        if (_selectedRoadTypeSetter is null)
442            return false;
443
444        try
445        {
446            return _selectedRoadTypeSetter(roadType);
447        }
448        catch (ObjectDisposedException)
449        {
450            _roadTypeSelectorAvailable = false;
451            ApplyRoadTypeAvailability();
452            return false;
453        }
454    }
455
456    private void SyncSelectedRoadTypeButtons()
457    {
458        if (_roadTypeButtons.Count == 0)
459            return;
460
461        RoadType selectedRoadType = RoadType.Street;
462        if (_selectedRoadTypeGetter is not null)
463        {
464            try
465            {
466                RoadType candidate = _selectedRoadTypeGetter();
467                if (RoadTypeContract.IsDefined(candidate))
468                    selectedRoadType = candidate;
469            }
470            catch (ObjectDisposedException)
471            {
472                _roadTypeSelectorAvailable = false;
473            }
474        }
475
476        foreach (RoadType roadType in RoadTypeOrder)
477            _roadTypeButtons[roadType].ButtonPressed = roadType == selectedRoadType;
478    }

491    private static string ResolveShortcutHint(ToolType toolType, string fallback)
492    {
493        if (GodotObject.IsInstanceValid(InputBindingManager.Instance) &&
494            InputBindingManager.TryGetToolAction(toolType, out string actionName))
495            return InputBindingManager.Instance.GetBindingText(actionName);
496
497        return fallback;
498    }

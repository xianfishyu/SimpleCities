1using Godot;
2using System.Collections.Generic;
3
4/// <summary>
5/// UI 面板生命周期管理器。每个 GameHUD 拥有独立实例，避免切换场景后残留全局 UI 状态。
6/// 只管理需要按名称显示、隐藏或模态化的面板；底部 ConstructionDock 始终可见，因此不在此注册。
7/// </summary>
8public partial class UIManager : Node
9{
10    private readonly Dictionary<string, Control> _panels = new();
11    private readonly Stack<string> _modalStack = new();
12
13    /// <summary>是否有模态面板正在阻塞游戏输入。</summary>
14    public bool IsModalActive => _modalStack.Count > 0;
15
16    public override void _ExitTree()
17    {
18        _panels.Clear();
19        _modalStack.Clear();
20    }
21
22    // ── 注册 ──────────────────────────────────────────────
23
24    /// <summary>
25    /// 注册一个 UI 面板到管理器。
26    /// 面板在 _Ready 中调用此方法注册自己。
27    /// </summary>
28    public void Register(string name, Control panel)
29    {
30        _panels[name] = panel;
31    }
32
33    /// <summary>注销面板。</summary>
34    public void Unregister(string name)
35    {
36        _panels.Remove(name);
37    }
38
39    // ── 可见性控制 ────────────────────────────────────────
40
41    /// <summary>显示指定面板。</summary>
42    public void Show(string name)
43    {
44        if (_panels.TryGetValue(name, out var panel))
45            panel.Visible = true;
46    }
47
48    /// <summary>隐藏指定面板。</summary>
49    public void Hide(string name)
50    {
51        if (_panels.TryGetValue(name, out var panel))
52            panel.Visible = false;
53    }
54
55    /// <summary>切换面板可见性。</summary>
56    public void Toggle(string name)
57    {
58        if (_panels.TryGetValue(name, out var panel))
59            panel.Visible = !panel.Visible;
60    }
61
62    /// <summary>查询面板是否可见。</summary>
63    public bool IsVisible(string name)
64    {
65        return _panels.TryGetValue(name, out var panel) && panel.Visible;
66    }
67
68    /// <summary>隐藏所有已注册面板。</summary>
69    public void HideAll()
70    {
71        foreach (var (_, panel) in _panels)
72            panel.Visible = false;
73    }
74
75    // ── 模态面板 ──────────────────────────────────────────
76
77    /// <summary>
78    /// 推入一个模态面板——显示该面板并阻塞游戏输入（例如设置弹窗、离线报告）。
79    /// </summary>
80    public void PushModal(string name)
81    {
82        if (_panels.TryGetValue(name, out var panel))
83        {
84            panel.Visible = true;
85            // 栈顶永远代表当前需要优先关闭的面板，支持之后扩展为嵌套弹窗。
86            _modalStack.Push(name);
87        }
88    }
89
90    /// <summary>关闭最顶层的模态面板。</summary>
91    public void PopModal()
92    {
93        if (_modalStack.Count == 0) return;
94        var name = _modalStack.Pop();
95        if (_panels.TryGetValue(name, out var panel))
96            panel.Visible = false;
97    }
98
99    // ── 查询 ──────────────────────────────────────────────
100
101    /// <summary>获取已注册面板（用于外部直接操作）。</summary>
102    public T? GetPanel<T>(string name) where T : Control
103    {
104        return _panels.TryGetValue(name, out var panel) ? panel as T : null;
105    }
106
107    public Control? GetPanel(string name)
108    {
109        return _panels.TryGetValue(name, out var panel) ? panel : null;
110    }
111}

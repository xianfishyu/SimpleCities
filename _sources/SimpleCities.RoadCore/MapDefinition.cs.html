1namespace SimpleCities.RoadCore;

3/// <summary>固定米制地图参数；调整格长需要创建另一张地图。</summary>
4public sealed record MapDefinition
5{
6    public MapDefinition(int cellSizeMetres = 100)
7    {
8        if (cellSizeMetres is not (25 or 50 or 100 or 200))
9            throw new ArgumentOutOfRangeException(nameof(cellSizeMetres), "Cell size must be 25, 50, 100 or 200 metres.");
10        CellSizeMetres = cellSizeMetres;
11    }
12
13    public int CellSizeMetres { get; }
14    public int MinimumMetres => -4000;
15    public int MaximumMetres => 4000;
16
17    public bool IsPrimaryPoint(RoadPoint point) => point.IsFinite &&
18        point.X >= MinimumMetres && point.X <= MaximumMetres && point.Y >= MinimumMetres && point.Y <= MaximumMetres &&
19        point.X % CellSizeMetres == 0 && point.Y % CellSizeMetres == 0;
20
21    public bool IsCellCenter(RoadPoint point) => point.IsFinite &&
22        point.X >= MinimumMetres && point.X <= MaximumMetres && point.Y >= MinimumMetres && point.Y <= MaximumMetres &&
23        Math.Abs(point.X % CellSizeMetres) == CellSizeMetres / 2d &&
24        Math.Abs(point.Y % CellSizeMetres) == CellSizeMetres / 2d;
25
26    public bool IsBuildPoint(RoadPoint point) => IsPrimaryPoint(point) || IsCellCenter(point);
27
28    internal bool IsBuildSegment(RoadPoint a, RoadPoint b) =>
29        IsBuildPoint(a) && IsBuildPoint(b) && a != b && IsEightDirection(a, b) &&
30        (!(IsCellCenter(a) || IsCellCenter(b)) || Math.Abs(b.X - a.X) == Math.Abs(b.Y - a.Y));
31
32    internal bool IsEightDirection(RoadPoint a, RoadPoint b) =>
33        a.X == b.X || a.Y == b.Y || Math.Abs(b.X - a.X) == Math.Abs(b.Y - a.Y);
34
35    public RoadPoint SnapPrimary(RoadPoint point) => new(
36        Math.Clamp(Math.Round(point.X / CellSizeMetres, MidpointRounding.AwayFromZero) * CellSizeMetres, MinimumMetres, MaximumMetres),
37        Math.Clamp(Math.Round(point.Y / CellSizeMetres, MidpointRounding.AwayFromZero) * CellSizeMetres, MinimumMetres, MaximumMetres));
38
39    /// <summary>选择地图内最近的主格点或格心；两组格点等距时优先主格点。</summary>
40    public RoadPoint SnapBuildPoint(RoadPoint point)
41    {
42        if (!point.IsFinite) return point;
43        RoadPoint primary = SnapPrimary(point);
44        double half = CellSizeMetres / 2d;
45        var center = new RoadPoint(
46            Math.Clamp(Math.Round((point.X - half) / CellSizeMetres, MidpointRounding.AwayFromZero) * CellSizeMetres + half,
47                MinimumMetres + half, MaximumMetres - half),
48            Math.Clamp(Math.Round((point.Y - half) / CellSizeMetres, MidpointRounding.AwayFromZero) * CellSizeMetres + half,
49                MinimumMetres + half, MaximumMetres - half));
50        double primaryDistance = Math.Pow(primary.X - point.X, 2) + Math.Pow(primary.Y - point.Y, 2);
51        double centerDistance = Math.Pow(center.X - point.X, 2) + Math.Pow(center.Y - point.Y, 2);
52        return primaryDistance <= centerDistance ? primary : center;
53    }
54
55    /// <summary>按最近八方向投影；后续核心校验方向合法性，不将格心横竖拖动偷偷改为对角。</summary>
56    public RoadPoint SnapDragEnd(RoadPoint start, RoadPoint cursor)
57    {
58        if (!IsBuildPoint(start) || !cursor.IsFinite)
59            return start;
60        double dx = cursor.X - start.X;
61        double dy = cursor.Y - start.Y;
62        double angle = Math.Atan2(dy, dx);
63        int direction = (int)Math.Round(angle / (Math.PI / 4));
64        int sx = Math.Sign(Math.Round(Math.Cos(direction * Math.PI / 4)));
65        int sy = Math.Sign(Math.Round(Math.Sin(direction * Math.PI / 4)));
66        double step = sx != 0 && sy != 0 ? CellSizeMetres / 2d : CellSizeMetres;
67        double projectedSteps = (dx * sx + dy * sy) / ((sx * sx + sy * sy) * step);
68        double maximum = double.PositiveInfinity;
69        if (sx != 0) maximum = Math.Min(maximum, (sx > 0 ? MaximumMetres - start.X : start.X - MinimumMetres) / step);
70        if (sy != 0) maximum = Math.Min(maximum, (sy > 0 ? MaximumMetres - start.Y : start.Y - MinimumMetres) / step);
71        double steps = Math.Clamp(Math.Round(projectedSteps, MidpointRounding.AwayFromZero), 0, Math.Floor(maximum));
72        return new RoadPoint(start.X + sx * steps * step, start.Y + sy * steps * step);
73    }
74}

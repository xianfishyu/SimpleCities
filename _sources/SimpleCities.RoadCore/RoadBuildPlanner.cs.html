5/// <summary>在私有内容中切分点交叉并恢复规范边，完成验证后才产生可发布计划。</summary>
6internal static class RoadBuildPlanner
7{
8
9    internal static RoadBuildResult Plan(RoadNetwork owner, RoadSnapshot source, RoadBuildRequest request,
10        CancellationToken cancellationToken)
11    {
12        var draftCuts = new HashSet<RoadPoint> { request.Start, request.End };
13        var edgeCuts = new Dictionary<EdgeId, HashSet<RoadPoint>>();
14        foreach (RoadEdge edge in source.Edges)
15        {
16            HashSet<RoadPoint>? cuts = null;
17            for (int i = 1; i < edge.Points.Count; i++)
18            {
19                cancellationToken.ThrowIfCancellationRequested();
20                RoadPoint? intersection = Intersect(request.Start, request.End, edge.Points[i - 1], edge.Points[i]);
21                if (!intersection.HasValue) continue;
22                if (!source.Map.IsBuildPoint(intersection.Value))
23                    throw new InvalidDataException("道路交叉位置必须是地图内的主格点或格心");
24                draftCuts.Add(intersection.Value);
25                // Endpoint connections still split the new road, but do not split
26                // the old edge. Only interior cuts need a mutable cutting set.
27                if (intersection.Value == edge.Points[0] || intersection.Value == edge.Points[^1]) continue;
28                cuts ??= new HashSet<RoadPoint> { edge.Points[0], edge.Points[^1] };
29                cuts.Add(intersection.Value);
30            }
31            if (cuts is not null) edgeCuts.Add(edge.Id, cuts);
32        }
33
34        long nextNode = source.NextNodeId, nextEdge = source.NextEdgeId;
35        var nodes = source.Nodes.ToDictionary(node => node.Position);
36        // 端点优先，再沿草稿顺序分配交点身份；不依赖集合的枚举顺序。
37        IEnumerable<RoadPoint> allocationOrder = new[] { request.Start, request.End }.Concat(draftCuts
38            .Where(point => point != request.Start && point != request.End)
39            .OrderBy(point => Parameter(point, request.Start, request.End)));
40        foreach (RoadPoint position in allocationOrder)
41        {
42            cancellationToken.ThrowIfCancellationRequested();
43            if (!nodes.ContainsKey(position))
44                nodes.Add(position, new RoadNode(new NodeId(Allocate(ref nextNode)), position));
45        }
46        var pieces = new List<Piece>(Math.Max(4, source.EdgeCount + 1));
47        foreach (RoadEdge edge in source.Edges)
48        {
49            cancellationToken.ThrowIfCancellationRequested();
50            if (edgeCuts.TryGetValue(edge.Id, out HashSet<RoadPoint>? cuts))
51                Split(edge.Id, edge.Profile, edge.Points, cuts, nodes, pieces, cancellationToken);
52            else
53                pieces.Add(new(edge.Id, edge.Start, edge.End, edge.Profile, edge.Points));
54        }
55        Split(null, request.Profile, [request.Start, request.End], draftCuts, nodes, pieces, cancellationToken);
56        Normalize(nodes, pieces, cancellationToken);
57
58        return new(RoadBuildStatus.Ready, Freeze(owner, source, nodes, pieces, nextNode, nextEdge, cancellationToken), "");
59    }
60
61    private static RoadPoint? Intersect(RoadPoint a, RoadPoint b, RoadPoint c, RoadPoint d)
62    {
63        double rx = b.X - a.X, ry = b.Y - a.Y, sx = d.X - c.X, sy = d.Y - c.Y;
64        double denominator = Cross(rx, ry, sx, sy);
65        if (denominator == 0)
66        {
67            // 调用入口已优先拒绝正长度共线重叠，此处仅余端点触碰。
68            if (OnSegment(a, c, d)) return a;
69            if (OnSegment(b, c, d)) return b;
70            return null;
71        }
72        double numerator = Cross(c.X - a.X, c.Y - a.Y, sx, sy);
73        double other = Cross(c.X - a.X, c.Y - a.Y, rx, ry);
74        if (denominator > 0
75            ? numerator < 0 || numerator > denominator || other < 0 || other > denominator
76            : numerator > 0 || numerator < denominator || other > 0 || other < denominator)
77            return null;
78        // 四档格长在 ±4000 m 内仅产生整数或半整数米坐标，行列式及乘加均可精确表示。
79        // 先相乘相加后相除，避免 t 插值引入格点误差。
80        return new RoadPoint((a.X * denominator + rx * numerator) / denominator,
81            (a.Y * denominator + ry * numerator) / denominator);
82    }

27/// <summary>V4 主格点、格心及闭环路网 schema 6；有界 stream 入口，不迁移旧调试格式。</summary>
28public static class RoadCodec
29{
30    public const int MaximumPayloadBytes = 32 * 1024 * 1024;
31    public const int SchemaVersion = 6;
32
33    public static void Write(Stream destination, RoadSnapshot snapshot)
34    {
35        ArgumentNullException.ThrowIfNull(snapshot);
36        using var writer = new Utf8JsonWriter(destination);
37        writer.WriteStartObject();
38        writer.WriteString("formatFamily", "simple-cities-v4");
39        writer.WriteString("payloadType", "road-network");
40        writer.WriteNumber("schemaVersion", SchemaVersion);
41        writer.WriteNumber("contentRevision", snapshot.Token.ContentRevision);
42        writer.WriteNumber("nextNodeId", snapshot.NextNodeId);
43        writer.WriteNumber("nextEdgeId", snapshot.NextEdgeId);
44        writer.WriteNumber("profileCatalogVersion", 1);
45        writer.WriteStartObject("map");
46        writer.WriteNumber("widthMetres", 8000);
47        writer.WriteNumber("heightMetres", 8000);
48        writer.WriteString("origin", "center");
49        writer.WriteNumber("metresPerUnit", 1);
50        writer.WriteString("grid", "square-eight");
51        writer.WriteNumber("cellSizeMetres", snapshot.Map.CellSizeMetres);
52        writer.WriteEndObject();
53        writer.WriteStartArray("nodes");
54        foreach (RoadNode node in snapshot.Nodes.OrderBy(node => node.Id.Value))
55        {
56            writer.WriteStartObject();
57            writer.WriteNumber("id", node.Id.Value);
58            writer.WriteNumber("x", node.Position.X);
59            writer.WriteNumber("y", node.Position.Y);
60            writer.WriteEndObject();
61        }
62        writer.WriteEndArray();
63        writer.WriteStartArray("edges");
64        foreach (RoadEdge edge in snapshot.Edges.OrderBy(edge => edge.Id.Value))
65        {
66            writer.WriteStartObject();
67            writer.WriteNumber("id", edge.Id.Value);
68            writer.WriteNumber("startNodeId", edge.Start.Value);
69            writer.WriteNumber("endNodeId", edge.End.Value);
70            writer.WriteString("profile", edge.Profile.Value);
71            writer.WriteStartArray("points");
72            foreach (RoadPoint point in edge.Points)
73            {
74                writer.WriteStartObject();
75                writer.WriteNumber("x", point.X);
76                writer.WriteNumber("y", point.Y);
77                writer.WriteEndObject();
78            }
79            writer.WriteEndArray();
80            writer.WriteEndObject();
81        }
82        writer.WriteEndArray();
83        writer.WriteEndObject();
84        writer.Flush();
85    }

87    public static PreparedRoadState Read(Stream source)
88    {
89        ArgumentNullException.ThrowIfNull(source);
90        // Grow only with actual input; small saves do not reserve the full admission budget.
91        byte[] bytes = new byte[4096];
92        int count = 0;
93        while (count < MaximumPayloadBytes)
94        {
95            if (count == bytes.Length)
96                Array.Resize(ref bytes, Math.Min(bytes.Length * 2, MaximumPayloadBytes));
97            int read = source.Read(bytes, count, bytes.Length - count);
98            if (read == 0)
99                break;
100            count += read;
101        }
102        // Exactly one sentinel byte detects excess, including non-seekable streams.
103        if (count == MaximumPayloadBytes && source.ReadByte() != -1)
104            throw new InvalidDataException($"V4 road payload exceeds {MaximumPayloadBytes} bytes.");
105        using JsonDocument document = JsonDocument.Parse(bytes.AsMemory(0, count),
106            new JsonDocumentOptions { MaxDepth = 6 });
107        JsonElement root = document.RootElement;
108        Fields(root, "formatFamily", "payloadType", "schemaVersion", "contentRevision",
109            "nextNodeId", "nextEdgeId", "profileCatalogVersion", "map", "nodes", "edges");
110        RequireText(root, "formatFamily", "simple-cities-v4");
111        RequireText(root, "payloadType", "road-network");
112        RequireNumber(root, "schemaVersion", SchemaVersion);
113        RequireNumber(root, "profileCatalogVersion", 1);
114        JsonElement map = root.GetProperty("map");
115        Fields(map, "widthMetres", "heightMetres", "origin", "metresPerUnit", "grid", "cellSizeMetres");
116        RequireNumber(map, "widthMetres", 8000);
117        RequireNumber(map, "heightMetres", 8000);
118        RequireText(map, "origin", "center");
119        RequireNumber(map, "metresPerUnit", 1);
120        RequireText(map, "grid", "square-eight");
121        long cell = Number(map, "cellSizeMetres");
122        if (cell is not (25 or 50 or 100 or 200))
123            throw new InvalidDataException("V4 cellSizeMetres must be 25, 50, 100 or 200.");
124        var definition = new MapDefinition((int)cell);
125        long nextNode = Positive(root, "nextNodeId");
126        long nextEdge = Positive(root, "nextEdgeId");
127        JsonElement nodesArray = root.GetProperty("nodes");
128        JsonElement edgesArray = root.GetProperty("edges");
129        if (nodesArray.ValueKind != JsonValueKind.Array || edgesArray.ValueKind != JsonValueKind.Array ||
130            nodesArray.GetArrayLength() > RoadTopology.MaximumNodes ||
131            edgesArray.GetArrayLength() > RoadTopology.MaximumEdges)
132            throw new InvalidDataException("V4 nodes or edges exceed the supported road topology budget.");
133        var nodes = new List<RoadNode>();
134        foreach (JsonElement node in nodesArray.EnumerateArray())
135        {
136            Fields(node, "id", "x", "y");
137            long id = Positive(node, "id");
138            var point = new RoadPoint(Coordinate(node, "x"), Coordinate(node, "y"));
139            if (id >= nextNode || (nodes.Count != 0 && id <= nodes[^1].Id.Value))
140                throw new InvalidDataException("V4 nodes must have sorted unique IDs below the watermark.");
141            nodes.Add(new RoadNode(new NodeId(id), point));
142        }
143        var edges = new List<RoadEdge>();
144        int pointCount = 0;
145        foreach (JsonElement edge in edgesArray.EnumerateArray())
146        {
147            Fields(edge, "id", "startNodeId", "endNodeId", "profile", "points");
148            long id = Positive(edge, "id");
149            long startId = Positive(edge, "startNodeId");
150            long endId = Positive(edge, "endNodeId");
151            JsonElement profile = edge.GetProperty("profile");
152            string? name = profile.ValueKind == JsonValueKind.String ? profile.GetString() : null;
153            if (id >= nextEdge || (edges.Count != 0 && id <= edges[^1].Id.Value) ||
154                name is not ("dirt" or "street" or "arterial" or "highway"))
155                throw new InvalidDataException("V4 edges must have sorted unique IDs below the watermark and a known profile.");
156            JsonElement pointsArray = edge.GetProperty("points");
157            if (pointsArray.ValueKind != JsonValueKind.Array || pointsArray.GetArrayLength() < 2 ||
158                pointsArray.GetArrayLength() > RoadTopology.MaximumPoints - pointCount)
159                throw new InvalidDataException("V4 edge points must contain endpoints and fit the road topology budget.");
160            var points = new List<RoadPoint>(pointsArray.GetArrayLength());
161            foreach (JsonElement point in pointsArray.EnumerateArray())
162            {
163                Fields(point, "x", "y");
164                points.Add(new RoadPoint(Coordinate(point, "x"), Coordinate(point, "y")));
165            }
166            pointCount += points.Count;
167            edges.Add(new RoadEdge(new EdgeId(id), new NodeId(startId), new NodeId(endId), new RoadProfileId(name), points));
168        }
169        RoadTopology.Validate(definition, nodes, edges, nextNode, nextEdge);
170        return new PreparedRoadState(definition, Positive(root, "contentRevision"), nextNode, nextEdge, nodes, edges);
171    }

173    private static double Coordinate(JsonElement value, string name)
174    {
175        JsonElement field = value.GetProperty(name);
176        if (field.ValueKind != JsonValueKind.Number || !field.TryGetDouble(out double number) || !double.IsFinite(number))
177            throw new InvalidDataException($"V4 coordinate '{name}' must be finite.");
178        return number;
179    }
180
181    private static void Fields(JsonElement value, params string[] names)
182    {
183        if (value.ValueKind != JsonValueKind.Object)
184            throw new InvalidDataException("Expected a V4 object.");
185        var remaining = new HashSet<string>(names, StringComparer.Ordinal);
186        foreach (JsonProperty property in value.EnumerateObject())
187            if (!remaining.Remove(property.Name))
188                throw new InvalidDataException($"Unknown or duplicate V4 field '{property.Name}'.");
189        if (remaining.Count != 0)
190            throw new InvalidDataException($"Missing V4 field '{remaining.First()}'.");
191    }
192
193    private static long Number(JsonElement value, string name)
194    {
195        JsonElement field = value.GetProperty(name);
196        if (field.ValueKind != JsonValueKind.Number || !field.TryGetInt64(out long number))
197            throw new InvalidDataException($"V4 field '{name}' must be an integer.");
198        return number;
199    }
200
201    private static long Positive(JsonElement value, string name)
202    {
203        long number = Number(value, name);
204        if (number <= 0 || number == long.MaxValue)
205            throw new InvalidDataException($"V4 field '{name}' must be positive and leave allocation headroom.");
206        return number;
207    }
208
209    private static void RequireNumber(JsonElement value, string name, long expected)
210    {
211        if (Number(value, name) != expected)
212            throw new InvalidDataException($"Unsupported V4 {name}; expected {expected}.");
213    }
214
215    private static void RequireText(JsonElement value, string name, string expected)
216    {
217        JsonElement field = value.GetProperty(name);
218        if (field.ValueKind != JsonValueKind.String || field.GetString() != expected)
219            throw new InvalidDataException($"Unsupported V4 {name}; expected '{expected}'.");
220    }

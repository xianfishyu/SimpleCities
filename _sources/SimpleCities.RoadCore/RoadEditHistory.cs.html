1namespace SimpleCities.RoadCore;

3/// <summary>当前会话的不可变领域变更历史；撤销与重做共用最多 64 条记录。</summary>
4public sealed class RoadEditHistory
5{
6    public const int Capacity = 64;
7    private readonly record struct Entry(RoadChangeSet Change, long EstimatedBytes);
8    private readonly Entry[] _changes;
9
10    private RoadEditHistory(Entry[] changes, int cursor)
11    {
12        _changes = changes;
13        UndoCount = cursor;
14        // Deltas are immutable. Cursor moves and trimming sum at most 64 cached
15        // scalars; they must not rescan all retained entity payloads without cancellation.
16        EstimatedBytes = changes.Sum(entry => entry.EstimatedBytes);
17    }
18
19    internal static RoadEditHistory Empty { get; } = new([], 0);
20    public int UndoCount { get; }
21    public int RedoCount => _changes.Length - UndoCount;
22    public int RetainedCount => _changes.Length;
23
24    /// <summary>
25    /// 领域载荷估算，非 .NET 堆实测：每条 192 字节元数据，每个变更槽 32 字节，
26    /// 节点每端 48 字节，边每端 80 字节加每个几何点 16 字节。
27    /// 跨条目共享实体重复计数；不包含活动快照、派生索引或绘图资源，不作为准入预算。
28    /// </summary>
29    public long EstimatedBytes { get; }
30
31    internal RoadChangeSet? UndoChange => UndoCount == 0 ? null : _changes[UndoCount - 1].Change;
32    internal RoadChangeSet? RedoChange => RedoCount == 0 ? null : _changes[UndoCount].Change;
33    internal RoadEditHistory Move(bool undo) => new(_changes, UndoCount + (undo ? -1 : 1));
34
35    internal RoadEditHistory Append(RoadChangeSet change)
36    {
37        int first = Math.Max(0, UndoCount + 1 - Capacity);
38        Entry[] retained = _changes.Skip(first).Take(UndoCount - first).Append(new Entry(change, Estimate(change))).ToArray();
39        return new(retained, retained.Length);
40    }
41
42    private static long Estimate(RoadChangeSet change) => 192L +
43        change.Nodes.Sum(node => 32L + (node.Before is null ? 0 : 48) + (node.After is null ? 0 : 48)) +
44        change.Edges.Sum(edge => 32L + EdgeBytes(edge.Before) + EdgeBytes(edge.After));
45
46    private static long EdgeBytes(RoadEdge? edge) => edge is null ? 0 : 80L + 16L * edge.Points.Count;
47}

9/// <summary>带来源版本的端接和几何转向；不表达交通许可、信号或容量。</summary>
10public sealed class RoadJunctionReadModel
11{
12    internal RoadJunctionReadModel(RoadStateToken source, RoadNode node, IEnumerable<RoadIncidence> incidences, IEnumerable<RoadTurnMovement> turns)
13    {
14        Source = source;
15        Node = node;
16        Incidences = Array.AsReadOnly(incidences.ToArray());
17        Turns = Array.AsReadOnly(turns.ToArray());
18    }
19    public RoadStateToken Source { get; }
20    public RoadNode Node { get; }
21    public IReadOnlyList<RoadIncidence> Incidences { get; }
22    public IReadOnlyList<RoadTurnMovement> Turns { get; }
23}
24
25public static class RoadJunctionQuery
26{
27    public static RoadJunctionReadModel? Read(RoadSnapshot snapshot, NodeId nodeId)
28    {
29        RoadNode? node = snapshot.QueryData.FindNode(nodeId);
30        if (node is null) return null;
31        var incidences = new List<RoadIncidence>();
32        foreach (RoadEdge edge in snapshot.QueryData.Incident(nodeId))
33        {
34            if (edge.Start == nodeId) Add(edge, RoadEndRole.Start, edge.Points[1]);
35            if (edge.End == nodeId) Add(edge, RoadEndRole.End, edge.Points[^2]);
36        }
37        RoadIncidence[] ordered = incidences.OrderBy(item => item.BearingRadians)
38            .ThenBy(item => item.Key.Edge.Value).ThenBy(item => item.Key.Role).ToArray();
39        var turns = new List<RoadTurnMovement>();
40        foreach (RoadIncidence from in ordered)
41            foreach (RoadIncidence to in ordered)
42            {
43                // Incoming travel points toward the node; outgoing points away from it.
44                double x = -from.Outward.X, y = -from.Outward.Y;
45                double angle = Math.Atan2(x * to.Outward.Y - y * to.Outward.X, x * to.Outward.X + y * to.Outward.Y);
46                if (angle == -Math.PI) angle = Math.PI;
47                turns.Add(new RoadTurnMovement(from.Key, to.Key, angle));
48            }
49        return new RoadJunctionReadModel(snapshot.Token, node, ordered, turns);
50
51        void Add(RoadEdge edge, RoadEndRole role, RoadPoint next)
52        {
53            double length = node.Position.DistanceTo(next);
54            var outward = new RoadVector((next.X - node.Position.X) / length, (next.Y - node.Position.Y) / length);
55            double bearing = Math.Atan2(outward.Y, outward.X);
56            if (bearing < 0) bearing += Math.Tau;
57            incidences.Add(new RoadIncidence(new RoadIncidenceKey(edge.Id, role), edge.Profile, outward, bearing));
58        }
59    }

46public sealed record RoadEdge
47{
48    public RoadEdge(EdgeId id, NodeId start, NodeId end, RoadProfileId profile, IEnumerable<RoadPoint> points)
49    {
50        Id = id;
51        Start = start;
52        End = end;
53        Profile = profile;
54        RoadPoint[] ownedPoints = points.ToArray();
55        Points = Array.AsReadOnly(ownedPoints);
56        // Points are owned and immutable. Keep the original left-to-right sum,
57        // but compute it once instead of allocating a LINQ pipeline per read.
58        double length = 0;
59        for (int i = 1; i < ownedPoints.Length; i++) length += ownedPoints[i - 1].DistanceTo(ownedPoints[i]);
60        Length = length;
61    }
62    public EdgeId Id { get; }
63    public NodeId Start { get; }
64    public NodeId End { get; }
65    public RoadProfileId Profile { get; }
66    public IReadOnlyList<RoadPoint> Points { get; }
67    public double Length { get; }
68
69    public RoadPoint PointAt(double parameter)
70    {
71        double distance = parameter * Length;
72        for (int i = 1; i < Points.Count; i++)
73        {
74            RoadPoint a = Points[i - 1], b = Points[i];
75            double length = a.DistanceTo(b);
76            if (distance <= length || i == Points.Count - 1)
77            {
78                double t = Math.Clamp(distance / length, 0, 1);
79                return new RoadPoint(a.X + (b.X - a.X) * t, a.Y + (b.Y - a.Y) * t);
80            }
81            distance -= length;
82        }
83        return Points[^1];
84    }
85}

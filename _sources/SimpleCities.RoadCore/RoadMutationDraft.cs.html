10    internal static RoadPlan Freeze(RoadNetwork owner, RoadSnapshot source, Dictionary<RoadPoint, RoadNode> nodes,
11        List<Piece> pieces, long nextNode, long nextEdge, CancellationToken cancellationToken)
12    {
13        var edges = new List<RoadEdge>(pieces.Count);
14        foreach (Piece piece in pieces)
15        {
16            cancellationToken.ThrowIfCancellationRequested();
17            EdgeId id = piece.Id ?? new EdgeId(Allocate(ref nextEdge));
18            bool forward = piece.Start.Value < piece.End.Value ||
19                (piece.Start == piece.End && RoadTopology.IsCanonicalLoopDirection(piece.Points));
20            RoadEdge? previous = source.FindEdge(id);
21            if (previous is not null && previous.Start == (forward ? piece.Start : piece.End) &&
22                previous.End == (forward ? piece.End : piece.Start) && previous.Profile == piece.Profile &&
23                SamePoints(previous.Points, piece.Points, forward))
24            {
25                edges.Add(previous);
26                continue;
27            }
28            edges.Add(forward
29                ? new RoadEdge(id, piece.Start, piece.End, piece.Profile, piece.Points)
30                : new RoadEdge(id, piece.End, piece.Start, piece.Profile, piece.Points.AsEnumerable().Reverse()));
31        }
32        RoadNode[] orderedNodes = nodes.Values.OrderBy(node => node.Id.Value).ToArray();
33        edges.Sort((a, b) => a.Id.Value.CompareTo(b.Id.Value));
34        RoadTopology.Validate(source.Map, orderedNodes, edges, nextNode, nextEdge, cancellationToken);
35        RoadStateToken token = source.Token with
36        {
37            ContentRevision = owner.NextContentRevision(source),
38            ChangeSequence = source.Token.ChangeSequence + 1,
39        };
40        var target = new RoadSnapshot(source.Map, token, nextNode, nextEdge, orderedNodes, edges,
41            source, cancellationToken);
42        cancellationToken.ThrowIfCancellationRequested();
43        var plan = new RoadPlan(owner, source, target, checked((int)(nextNode - source.NextNodeId)),
44            checked((int)(nextEdge - source.NextEdgeId)), cancellationToken);
45        cancellationToken.ThrowIfCancellationRequested();
46        return plan;
47    }

49    private static bool SamePoints(IReadOnlyList<RoadPoint> before, IReadOnlyList<RoadPoint> after, bool forward)
50    {
51        if (forward && ReferenceEquals(before, after)) return true;
52        if (before.Count != after.Count) return false;
53        for (int i = 0; i < before.Count; i++)
54            if (before[i] != after[forward ? i : after.Count - 1 - i]) return false;
55        return true;
56    }

58    internal static long Allocate(ref long next)
59    {
60        if (next >= long.MaxValue - 1) throw new InvalidDataException("道路身份或版本已耗尽");
61        return next++;
62    }

64    internal static void Split(EdgeId? originalId, RoadProfileId profile, IReadOnlyList<RoadPoint> points,
65        HashSet<RoadPoint> cuts, Dictionary<RoadPoint, RoadNode> nodes, List<Piece> destination,
66        CancellationToken cancellationToken)
67    {
68        var current = new List<RoadPoint> { points[0] };
69        EdgeId? id = originalId;
70        for (int i = 1; i < points.Count; i++)
71        {
72            cancellationToken.ThrowIfCancellationRequested();
73            RoadPoint a = points[i - 1], b = points[i];
74            IEnumerable<RoadPoint> ordered = cuts.Where(point => point != a && OnSegment(point, a, b))
75                .Append(b).Distinct().OrderBy(point => Parameter(point, a, b));
76            foreach (RoadPoint point in ordered)
77            {
78                current.Add(point);
79                if (!cuts.Contains(point)) continue;
80                destination.Add(new Piece(id, nodes[current[0]].Id, nodes[point].Id, profile, current));
81                id = null;
82                current = [point];
83            }
84        }
85    }

87    internal static void Normalize(Dictionary<RoadPoint, RoadNode> nodes, List<Piece> pieces,
88        CancellationToken cancellationToken)
89    {
90        cancellationToken.ThrowIfCancellationRequested();
91        if (pieces.Count == 0)
92        {
93            nodes.Clear();
94            return;
95        }
96        var incident = new Dictionary<NodeId, IncidentPieces>(nodes.Count);
97        foreach (RoadNode node in nodes.Values)
98        {
99            cancellationToken.ThrowIfCancellationRequested();
100            incident.Add(node.Id, default);
101        }
102        for (int i = 0; i < pieces.Count; i++)
103        {
104            cancellationToken.ThrowIfCancellationRequested();
105            // Endpoints were allocated before the pieces; the map cannot grow here.
106            CollectionsMarshal.GetValueRefOrNullRef(incident, pieces[i].Start).Add(i);
107            CollectionsMarshal.GetValueRefOrNullRef(incident, pieces[i].End).Add(i);
108        }
109        List<RoadPoint>? unused = null;
110        var candidates = new List<RoadNode>();
111        foreach (RoadNode node in nodes.Values)
112        {
113            cancellationToken.ThrowIfCancellationRequested();
114            IncidentPieces ends = incident[node.Id];
115            if (ends.Count == 0)
116            {
117                (unused ??= []).Add(node.Position);
118                continue;
119            }
120            if (ends.Count == 2 && !ReferenceEquals(pieces[ends.First], pieces[ends.Second]) &&
121                pieces[ends.First].Profile == pieces[ends.Second].Profile)
122                candidates.Add(node);
123        }
124        if (unused is not null)
125            foreach (RoadPoint position in unused)
126            {
127                cancellationToken.ThrowIfCancellationRequested();
128                incident.Remove(nodes[position].Id);
129                nodes.Remove(position);
130            }
131        if (candidates.Count == 0) return;
132        // Contraction preserves every other node's degree and profiles. It cannot
133        // create a new candidate, but can turn one into a non-removable self-loop.
134        // Higher IDs disappear first, retaining the smallest seam of an isolated ring.
135        candidates.Sort((a, b) => b.Id.Value.CompareTo(a.Id.Value));
136        var work = new List<Piece?>(checked(pieces.Count * 2));
137        work.AddRange(pieces);
138        pieces.Clear();
139        foreach (RoadNode removable in candidates)
140        {
141            cancellationToken.ThrowIfCancellationRequested();
142            IncidentPieces ends = incident[removable.Id];
143            Piece a = work[ends.First] ?? throw new InvalidOperationException("Missing normalization incidence.");
144            Piece b = work[ends.Second] ?? throw new InvalidOperationException("Missing normalization incidence.");
145            if (ReferenceEquals(a, b)) continue;
146            NodeId start = a.Start == removable.Id ? a.End : a.Start;
147            NodeId end = b.Start == removable.Id ? b.End : b.Start;
148            IEnumerable<RoadPoint> first = a.End == removable.Id ? a.Points : a.Points.AsEnumerable().Reverse();
149            IEnumerable<RoadPoint> second = b.Start == removable.Id ? b.Points : b.Points.AsEnumerable().Reverse();
150            var joined = new List<RoadPoint>();
151            foreach (RoadPoint point in first.Concat(second.Skip(1)))
152            {
153                cancellationToken.ThrowIfCancellationRequested();
154                while (joined.Count >= 2 && RoadTopology.IsForwardCollinear(joined[^2], joined[^1], point))
155                    joined.RemoveAt(joined.Count - 1);
156                joined.Add(point);
157            }
158            EdgeId? retained = a.Id.HasValue && b.Id.HasValue
159                ? (a.Id.Value.Value < b.Id.Value.Value ? a.Id : b.Id) : a.Id ?? b.Id;
160            int replacement = work.Count;
161            work[ends.First] = null;
162            work[ends.Second] = null;
163            work.Add(new Piece(retained, start, end, a.Profile, joined));
164            CollectionsMarshal.GetValueRefOrNullRef(incident, start).Replace(ends.First, ends.Second, replacement);
165            if (end != start)
166                CollectionsMarshal.GetValueRefOrNullRef(incident, end).Replace(ends.First, ends.Second, replacement);
167            incident.Remove(removable.Id);
168            nodes.Remove(removable.Position);
169        }
170        // Surviving original pieces retain their order; merged pieces were appended
171        // in the same order as before. Freeze therefore assigns the same new IDs.
172        foreach (Piece? piece in work)
173        {
174            cancellationToken.ThrowIfCancellationRequested();
175            if (piece is not null) pieces.Add(piece);
176        }
177    }

179    private struct IncidentPieces
180    {
181        internal int Count;
182        internal int First;
183        internal int Second;
184        internal void Add(int index)
185        {
186            if (Count == 0) First = index;
187            else if (Count == 1) Second = index;
188            Count++;
189        }
190        internal void Replace(int a, int b, int replacement)
191        {
192            // Only degree-two nodes can be considered later. Their degree and
193            // profiles do not change; keep the newly appended piece last.
194            if (Count != 2) return;
195            if (First == a || First == b) First = replacement;
196            if (Second == a || Second == b) Second = replacement;
197            if (First > Second) (First, Second) = (Second, First);
198        }
199    }

201    internal static double Parameter(RoadPoint p, RoadPoint a, RoadPoint b) =>
202        a.X != b.X ? (p.X - a.X) / (b.X - a.X) : (p.Y - a.Y) / (b.Y - a.Y);
203
204    internal static bool OnSegment(RoadPoint p, RoadPoint a, RoadPoint b) =>
205        Cross(p.X - a.X, p.Y - a.Y, b.X - a.X, b.Y - a.Y) == 0 &&
206        p.X >= Math.Min(a.X, b.X) && p.X <= Math.Max(a.X, b.X) &&
207        p.Y >= Math.Min(a.Y, b.Y) && p.Y <= Math.Max(a.Y, b.Y);
208
209    internal static double Cross(double ax, double ay, double bx, double by) => ax * by - ay * bx;

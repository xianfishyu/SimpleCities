6/// <summary>只读权威内容，规范道路边可跨多个直线格段和转弯。</summary>
7public sealed class RoadSnapshot
8{
9    internal RoadSnapshot(MapDefinition map, RoadStateToken token, long nextNodeId = 1, long nextEdgeId = 1,
10        IEnumerable<RoadNode>? nodes = null, IEnumerable<RoadEdge>? edges = null,
11        RoadSnapshot? previous = null, CancellationToken cancellationToken = default)
12    {
13        cancellationToken.ThrowIfCancellationRequested();
14        Map = map;
15        Token = token;
16        NextNodeId = nextNodeId;
17        NextEdgeId = nextEdgeId;
18        Nodes = Array.AsReadOnly(nodes?.ToArray() ?? []);
19        cancellationToken.ThrowIfCancellationRequested();
20        Edges = Array.AsReadOnly(edges?.ToArray() ?? []);
21        QueryData = new RoadSnapshotQueryData(this, previous?.QueryData, cancellationToken);
22    }
23
24    private RoadSnapshot(RoadSnapshot content, RoadStateToken token, long nextNodeId, long nextEdgeId)
25    {
26        Map = content.Map;
27        Token = token;
28        NextNodeId = nextNodeId;
29        NextEdgeId = nextEdgeId;
30        Nodes = content.Nodes;
31        Edges = content.Edges;
32        QueryData = content.QueryData;
33    }
34
35    internal RoadSnapshot Republish(RoadStateToken token, long nextNodeId, long nextEdgeId) =>
36        new(this, token, nextNodeId, nextEdgeId);
37
38    public MapDefinition Map { get; }
39    public RoadStateToken Token { get; }
40    public long NextNodeId { get; }
41    public long NextEdgeId { get; }
42    public IReadOnlyList<RoadNode> Nodes { get; }
43    public IReadOnlyList<RoadEdge> Edges { get; }
44    public int NodeCount => Nodes.Count;
45    public int EdgeCount => Edges.Count;
46    internal RoadSnapshotQueryData QueryData { get; }
47
48    public RoadEdge? FindEdge(EdgeId id) => QueryData.FindEdge(id);

50    public RoadPoint? Resolve(RoadLocation location)
51    {
52        if (location.Source != Token || !double.IsFinite(location.Parameter) || location.Parameter < 0 || location.Parameter > 1)
53            return null;
54        return QueryData.Resolve(location);

74    internal long NextContentRevision(RoadSnapshot source)
75    {
76        RoadPublishedState published = _published;
77        long watermark = ReferenceEquals(source, published.Snapshot)
78            ? published.RevisionWatermark : source.Token.ContentRevision;
79        if (watermark >= long.MaxValue - 1)
80            throw new InvalidDataException("道路身份或版本已耗尽");
81        return watermark + 1;
82    }

87    private RoadEditResult PlanHistory(bool undo, CancellationToken cancellationToken)
88    {
89        cancellationToken.ThrowIfCancellationRequested();
90        RoadPublishedState published = _published;
91        RoadSnapshot before = published.Snapshot;
92        RoadChangeSet? change = undo ? published.History.UndoChange : published.History.RedoChange;
93        if (change is null) return new(RoadEditStatus.NoChange, null, "");
94        RoadChangeMetadata expected = undo ? change.After : change.Before;
95        if (expected.Token.NetworkInstance != before.Token.NetworkInstance || expected.Token.Lineage != before.Token.Lineage ||
96            expected.Token.ContentRevision != before.Token.ContentRevision)
97            return new(RoadEditStatus.Rejected, null, "道路历史来源已过期");
98        if (before.Token.ChangeSequence == long.MaxValue)
99            return new(RoadEditStatus.Rejected, null, "道路身份或版本已耗尽");
100
101        RoadChangeMetadata desired = undo ? change.Before : change.After;
102        RoadStateToken token = before.Token with
103        {
104            ContentRevision = desired.Token.ContentRevision,
105            ChangeSequence = before.Token.ChangeSequence + 1,
106        };
107        long nextNode = Math.Max(before.NextNodeId, desired.NextNodeId), nextEdge = Math.Max(before.NextEdgeId, desired.NextEdgeId);
108        RoadSnapshot target;
109        // A display or reader may still own the preceding content. The network
110        // itself keeps only a weak reference, never a chain of old derived data.
111        if (published.PreviousContent is not null && published.PreviousContent.TryGetTarget(out RoadSnapshot? observed) &&
112            observed.Token.NetworkInstance == token.NetworkInstance && observed.Token.Lineage == token.Lineage &&
113            observed.Token.ContentRevision == token.ContentRevision && observed.Map == desired.Map)
114        {
115            target = observed.Republish(token, nextNode, nextEdge);
116        }
117        else
118        {
119            var nodes = new Dictionary<NodeId, RoadNode>(before.NodeCount);
120            foreach (RoadNode node in before.Nodes)
121            {
122                cancellationToken.ThrowIfCancellationRequested();
123                nodes.Add(node.Id, node);
124            }
125            foreach (RoadNodeChange node in change.Nodes)
126            {
127                cancellationToken.ThrowIfCancellationRequested();
128                nodes.Remove(node.Id);
129                RoadNode? restored = undo ? node.Before : node.After;
130                if (restored is not null) nodes.Add(node.Id, restored);
131            }
132            var edges = new Dictionary<EdgeId, RoadEdge>(before.EdgeCount);
133            foreach (RoadEdge edge in before.Edges)
134            {
135                cancellationToken.ThrowIfCancellationRequested();
136                edges.Add(edge.Id, edge);
137            }
138            foreach (RoadEdgeChange edge in change.Edges)
139            {
140                cancellationToken.ThrowIfCancellationRequested();
141                edges.Remove(edge.Id);
142                RoadEdge? restored = undo ? edge.Before : edge.After;
143                if (restored is not null) edges.Add(edge.Id, restored);
144            }
145            target = new RoadSnapshot(desired.Map, token, nextNode, nextEdge,
146                nodes.Values.OrderBy(node => node.Id.Value), edges.Values.OrderBy(edge => edge.Id.Value),
147                before, cancellationToken);
148        }
149        cancellationToken.ThrowIfCancellationRequested();
150        var plan = new RoadPlan(this, published, target, published.History.Move(undo), published.RevisionWatermark,
151            new RoadChangeSet(before, target, change, undo, cancellationToken));
152        cancellationToken.ThrowIfCancellationRequested();
153        return new(RoadEditStatus.Ready, plan, "");
154    }

168    public RoadPlan PlanLoad(PreparedRoadState prepared)
169    {
170        ArgumentNullException.ThrowIfNull(prepared);
171        RoadPublishedState published = _published;
172        RoadSnapshot before = published.Snapshot;
173        var token = new RoadStateToken(before.Token.NetworkInstance, Guid.NewGuid(),
174            prepared.ContentRevision, checked(before.Token.ChangeSequence + 1));
175        return new RoadPlan(this, published, new RoadSnapshot(prepared.Map, token,
176            prepared.NextNodeId, prepared.NextEdgeId, prepared.Nodes, prepared.Edges),
177            RoadEditHistory.Empty, prepared.ContentRevision);
178    }

180    public RoadBuildResult PlanBuild(RoadBuildRequest request, CancellationToken cancellationToken = default)
181    {
182        cancellationToken.ThrowIfCancellationRequested();
183        RoadSnapshot before = Snapshot;
184        if (request.Source != before.Token)
185            return new(RoadBuildStatus.Rejected, null, "道路来源版本已过期");
186        if (!request.Profile.IsValid || !before.Map.IsBuildPoint(request.Start) || !before.Map.IsBuildPoint(request.End))
187            return new(RoadBuildStatus.Rejected, null, "请选择地图内的主格点或格心和有效道路类型");
188        if (request.Start == request.End)
189            return new(RoadBuildStatus.NoChange, null, "");
190        if (!before.Map.IsBuildSegment(request.Start, request.End))
191            return new(RoadBuildStatus.Rejected, null, "主格点道路仅支持米字网格八方向，格心仅允许对角方向");
192        IReadOnlyList<RoadConflictSpan> conflicts = RoadOverlap.Find(before, request.Start, request.End, cancellationToken);
193        if (conflicts.Count != 0)
194            return new(RoadBuildStatus.Rejected, null, "与已有道路重叠，整笔不可建造", conflicts);
195        if (before.Token.ContentRevision >= long.MaxValue - 1 || before.Token.ChangeSequence == long.MaxValue)
196            return new(RoadBuildStatus.Rejected, null, "道路身份或版本已耗尽");
197        try
198        {
199            return RoadBuildPlanner.Plan(this, before, request, cancellationToken);
200        }
201        catch (InvalidDataException exception)
202        {
203            return new(RoadBuildStatus.Rejected, null, exception.Message);
204        }
205    }

207    public bool CanCommit(RoadPlan plan) =>
208        ReferenceEquals(plan.Owner, this) && ReferenceEquals(plan.Source, Snapshot) &&
209        ReferenceEquals(plan.SourceState, _published);
210
211    /// <summary>单写者在预检后发布；过期计划无副作用，成功时仅交换引用。</summary>
212    public bool TryCommit(RoadPlan plan)
213    {
214        if (!CanCommit(plan))
215            return false;
216        _published = plan.TargetState;
217        return true;
218    }

221public sealed class RoadPlan
222{
223    internal RoadPlan(RoadNetwork owner, RoadSnapshot source, RoadSnapshot target, int addedNodeCount = 0,
224        int addedEdgeCount = 0, CancellationToken cancellationToken = default)
225    {
226        Owner = owner;
227        Source = source;
228        Target = target;
229        AddedNodeCount = addedNodeCount;
230        AddedEdgeCount = addedEdgeCount;
231        ChangeSet = new RoadChangeSet(source, target, cancellationToken);
232        SourceState = owner.Published;
233        TargetState = new(target, SourceState.History.Append(ChangeSet),
234            Math.Max(SourceState.RevisionWatermark, target.Token.ContentRevision), new(source));
235    }
236
237    internal RoadPlan(RoadNetwork owner, RoadPublishedState source, RoadSnapshot target,
238        RoadEditHistory history, long revisionWatermark, RoadChangeSet? replayedChange = null)
239    {
240        Owner = owner;
241        Source = source.Snapshot;
242        SourceState = source;
243        Target = target;
244        ChangeSet = replayedChange ?? new RoadChangeSet(Source, target);
245        TargetState = new(target, history, revisionWatermark,
246            Source.Token.NetworkInstance == target.Token.NetworkInstance && Source.Token.Lineage == target.Token.Lineage ? new(Source) : null);
247    }

250    internal RoadSnapshot Source { get; }
251    internal RoadPublishedState SourceState { get; }
252    internal RoadPublishedState TargetState { get; }

255    public RoadStateToken SourceToken => Source.Token;

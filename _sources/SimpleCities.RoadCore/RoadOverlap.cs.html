3/// <summary>只计算本次直线草稿与已发布道路的正长度共线覆盖。</summary>
4internal static class RoadOverlap
5{
6    // 调用方已验证非零合法米字网格草稿；快照中的道路也已通过拓扑验证。
7    // 按半格轴向间隔覆盖主格点和格心，8 km / 12.5 m 使差分数组最多含 641 项，不建立逐格道路实体。
8    internal static IReadOnlyList<RoadConflictSpan> Find(RoadSnapshot snapshot, RoadPoint start, RoadPoint end,
9        CancellationToken cancellationToken)
10    {
11        double dx = end.X - start.X, dy = end.Y - start.Y;
12        bool useX = dx != 0;
13        double origin = useX ? start.X : start.Y;
14        double step = Math.Sign(useX ? dx : dy) * (snapshot.Map.CellSizeMetres / 2d);
15        int count = (int)((useX ? dx : dy) / step);
16        var changes = new int[count + 1];
17        foreach (RoadEdge edge in snapshot.Edges)
18        {
19            for (int i = 1; i < edge.Points.Count; i++)
20            {
21                cancellationToken.ThrowIfCancellationRequested();
22                RoadPoint a = edge.Points[i - 1], b = edge.Points[i];
23                // 当前有界整数或半整数米坐标的乘积可由 double 精确表达，不需扩大为容差带。
24                if (dx * (b.Y - a.Y) - dy * (b.X - a.X) != 0 ||
25                    dx * (a.Y - start.Y) - dy * (a.X - start.X) != 0)
26                    continue;
27                int aIndex = (int)(((useX ? a.X : a.Y) - origin) / step);
28                int bIndex = (int)(((useX ? b.X : b.Y) - origin) / step);
29                int lower = Math.Max(0, Math.Min(aIndex, bIndex));
30                int upper = Math.Min(count, Math.Max(aIndex, bIndex));
31                if (lower >= upper) continue;
32                changes[lower]++;
33                changes[upper]--;
34            }
35        }
36
37        var conflicts = new List<RoadConflictSpan>();
38        int coverage = 0, startIndex = 0;
39        for (int i = 0; i <= count; i++)
40        {
41            cancellationToken.ThrowIfCancellationRequested();
42            int previous = coverage;
43            coverage += changes[i];
44            if (previous == 0 && coverage > 0) startIndex = i;
45            if (previous > 0 && coverage == 0)
46                conflicts.Add(new RoadConflictSpan((double)startIndex / count, (double)i / count));
47        }
48        return conflicts;
49    }
50}

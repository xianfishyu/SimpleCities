23    public static RoadProfile Get(RoadProfileId id)
24    {
25        // The fixed, read-only catalog contains one entry per valid ID.
26        // Indexing avoids a captured predicate and interface enumerator per lookup.
27        for (int i = 0; i < All.Count; i++)
28        {
29            RoadProfile profile = All[i];
30            if (profile.Id == id) return profile;
31        }
32        // Preserve the existing no-match exception (including its runtime message).
33        return All.Single(static _ => false);

76    private sealed record EndConnection(RoadEdge Edge, RoadEndRole Role)
77    {
78        public RoadPoint Next => Role == RoadEndRole.Start ? Edge.Points[1] : Edge.Points[^2];
79        public double Parameter => Role == RoadEndRole.Start ? 0 : 1;
80    }

82    public static RoadSurfaceData? Prepare(IReadOnlyList<RoadNode> nodes, IReadOnlyList<RoadEdge> edges, CancellationToken cancellationToken = default)
83    {
84        cancellationToken.ThrowIfCancellationRequested();
85        if (edges.Count == 0) return null;
86        var incident = nodes.ToDictionary(node => node.Id, _ => new List<EndConnection>());
87        foreach (RoadEdge edge in edges)
88        {
89            incident[edge.Start].Add(new EndConnection(edge, RoadEndRole.Start));
90            incident[edge.End].Add(new EndConnection(edge, RoadEndRole.End));
91        }
92        var pieces = new List<RoadSurfacePiece>();
93        var joins = new List<RoadSurfacePiece>();
94        foreach (RoadEdge edge in edges)
95            PrepareEdge(edge, incident[edge.Start].Count, incident[edge.End].Count, pieces, joins, cancellationToken);
96        foreach (RoadNode node in nodes)
97        {
98            cancellationToken.ThrowIfCancellationRequested();
99            PrepareNode(node, incident[node.Id], joins);
100        }
101        pieces.AddRange(joins);
102        return new RoadSurfaceData(nodes, edges, pieces);
103    }
104
105    internal static void PrepareEdge(RoadEdge edge, int startDegree, int endDegree,
106        List<RoadSurfacePiece> pieces, List<RoadSurfacePiece> joins, CancellationToken cancellationToken)
107    {
108        double total = edge.Length;
109        if (!double.IsFinite(total) || total <= 0)
110            throw new InvalidOperationException("Cannot prepare a zero-length or non-finite road surface.");
111        double half = RoadProfiles.Get(edge.Profile).DisplayWidth / 2;
112        double distance = 0;
113        for (int i = 1; i < edge.Points.Count; i++)
114        {
115            cancellationToken.ThrowIfCancellationRequested();
116            RoadPoint a = edge.Points[i - 1], b = edge.Points[i];
117            double length = a.DistanceTo(b);
118            if (!double.IsFinite(length) || length <= 0)
119                throw new InvalidOperationException("Cannot prepare a zero-length or non-finite road segment.");
120            double tx = (b.X - a.X) / length, ty = (b.Y - a.Y) / length;
121            RoadPoint normal = new(-ty * half, tx * half);
122            RoadPoint capStart = i == 1 && startDegree == 1 ? new(a.X - tx * half, a.Y - ty * half) : a;
123            RoadPoint capEnd = i == edge.Points.Count - 1 && endDegree == 1 ? new(b.X + tx * half, b.Y + ty * half) : b;
124            double startParameter = distance / total;
125            distance += length;
126            double endParameter = i == edge.Points.Count - 1 ? 1 : distance / total;
127            pieces.Add(new RoadSurfacePiece(edge, a, b, startParameter, endParameter,
128                [Add(capStart, normal), Add(capEnd, normal), Subtract(capEnd, normal), Subtract(capStart, normal)]));
129            if (i < edge.Points.Count - 1)
130                AddJoin(joins, b, a, edge.Points[i + 1], edge, edge, endParameter, endParameter);
131        }
132    }

134    internal static void PrepareNode(RoadSnapshot snapshot, NodeId id, List<RoadSurfacePiece> pieces)
135    {
136        RoadNode? node = snapshot.QueryData.FindNode(id);
137        if (node is null) return;
138        var connected = new List<EndConnection>();
139        foreach (RoadEdge edge in snapshot.QueryData.Incident(id))
140        {
141            if (edge.Start == id) connected.Add(new(edge, RoadEndRole.Start));
142            if (edge.End == id) connected.Add(new(edge, RoadEndRole.End));
143        }
144        PrepareNode(node, connected, pieces);
145    }
146
147    private static void PrepareNode(RoadNode node, IReadOnlyList<EndConnection> connected, List<RoadSurfacePiece> pieces)
148    {
149        if (connected.Count >= 3) { AddJunction(pieces, node, connected); return; }
150        if (connected.Count != 2) return;
151        EndConnection a = connected[0], b = connected[1];
152        AddJoin(pieces, node.Position, a.Next, b.Next, a.Edge, b.Edge, a.Parameter, b.Parameter);
153    }

155    private readonly record struct JunctionCorner(RoadPoint Point, EdgeId Edge, double Parameter) : IComparable<JunctionCorner>
156    {
157        public int CompareTo(JunctionCorner other)
158        {
159            int order = Point.X.CompareTo(other.Point.X);
160            if (order == 0) order = Point.Y.CompareTo(other.Point.Y);
161            if (order == 0) order = Edge.Value.CompareTo(other.Edge.Value);
162            return order == 0 ? Parameter.CompareTo(other.Parameter) : order;
163        }
164    }

166    private static void AddJunction(List<RoadSurfacePiece> pieces, RoadNode node, IReadOnlyList<EndConnection> connections)
167    {
168        RoadPoint center = node.Position;
169        double reach = connections.Max(connection => RoadProfiles.Get(connection.Edge.Profile).DisplayWidth / 2);
170        int capacity = checked(connections.Count * 4);
171        // Eight-direction junctions fit in bounded stack scratch. Larger
172        // presentation inputs keep the same algorithm through a heap fallback.
173        Span<JunctionCorner> corners = capacity <= 64 ? stackalloc JunctionCorner[capacity] : new JunctionCorner[capacity];
174        int cornerCount = 0;
175        foreach (EndConnection connection in connections)
176        {
177            RoadEdge edge = connection.Edge;
178            RoadPoint next = connection.Next;
179            double length = center.DistanceTo(next);
180            double distance = Math.Min(reach, length / 2);
181            // Build the junction in center-relative coordinates. Adding the world offset
182            // before mouth +/- normal can turn identical corners into one-ULP slivers
183            // which collapse when the renderer converts them to binary32.
184            RoadPoint mouth = new((next.X - center.X) / length * distance,
185                (next.Y - center.Y) / length * distance);
186            RoadPoint normal = Normal(center, next, RoadProfiles.Get(edge.Profile).DisplayWidth / 2);
187            double parameter = connection.Parameter;
188            corners[cornerCount++] = new(normal, edge.Id, parameter);
189            corners[cornerCount++] = new(new(-normal.X, -normal.Y), edge.Id, parameter);
190            corners[cornerCount++] = new(Add(mouth, normal), edge.Id, parameter);
191            corners[cornerCount++] = new(Subtract(mouth, normal), edge.Id, parameter);
192        }
193
194        // The convex envelope joins unequal-width mouths without unbounded miters.
195        // Including both sides at the node also encloses its center for one-sided forks.
196        corners.Sort();
197        int uniqueCount = 0;
198        for (int i = 0; i < cornerCount; i++)
199            if (uniqueCount == 0 || corners[i].Point != corners[uniqueCount - 1].Point)
200                corners[uniqueCount++] = corners[i];
201        Span<RoadPoint> hull = capacity <= 64 ? stackalloc RoadPoint[capacity * 2] : new RoadPoint[checked(capacity * 2)];
202        Span<RoadPoint> upper = hull[capacity..];
203        int lowerCount = 0, upperCount = 0;
204        for (int i = 0; i < uniqueCount; i++) Append(hull, ref lowerCount, corners[i].Point);
205        for (int i = uniqueCount - 1; i >= 0; i--) Append(upper, ref upperCount, corners[i].Point);
206        lowerCount--;
207        upperCount--;
208        upper[..upperCount].CopyTo(hull[lowerCount..]);
209        int hullCount = lowerCount + upperCount;
210        // Partition the existing envelope by angular end-role ownership. Several
211        // mouths can contribute the same hull corner; choosing its first EdgeId
212        // would erase another incidence (including the other end of a self-loop).
213        var owners = connections.Select(connection =>
214        {
215            double length = center.DistanceTo(connection.Next);
216            var outward = new RoadPoint((connection.Next.X - center.X) / length,
217                (connection.Next.Y - center.Y) / length);
218            double angle = Math.Atan2(outward.Y, outward.X);
219            if (angle < 0) angle += Math.Tau;
220            return (Connection: connection, Outward: outward, Angle: angle);
221        }).OrderBy(owner => owner.Angle).ThenBy(owner => owner.Connection.Edge.Id.Value)
222            .ThenBy(owner => owner.Connection.Role).ToArray();
223        var boundaries = new List<RoadPoint>();
224        for (int i = 0; i < owners.Length; i++)
225        {
226            double nextAngle = i + 1 == owners.Length ? owners[0].Angle + Math.Tau : owners[i + 1].Angle;
227            double middleAngle = (owners[i].Angle + nextAngle) / 2;
228            boundaries.Add(new RoadPoint(Math.Cos(middleAngle), Math.Sin(middleAngle)));
229        }
230        for (int i = 0; i < hullCount; i++)
231        {
232            RoadPoint a = hull[i], b = hull[(i + 1) % hullCount];
233            if (Cross(default, a, b) == 0) continue;
234            var cuts = new List<double> { 0, 1 };
235            RoadPoint relative = a, direction = Subtract(b, a);
236            foreach (RoadPoint ray in boundaries)
237            {
238                double denominator = direction.X * ray.Y - direction.Y * ray.X;
239                if (Math.Abs(denominator) < 1e-12) continue;
240                double t = (relative.Y * ray.X - relative.X * ray.Y) / denominator;
241                if (t <= 1e-12 || t >= 1 - 1e-12) continue;
242                RoadPoint hit = new(relative.X + direction.X * t, relative.Y + direction.Y * t);
243                if (hit.X * ray.X + hit.Y * ray.Y > 0) cuts.Add(t);
244            }
245            cuts.Sort();
246            for (int j = 1; j < cuts.Count; j++)
247            {
248                RoadPoint start = new(a.X + direction.X * cuts[j - 1], a.Y + direction.Y * cuts[j - 1]);
249                RoadPoint end = new(a.X + direction.X * cuts[j], a.Y + direction.Y * cuts[j]);
250                if (Cross(default, start, end) == 0) continue;
251                RoadPoint middle = new((start.X + end.X) / 2, (start.Y + end.Y) / 2);
252                // Select the same lexicographic first owner without allocating a
253                // sorting pipeline for every emitted triangle. Keep exact ties stable.
254                var selected = owners[0];
255                double selectedScore = middle.X * selected.Outward.X + middle.Y * selected.Outward.Y;
256                for (int candidateIndex = 1; candidateIndex < owners.Length; candidateIndex++)
257                {
258                    var candidate = owners[candidateIndex];
259                    double score = middle.X * candidate.Outward.X + middle.Y * candidate.Outward.Y;
260                    int order = score.CompareTo(selectedScore);
261                    if (order > 0 || (order == 0 &&
262                        (candidate.Connection.Edge.Id.Value < selected.Connection.Edge.Id.Value ||
263                         (candidate.Connection.Edge.Id == selected.Connection.Edge.Id &&
264                          (int)candidate.Connection.Role < (int)selected.Connection.Role))))
265                    {
266                        selected = candidate;
267                        selectedScore = score;
268                    }
269                }
270                EndConnection owner = selected.Connection;
271                pieces.Add(new RoadSurfacePiece(owner.Edge, center, center, owner.Parameter, owner.Parameter,
272                    [center, Add(center, start), Add(center, end)], node.Id));
273            }
274        }

276        static void Append(Span<RoadPoint> chain, ref int count, RoadPoint point)
277        {
278            while (count >= 2 && Cross(chain[count - 2], chain[count - 1], point) <= 0) count--;
279            chain[count++] = point;
280        }
281    }

283    private static double Cross(RoadPoint a, RoadPoint b, RoadPoint c) =>
284        (b.X - a.X) * (c.Y - a.Y) - (b.Y - a.Y) * (c.X - a.X);

286    private static void AddJoin(List<RoadSurfacePiece> pieces, RoadPoint center, RoadPoint aNext, RoadPoint bNext,
287        RoadEdge a, RoadEdge b, double aParameter, double bParameter)
288    {
289        RoadPoint aNormal = Normal(center, aNext, RoadProfiles.Get(a.Profile).DisplayWidth / 2);
290        RoadPoint bNormal = Normal(center, bNext, RoadProfiles.Get(b.Profile).DisplayWidth / 2);
291        AddSide(Add(center, aNormal), Subtract(center, bNormal));
292        AddSide(Subtract(center, aNormal), Add(center, bNormal));
293
294        void AddSide(RoadPoint aCorner, RoadPoint bCorner)
295        {
296            double cross = (aCorner.X - center.X) * (bCorner.Y - center.Y) - (aCorner.Y - center.Y) * (bCorner.X - center.X);
297            if (cross == 0) return;
298            RoadPoint middle = new((aCorner.X + bCorner.X) / 2, (aCorner.Y + bCorner.Y) / 2);
299            pieces.Add(new RoadSurfacePiece(a, center, center, aParameter, aParameter, [center, aCorner, middle]));
300            pieces.Add(new RoadSurfacePiece(b, center, center, bParameter, bParameter, [center, middle, bCorner]));
301        }
302    }

304    private static RoadPoint Normal(RoadPoint start, RoadPoint end, double half)
305    {
306        double length = start.DistanceTo(end);
307        return new(-(end.Y - start.Y) / length * half, (end.X - start.X) / length * half);
308    }
309    private static RoadPoint Add(RoadPoint a, RoadPoint b) => new(a.X + b.X, a.Y + b.Y);
310    private static RoadPoint Subtract(RoadPoint a, RoadPoint b) => new(a.X - b.X, a.Y - b.Y);

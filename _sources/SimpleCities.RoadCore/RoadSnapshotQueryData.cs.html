3/// <summary>随快照建立的只读查找和弧长数据；查询不扫描路网或重算整边长度。</summary>
4internal sealed class RoadSnapshotQueryData
5{
6    internal readonly record struct Boundary(double Parameter, RoadPoint Point);
7    private sealed record ArcData(Boundary[] Vertices, double Length, double[] Prefix, double[] SegmentLengths);
8    private readonly record struct EdgeLookup(RoadEdge Edge, ArcData Arc);
9    private readonly record struct NodeLookup(RoadNode Node, RoadEdge[] Incident, int Degree);
10    private readonly Dictionary<EdgeId, EdgeLookup> _edges;
11    private readonly Dictionary<NodeId, NodeLookup> _nodes;
12
13    internal RoadSnapshotQueryData(RoadSnapshot snapshot, RoadSnapshotQueryData? previous,
14        CancellationToken cancellationToken)
15    {
16        cancellationToken.ThrowIfCancellationRequested();
17        _edges = new(snapshot.EdgeCount);
18        _nodes = new(snapshot.NodeCount);
19        var changedNodes = new HashSet<NodeId>();
20        foreach (RoadNode node in snapshot.Nodes)
21        {
22            cancellationToken.ThrowIfCancellationRequested();
23            _nodes.Add(node.Id, new(node, [], 0));
24        }
25        foreach (RoadEdge edge in snapshot.Edges)
26        {
27            cancellationToken.ThrowIfCancellationRequested();
28            RoadEdge? old = previous?.FindEdge(edge.Id);
29            // Arc data contains only geometry, so profile-only changes may share it
30            // too. Never attach it to an entity retained by edit history.
31            ArcData arc = previous is not null && old is not null && SameGeometry(old, edge, cancellationToken)
32                ? previous._edges[edge.Id].Arc : PrepareArc(edge, cancellationToken);
33            _edges.Add(edge.Id, new(edge, arc));
34            if (ReferenceEquals(old, edge)) continue;
35            changedNodes.Add(edge.Start);
36            changedNodes.Add(edge.End);
37            if (old is not null)
38            {
39                changedNodes.Add(old.Start);
40                changedNodes.Add(old.End);
41            }
42        }
43        if (previous is not null)
44            foreach (EdgeLookup lookup in previous._edges.Values)
45            {
46                cancellationToken.ThrowIfCancellationRequested();
47                RoadEdge old = lookup.Edge;
48                if (_edges.ContainsKey(old.Id)) continue;
49                changedNodes.Add(old.Start);
50                changedNodes.Add(old.End);
51            }
52
53        var incident = new Dictionary<NodeId, List<RoadEdge>>();
54        foreach (RoadEdge edge in snapshot.Edges)
55        {
56            cancellationToken.ThrowIfCancellationRequested();
57            AddIncident(edge.Start, edge);
58            if (edge.End != edge.Start) AddIncident(edge.End, edge);
59        }
60        foreach (RoadNode node in snapshot.Nodes)
61        {
62            cancellationToken.ThrowIfCancellationRequested();
63            if (incident.TryGetValue(node.Id, out List<RoadEdge>? rebuilt))
64            {
65                int degree = 0;
66                foreach (RoadEdge edge in rebuilt)
67                    degree += edge.Start == edge.End ? 2 : 1;
68                _nodes[node.Id] = new(node, rebuilt.ToArray(), degree);
69            }
70            else if (!changedNodes.Contains(node.Id) && previous is not null &&
71                previous._nodes.TryGetValue(node.Id, out NodeLookup unchanged))
72            {
73                // Share immutable leaves only; no previous snapshot or cache root
74                // is retained. Removed nodes/edges cannot survive in these maps.
75                _nodes[node.Id] = new(node, unchanged.Incident, unchanged.Degree);
76            }
77        }
78        cancellationToken.ThrowIfCancellationRequested();
79
80        void AddIncident(NodeId node, RoadEdge edge)
81        {
82            if (!changedNodes.Contains(node)) return;
83            if (!incident.TryGetValue(node, out List<RoadEdge>? edges)) incident.Add(node, edges = []);
84            edges.Add(edge);
85        }
86    }

88    private static bool SameGeometry(RoadEdge before, RoadEdge after, CancellationToken cancellationToken)
89    {
90        if (ReferenceEquals(before, after)) return true;
91        if (before.Points.Count != after.Points.Count) return false;
92        for (int i = 0; i < before.Points.Count; i++)
93        {
94            cancellationToken.ThrowIfCancellationRequested();
95            if (before.Points[i] != after.Points[i]) return false;
96        }
97        return true;
98    }
99
100    private static ArcData PrepareArc(RoadEdge edge, CancellationToken cancellationToken)
101    {
102        var vertices = new Boundary[edge.Points.Count];
103        var prefix = new double[edge.Points.Count];
104        var segmentLengths = new double[edge.Points.Count - 1];
105        double length = edge.Length, distance = 0;
106        vertices[0] = new(0, edge.Points[0]);
107        for (int i = 1; i < vertices.Length; i++)
108        {
109            cancellationToken.ThrowIfCancellationRequested();
110            segmentLengths[i - 1] = edge.Points[i - 1].DistanceTo(edge.Points[i]);
111            distance += segmentLengths[i - 1];
112            prefix[i] = distance;
113            vertices[i] = new(i == vertices.Length - 1 ? 1 : distance / length, edge.Points[i]);
114        }
115        return new(vertices, length, prefix, segmentLengths);
116    }

118    internal RoadEdge? FindEdge(EdgeId id) => _edges.GetValueOrDefault(id).Edge;
119    internal RoadNode? FindNode(NodeId id) => _nodes.GetValueOrDefault(id).Node;
120    internal IReadOnlyList<RoadEdge> Incident(NodeId id) => _nodes.TryGetValue(id, out NodeLookup node) ? node.Incident : [];
121    internal int Degree(NodeId id) => _nodes.GetValueOrDefault(id).Degree;
122    internal Boundary[] Vertices(EdgeId id) => _edges[id].Arc.Vertices;
123    internal static int Segment(Boundary[] vertices, double parameter)
124    {
125        int low = 0, high = vertices.Length - 1;
126        while (low + 1 < high)
127        {
128            int middle = (low + high) / 2;
129            if (vertices[middle].Parameter <= parameter) low = middle;
130            else high = middle;
131        }
132        return Math.Min(low, vertices.Length - 2);
133    }
134
135    internal RoadPoint? Resolve(RoadLocation location)
136    {
137        if (!_edges.TryGetValue(location.Edge, out EdgeLookup edge)) return null;
138        ArcData arc = edge.Arc;
139        Boundary[] vertices = arc.Vertices;
140        double distance = location.Parameter * arc.Length;
141        int low = 0, high = vertices.Length - 1;
142        while (low + 1 < high)
143        {
144            int middle = (low + high) / 2;
145            if (arc.Prefix[middle] < distance) low = middle;
146            else high = middle;
147        }
148        int segment = low;
149        Boundary a = vertices[segment], b = vertices[segment + 1];
150        // Resolve in metres, as RoadEdge.PointAt does. Dividing both prefix values into
151        // normalized parameters first adds rounding even at exact positions such as (300, 50).
152        double fraction = Math.Clamp((distance - arc.Prefix[segment]) / arc.SegmentLengths[segment], 0, 1);
153        return new RoadPoint(a.Point.X + (b.Point.X - a.Point.X) * fraction,
154            a.Point.Y + (b.Point.Y - a.Point.Y) * fraction);
155    }

5/// <summary>冻结整笔格段选择，在来源快照的私有草稿中完成编辑和规范化。</summary>
6internal static class RoadSpanEditPlanner
7{
8    internal static RoadEditResult Plan(RoadNetwork owner, RoadSnapshot source, IReadOnlyList<RoadGridSpan> spans, RoadProfileId? profile,
9        CancellationToken cancellationToken)
10    {
11        ArgumentNullException.ThrowIfNull(spans);
12        cancellationToken.ThrowIfCancellationRequested();
13        if (profile.HasValue && !profile.Value.IsValid)
14            return new(RoadEditStatus.Rejected, null, "请选择有效道路类型");
15        var selection = new Dictionary<RoadSpanKey, RoadGridSpan>();
16        foreach (RoadGridSpan span in spans.ToArray())
17        {
18            cancellationToken.ThrowIfCancellationRequested();
19            if (span is null || span.Source != source.Token)
20                return new(RoadEditStatus.Rejected, null, "道路来源版本已过期");
21            RoadEdge? selected = source.FindEdge(span.Edge);
22            if (selected is null)
23                return new(RoadEditStatus.Rejected, null, "道路格段不存在");
24            // 先核对每个原始条目，再去重或跳过同类型；同 token 的候选内容也必须一致。
25            RoadSpanRange first = span.Ranges[0];
26            RoadGridSpan? current = RoadSpanQuery.Pick(source,
27                new(source.Token, selected.Id, (first.StartParameter + first.EndParameter) / 2));
28            if (current is null || current.Key != span.Key || !current.Ranges.SequenceEqual(span.Ranges) ||
29                !current.Points.SequenceEqual(span.Points))
30                return new(RoadEditStatus.Rejected, null, "道路格段与当前内容不一致");
31            if (profile != selected.Profile) selection.TryAdd(span.Key, span);
32        }
33        if (selection.Count == 0)
34            return new(RoadEditStatus.NoChange, null, "");
35        if (source.Token.ContentRevision >= long.MaxValue - 1 || source.Token.ChangeSequence == long.MaxValue)
36            return new(RoadEditStatus.Rejected, null, "道路身份或版本已耗尽");
37        try
38        {
39            return Prepare(owner, source, selection.Values, profile, cancellationToken);
40        }
41        catch (InvalidDataException exception)
42        {
43            return new(RoadEditStatus.Rejected, null, exception.Message);
44        }
45    }
46
47    private static RoadEditResult Prepare(RoadNetwork owner, RoadSnapshot source, IEnumerable<RoadGridSpan> selection,
48        RoadProfileId? profile, CancellationToken cancellationToken)
49    {
50        long nextNode = source.NextNodeId, nextEdge = source.NextEdgeId;
51        var nodes = source.Nodes.ToDictionary(node => node.Position);
52        var byEdge = selection.GroupBy(span => span.Edge).ToDictionary(group => group.Key, group => group.ToArray());
53        var pieces = new List<Piece>(Math.Max(4, source.EdgeCount));
54        foreach (RoadEdge selected in source.Edges)
55        {
56            cancellationToken.ThrowIfCancellationRequested();
57            if (!byEdge.TryGetValue(selected.Id, out RoadGridSpan[]? spans))
58            {
59                pieces.Add(new(selected.Id, selected.Start, selected.End, selected.Profile, selected.Points));
60                continue;
61            }
62            var cuts = new HashSet<RoadPoint>();
63            foreach (RoadGridSpan span in spans)
64                foreach (RoadPoint point in new[] { span.Points[0], span.Points[^1] })
65                    // 相邻已选格段的共同边界不需要分配节点；端点用查询给出的精确格坐标。
66                    if (!cuts.Add(point)) cuts.Remove(point);
67            cuts.Add(selected.Points[0]);
68            cuts.Add(selected.Points[^1]);
69            foreach (RoadPoint point in cuts.OrderBy(point => point.X).ThenBy(point => point.Y))
70                if (!nodes.ContainsKey(point)) nodes.Add(point, new(new(Allocate(ref nextNode)), point));
71
72            var split = new List<Piece>();
73            Split(selected.Id, selected.Profile, selected.Points, cuts, nodes, split, cancellationToken);
74            RoadSpanRange[] ranges = spans.SelectMany(span => span.Ranges).ToArray();
75            double distance = 0, total = selected.Length;
76            foreach (Piece piece in split)
77            {
78                cancellationToken.ThrowIfCancellationRequested();
79                double length = piece.Points.Zip(piece.Points.Skip(1), (a, b) => a.DistanceTo(b)).Sum();
80                double middle = (distance + length / 2) / total;
81                distance += length;
82                if (!ranges.Any(range => middle >= range.StartParameter && middle <= range.EndParameter))
83                    pieces.Add(piece);
84                else if (profile.HasValue)
85                    pieces.Add(piece with { Profile = profile.Value });
86            }
87        }
88        Normalize(nodes, pieces, cancellationToken);
89
90        return new(RoadEditStatus.Ready, Freeze(owner, source, nodes, pieces, nextNode, nextEdge, cancellationToken), "");
91    }

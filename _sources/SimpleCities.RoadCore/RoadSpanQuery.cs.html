29public static class RoadSpanQuery
30{
31    /// <summary>定位命中折线片后只求相邻主格边界，保留普通格心转折，不依赖渲染表面。</summary>
32    public static RoadGridSpan? Pick(RoadSnapshot snapshot, RoadLocation location)
33    {
34        ArgumentNullException.ThrowIfNull(snapshot);
35        if (location.Source != snapshot.Token || !double.IsFinite(location.Parameter) ||
36            location.Parameter < 0 || location.Parameter > 1) return null;
37        RoadEdge? edge = snapshot.FindEdge(location.Edge);
38        if (edge is null) return null;
39        if ((location.Parameter == 0 && snapshot.QueryData.Degree(edge.Start) >= 3) ||
40            (location.Parameter == 1 && snapshot.QueryData.Degree(edge.End) >= 3)) return null;
41
42        Boundary[] vertices = snapshot.QueryData.Vertices(edge.Id);
43        int segment = RoadSnapshotQueryData.Segment(vertices, location.Parameter);
44        Boundary start = FindPrimary(snapshot.Map, vertices, segment, location.Parameter, backwards: true) ?? vertices[0];
45        Boundary end = FindPrimary(snapshot.Map, vertices, segment, location.Parameter, backwards: false) ?? vertices[^1];
46
47        bool transparentSeam = edge.Start == edge.End && snapshot.QueryData.Degree(edge.Start) == 2 &&
48            !snapshot.Map.IsPrimaryPoint(edge.Points[0]);
49        if (transparentSeam && (start.Parameter == 0 || end.Parameter == 1))
50        {
51            Boundary tail = FindPrimary(snapshot.Map, vertices, vertices.Length - 2, 1, backwards: true) ?? vertices[0];
52            Boundary head = FindPrimary(snapshot.Map, vertices, 0, 0, backwards: false) ?? vertices[^1];
53            return new(snapshot.Token, edge.Id,
54                [new(tail.Parameter, 1), new(0, head.Parameter)],
55                Slice(tail, vertices[^1], vertices).Concat(Slice(vertices[0], head, vertices).Skip(1)));
56        }
57        return new(snapshot.Token, edge.Id, [new(start.Parameter, end.Parameter)], Slice(start, end, vertices));
58    }
59
60    private static Boundary? FindPrimary(MapDefinition map, Boundary[] vertices, int segment,
61        double parameter, bool backwards)
62    {
63        for (int i = segment; i >= 0 && i < vertices.Length - 1; i += backwards ? -1 : 1)
64        {
65            Boundary a = vertices[i], b = vertices[i + 1];
66            bool useX = a.Point.X != b.Point.X;
67            double from = useX ? a.Point.X : a.Point.Y, to = useX ? b.Point.X : b.Point.Y;
68            double fraction = Math.Clamp((parameter - a.Parameter) / (b.Parameter - a.Parameter), 0, 1);
69            double grid = Math.Floor((from + (to - from) * fraction) / map.CellSizeMetres);
70            Boundary? nearest = null;
71            // Test adjacent grid coordinates against their actual source parameters. This also
72            // handles an exactly-hit boundary whose inverse interpolation rounded to either side.
73            for (int offset = -1; offset <= 2; offset++)
74            {
75                double coordinate = (grid + offset) * map.CellSizeMetres;
76                double along = (coordinate - from) / (to - from);
77                if (along < 0 || along > 1) continue;
78                double candidateParameter = along == 0 ? a.Parameter : along == 1 ? b.Parameter :
79                    a.Parameter + (b.Parameter - a.Parameter) * along;
80                if (backwards ? candidateParameter > parameter || candidateParameter == 1 : candidateParameter <= parameter)
81                    continue;
82                RoadPoint point = useX
83                    ? new(coordinate, a.Point.Y + Math.Sign(b.Point.Y - a.Point.Y) * Math.Abs(coordinate - from))
84                    : new(a.Point.X, coordinate);
85                if (!map.IsPrimaryPoint(point)) continue;
86                if (nearest is null || (backwards ? candidateParameter > nearest.Value.Parameter : candidateParameter < nearest.Value.Parameter))
87                    nearest = new(candidateParameter, point);
88            }
89            if (nearest is not null) return nearest;
90        }
91        return null;
92    }
93
94    private static IEnumerable<RoadPoint> Slice(Boundary start, Boundary end, Boundary[] vertices)
95    {
96        yield return start.Point;
97        int first = RoadSnapshotQueryData.Segment(vertices, start.Parameter) + 1;
98        for (int i = first; i < vertices.Length && vertices[i].Parameter < end.Parameter; i++)
99            if (vertices[i].Parameter > start.Parameter) yield return vertices[i].Point;
100        yield return end.Point;
101    }

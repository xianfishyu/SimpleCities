1namespace SimpleCities.RoadCore;

3/// <summary>只读选择手势：预选独立于已选，抬起保留区间，版本变化使旧会话失效。</summary>
4public sealed class RoadSpanSelectionSession
5{
6    private readonly List<RoadGridSpan> _selected = [];
7    private readonly HashSet<RoadSpanKey> _keys = [];
8
9    public RoadSpanSelectionSession() => Selected = _selected.AsReadOnly();
10
11    public RoadStateToken? Source { get; private set; }
12    public RoadGridSpan? Hovered { get; private set; }
13    public IReadOnlyList<RoadGridSpan> Selected { get; }
14    public bool IsSelecting { get; private set; }
15
16    public bool Hover(RoadSnapshot snapshot, RoadGridSpan? span)
17    {
18        ArgumentNullException.ThrowIfNull(snapshot);
19        if (!Matches(snapshot) || (span is not null && span.Source != snapshot.Token))
20        {
21            Clear();
22            return false;
23        }
24        Source = snapshot.Token;
25        Hovered = span;
26        return true;
27    }
28
29    public void Begin(RoadSnapshot snapshot)
30    {
31        ArgumentNullException.ThrowIfNull(snapshot);
32        RoadGridSpan? hover = Hovered?.Source == snapshot.Token ? Hovered : null;
33        Clear();
34        Source = snapshot.Token;
35        Hovered = hover;
36        IsSelecting = true;
37    }
38
39    public bool Accumulate(RoadSnapshot snapshot, IEnumerable<RoadGridSpan> spans)
40    {
41        ArgumentNullException.ThrowIfNull(snapshot);
42        ArgumentNullException.ThrowIfNull(spans);
43        if (!Matches(snapshot))
44        {
45            Clear();
46            return false;
47        }
48        if (!IsSelecting) return false;
49        RoadGridSpan[] additions = spans.ToArray();
50        if (additions.Any(span => span.Source != snapshot.Token))
51        {
52            Clear();
53            return false;
54        }
55        foreach (RoadGridSpan span in additions)
56            if (_keys.Add(span.Key)) _selected.Add(span);
57        return true;
58    }
59
60    public void End() => IsSelecting = false;
61
62    public void Clear()
63    {
64        _selected.Clear();
65        _keys.Clear();
66        Source = null;
67        Hovered = null;
68        IsSelecting = false;
69    }
70
71    private bool Matches(RoadSnapshot snapshot) => Source is null || Source == snapshot.Token;
72}

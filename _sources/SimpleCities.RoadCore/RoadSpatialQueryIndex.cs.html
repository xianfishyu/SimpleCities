7/// <summary>由不可变快照重建的中心线查询；精确检查只使用局部有界片段。</summary>
8public sealed class RoadSpatialQueryIndex
9{
10    private readonly RoadSnapshot _snapshot;
11    private readonly SpatialQueryIndex<RoadQueryFragment> _index;
12    public RoadSpatialQueryIndex(RoadSnapshot snapshot)
13    {
14        ArgumentNullException.ThrowIfNull(snapshot);
15        _snapshot = snapshot;
16        var fragments = new List<SpatialFragment<RoadQueryFragment>>();
17        foreach (RoadEdge edge in snapshot.Edges)
18        {
19            RoadSnapshotQueryData.Boundary[] vertices = snapshot.QueryData.Vertices(edge.Id);
20            for (int geometry = 0; geometry < vertices.Length - 1; geometry++)
21            {
22                var a = vertices[geometry]; var b = vertices[geometry + 1];
23                int count = Math.Max(1, (int)Math.Ceiling(a.Point.DistanceTo(b.Point) / 100));
24                for (int part = 0; part < count; part++)
25                {
26                    double start = (double)part / count, end = (double)(part + 1) / count;
27                    RoadPoint first = Interpolate(a.Point, b.Point, start), last = Interpolate(a.Point, b.Point, end);
28                    var fragment = new RoadQueryFragment(edge.Id, geometry, first, last,
29                        a.Parameter + (b.Parameter - a.Parameter) * start,
30                        a.Parameter + (b.Parameter - a.Parameter) * end,
31                        geometry == 0 && part == 0, geometry == vertices.Length - 2 && part == count - 1);
32                    SpatialBounds bounds = SpatialBounds.Between(first, last);
33                    // Conservative broad-phase padding only; it never changes exact geometry decisions.
34                    fragments.Add(new(fragment, new(bounds.MinX - 1e-9, bounds.MinY - 1e-9,
35                        bounds.MaxX + 1e-9, bounds.MaxY + 1e-9)));
36                }
37            }
38        }
39        _index = new(snapshot.Token, fragments);
40    }

42    public SpatialQueryResult<RoadQueryHit> QueryNearby(RoadPoint point, double radius, SpatialQueryBudget? budget = null)
43    {
44        if (!point.IsFinite || !double.IsFinite(radius) || radius < 0)
45            return new(_snapshot.Token, SpatialQueryStatus.InvalidParameters, [], default, "查询位置或半径非法");
46        SpatialQueryBudget limits = budget ?? SpatialQueryBudget.Default;
47        SpatialQueryResult<RoadQueryFragment> candidates = _index.QueryBounds(
48            new(point.X - radius, point.Y - radius, point.X + radius, point.Y + radius), limits);
49        if (candidates.Status != SpatialQueryStatus.Ready)
50            return new(candidates.Source, candidates.Status, [], candidates.Metrics, candidates.Reason);
51        var hits = new List<RoadQueryHit>();
52        int exact = 0;
53        foreach (RoadQueryFragment fragment in candidates.Results)
54        {
55            if (exact >= limits.MaxExactTests)
56                return new(candidates.Source, SpatialQueryStatus.BudgetExceeded, [], candidates.Metrics with { ExactGeometryTests = exact }, "空间查询精确检查预算已耗尽");
57            exact++;
58            SourceSegment source = ReadSource(fragment);
59            double dx = source.End.X - source.Start.X, dy = source.End.Y - source.Start.Y;
60            double px = point.X - source.Start.X, py = point.Y - source.Start.Y;
61            double projection = (px * dx + py * dy) / (dx * dx + dy * dy);
62            double fraction = Math.Clamp(projection, source.StartFraction, source.EndFraction);
63            double distance = fraction == projection ? Math.Abs(px * dy - py * dx) / Math.Sqrt(dx * dx + dy * dy)
64                : point.DistanceTo(Interpolate(source.Start, source.End, fraction));
65            if (distance > radius) continue;
66            double parameter = source.Parameter(fraction);
67            hits.Add(new(new(_snapshot.Token, fragment.Edge, parameter), fragment, distance));
68        }
69        return new(_snapshot.Token, SpatialQueryStatus.Ready, hits.AsReadOnly(),
70            candidates.Metrics with { ExactGeometryTests = exact, Hits = hits.Select(hit => hit.Location.Edge).Distinct().Count() }, "");
71    }

73    public SpatialQueryResult<RoadQueryHit> QueryBounds(SpatialBounds bounds, SpatialQueryBudget? budget = null)
74    {
75        SpatialQueryBudget limits = budget ?? SpatialQueryBudget.Default;
76        return Refine(_index.QueryBounds(bounds, limits), limits, fragment =>
77        {
78            SourceSegment source = ReadSource(fragment);
79            double start = source.StartFraction, end = source.EndFraction;
80            if (!ClipAxis(source.Start.X, source.End.X - source.Start.X, bounds.MinX, bounds.MaxX, ref start, ref end) ||
81                !ClipAxis(source.Start.Y, source.End.Y - source.Start.Y, bounds.MinY, bounds.MaxY, ref start, ref end)) return null;
82            return source.Parameter(start);
83        });
84    }

86    public SpatialQueryResult<RoadQueryHit> QuerySegment(RoadPoint from, RoadPoint to, SpatialQueryBudget? budget = null)
87    {
88        SpatialQueryBudget limits = budget ?? SpatialQueryBudget.Default;
89        return Refine(_index.QuerySegment(from, to, limits), limits, fragment =>
90        {
91            SourceSegment source = ReadSource(fragment);
92            double rx = source.End.X - source.Start.X, ry = source.End.Y - source.Start.Y;
93            double sx = to.X - from.X, sy = to.Y - from.Y;
94            double qx = from.X - source.Start.X, qy = from.Y - source.Start.Y;
95            double denominator = rx * sy - ry * sx;
96            if (denominator == 0)
97            {
98                if (qx * ry - qy * rx != 0) return null;
99                double lengthSquared = rx * rx + ry * ry;
100                double first = (qx * rx + qy * ry) / lengthSquared;
101                double last = first + (sx * rx + sy * ry) / lengthSquared;
102                double start = Math.Max(source.StartFraction, Math.Min(first, last)), end = Math.Min(source.EndFraction, Math.Max(first, last));
103                return start <= end ? source.Parameter(start) : null;
104            }
105            double alongFragment = (qx * sy - qy * sx) / denominator;
106            double alongQuery = (qx * ry - qy * rx) / denominator;
107            return alongFragment >= source.StartFraction && alongFragment <= source.EndFraction && alongQuery >= 0 && alongQuery <= 1
108                ? source.Parameter(alongFragment) : null;
109        });
110    }

112    private SpatialQueryResult<RoadQueryHit> Refine(SpatialQueryResult<RoadQueryFragment> candidates,
113        SpatialQueryBudget budget, Func<RoadQueryFragment, double?> intersect)
114    {
115        if (candidates.Status != SpatialQueryStatus.Ready)
116            return new(candidates.Source, candidates.Status, [], candidates.Metrics, candidates.Reason);
117        var hits = new List<RoadQueryHit>();
118        int exact = 0;
119        foreach (RoadQueryFragment fragment in candidates.Results)
120        {
121            if (exact >= budget.MaxExactTests)
122                return new(candidates.Source, SpatialQueryStatus.BudgetExceeded, [],
123                    candidates.Metrics with { ExactGeometryTests = exact }, "空间查询精确检查预算已耗尽");
124            exact++;
125            if (intersect(fragment) is not double parameter) continue;
126            hits.Add(new(new(_snapshot.Token, fragment.Edge, parameter), fragment, 0));
127        }
128        return new(_snapshot.Token, SpatialQueryStatus.Ready, hits.AsReadOnly(),
129            candidates.Metrics with { ExactGeometryTests = exact, Hits = hits.Select(hit => hit.Location.Edge).Distinct().Count() }, "");
130    }

132    private readonly record struct SourceSegment(RoadPoint Start, RoadPoint End, double StartParameter, double EndParameter,
133        double StartFraction, double EndFraction)
134    {
135        internal double Parameter(double fraction) => StartParameter + (EndParameter - StartParameter) * fraction;
136    }
137
138    private SourceSegment ReadSource(RoadQueryFragment fragment)
139    {
140        // The fragment is an indexing detail: exact queries address two authoritative grid points
141        // directly, then restrict their parameter interval. No full canonical edge is enumerated.
142        RoadSnapshotQueryData.Boundary[] vertices = _snapshot.QueryData.Vertices(fragment.Edge);
143        var a = vertices[fragment.GeometryIndex]; var b = vertices[fragment.GeometryIndex + 1];
144        return new(a.Point, b.Point, a.Parameter, b.Parameter,
145            (fragment.StartParameter - a.Parameter) / (b.Parameter - a.Parameter),
146            (fragment.EndParameter - a.Parameter) / (b.Parameter - a.Parameter));
147    }
148
149    private static bool ClipAxis(double origin, double direction, double minimum, double maximum, ref double start, ref double end)
150    {
151        if (direction == 0) return origin >= minimum && origin <= maximum;
152        double first = (minimum - origin) / direction, last = (maximum - origin) / direction;
153        start = Math.Max(start, Math.Min(first, last));
154        end = Math.Min(end, Math.Max(first, last));
155        return start <= end;
156    }
157
158    private static RoadPoint Interpolate(RoadPoint start, RoadPoint end, double fraction) =>
159        new(start.X + (end.X - start.X) * fraction, start.Y + (end.Y - start.Y) * fraction);

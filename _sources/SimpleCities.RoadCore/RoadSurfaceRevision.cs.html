7/// <summary>绘制顺序：全部带状片、全部边内转弯、全部节点连接；层内按规范实体身份。</summary>
8public readonly record struct RoadSurfaceGroupKey(RoadSurfaceLayer Layer, long Owner) : IComparable<RoadSurfaceGroupKey>
9{
10    public int CompareTo(RoadSurfaceGroupKey other)
11    {
12        int layer = ((int)Layer).CompareTo((int)other.Layer);
13        return layer == 0 ? Owner.CompareTo(other.Owner) : layer;
14    }
15}

17/// <summary>已完成纯核心预检的不可变表面组；不持有 Godot 对象或祖先快照。</summary>
18public sealed class RoadSurfaceGroup
19{
20    internal RoadSurfaceGroup(RoadSurfaceGroupKey key, RoadSurfacePiece[] pieces)
21    {
22        Key = key;
23        Pieces = Array.AsReadOnly(pieces);
24        VertexCount = pieces.Sum(piece => piece.Corners.Count);
25        IndexCount = pieces.Sum(piece => checked(3 * (piece.Corners.Count - 2)));
26    }
27    public RoadSurfaceGroupKey Key { get; }
28    public IReadOnlyList<RoadSurfacePiece> Pieces { get; }
29    public int VertexCount { get; }
30    public int IndexCount { get; }
31}

35/// <summary>随发布版本持有的纯表面。局部编辑仅替换一跳端点邻域，未变组和有序树节点共享。</summary>
36public sealed class RoadSurfaceRevision
37{
38    private static readonly double MaximumWidth = RoadProfiles.All.Max(profile => profile.DisplayWidth);
39    private readonly ImmutableSortedDictionary<RoadSurfaceGroupKey, RoadSurfaceGroup> _groups;

41    private RoadSurfaceRevision(RoadSnapshot snapshot, ImmutableSortedDictionary<RoadSurfaceGroupKey, RoadSurfaceGroup> groups,
42        int pieces, int vertices, int indices)
43    {
44        Snapshot = snapshot;
45        _groups = groups;
46        PieceCount = pieces;
47        VertexCount = vertices;
48        IndexCount = indices;

50    public RoadSnapshot Snapshot { get; }
51    public IReadOnlyDictionary<RoadSurfaceGroupKey, RoadSurfaceGroup> Groups => _groups;
52    public int PieceCount { get; }
53    public int VertexCount { get; }
54    public int IndexCount { get; }
55    public RoadSurfaceGroup? FindGroup(RoadSurfaceGroupKey key) => _groups.GetValueOrDefault(key);

57    public RoadSurfaceRevision Rebind(RoadSnapshot snapshot)
58    {
59        if (Snapshot.Token.NetworkInstance != snapshot.Token.NetworkInstance || Snapshot.Token.Lineage != snapshot.Token.Lineage ||
60            Snapshot.Token.ContentRevision != snapshot.Token.ContentRevision || Snapshot.Map != snapshot.Map ||
61            !ReferenceEquals(Snapshot.Nodes, snapshot.Nodes) || !ReferenceEquals(Snapshot.Edges, snapshot.Edges))
62            throw new InvalidOperationException("V4 surface content cannot be rebound to a different revision.");
63        return new(snapshot, _groups, PieceCount, VertexCount, IndexCount);
64    }

66    public static RoadSurfaceRevision Prepare(RoadSnapshot snapshot, CancellationToken cancellationToken = default)
67    {
68        cancellationToken.ThrowIfCancellationRequested();
69        var builder = ImmutableSortedDictionary.CreateBuilder<RoadSurfaceGroupKey, RoadSurfaceGroup>();
70        int pieces = 0, vertices = 0, indices = 0;
71        foreach (RoadEdge edge in snapshot.Edges)
72        {
73            cancellationToken.ThrowIfCancellationRequested();
74            var ribbons = new List<RoadSurfacePiece>(edge.Points.Count - 1);
75            var bends = new List<RoadSurfacePiece>();
76            RoadPresentation.PrepareEdge(edge, snapshot.QueryData.Degree(edge.Start), snapshot.QueryData.Degree(edge.End),
77                ribbons, bends, cancellationToken);
78            Add(new(RoadSurfaceLayer.Ribbon, edge.Id.Value), ribbons);
79            Add(new(RoadSurfaceLayer.Bend, edge.Id.Value), bends);
80        }
81        foreach (RoadNode node in snapshot.Nodes)
82        {
83            cancellationToken.ThrowIfCancellationRequested();
84            var joins = new List<RoadSurfacePiece>();
85            RoadPresentation.PrepareNode(snapshot, node.Id, joins);
86            Add(new(RoadSurfaceLayer.NodeJoin, node.Id.Value), joins);
87        }
88        return new(snapshot, builder.ToImmutable(), pieces, vertices, indices);
89
90        void Add(RoadSurfaceGroupKey key, List<RoadSurfacePiece> values)
91        {
92            if (values.Count == 0)
93            {
94                if (key.Layer == RoadSurfaceLayer.Ribbon) throw new InvalidOperationException("V4 road surface is missing a ribbon.");
95                return;
96            }
97            RoadSurfaceGroup group = ValidateGroup(snapshot, key, values.ToArray(), cancellationToken);
98            builder.Add(key, group);
99            pieces = checked(pieces + group.Pieces.Count);
100            vertices = checked(vertices + group.VertexCount);
101            indices = checked(indices + group.IndexCount);
102        }
103    }

105    /// <summary>Compatibility/preflight entry for an independently prepared full surface; never trusts its owner metadata.</summary>
106    public static RoadSurfaceRevision FromSurface(RoadSnapshot snapshot, RoadSurfaceData? surface, CancellationToken cancellationToken = default)
107    {
108        cancellationToken.ThrowIfCancellationRequested();
109        if (surface is null)
110        {
111            if (snapshot.EdgeCount != 0) throw new InvalidOperationException("Missing V4 road surface.");
112            return new(snapshot, ImmutableSortedDictionary<RoadSurfaceGroupKey, RoadSurfaceGroup>.Empty, 0, 0, 0);
113        }
114        if (!surface.Edges.SequenceEqual(snapshot.Edges) || !surface.Nodes.SequenceEqual(snapshot.Nodes) || surface.Pieces.Count == 0)
115            throw new InvalidOperationException("V4 surface does not match its target snapshot.");
116        var builder = ImmutableSortedDictionary.CreateBuilder<RoadSurfaceGroupKey, RoadSurfaceGroup>();
117        RoadSurfaceGroupKey? active = null;
118        var values = new List<RoadSurfacePiece>();
119        int vertices = 0, indices = 0;
120        foreach (RoadSurfacePiece piece in surface.Pieces)
121        {
122            cancellationToken.ThrowIfCancellationRequested();
123            RoadSurfaceGroupKey key = GroupKey(piece);
124            if (active is RoadSurfaceGroupKey previous && previous != key)
125            {
126                if (previous.CompareTo(key) > 0) throw new InvalidOperationException("V4 surface draw order is invalid.");
127                Flush(previous);
128            }
129            active = key;
130            values.Add(piece);
131        }
132        if (active is RoadSurfaceGroupKey last) Flush(last);
133        foreach (RoadEdge edge in snapshot.Edges)
134        {
135            cancellationToken.ThrowIfCancellationRequested();
136            if (!builder.ContainsKey(new(RoadSurfaceLayer.Ribbon, edge.Id.Value)))
137                throw new InvalidOperationException("V4 road surface is missing a ribbon.");
138        }
139        return new(snapshot, builder.ToImmutable(), surface.Pieces.Count, vertices, indices);
140
141        void Flush(RoadSurfaceGroupKey key)
142        {
143            RoadSurfaceGroup group = ValidateGroup(snapshot, key, values.ToArray(), cancellationToken);
144            builder.Add(key, group);
145            vertices = checked(vertices + group.VertexCount);
146            indices = checked(indices + group.IndexCount);
147            values.Clear();
148        }
149    }

151    public RoadSurfaceUpdate Apply(RoadPlan plan, CancellationToken cancellationToken = default)
152    {
153        cancellationToken.ThrowIfCancellationRequested();
154        if (!ReferenceEquals(Snapshot, plan.Source) || Snapshot.Token.NetworkInstance != plan.Target.Token.NetworkInstance ||
155            Snapshot.Token.Lineage != plan.Target.Token.Lineage || Snapshot.Map != plan.Target.Map)
156            throw new InvalidOperationException("V4 surface source does not match its edit plan.");
157        RoadSnapshot target = plan.Target;
158        var edges = new HashSet<EdgeId>();
159        var nodes = new HashSet<NodeId>();
160        foreach (RoadEdgeChange change in plan.ChangeSet.Edges)
161        {
162            cancellationToken.ThrowIfCancellationRequested();
163            edges.Add(change.Id);
164            if (change.Before is RoadEdge before) { nodes.Add(before.Start); nodes.Add(before.End); }
165            if (change.After is RoadEdge after) { nodes.Add(after.Start); nodes.Add(after.End); }
166        }
167        foreach (RoadNodeChange change in plan.ChangeSet.Nodes) nodes.Add(change.Id);
168        foreach (NodeId node in nodes)
169        {
170            cancellationToken.ThrowIfCancellationRequested();
171            foreach (RoadEdge edge in Snapshot.QueryData.Incident(node)) edges.Add(edge.Id);
172            foreach (RoadEdge edge in target.QueryData.Incident(node)) edges.Add(edge.Id);
173        }
174        var builder = _groups.ToBuilder();
175        var changed = new List<RoadSurfaceGroupKey>();
176        int pieces = PieceCount, vertices = VertexCount, indices = IndexCount;
177        foreach (EdgeId id in edges)
178        {
179            cancellationToken.ThrowIfCancellationRequested();
180            var ribbons = new List<RoadSurfacePiece>();
181            var bends = new List<RoadSurfacePiece>();
182            if (target.FindEdge(id) is RoadEdge edge)
183                RoadPresentation.PrepareEdge(edge, target.QueryData.Degree(edge.Start), target.QueryData.Degree(edge.End),
184                    ribbons, bends, cancellationToken);
185            Replace(new(RoadSurfaceLayer.Ribbon, id.Value), ribbons);
186            Replace(new(RoadSurfaceLayer.Bend, id.Value), bends);
187        }
188        foreach (NodeId id in nodes)
189        {
190            cancellationToken.ThrowIfCancellationRequested();
191            var joins = new List<RoadSurfacePiece>();
192            RoadPresentation.PrepareNode(target, id, joins);
193            Replace(new(RoadSurfaceLayer.NodeJoin, id.Value), joins);
194        }
195        cancellationToken.ThrowIfCancellationRequested();
196        return new(new(target, builder.ToImmutable(), pieces, vertices, indices), changed.AsReadOnly());
197
198        void Replace(RoadSurfaceGroupKey key, List<RoadSurfacePiece> values)
199        {
200            if (key.Layer == RoadSurfaceLayer.Ribbon && values.Count == 0 && target.FindEdge(new(key.Owner)) is not null)
201                throw new InvalidOperationException("V4 road surface is missing a ribbon.");
202            if (builder.TryGetValue(key, out RoadSurfaceGroup? old))
203            {
204                pieces -= old.Pieces.Count;
205                vertices -= old.VertexCount;
206                indices -= old.IndexCount;
207                builder.Remove(key);
208            }
209            if (values.Count > 0)
210            {
211                RoadSurfaceGroup group = ValidateGroup(target, key, values.ToArray(), cancellationToken);
212                builder.Add(key, group);
213                pieces = checked(pieces + group.Pieces.Count);
214                vertices = checked(vertices + group.VertexCount);
215                indices = checked(indices + group.IndexCount);
216            }
217            if (old is not null || values.Count > 0) changed.Add(key);
218        }
219    }

221    private static RoadSurfaceGroupKey GroupKey(RoadSurfacePiece piece) => piece.StartParameter != piece.EndParameter
222        ? new(RoadSurfaceLayer.Ribbon, piece.Edge.Id.Value)
223        : piece.StartParameter is 0 or 1
224            ? new(RoadSurfaceLayer.NodeJoin, (piece.StartParameter == 0 ? piece.Edge.Start : piece.Edge.End).Value)
225            : new(RoadSurfaceLayer.Bend, piece.Edge.Id.Value);
226
227    private static RoadSurfaceGroup ValidateGroup(RoadSnapshot snapshot, RoadSurfaceGroupKey key, RoadSurfacePiece[] pieces,
228        CancellationToken cancellationToken)
229    {
230        RoadEdge? owner = key.Layer == RoadSurfaceLayer.NodeJoin ? null : snapshot.FindEdge(new(key.Owner));
231        int degree = key.Layer == RoadSurfaceLayer.NodeJoin ? snapshot.QueryData.Degree(new(key.Owner)) : 0;
232        long maximum = key.Layer switch
233        {
234            RoadSurfaceLayer.Ribbon => owner?.Points.Count - 1 ?? 0,
235            RoadSurfaceLayer.Bend => 4L * ((owner?.Points.Count ?? 2) - 2),
236            RoadSurfaceLayer.NodeJoin => degree >= 3 ? 4L * degree * (degree + 1) : degree == 2 ? 4 : 0,
237            _ => 0,
238        };
239        if (pieces.Length > maximum || (key.Layer == RoadSurfaceLayer.Ribbon && pieces.Length != maximum))
240            throw new InvalidOperationException("V4 road surface exceeds its target resource budget or is missing a ribbon.");
241        for (int index = 0; index < pieces.Length; index++)
242        {
243            cancellationToken.ThrowIfCancellationRequested();
244            RoadSurfacePiece piece = pieces[index];
245            if (!ReferenceEquals(snapshot.FindEdge(piece.Edge.Id), piece.Edge) || GroupKey(piece) != key ||
246                !double.IsFinite(piece.StartParameter) || !double.IsFinite(piece.EndParameter) ||
247                piece.StartParameter < 0 || piece.EndParameter > 1 || piece.StartParameter > piece.EndParameter)
248                throw new InvalidOperationException("V4 road surface owner or location is invalid.");
249            RoadSnapshotQueryData.Boundary[] boundaries = snapshot.QueryData.Vertices(piece.Edge.Id);
250            int start = BoundaryIndex(boundaries, piece.StartParameter), end = BoundaryIndex(boundaries, piece.EndParameter);
251            if (start < 0 || end < 0 || boundaries[start].Point != piece.Start || boundaries[end].Point != piece.End)
252                throw new InvalidOperationException("V4 road surface location geometry is invalid.");
253            if (key.Layer == RoadSurfaceLayer.Ribbon)
254            {
255                if (start != index || end != index + 1 || piece.Corners.Count != 4 || piece.JunctionNode is not null)
256                    throw new InvalidOperationException("V4 road ribbon coverage is invalid.");
257            }
258            else if (piece.Corners.Count != 3 || start != end ||
259                (key.Layer == RoadSurfaceLayer.Bend && piece.JunctionNode is not null) ||
260                (key.Layer == RoadSurfaceLayer.NodeJoin && (degree >= 3 ? piece.JunctionNode != new NodeId(key.Owner) : piece.JunctionNode is not null)))
261                throw new InvalidOperationException("V4 road join ownership is invalid.");
262            foreach (RoadPoint corner in piece.Corners)
263                if (!corner.IsFinite || corner.X < Math.Min(piece.Start.X, piece.End.X) - MaximumWidth ||
264                    corner.X > Math.Max(piece.Start.X, piece.End.X) + MaximumWidth ||
265                    corner.Y < Math.Min(piece.Start.Y, piece.End.Y) - MaximumWidth || corner.Y > Math.Max(piece.Start.Y, piece.End.Y) + MaximumWidth)
266                    throw new InvalidOperationException("V4 road surface vertex is outside its source envelope.");
267        }
268        return new(key, pieces);
269    }
270
271    private static int BoundaryIndex(RoadSnapshotQueryData.Boundary[] boundaries, double parameter)
272    {
273        int low = 0, high = boundaries.Length - 1;
274        while (low <= high)
275        {
276            int middle = low + (high - low) / 2;
277            double value = boundaries[middle].Parameter;
278            if (value == parameter) return middle;
279            if (value < parameter) low = middle + 1;
280            else high = middle - 1;
281        }
282        return -1;
283    }

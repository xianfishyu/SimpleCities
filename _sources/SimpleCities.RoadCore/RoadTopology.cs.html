13    internal static bool IsForwardCollinear(RoadPoint a, RoadPoint b, RoadPoint c) =>
14        Cross(b.X - a.X, b.Y - a.Y, c.X - b.X, c.Y - b.Y) == 0 &&
15        (b.X - a.X) * (c.X - b.X) + (b.Y - a.Y) * (c.Y - b.Y) > 0;

17    internal static bool IsCanonicalLoopDirection(IReadOnlyList<RoadPoint> points)
18    {
19        for (int i = 1; i < points.Count - 1; i++)
20        {
21            RoadPoint forward = points[i], reverse = points[points.Count - 1 - i];
22            int comparison = forward.X.CompareTo(reverse.X);
23            if (comparison == 0) comparison = forward.Y.CompareTo(reverse.Y);
24            if (comparison != 0) return comparison < 0;
25        }
26        return true;
27    }

29    internal static void Validate(MapDefinition map, IReadOnlyList<RoadNode> nodes, IReadOnlyList<RoadEdge> edges, long nextNodeId, long nextEdgeId,
30        CancellationToken cancellationToken = default)
31    {
32        cancellationToken.ThrowIfCancellationRequested();
33        if (nextNodeId <= 0 || nextNodeId == long.MaxValue || nextEdgeId <= 0 || nextEdgeId == long.MaxValue)
34            throw new InvalidDataException("道路身份水位无效");
35        if (nodes.Count > MaximumNodes || edges.Count > MaximumEdges)
36            throw new InvalidDataException("超过当前道路路网的资源上限");
37        int pointCount = 0, segmentCount = 0;
38        foreach (RoadEdge edge in edges)
39        {
40            cancellationToken.ThrowIfCancellationRequested();
41            if (edge.Points.Count > MaximumPoints - pointCount)
42                throw new InvalidDataException("超过当前道路路网的资源上限");
43            pointCount += edge.Points.Count;
44            segmentCount += Math.Max(0, edge.Points.Count - 1);
45        }
46        if (nodes.Count == 0 && edges.Count == 0) return;
47        var byId = new Dictionary<NodeId, NodeUse>(nodes.Count);
48        var positions = new HashSet<RoadPoint>(nodes.Count);
49        long lastId = 0;
50        foreach (RoadNode node in nodes)
51        {
52            cancellationToken.ThrowIfCancellationRequested();
53            if (node.Id.Value <= lastId || node.Id.Value >= nextNodeId || !map.IsBuildPoint(node.Position) || !positions.Add(node.Position))
54                throw new InvalidDataException("节点身份、排序、水位或主格点/格心位置无效");
55            byId.Add(node.Id, new(node.Position));
56            lastId = node.Id.Value;
57        }
58        using var segments = new PooledBuffer<Segment>(segmentCount);
59        lastId = 0;
60        foreach (RoadEdge edge in edges)
61        {
62            cancellationToken.ThrowIfCancellationRequested();
63            if (edge.Id.Value <= lastId || edge.Id.Value >= nextEdgeId || edge.Start.Value > edge.End.Value || !edge.Profile.IsValid ||
64                !byId.TryGetValue(edge.Start, out NodeUse start) || !byId.TryGetValue(edge.End, out NodeUse end) ||
65                edge.Points.Count < 2 || edge.Points[0] != start.Position || edge.Points[^1] != end.Position)
66                throw new InvalidDataException("道路身份、端点、方向或类型无效");
67            if (edge.Start == edge.End && !IsCanonicalLoopDirection(edge.Points))
68                throw new InvalidDataException("自环必须在既定 seam 上采用规范方向");
69            lastId = edge.Id.Value;
70            // Both nodes exist and no more dictionary entries are added. Updating
71            // separately also counts the two endpoint roles of a self-loop.
72            CollectionsMarshal.GetValueRefOrNullRef(byId, edge.Start).Add(edge);
73            CollectionsMarshal.GetValueRefOrNullRef(byId, edge.End).Add(edge);
74            for (int i = 0; i < edge.Points.Count; i++)
75            {
76                cancellationToken.ThrowIfCancellationRequested();
77                if (!map.IsBuildPoint(edge.Points[i])) throw new InvalidDataException("道路折点不在地图内的主格点或格心上");
78                if (i == 0) continue;
79                RoadPoint a = edge.Points[i - 1], b = edge.Points[i];
80                if (!map.IsBuildSegment(a, b)) throw new InvalidDataException("道路格段必须具有正长度，主格点沿八方向，格心仅沿对角方向");
81                if (i >= 2 && IsForwardCollinear(edge.Points[i - 2], a, b)) throw new InvalidDataException("道路含多余共线折点");
82                segments.Add(new Segment(edge.Id, i, a, b, i == 1 ? edge.Start : null, i == edge.Points.Count - 1 ? edge.End : null));
83            }
84        }
85        foreach (NodeUse connected in byId.Values)
86        {
87            cancellationToken.ThrowIfCancellationRequested();
88            if (connected.InvalidDegree)
89                throw new InvalidDataException("结构节点必须是道路端点、真实路口或必要的道路类型分界");
90        }
91        ValidateIntersections(map, segments.Values, cancellationToken);
92    }

94    private static void ValidateIntersections(MapDefinition map, ReadOnlySpan<Segment> segments, CancellationToken cancellationToken)
95    {
96        // Valid square-eight segments intersect only on the half-cell lattice. Every
97        // visited point includes both endpoints; exact pair validation remains authoritative.
98        // This candidate rule must be revisited if free-angle or curve geometry is admitted.
99        double half = map.CellSizeMetres / 2d;
100        var byPoint = new Dictionary<(int X, int Y), Occupants>(segments.Length);
101        using var occupants = new PooledBuffer<Occupant>(segments.Length * 2);
102        var compared = new HashSet<int>();
103        for (int i = 0; i < segments.Length; i++)
104        {
105            ref readonly Segment segment = ref segments[i];
106            int x = (int)(segment.A.X / half), y = (int)(segment.A.Y / half);
107            int dx = (int)(segment.B.X / half) - x, dy = (int)(segment.B.Y / half) - y;
108            int steps = Math.Max(Math.Abs(dx), Math.Abs(dy));
109            int sx = Math.Sign(dx), sy = Math.Sign(dy);
110            compared.Clear();
111            for (int step = 0; step <= steps; step++)
112            {
113                cancellationToken.ThrowIfCancellationRequested();
114                var point = (x + step * sx, y + step * sy);
115                if (byPoint.TryGetValue(point, out Occupants bucket))
116                {
117                    // Linked value entries preserve the original insertion order,
118                    // including which invalid pair reports an error first.
119                    for (int link = bucket.First; link >= 0; link = occupants[link].Next)
120                    {
121                        int other = occupants[link].Segment;
122                        if (compared.Add(other)) ValidatePair(in segments[other], in segment);
123                    }
124                    int next = occupants.Count;
125                    occupants.Add(new(i, -1));
126                    // Add may grow the buffer; acquire its ref only afterwards.
127                    occupants[bucket.Last] = occupants[bucket.Last] with { Next = next };
128                    byPoint[point] = bucket with { Last = next };
129                }
130                else
131                {
132                    int first = occupants.Count;
133                    occupants.Add(new(i, -1));
134                    byPoint.Add(point, new(first, first));
135                }
136            }
137        }
138    }

140    private struct NodeUse(RoadPoint position)
141    {
142        internal readonly RoadPoint Position = position;
143        private RoadEdge? _first;
144        private int _degree;
145        private bool _redundantPair;
146        internal readonly bool InvalidDegree => _degree == 0 || (_degree == 2 && _redundantPair);
147        internal void Add(RoadEdge edge)
148        {
149            if (_first is null) _first = edge;
150            else if (_degree == 1)
151                _redundantPair = _first.Profile == edge.Profile && !ReferenceEquals(_first, edge);
152            _degree++;
153        }
154    }

160    /// <summary>Operation-owned scratch; leases return on success, rejection and cancellation.</summary>
161    private sealed class PooledBuffer<T>(int capacity) : IDisposable where T : struct
162    {
163        private T[] _items = ArrayPool<T>.Shared.Rent(capacity);
164        internal int Count { get; private set; }
165        internal ReadOnlySpan<T> Values => _items.AsSpan(0, Count);
166        internal ref T this[int index] => ref _items[index];
167        internal void Add(T value)
168        {
169            if (Count == _items.Length)
170            {
171                T[] grown = ArrayPool<T>.Shared.Rent(Math.Max(4, checked(_items.Length * 2)));
172                _items.AsSpan(0, Count).CopyTo(grown);
173                ArrayPool<T>.Shared.Return(_items, clearArray: true);
174                _items = grown;
175            }
176            _items[Count++] = value;
177        }
178        public void Dispose()
179        {
180            if (_items.Length == 0) return;
181            ArrayPool<T>.Shared.Return(_items, clearArray: true);
182            _items = [];
183            Count = 0;
184        }
185    }

187    private static double Cross(double ax, double ay, double bx, double by) => ax * by - ay * bx;
188    private static void ValidatePair(in Segment a, in Segment b)
189    {
190        double rx = a.B.X - a.A.X, ry = a.B.Y - a.A.Y;
191        double sx = b.B.X - b.A.X, sy = b.B.Y - b.A.Y;
192        double qx = b.A.X - a.A.X, qy = b.A.Y - a.A.Y;
193        double denominator = Cross(rx, ry, sx, sy);
194        if (denominator == 0)
195        {
196            if (Cross(qx, qy, rx, ry) != 0) return;
197            double a0 = rx != 0 ? a.A.X : a.A.Y, a1 = rx != 0 ? a.B.X : a.B.Y;
198            double b0 = rx != 0 ? b.A.X : b.A.Y, b1 = rx != 0 ? b.B.X : b.B.Y;
199            double lower = Math.Max(Math.Min(a0, a1), Math.Min(b0, b1));
200            double upper = Math.Min(Math.Max(a0, a1), Math.Max(b0, b1));
201            if (lower > upper) return;
202            if (lower < upper) throw new InvalidDataException("道路中心线存在共线重叠");
203        }
204        else
205        {
206            double t = Cross(qx, qy, sx, sy) / denominator;
207            double u = Cross(qx, qy, rx, ry) / denominator;
208            if (t < 0 || t > 1 || u < 0 || u > 1) return;
209        }
210        bool adjacent = a.Edge == b.Edge && Math.Abs(a.Index - b.Index) == 1;
211        bool joined = (a.A == b.A && a.Start.HasValue && a.Start == b.Start) ||
212            (a.A == b.B && a.Start.HasValue && a.Start == b.End) ||
213            (a.B == b.A && a.End.HasValue && a.End == b.Start) ||
214            (a.B == b.B && a.End.HasValue && a.End == b.End);
215        if (!adjacent && !joined) throw new InvalidDataException("道路交叉或接入位置缺少共同结构节点");
216    }

30/// <summary>不可变、有界片段的粗筛索引。直线路径逐格遍历，不扫描路径包围盒或全表。</summary>
31public sealed class SpatialQueryIndex<T>
32{
33    private readonly Dictionary<(int X, int Y), int[]> _buckets;
34    private readonly SpatialFragment<T>[] _fragments;
35    private readonly double _bucketSize;
36    private readonly RoadStateToken _source;
37
38    public SpatialQueryIndex(RoadStateToken source, IEnumerable<SpatialFragment<T>> fragments, double bucketSize = 100,
39        CancellationToken cancellationToken = default)
40    {
41        ArgumentNullException.ThrowIfNull(fragments);
42        if (!double.IsFinite(bucketSize) || bucketSize <= 0) throw new ArgumentOutOfRangeException(nameof(bucketSize));
43        _source = source;
44        _bucketSize = bucketSize;
45        cancellationToken.ThrowIfCancellationRequested();
46        if (!cancellationToken.CanBeCanceled) _fragments = fragments.ToArray();
47        else
48        {
49            var materialized = new List<SpatialFragment<T>>(fragments.TryGetNonEnumeratedCount(out int count) ? count : 0);
50            foreach (var fragment in fragments)
51            {
52                cancellationToken.ThrowIfCancellationRequested();
53                materialized.Add(fragment);
54            }
55            _fragments = materialized.ToArray();
56        }
57        var buckets = new Dictionary<(int, int), List<int>>();
58        for (int i = 0; i < _fragments.Length; i++)
59        {
60            cancellationToken.ThrowIfCancellationRequested();
61            SpatialBounds bounds = _fragments[i].Bounds;
62            if (!ValidBounds(bounds)) throw new ArgumentException("Invalid fragment bounds.", nameof(fragments));
63            int left = Bucket(bounds.MinX), right = Bucket(bounds.MaxX), top = Bucket(bounds.MinY), bottom = Bucket(bounds.MaxY);
64            // Every caller must fragment long geometry before indexing; never expand an unbounded rectangle.
65            if (((long)right - left + 1) * ((long)bottom - top + 1) > 64)
66                throw new ArgumentException("Spatial fragment exceeds its bounded bucket coverage.", nameof(fragments));
67            for (int x = left; x <= right; x++)
68                for (int y = top; y <= bottom; y++)
69                {
70                    if (!buckets.TryGetValue((x, y), out List<int>? entries)) buckets.Add((x, y), entries = []);
71                    entries.Add(i);
72                }
73        }
74        _buckets = new(buckets.Count);
75        foreach (var pair in buckets)
76        {
77            cancellationToken.ThrowIfCancellationRequested();
78            _buckets.Add(pair.Key, pair.Value.ToArray());
79        }
80        cancellationToken.ThrowIfCancellationRequested();
81    }

83    public SpatialQueryResult<T> QueryBounds(SpatialBounds bounds, SpatialQueryBudget? budget = null)
84    {
85        SpatialQueryBudget limits = budget ?? SpatialQueryBudget.Default;
86        if (!limits.IsValid || !ValidBounds(bounds))
87            return new(_source, SpatialQueryStatus.InvalidParameters, [], default, "空间查询参数非法");
88        int left = Bucket(bounds.MinX), right = Bucket(bounds.MaxX);
89        int top = Bucket(bounds.MinY), bottom = Bucket(bounds.MaxY);
90        if (left == right && top == bottom) return QuerySingleBucket(bounds, left, top, limits);
91        var query = new Query(this, limits, bounds);
92        for (int x = left; x <= right; x++)
93            for (int y = top; y <= bottom; y++)
94                if (!query.Visit(x, y)) return query.Result();
95        return query.Result();
96    }
97
98    private SpatialQueryResult<T> QuerySingleBucket(SpatialBounds bounds, int x, int y, SpatialQueryBudget limits)
99    {
100        // Construction adds each fragment index only once to a given bucket.
101        // A one-bucket query therefore needs neither bucket nor fragment deduplication sets.
102        List<T>? results = null;
103        int candidates = 0;
104        if (_buckets.TryGetValue((x, y), out int[]? entries))
105            foreach (int index in entries)
106            {
107                if (candidates >= limits.MaxCandidates)
108                    return new(_source, SpatialQueryStatus.BudgetExceeded, [], new(1, candidates, 0, 0, 0),
109                        "空间查询候选或分桶预算已耗尽");
110                candidates++;
111                SpatialFragment<T> fragment = _fragments[index];
112                if (fragment.Bounds.Intersects(bounds)) (results ??= []).Add(fragment.Value);
113            }
114        return new(_source, SpatialQueryStatus.Ready, results is null ? [] : results.AsReadOnly(),
115            new(1, candidates, 0, 0, 0), "");
116    }

118    public SpatialQueryResult<T> QuerySegment(RoadPoint from, RoadPoint to, SpatialQueryBudget? budget = null)
119    {
120        var query = new Query(this, budget ?? SpatialQueryBudget.Default, SpatialBounds.Between(from, to));
121        if (!from.IsFinite || !to.IsFinite || !query.IsValid) return query.Invalid();
122        int x = Bucket(from.X), y = Bucket(from.Y), endX = Bucket(to.X), endY = Bucket(to.Y);
123        int stepX = Math.Sign(to.X - from.X), stepY = Math.Sign(to.Y - from.Y);
124        double dx = to.X - from.X, dy = to.Y - from.Y;
125        double nextX = stepX == 0 ? double.PositiveInfinity : ((x + (stepX > 0 ? 1d : 0d)) * _bucketSize - from.X) / dx;
126        double nextY = stepY == 0 ? double.PositiveInfinity : ((y + (stepY > 0 ? 1d : 0d)) * _bucketSize - from.Y) / dy;
127        double deltaX = stepX == 0 ? double.PositiveInfinity : _bucketSize / Math.Abs(dx);
128        double deltaY = stepY == 0 ? double.PositiveInfinity : _bucketSize / Math.Abs(dy);
129        while (true)
130        {
131            if (!query.Visit(x, y) || (x == endX && y == endY)) return query.Result();
132            // The terminal point belongs to floor(coordinate). A negative direction must not
133            // cross its final grid boundary again when the other axis reaches the endpoint.
134            if (x == endX) nextX = double.PositiveInfinity;
135            if (y == endY) nextY = double.PositiveInfinity;
136            if (nextX < nextY) { x += stepX; nextX += deltaX; }
137            else if (nextY < nextX) { y += stepY; nextY += deltaY; }
138            else
139            {
140                // A corner touch includes both side buckets, so surface endpoint ownership is retained.
141                if (!query.Visit(x + stepX, y) || !query.Visit(x, y + stepY)) return query.Result();
142                x += stepX; y += stepY; nextX += deltaX; nextY += deltaY;
143            }
144        }
145    }

147    private int Bucket(double coordinate) => (int)Math.Floor(coordinate / _bucketSize);
148    private bool ValidBounds(SpatialBounds bounds) => bounds.IsValid &&
149        Math.Abs(bounds.MinX / _bucketSize) < int.MaxValue / 2d && Math.Abs(bounds.MaxX / _bucketSize) < int.MaxValue / 2d &&
150        Math.Abs(bounds.MinY / _bucketSize) < int.MaxValue / 2d && Math.Abs(bounds.MaxY / _bucketSize) < int.MaxValue / 2d;

152    private sealed class Query(SpatialQueryIndex<T> owner, SpatialQueryBudget budget, SpatialBounds bounds)
153    {
154        private readonly HashSet<(int, int)> _visited = [];
155        private readonly HashSet<int> _seen = [];
156        private readonly List<T> _results = [];
157        private SpatialQueryStatus _status = SpatialQueryStatus.Ready;
158        internal bool IsValid => budget.IsValid && owner.ValidBounds(bounds);
159        internal bool Visit(int x, int y)
160        {
161            if (_visited.Contains((x, y))) return true;
162            if (_visited.Count >= budget.MaxBuckets) { _status = SpatialQueryStatus.BudgetExceeded; return false; }
163            _visited.Add((x, y));
164            if (!owner._buckets.TryGetValue((x, y), out int[]? entries)) return true;
165            foreach (int index in entries)
166            {
167                if (_seen.Contains(index)) continue;
168                if (_seen.Count >= budget.MaxCandidates) { _status = SpatialQueryStatus.BudgetExceeded; return false; }
169                _seen.Add(index);
170                SpatialFragment<T> fragment = owner._fragments[index];
171                if (fragment.Bounds.Intersects(bounds)) _results.Add(fragment.Value);
172            }
173            return true;
174        }
175        internal SpatialQueryResult<T> Invalid() => new(owner._source, SpatialQueryStatus.InvalidParameters, [], default, "空间查询参数非法");
176        internal SpatialQueryResult<T> Result()
177        {
178            if (!IsValid) return Invalid();
179            return new(owner._source, _status, _status == SpatialQueryStatus.Ready ? _results.AsReadOnly() : [],
180                new(_visited.Count, _seen.Count, 0, 0, 0), _status == SpatialQueryStatus.Ready ? "" : "空间查询候选或分桶预算已耗尽");
181        }
182    }

251    private static string ExtractNodeBlock(string scene, string nodeName)
252    {
253        string header = $"[node name=\"{nodeName}\"";
254        int start = scene.IndexOf(header, StringComparison.Ordinal);
255        Assert.True(start >= 0, $"Missing node block for {nodeName}");
256
257        int next = scene.IndexOf("\n[node ", start + header.Length, StringComparison.Ordinal);
258        return next < 0 ? scene[start..] : scene[start..next];
259    }
260
261    private static string ExtractUniqueExtResourceId(string source, string path)
262    {
263        string marker = $"path=\"{path}\"";
264        string declaration = Assert.Single(
265            source.Split('\n'),
266            line => line.StartsWith("[ext_resource ", StringComparison.Ordinal) && line.Contains(marker, StringComparison.Ordinal));
267        return ExtractHeaderAttribute(declaration, "id");
268    }
269
270    private static string ExtractHeaderAttribute(string header, string attribute)
271    {
272        string marker = $" {attribute}=\"";
273        int valueStart = header.IndexOf(marker, StringComparison.Ordinal);
274        Assert.True(valueStart >= 0, $"Missing {attribute} attribute in: {header}");
275        valueStart += marker.Length;
276        int valueEnd = header.IndexOf('"', valueStart);
277        Assert.True(valueEnd > valueStart, $"Malformed {attribute} attribute in: {header}");
278        return header[valueStart..valueEnd];
279    }
280
281    private static string[] ExtractSubResourceBlocks(string source)
282    {
283        return source.Split("\n[sub_resource ", StringSplitOptions.None)
284            .Skip(1)
285            .Select(block => "[sub_resource " + block.Split("\n[", 2, StringSplitOptions.None)[0])
286            .ToArray();
287    }
288
289    private static string[] ReadPropertyLines(string source, string propertyName)
290    {
291        string prefix = propertyName + " = ";
292        return source.Split('\n').Select(line => line.TrimEnd('\r')).Where(line => line.StartsWith(prefix, StringComparison.Ordinal)).ToArray();
293    }
294
295    private static string[] ExtractSubResourceReferences(string propertyLine)
296    {
297        return propertyLine.Split("SubResource(\"", StringSplitOptions.None).Skip(1).Select(part => part.Split('\"')[0]).ToArray();
298    }
299
300    private static int ReadScrollMode(string nodeBlock, string propertyName)
301    {
302        string line = Assert.Single(ReadPropertyLines(nodeBlock, propertyName));
303        string serializedValue = line[(propertyName.Length + 3)..];
304        Assert.True(int.TryParse(serializedValue, System.Globalization.NumberStyles.None, System.Globalization.CultureInfo.InvariantCulture, out int value),
305            $"{propertyName} must be one exact serialized Godot ScrollMode integer, but was '{serializedValue}'.");
306        return value;
307    }

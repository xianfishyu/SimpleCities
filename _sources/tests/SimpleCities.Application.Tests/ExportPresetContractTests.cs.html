8    [Fact]
9    public void WindowsDesktopQa_ExportsOnlyTheDedicatedSaveContractResourcesFromTests()
10    {
11        string projectRoot = FindProjectRoot();
12        IReadOnlyDictionary<string, string> preset = ReadPreset(projectRoot, "Windows Desktop QA");
13        Assert.Equal("all_resources", DecodeString(preset["export_filter"]));
14
15        string[] excludePatterns = DecodeString(preset["exclude_filter"])
16            .Split(',', StringSplitOptions.RemoveEmptyEntries | StringSplitOptions.TrimEntries);
17        Assert.Contains("tests/SimpleCities.Application.Tests/**", excludePatterns);
18        Assert.Contains("tests/SimpleCities.RoadCore.Performance/**", excludePatterns);
19        Assert.Contains("tests/SimpleCities.SaveLockProbe/**", excludePatterns);
20
21        string testsRoot = Path.Combine(projectRoot, "tests");
22        string[] exportedTestFiles = Directory
23            .EnumerateFiles(testsRoot, "*", SearchOption.AllDirectories)
24            .Select(path => Path.GetRelativePath(projectRoot, path).Replace('\\', '/'))
25            .Where(path => !excludePatterns.Any(pattern => GlobMatches(pattern, path)))
26            .Order(StringComparer.Ordinal)
27            .ToArray();
28
29        Assert.Equal(
30            [
31                "tests/godot/exported_save_runtime_contract.gd",
32                "tests/godot/exported_save_runtime_contract.gd.uid",
33                "tests/godot/exported_save_runtime_contract.tscn",
34                "tests/godot/road_save_fixture.gd",
35                "tests/godot/road_save_fixture.gd.uid",
36            ],
37            exportedTestFiles);
38    }

40    private static IReadOnlyDictionary<string, string> ReadPreset(
41        string projectRoot,
42        string presetName)
43    {
44        var sections = new List<Dictionary<string, string>>();
45        Dictionary<string, string>? current = null;
46        foreach (string rawLine in File.ReadLines(Path.Combine(projectRoot, "export_presets.cfg")))
47        {
48            string line = rawLine.Trim();
49            if (line.StartsWith("[preset.", StringComparison.Ordinal) &&
50                line.EndsWith(']') &&
51                !line.EndsWith(".options]", StringComparison.Ordinal))
52            {
53                current = new Dictionary<string, string>(StringComparer.Ordinal);
54                sections.Add(current);
55                continue;
56            }
57            if (current is null || line.Length == 0 || line.StartsWith(';'))
58                continue;
59            int separator = line.IndexOf('=');
60            if (separator > 0)
61                current[line[..separator]] = line[(separator + 1)..];
62        }
63
64        return Assert.Single(
65            sections,
66            section => section.TryGetValue("name", out string? value) &&
67                       string.Equals(DecodeString(value), presetName, StringComparison.Ordinal));
68    }
69
70    private static string DecodeString(string encoded) =>
71        JsonSerializer.Deserialize<string>(encoded) ??
72        throw new InvalidDataException("Export preset string cannot be null.");
73
74    private static bool GlobMatches(string pattern, string path)
75    {
76        string expression = "^" + Regex.Escape(pattern)
77            .Replace(@"\*\*", ".*", StringComparison.Ordinal)
78            .Replace(@"\*", "[^/]*", StringComparison.Ordinal)
79            .Replace(@"\?", "[^/]", StringComparison.Ordinal) + "$";
80        return Regex.IsMatch(path, expression, RegexOptions.CultureInvariant);
81    }
82
83    private static string FindProjectRoot()
84    {
85        DirectoryInfo? directory = new(AppContext.BaseDirectory);
86        while (directory is not null &&
87               !File.Exists(Path.Combine(directory.FullName, "SimpleCities.sln")))
88        {
89            directory = directory.Parent;
90        }
91        return directory?.FullName
92            ?? throw new DirectoryNotFoundException("SimpleCities project root was not found.");
93    }

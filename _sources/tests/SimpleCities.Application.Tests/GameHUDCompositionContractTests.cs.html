112    private static string ExtractNodeBlock(string scene, string nodeName)
113    {
114        string header = $"[node name=\"{nodeName}\"";
115        int start = scene.IndexOf(header, StringComparison.Ordinal);
116        Assert.True(start >= 0, $"Missing node block for {nodeName}");
117
118        int next = scene.IndexOf("\n[node ", start + header.Length, StringComparison.Ordinal);
119        return next < 0 ? scene[start..] : scene[start..next];
120    }

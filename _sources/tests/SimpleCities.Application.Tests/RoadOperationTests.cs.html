3    [Fact]
4    public void AcceptedCancellationRejectsLateCommitAndKeepsSingleWriterUntilCleanup()
5    {
6        var operation = new RoadOperation();
7        Assert.True(operation.TryBegin());
8        Assert.True(operation.TryCancel());
9        Assert.Equal("Cancelling", operation.Phase);
10        Assert.False(operation.TryBegin());
11        bool published = false;
12        Assert.False(operation.TryCommit(() => published = true));
13        Assert.False(published);
14        Assert.True(operation.IsBusy);
15        operation.Finish();
16        Assert.False(operation.IsBusy);
17        Assert.True(operation.TryBegin());
18    }

20    [Fact]
21    public void CommitWinningBeforeCancellationRemainsBusyUntilFirstDrawAndCannotBeUndoneByEsc()
22    {
23        var operation = new RoadOperation();
24        Assert.True(operation.TryBegin());
25        Assert.False(operation.RecordFirstDraw());
26        Assert.Equal(-1, operation.DrawnElapsedMilliseconds);
27        Assert.True(operation.TryCommit(() => true));
28        Assert.False(operation.TryCancel());
29        Assert.False(operation.TryBegin());
30        Assert.Equal("Committed", operation.Phase);
31        Assert.True(operation.RecordFirstDraw());
32        Assert.Equal("Drawn", operation.Phase);
33        Assert.False(operation.IsBusy);
34        Assert.True(operation.DrawnElapsedMilliseconds >= 0);
35        double firstDraw = operation.DrawnElapsedMilliseconds;
36        Assert.False(operation.RecordFirstDraw());
37        Assert.Equal(firstDraw, operation.DrawnElapsedMilliseconds);
38        Assert.False(operation.TryCancel());
39    }

180    [Fact]
181    public void Write_IsDeterministicUnindentedAndSortedByFileName()
182    {
183        var manifest = new SaveSlotManifest(
184            "manual-1",
185            "River City",
186            "2026-08-04T00:00:00.0000000Z",
187            "River City",
188            null,
189            12.50m,
190            null,
191            [
192                new SaveSlotManifestFile("road_network.json", 3, new string('b', 64)),
193                new SaveSlotManifestFile("economy.json", 2, new string('a', 64)),
194            ]);
195
196        string first = Write(manifest);
197        string second = Write(manifest);
198
199        Assert.Equal(first, second);
200        Assert.DoesNotContain('\n', first);
201        Assert.True(first.IndexOf("economy.json", StringComparison.Ordinal) <
202                    first.IndexOf("road_network.json", StringComparison.Ordinal));
203        SaveSlotManifest restored = Read(first);
204        Assert.Equal(manifest.SlotID, restored.SlotID);
205        Assert.Equal(manifest.DisplayName, restored.DisplayName);
206        Assert.Equal(manifest.Timestamp, restored.Timestamp);
207        Assert.Equal(manifest.CityName, restored.CityName);
208        Assert.Equal(manifest.Population, restored.Population);
209        Assert.Equal(manifest.Funds, restored.Funds);
210        Assert.Equal(manifest.ThumbnailFile, restored.ThumbnailFile);
211        Assert.Equal(
212            manifest.Files.OrderBy(file => file.Name, StringComparer.Ordinal),
213            restored.Files);
214    }

216    private static SaveSlotManifest Read(string json)
217    {
218        using var stream = new MemoryStream(Encoding.UTF8.GetBytes(json), writable: false);
219        return SaveSlotManifestCodec.Read(stream);
220    }
221
222    private static string Write(SaveSlotManifest manifest)
223    {
224        using var stream = new MemoryStream();
225        SaveSlotManifestCodec.Write(stream, manifest);
226        return Encoding.UTF8.GetString(stream.ToArray());
227    }
228
229    private static string CreateManifest(IReadOnlyList<SaveSlotManifestFile> files)
230    {
231        using var stream = new MemoryStream();
232        using var writer = new Utf8JsonWriter(stream);
233        writer.WriteStartObject();
234        writer.WriteString("formatFamily", "simple-cities-v3");
235        writer.WriteNumber("schemaVersion", 1);
236        writer.WriteString("slotId", "autosave");
237        writer.WriteString("displayName", "Autosave");
238        writer.WriteString("timestamp", "2026-08-04T00:00:00.0000000Z");
239        writer.WriteString("cityName", "My City");
240        writer.WriteNull("population");
241        writer.WriteNull("funds");
242        writer.WriteNull("thumbnailFile");
243        writer.WriteStartArray("files");
244        foreach (SaveSlotManifestFile file in files)
245        {
246            writer.WriteStartObject();
247            writer.WriteString("name", file.Name);
248            writer.WriteNumber("encodedLength", file.EncodedLength);
249            writer.WriteString("sha256", file.Sha256);
250            writer.WriteEndObject();
251        }
252        writer.WriteEndArray();
253        writer.WriteEndObject();
254        writer.Flush();
255        return Encoding.UTF8.GetString(stream.ToArray());
256    }

258    private sealed class ChunkedNonSeekableStream(byte[] data, int chunkSize) : Stream
259    {
260        private int _position;
261
262        internal int BytesRead { get; private set; }
263        public override bool CanRead => true;
264        public override bool CanSeek => false;
265        public override bool CanWrite => false;
266        public override long Length => throw new NotSupportedException();
267        public override long Position
268        {
269            get => throw new NotSupportedException();
270            set => throw new NotSupportedException();
271        }
272
273        public override int Read(byte[] buffer, int offset, int count) =>
274            Read(buffer.AsSpan(offset, count));
275
276        public override int Read(Span<byte> buffer)
277        {
278            int count = Math.Min(Math.Min(buffer.Length, chunkSize), data.Length - _position);
279            if (count == 0)
280                return 0;
281            data.AsSpan(_position, count).CopyTo(buffer);
282            _position += count;
283            BytesRead += count;
284            return count;
285        }
286
287        public override void Flush() { }
288        public override long Seek(long offset, SeekOrigin origin) => throw new NotSupportedException();
289        public override void SetLength(long value) => throw new NotSupportedException();
290        public override void Write(byte[] buffer, int offset, int count) => throw new NotSupportedException();
291    }

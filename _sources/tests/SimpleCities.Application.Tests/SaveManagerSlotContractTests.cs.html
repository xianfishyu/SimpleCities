485    {

488        string configuration = new DirectoryInfo(AppContext.BaseDirectory).Parent!.Name;
489        string probeDll = Path.Combine(
490            ProjectRoot(),
491            "tests",
492            "SimpleCities.SaveLockProbe",
493            "bin",
494            configuration,
495            "net10.0",
496            "SimpleCities.SaveLockProbe.dll");
497        Assert.True(File.Exists(probeDll), $"Lock probe was not built: {probeDll}");
498
499        var startInfo = new ProcessStartInfo("dotnet")
500        {
501            RedirectStandardInput = true,
502            RedirectStandardOutput = true,
503            RedirectStandardError = true,
504            UseShellExecute = false,
505            CreateNoWindow = true,
506        };
507        startInfo.ArgumentList.Add(probeDll);
508        startInfo.ArgumentList.Add(lockPath);
509        using Process process = Process.Start(startInfo)
510            ?? throw new InvalidOperationException("Failed to start the save lock probe.");
511        try
512        {
513            string? ready = await process.StandardOutput.ReadLineAsync()
514                .WaitAsync(TimeSpan.FromSeconds(10));
515            Assert.Equal("READY", ready);

518                () => CreateStore().ListSlots());
519            Assert.Contains("another operation or process", exception.Message, StringComparison.Ordinal);
520
521            await process.StandardInput.WriteLineAsync("RELEASE");
522            process.StandardInput.Close();
523            Assert.True(process.WaitForExit(10_000), "Save lock probe did not exit after release.");
524            Assert.Equal(0, process.ExitCode);
525            Assert.Empty(CreateStore().ListSlots());
526        }
527        finally
528        {
529            if (!process.HasExited)
530                process.Kill(entireProcessTree: true);
531        }

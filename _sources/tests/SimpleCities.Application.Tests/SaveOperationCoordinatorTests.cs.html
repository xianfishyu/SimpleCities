354    private sealed record Snapshot(int Value) : ISaveSnapshot;
355    private sealed record Prepared(int Value) : IPreparedSaveState;
356
357    private sealed class ThreadRecordingSaveable(int value)
358        : IStreamingSaveable
359    {
360        private readonly ThreadRecordingLoadReader _reader = new();
361        public string SaveFileName => "road_network";
362        public int Value { get; private set; } = value;
363        public int CaptureThreadID { get; private set; }
364        public int WriteThreadID { get; private set; }
365        public int PrepareCount { get; private set; }
366        public int CommitCount { get; private set; }
367        public int LoadReaderCaptureThreadID { get; private set; }
368        internal ThreadRecordingLoadReader Reader => _reader;
369
370        public ISaveSnapshot CaptureSnapshot()
371        {
372            CaptureThreadID = Environment.CurrentManagedThreadId;
373            return new Snapshot(Value);
374        }
375
376        public void WriteSnapshot(Stream destination, ISaveSnapshot snapshot)
377        {
378            WriteThreadID = Environment.CurrentManagedThreadId;
379            using var writer = new Utf8JsonWriter(destination);
380            writer.WriteStartObject();
381            writer.WriteNumber("value", Assert.IsType<Snapshot>(snapshot).Value);
382            writer.WriteEndObject();
383            writer.Flush();
384        }
385
386        public IStreamingLoadReader CaptureLoadReader()
387        {
388            LoadReaderCaptureThreadID = Environment.CurrentManagedThreadId;
389            _reader.Preparing = () => PrepareCount++;
390            return _reader;
391        }
392
393        public void CommitPreparedLoad(IPreparedSaveState preparedState)
394        {
395            Value = Assert.IsType<Prepared>(preparedState).Value;
396            CommitCount++;
397        }
398    }
399
400    internal sealed class ThreadRecordingLoadReader : IStreamingLoadReader
401    {
402        internal int PrepareThreadID { get; private set; }
403        internal Action? Preparing { get; set; }
404
405        public IPreparedSaveState PrepareLoad(Stream source)
406        {
407            PrepareThreadID = Environment.CurrentManagedThreadId;
408            Preparing?.Invoke();
409            using JsonDocument document = JsonDocument.Parse(source);
410            return new Prepared(document.RootElement.GetProperty("value").GetInt32());
411        }
412    }

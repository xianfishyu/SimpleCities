50internal static class PngTestData
51{
52    internal static byte[] CreateRgba(int width, int height, byte value)
53    {
54        byte[] scanlines = new byte[checked(height * (1 + width * 4))];
55        int offset = 0;
56        for (int y = 0; y < height; y++)
57        {
58            scanlines[offset++] = 0;
59            for (int x = 0; x < width; x++)
60            {
61                scanlines[offset++] = value;
62                scanlines[offset++] = value;
63                scanlines[offset++] = value;
64                scanlines[offset++] = byte.MaxValue;
65            }
66        }
67        return Create(width, height, scanlines);
68    }
69
70    internal static byte[] Create(int width, int height, byte[] decodedScanlines)
71    {
72        using var compressed = new MemoryStream();
73        using (var zlib = new ZLibStream(compressed, CompressionLevel.SmallestSize, leaveOpen: true))
74            zlib.Write(decodedScanlines);
75
76        using var png = new MemoryStream();
77        png.Write([137, 80, 78, 71, 13, 10, 26, 10]);
78        Span<byte> header = stackalloc byte[13];
79        BinaryPrimitives.WriteUInt32BigEndian(header[..4], checked((uint)width));
80        BinaryPrimitives.WriteUInt32BigEndian(header.Slice(4, 4), checked((uint)height));
81        header[8] = 8;
82        header[9] = 6;
83        WriteChunk(png, "IHDR", header);
84        WriteChunk(png, "IDAT", compressed.ToArray());
85        WriteChunk(png, "IEND", []);
86        return png.ToArray();
87    }
88
89    internal static byte[] CreateRgbaWithAncillaryChunk(string chunkType)
90    {
91        byte[] valid = CreateRgba(1, 1, 0x22);
92        using var output = new MemoryStream();
93        output.Write(valid.AsSpan(0, 33));
94        WriteChunk(output, chunkType, []);
95        output.Write(valid.AsSpan(33));
96        return output.ToArray();
97    }
98
99    private static void WriteChunk(Stream destination, string type, ReadOnlySpan<byte> data)
100    {
101        Span<byte> length = stackalloc byte[4];
102        BinaryPrimitives.WriteUInt32BigEndian(length, checked((uint)data.Length));
103        destination.Write(length);
104        byte[] typeBytes = Encoding.ASCII.GetBytes(type);
105        destination.Write(typeBytes);
106        destination.Write(data);
107        Span<byte> crc = stackalloc byte[4];
108        BinaryPrimitives.WriteUInt32BigEndian(crc, ComputeCrc32(typeBytes, data));
109        destination.Write(crc);
110    }
111
112    private static uint ComputeCrc32(ReadOnlySpan<byte> type, ReadOnlySpan<byte> data)
113    {
114        uint crc = uint.MaxValue;
115        foreach (byte value in type)
116            crc = Update(crc, value);
117        foreach (byte value in data)
118            crc = Update(crc, value);
119        return ~crc;
120    }
121
122    private static uint Update(uint crc, byte value)
123    {
124        crc ^= value;
125        for (int bit = 0; bit < 8; bit++)
126            crc = (crc & 1) != 0 ? (crc >> 1) ^ 0xedb88320u : crc >> 1;
127        return crc;
128    }

1using System.Text;
2using System.Text.Json;
3
4namespace SimpleCities.Application.Tests;
5
6public sealed class SavePublicationDescriptorCodecTests
7{
8    private const string OldDigest = "0123456789abcdef0123456789abcdef0123456789abcdef0123456789abcdef";
9    private const string NewDigest = "abcdef0123456789abcdef0123456789abcdef0123456789abcdef0123456789";
10    private const string OperationToken = "0123456789abcdef0123456789abcdef";
11
12    [Fact]
13    public void WriteRead_RoundTripsDeterministicCanonicalDescriptor()
14    {
15        var descriptor = new SavePublicationDescriptor(
16            "manual-1", OperationToken, OldDigest, NewDigest, "staging", "backup");
17
18        string first = Write(descriptor);
19        string second = Write(descriptor);
20        SavePublicationDescriptor restored = Read(first);
21
22        Assert.Equal(first, second);
23        Assert.DoesNotContain('\n', first);
24        Assert.Equal(descriptor, restored);
25    }
26
27    [Theory]
28    [InlineData("\"descriptorType\":\"publish\"", "\"DescriptorType\":\"publish\"")]
29    [InlineData("\"descriptorType\":\"publish\"", "\"descriptorType\":\"delete\"")]
30    [InlineData("\"stagingPath\":\"staging\"", "\"stagingPath\":\"../staging\"")]
31    [InlineData("\"backupPath\":\"backup\"", "\"backupPath\":\"Backup\"")]
32    [InlineData(OperationToken, "0123456789ABCDEF0123456789ABCDEF")]
33    public void Read_RejectsNonCanonicalIdentityOrPaths(string original, string replacement)
34    {
35        string json = Write(new SavePublicationDescriptor(
36            "manual-1", OperationToken, OldDigest, NewDigest, "staging", "backup"));
37
38        Assert.ThrowsAny<JsonException>(() => Read(
39            json.Replace(original, replacement, StringComparison.Ordinal)));
40    }
41
42    [Fact]
43    public void Read_RejectsDuplicateKnownProperty()
44    {
45        string json = Write(new SavePublicationDescriptor(
46            "manual-1", OperationToken, OldDigest, NewDigest, "staging", "backup"));
47        string duplicate = json.Replace(
48            "{\"formatFamily\"",
49            "{\"formatFamily\":\"simple-cities-v3\",\"formatFamily\"",
50            StringComparison.Ordinal);
51
52        Assert.Throws<JsonException>(() => Read(duplicate));
53    }
54
55    [Fact]
56    public void Read_RejectsDescriptorOverByteBudgetBeforeJsonAllocation()
57    {
58        string oversized = "{" + new string(' ', SavePublicationDescriptorCodec.MaximumEncodedBytes) + "}";
59
60        Assert.Throws<InvalidDataException>(() => Read(oversized));
61    }
62
63    private static string Write(SavePublicationDescriptor descriptor)
64    {
65        using var stream = new MemoryStream();
66        SavePublicationDescriptorCodec.Write(stream, descriptor);
67        return Encoding.UTF8.GetString(stream.ToArray());
68    }
69
70    private static SavePublicationDescriptor Read(string json)
71    {
72        using var stream = new MemoryStream(Encoding.UTF8.GetBytes(json), writable: false);
73        return SavePublicationDescriptorCodec.Read(stream);
74    }
75}

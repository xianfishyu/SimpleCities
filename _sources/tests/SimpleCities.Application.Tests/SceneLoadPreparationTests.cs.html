7    [Fact]
8    public async Task V4Slot_PreparesCoreAndPurePresentationBeforeCommit()
9    {
10        string directory = Path.Combine(Path.GetTempPath(), $"simple-cities-load-context-{Guid.NewGuid():N}");
11        try
12        {
13            var saved = RoadSaveTestFixture.Create(RoadProfileId.Street, new(100, 0));
14            var savedParticipant = new RoadSaveParticipant(saved);
15            byte[] expected = RoadSaveTestFixture.Capture(saved);
16            var store = new SaveSlotStore(directory);
17            string slotID = store.Create("Generic preparation", [savedParticipant]);
18
19            var active = RoadSaveTestFixture.Create(RoadProfileId.Dirt, new(0, 100));
20            var participant = new RoadSaveParticipant(active);
21            byte[] original = RoadSaveTestFixture.Capture(active);
22            RoadStateToken originalToken = active.Snapshot.Token;
23            var context = new SceneLoadPreparationContext(7, participant, CreatePreparer());
24            IReadOnlyList<CapturedLoadParticipant> readers = SaveSlotStore.CaptureLoadParticipants([participant]);
25
26            PreparedSceneLoad prepared = await Task.Run(() =>
27            {
28                PreparedSaveSlot slot = store.PrepareLoad(
29                    slotID, readers, new UncoordinatedStorageOperationLease(SaveOperationKind.Load));
30                IPreparedSaveState networkState = slot.GetPreparedState(context.NetworkTarget);
31                IPreparedScenePresentation presentation = context.PresentationPreparer.Prepare(networkState);
32                return new PreparedSceneLoad(context, slot, networkState, presentation, TimeSpan.Zero);
33            });
34
35            Assert.Equal(original, RoadSaveTestFixture.Capture(active));
36            Assert.Equal(originalToken, active.Snapshot.Token);
37            Assert.Equal(slotID, prepared.Slot.SlotID);
38            Assert.Equal(7, prepared.Context.Generation);
39            PreparedRoadPresentation presentation = Assert.IsType<PreparedRoadPresentation>(prepared.Presentation);
40            Assert.Equal(1, presentation.Target.EdgeCount);
41            Assert.Equal([new RoadPoint(0, 0), new RoadPoint(100, 0)], Assert.Single(presentation.Target.Edges).Points);
42            Assert.True(presentation.Geometry.UploadBytes > 0);
43
44            participant.CommitPreparedLoad(prepared.NetworkState);
45
46            Assert.Equal(expected, RoadSaveTestFixture.Capture(active));
47            Assert.NotEqual(originalToken.Lineage, active.Snapshot.Token.Lineage);
48        }
49        finally
50        {
51            if (Directory.Exists(directory))
52                Directory.Delete(directory, recursive: true);
53        }
54    }

56    [Fact]
57    public void NetworkParticipant_PreflightsWithoutPublishingAndCommitsOnce()
58    {
59        var saved = RoadSaveTestFixture.Create(RoadProfileId.Street, new(100, 0));
60        byte[] expected = RoadSaveTestFixture.Capture(saved);
61        var active = new RoadNetwork();
62        var participant = new RoadSaveParticipant(active);
63        byte[] original = RoadSaveTestFixture.Capture(active);
64        IPreparedSaveState loaded = RoadSaveTestFixture.Prepare(participant, saved);
65        using ISceneNetworkLoadAdmission admission = participant.BeginSceneLoadAdmission();
66        using INonThrowingLoadCommitPlan plan = admission.PreflightPreparedLoad(loaded, out IPreparedSaveState target);
67
68        Assert.Equal(original, RoadSaveTestFixture.Capture(active));
69        Assert.IsType<RoadSaveParticipant.Target>(target);
70        long sequence = active.Snapshot.Token.ChangeSequence;
71        using var aggregate = new PreparedAggregateLoad([plan]);
72        aggregate.Commit(new UncoordinatedStorageOperationLease(SaveOperationKind.Load));
73
74        Assert.Equal(expected, RoadSaveTestFixture.Capture(active));
75        Assert.Equal(sequence + 1, active.Snapshot.Token.ChangeSequence);
76        Assert.Throws<InvalidOperationException>(() => aggregate.Commit(new UncoordinatedStorageOperationLease(SaveOperationKind.Load)));
77        Assert.False(plan.IsGenerationCurrent);
78    }

80    [Fact]
81    public async Task NetworkParticipant_CanceledAggregateKeepsTheOriginalStateAndReleasesAdmission()
82    {
83        var active = new RoadNetwork();
84        var participant = new RoadSaveParticipant(active);
85        byte[] original = RoadSaveTestFixture.Capture(active);
86        RoadStateToken originalToken = active.Snapshot.Token;
87        var saved = RoadSaveTestFixture.Create(RoadProfileId.Street, new(100, 0));
88        IPreparedSaveState loaded = RoadSaveTestFixture.Prepare(participant, saved);
89        using var cancellation = new CancellationTokenSource();
90        await using var coordinator = new SaveOperationCoordinator();
91        SaveOperationAdmission operation = await coordinator.AdmitManualAsync(
92            SaveOperationKind.Load, "manual-1", cancellation.Token);
93        await using SaveOperationLease lease = Assert.IsType<SaveOperationLease>(operation.Lease);
94
95        using (ISceneNetworkLoadAdmission admission = participant.BeginSceneLoadAdmission())
96        using (var aggregate = new PreparedAggregateLoad([admission.PreflightPreparedLoad(loaded, out _)]))
97        {
98            cancellation.Cancel();
99
100            Assert.Throws<OperationCanceledException>(() => aggregate.Commit(lease));
101            Assert.Equal(original, RoadSaveTestFixture.Capture(active));
102            Assert.Equal(originalToken, active.Snapshot.Token);
103        }
104
105        using ISceneNetworkLoadAdmission next = participant.BeginSceneLoadAdmission();
106        RoadSaveTestFixture.Build(active, RoadProfileId.Dirt, new(0, 0), new(100, 0));
107    }

109    [Theory]
110    [InlineData(false)]
111    [InlineData(true)]
112    public void PreparedLoad_RejectsAStateOrTargetFromAnotherPreparation(bool replaceState)
113    {
114        var active = new RoadNetwork();
115        var participant = new RoadSaveParticipant(active);
116        var other = RoadSaveTestFixture.Create(RoadProfileId.Street, new(100, 0));
117        var otherParticipant = new RoadSaveParticipant(other);
118        IPreparedSaveState state = RoadSaveTestFixture.Prepare(participant, other);
119        var slot = new PreparedSaveSlot("manual-1", [new PreparedSaveParticipant(participant, state)]);
120        var context = new SceneLoadPreparationContext(7, replaceState ? participant : otherParticipant, CreatePreparer());
121        IPreparedSaveState suppliedState = replaceState
122            ? RoadSaveTestFixture.Prepare(participant, active)
123            : state;
124        IPreparedScenePresentation presentation = context.PresentationPreparer.Prepare(suppliedState);
125        byte[] original = RoadSaveTestFixture.Capture(active);
126
127        Assert.Throws<InvalidOperationException>(() =>
128            new PreparedSceneLoad(context, slot, suppliedState, presentation, TimeSpan.Zero));
129
130        Assert.Equal(original, RoadSaveTestFixture.Capture(active));
131    }

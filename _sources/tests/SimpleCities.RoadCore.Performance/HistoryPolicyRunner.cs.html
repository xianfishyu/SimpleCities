26    public static void Run(string datasetDirectory, string outputDirectory)
27    {
28#if DEBUG
29        throw new InvalidOperationException("Use the Release diagnostic build.");
30#else
31        if (Environment.GetEnvironmentVariable("DOTNET_TieredCompilation") != "0")
32            throw new InvalidOperationException("Set DOTNET_TieredCompilation=0 for stable diagnostic variable lifetimes.");
33        string output = Path.GetFullPath(outputDirectory);
34        if (Directory.Exists(output)) throw new IOException("Use a new evidence directory.");
35        Directory.CreateDirectory(Path.Combine(output, "datasets"));
36        var datasets = new List<object>();
37        var histories = new List<object>();
38        var cancellations = new List<object>();
39        var releases = new List<object>();
40        string reportPath = Path.Combine(output, "history-policy.json");
41        void Save(string status) => File.WriteAllText(reportPath, JsonSerializer.Serialize(new
42        {
43            status, measuredUtc = DateTimeOffset.UtcNow, runtime = RuntimeInformation.FrameworkDescription,
44            os = RuntimeInformation.OSDescription, tieredCompilation = "0", historyCapacity = RoadEditHistory.Capacity,
45            historyByteAdmission = "none", workerWarningMs = 10,
46            method = "Retain the identical current snapshot; release only its owning network/history. Shared-entity sets are discarded before live-heap measurements. Signed heap deltas and forced GC times are separate from reference release. Input-copy progress and timed cancellation do not identify a particular internal topology phase. No UI/GPU latency assertion.",
47            datasets, histories, cancellations, releases,
48        }, Json));
49        try
50        {
51            foreach (string name in new[] { "target-25", "target-50", "target-100", "target-200", "polylines-200", "high-chain-25", "limit-25" })
52            {
53                byte[] payload = name switch
54                {
55                    "high-chain-25" => HighChainPayload(),
56                    "limit-25" => Encode(CreateLimitLattice().Snapshot),
57                    _ => File.ReadAllBytes(Path.Combine(datasetDirectory, name + ".json")),
58                };
59                RoadSnapshot validated = Decode(payload).Snapshot;
60                Require(payload.SequenceEqual(Encode(validated)), "Strict Codec roundtrip changed the dataset.");
61                File.WriteAllBytes(Path.Combine(output, "datasets", name + ".json"), payload);
62                datasets.Add(new { name, validated.EdgeCount, validated.NodeCount,
63                    chainPoints = validated.Edges.Sum(edge => edge.Points.Count),
64                    maximumEdgePoints = validated.Edges.Select(edge => edge.Points.Count).DefaultIfEmpty().Max(),
65                    bytes = payload.Length, sha256 = Convert.ToHexString(SHA256.HashData(payload)) });
66                Save("running");
67                string[] scenarios = name == "limit-25" ? ["all"] : name is "target-200" or "high-chain-25"
68                    ? ["single", "batch128", "delete3", "all"] : ["single", "batch128", "delete3", "all", "build"];
69                foreach (string scenario in scenarios)
70                {
71                    _ = MeasureHistory(payload, scenario, name);
72                    for (int repeat = 0; repeat < 3; repeat++)
73                    {
74                        HistoryMeasurement measurement = MeasureHistory(payload, scenario, name);
75                        if (name == "limit-25")
76                            Require(measurement.MaximumEntryBytes > 4L * 1024 * 1024 && measurement.EstimatedBytes > 256L * 1024 * 1024,
77                                "Limit fixture must exercise both discarded byte-budget suggestions.");
78                        histories.Add(new { name, scenario, repeat, measurement });
79                        Save("running");
80                        Console.WriteLine($"HISTORY {name} {scenario} {repeat} estimated={measurement.EstimatedBytes} live={measurement.MarginalLiveBytes}");
81                    }
82                }
83                VerifyEviction(payload);
84                // Keep one unchanged network alive throughout repeated cancellation and abandoned plans.
85                RoadNetwork network = Decode(payload);
86                RoadGridSpan[] spans = RoadBaselineFixtures.Spans(network.Snapshot);
87                foreach (int delay in new[] { 0, 1, 5, 15 })
88                    for (int repeat = 0; repeat < 10; repeat++)
89                    {
90                        CancellationMeasurement measurement = MeasureCancellation(network, spans, delay);
91                        cancellations.Add(new { name, repeat, measurement });
92                    }
93                for (int repeat = 0; repeat < 10; repeat++)
94                    releases.Add(new { name, repeat, measurement = MeasureRelease(network, spans) });
95                Save("running");
96                Console.WriteLine($"LIFETIME {name} 40 cancellation attempts, 10 abandoned candidates, eviction verified");
97            }
98            Save("complete");
99        }
100        catch { Save("failed"); throw; }
101#endif

104    [MethodImpl(MethodImplOptions.NoInlining)]
105    private static HistoryHolder Populate(byte[] payload, string scenario, string name)
106    {
107        RoadNetwork network = Decode(payload);
108        var changes = new List<RoadChangeSet>();
109        long maximum = 0;
110        for (int i = 0; i < 64; i++)
111        {
112            RoadPlan plan;
113            if (scenario == "build")
114            {
115                int cell = network.Snapshot.Map.CellSizeMetres;
116                var start = name == "polylines-200"
117                    ? new RoadPoint(2400 + 400 * (i % 4), -4000 + 400 * (i / 4))
118                    : new RoadPoint(-4000 + 2 * cell * (i % 4), cell + 2 * cell * (i / 4));
119                plan = network.PlanBuild(new(network.Snapshot.Token, start, new(start.X + cell, start.Y), RoadProfileId.Street)).Plan
120                    ?? throw new InvalidOperationException("Build fixture was rejected.");
121            }
122            else
123            {
124                RoadGridSpan[] spans = RoadBaselineFixtures.Spans(network.Snapshot);
125                spans = scenario switch { "single" => [spans[0]], "batch128" => spans.Take(128).ToArray(), "delete3" => spans.Take(3).ToArray(), _ => spans };
126                RoadEditResult edit = scenario == "delete3" ? network.PlanRemove(spans) :
127                    network.PlanChangeProfile(spans, i % 2 == 0 ? RoadProfileId.Highway : RoadProfileId.Dirt);
128                plan = edit.Plan ?? throw new InvalidOperationException(edit.Reason);
129            }
130            long before = network.History.EstimatedBytes;
131            Require(network.TryCommit(plan), "Prepared history edit failed to commit.");
132            maximum = Math.Max(maximum, network.History.EstimatedBytes - before);
133            changes.Add(plan.ChangeSet);
134        }
135        Require(network.History.RetainedCount == 64, "History did not retain 64 entries.");
136        return new(network, maximum, CountSharing(changes, network.Snapshot));
137    }

139    private static Sharing CountSharing(IReadOnlyList<RoadChangeSet> changes, RoadSnapshot current)
140    {
141        var nodes = new HashSet<RoadNode>(ReferenceEqualityComparer.Instance);
142        var edges = new HashSet<RoadEdge>(ReferenceEqualityComparer.Instance);
143        long nodeSlots = 0, edgeSlots = 0;
144        foreach (RoadChangeSet change in changes)
145        {
146            foreach (RoadNodeChange item in change.Nodes)
147            {
148                if (item.Before is not null) { nodeSlots++; nodes.Add(item.Before); }
149                if (item.After is not null) { nodeSlots++; nodes.Add(item.After); }
150            }
151            foreach (RoadEdgeChange item in change.Edges)
152            {
153                if (item.Before is not null) { edgeSlots++; edges.Add(item.Before); }
154                if (item.After is not null) { edgeSlots++; edges.Add(item.After); }
155            }
156        }
157        return new(nodeSlots, edgeSlots, nodes.Count, edges.Count,
158            current.Nodes.Count(nodes.Contains), current.Edges.Count(edges.Contains));
159    }

161    [MethodImpl(MethodImplOptions.NoInlining)]
162    private static HistoryMeasurement MeasureHistory(byte[] payload, string scenario, string name)
163    {
164        HistoryHolder holder = Populate(payload, scenario, name);
165        long estimated = holder.Network!.History.EstimatedBytes;
166        int retained = holder.Network.History.RetainedCount;
167        var weak = new WeakReference(holder.Network.History);
168        long stamp = Stopwatch.GetTimestamp(), before = Collect();
169        double beforeGc = Stopwatch.GetElapsedTime(stamp).TotalMilliseconds;
170        stamp = Stopwatch.GetTimestamp();
171        holder.Release();
172        double release = Stopwatch.GetElapsedTime(stamp).TotalMilliseconds;
173        stamp = Stopwatch.GetTimestamp();
174        long after = Collect();
175        double afterGc = Stopwatch.GetElapsedTime(stamp).TotalMilliseconds;
176        Require(!weak.IsAlive, "Released history remains rooted.");
177        GC.KeepAlive(holder.Current);
178        return new(estimated, holder.Maximum, before, after, before - after, release, beforeGc, afterGc,
179            !weak.IsAlive, retained, holder.Current.EdgeCount, holder.Sharing);
180    }

182    [MethodImpl(MethodImplOptions.NoInlining)]
183    private static CancellationMeasurement MeasureCancellation(RoadNetwork network, RoadGridSpan[] spans, int delay)
184    {
185        RoadSnapshot before = network.Snapshot;
186        RoadEditHistory history = network.History;
187        using var copied = new ManualResetEventSlim();
188        using var cancellation = new CancellationTokenSource();
189        long copiedAt = 0, exitedAt = 0, allocated = 0;
190        bool observed = false;
191        var input = new ObservedSpans(spans, () => { Volatile.Write(ref copiedAt, Stopwatch.GetTimestamp()); copied.Set(); });
192        Task task = Task.Run(() =>
193        {
194            long allocationStart = GC.GetAllocatedBytesForCurrentThread();
195            try { _ = network.PlanChangeProfile(input, RoadProfileId.Highway, cancellation.Token); }
196            catch (OperationCanceledException) when (cancellation.IsCancellationRequested) { observed = true; }
197            finally
198            {
199                allocated = GC.GetAllocatedBytesForCurrentThread() - allocationStart;
200                Volatile.Write(ref exitedAt, Stopwatch.GetTimestamp());
201            }
202        });
203        Require(copied.Wait(TimeSpan.FromSeconds(10)), "Worker did not finish copying the selection.");
204        if (delay > 0) Thread.Sleep(delay);
205        long requestedAt = Stopwatch.GetTimestamp();
206        cancellation.Cancel();
207        task.GetAwaiter().GetResult();
208        bool early = exitedAt <= requestedAt;
209        double elapsed = early ? 0 : Stopwatch.GetElapsedTime(requestedAt, exitedAt).TotalMilliseconds;
210        Require(ReferenceEquals(before, network.Snapshot) && ReferenceEquals(history, network.History), "Cancellation changed active state.");
211        return new(delay, Stopwatch.GetElapsedTime(copiedAt, requestedAt).TotalMilliseconds,
212            early ? "ExitedBeforeRequest" : observed ? "ObservedCancellation" : "CompletedAfterRequest",
213            observed, elapsed, elapsed > 10, allocated, true);
214    }

220    [MethodImpl(MethodImplOptions.NoInlining)]
221    private static ReleaseMeasurement MeasureRelease(RoadNetwork network, RoadGridSpan[] spans)
222    {
223        RoadSnapshot before = network.Snapshot;
224        RoadEditHistory history = network.History;
225        CandidateHolder holder = PrepareCandidate(network, spans);
226        var plan = new WeakReference(holder.Plan);
227        var snapshot = new WeakReference(holder.Plan!.Target);
228        var change = new WeakReference(holder.Plan.ChangeSet);
229        long liveBefore = Collect(), started = Stopwatch.GetTimestamp();
230        holder.Release();
231        double referenceMs = Stopwatch.GetElapsedTime(started).TotalMilliseconds;
232        started = Stopwatch.GetTimestamp();
233        long liveAfter = Collect();
234        double collectionMs = Stopwatch.GetElapsedTime(started).TotalMilliseconds;
235        Require(!plan.IsAlive && !snapshot.IsAlive && !change.IsAlive, "An abandoned plan, target or change set is still rooted.");
236        Require(ReferenceEquals(before, network.Snapshot) && ReferenceEquals(history, network.History), "Candidate release changed state.");
237        GC.KeepAlive(network);
238        return new(liveBefore - liveAfter, referenceMs, collectionMs, true, true, true, true);
239    }

241    private static void VerifyEviction(byte[] payload)
242    {
243        RoadNetwork network = Decode(payload);
244        long oldestRevision = 0;
245        for (int i = 0; i < 65; i++)
246        {
247            Require(network.TryCommit(network.PlanChangeProfile(RoadBaselineFixtures.Spans(network.Snapshot),
248                i % 2 == 0 ? RoadProfileId.Highway : RoadProfileId.Dirt).Plan!), "Eviction edit failed.");
249            if (i == 0) oldestRevision = network.Snapshot.Token.ContentRevision;
250        }
251        long finalRevision = network.Snapshot.Token.ContentRevision;
252        Require(network.History.RetainedCount == 64, "Entry 65 did not evict only the oldest entry.");
253        for (int i = 0; i < 64; i++) Require(network.TryCommit(network.PlanUndo().Plan!), "Undo failed.");
254        Require(network.Snapshot.Token.ContentRevision == oldestRevision && network.PlanUndo().Status == RoadEditStatus.NoChange,
255            "Oldest reachable content does not match count-only eviction.");
256        for (int i = 0; i < 64; i++) Require(network.TryCommit(network.PlanRedo().Plan!), "Redo failed.");
257        Require(network.Snapshot.Token.ContentRevision == finalRevision && network.History.UndoCount == 64, "Redo lost retained history.");
258    }

260    private static RoadNetwork CreateLimitLattice()
261    {
262        var network = new RoadNetwork(new MapDefinition(25));
263        const int size = 52;
264        void Line(int ax, int ay, int bx, int by) => RoadBaselineFixtures.Build(network,
265            new(-4000 + ax * 25, -4000 + ay * 25), new(-4000 + bx * 25, -4000 + by * 25));
266        for (int i = 0; i <= size; i++) { Line(0, i, size, i); Line(i, 0, i, size); }
267        for (int offset = 1 - size; offset < size; offset++)
268        {
269            int x = Math.Max(0, -offset), y = Math.Max(0, offset), length = size - Math.Abs(offset);
270            Line(x, y, x + length, y + length);
271        }
272        for (int sum = 1; sum < 2 * size; sum++)
273            Line(Math.Max(0, sum - size), Math.Min(size, sum), Math.Min(size, sum), Math.Max(0, sum - size));
274        for (int i = 0; i < 56; i++)
275            RoadBaselineFixtures.Build(network, new(-4000 + 50 * (i % 8), 25 + 50 * (i / 8)), new(-3975 + 50 * (i % 8), 25 + 50 * (i / 8)));
276        Require(network.Snapshot.EdgeCount == 16384, "Limit lattice must have exactly 16,384 canonical edges.");
277        return network;
278    }

280    private static byte[] HighChainPayload()
281    {
282        JsonObject document = JsonNode.Parse(Encode(new RoadNetwork(new MapDefinition(25)).Snapshot))!.AsObject();
283        var points = new List<RoadPoint> { new(-4000, -4000) };
284        for (int row = 0; row < 32; row++)
285        {
286            int startX = row % 2 == 0 ? -4000 : 4000, direction = row % 2 == 0 ? 1 : -1, y = -4000 + row * 50;
287            if (row > 0) points.Add(new(startX, y));
288            for (int step = 1; step <= 320; step++) points.Add(new(startX + direction * step * 25, y + step % 2 * 25));
289        }
290        document["nextNodeId"] = 3;
291        document["nextEdgeId"] = 2;
292        document["nodes"] = new JsonArray(new JsonObject { ["id"] = 1, ["x"] = points[0].X, ["y"] = points[0].Y },
293            new JsonObject { ["id"] = 2, ["x"] = points[^1].X, ["y"] = points[^1].Y });
294        document["edges"] = new JsonArray(new JsonObject
295        {
296            ["id"] = 1, ["startNodeId"] = 1, ["endNodeId"] = 2, ["profile"] = RoadProfileId.Street.Value,
297            ["points"] = new JsonArray(points.Select(point => (JsonNode)new JsonObject { ["x"] = point.X, ["y"] = point.Y }).ToArray()),
298        });
299        byte[] payload = System.Text.Encoding.UTF8.GetBytes(document.ToJsonString());
300        return Encode(Decode(payload).Snapshot);
301    }

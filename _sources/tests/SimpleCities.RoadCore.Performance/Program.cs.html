83static byte[] Encode(RoadSnapshot snapshot)
84{
85    using var stream = new MemoryStream();
86    RoadCodec.Write(stream, snapshot);
87    return stream.ToArray();
88}

90static RoadNetwork Decode(byte[] payload)
91{
92    using var stream = new MemoryStream(payload, writable: false);
93    var network = new RoadNetwork();
94    if (!network.TryCommit(network.PlanLoad(RoadCodec.Read(stream)))) throw new InvalidOperationException("Load rejected.");
95    return network;
96}

98[MethodImpl(MethodImplOptions.NoInlining)]
99static HistoryHolder PopulateHistory(byte[] payload, string scenario)
100{
101    var network = Decode(payload);
102    long start = Stopwatch.GetTimestamp();
103    long allocated = GC.GetTotalAllocatedBytes(true);
104    long maximumEntry = 0;
105    for (int i = 0; i < 64; i++)
106    {
107        RoadGridSpan[] spans = RoadBaselineFixtures.Spans(network.Snapshot);
108        if (scenario == "singleSpan") spans = [spans[0]];
109        if (scenario == "batchDelete") spans = spans.Take(3).ToArray();
110        long estimate = network.History.EstimatedBytes;
111        RoadEditResult result = scenario == "batchDelete" ? network.PlanRemove(spans) :
112            network.PlanChangeProfile(spans, i % 2 == 0 ? RoadProfileId.Dirt : RoadProfileId.Street);
113        if (result.Plan is null || !network.TryCommit(result.Plan))
114            throw new InvalidOperationException($"History {scenario} {i}: {result.Reason}");
115        maximumEntry = Math.Max(maximumEntry, network.History.EstimatedBytes - estimate);
116    }
117    if (network.History.RetainedCount != 64) throw new InvalidOperationException("History did not retain 64 edits.");
118    return new HistoryHolder(network, maximumEntry, Stopwatch.GetElapsedTime(start).TotalMilliseconds,
119        GC.GetTotalAllocatedBytes(true) - allocated);
120}

122[MethodImpl(MethodImplOptions.NoInlining)]
123static object MeasureHistory(byte[] payload, string scenario)
124{
125    HistoryHolder holder = PopulateHistory(payload, scenario);
126    var historyWeak = new WeakReference(holder.Network!.History);
127    long estimatedBytes = holder.Network.History.EstimatedBytes;
128    long collectStart = Stopwatch.GetTimestamp();
129    long before = CollectLiveBytes();
130    double beforeGcMs = Stopwatch.GetElapsedTime(collectStart).TotalMilliseconds;
131    long releaseStart = Stopwatch.GetTimestamp();
132    holder.ReleaseNetwork();
133    double releaseReferenceMs = Stopwatch.GetElapsedTime(releaseStart).TotalMilliseconds;
134    collectStart = Stopwatch.GetTimestamp();
135    long after = CollectLiveBytes();
136    double afterGcMs = Stopwatch.GetElapsedTime(collectStart).TotalMilliseconds;
137    bool collected = !historyWeak.IsAlive;
138    if (!collected) throw new InvalidOperationException("Released history is still rooted. Use DOTNET_TieredCompilation=0.");
139    GC.KeepAlive(holder.Snapshot);
140    return new { estimatedBytes, maximumEntryEstimatedBytes = holder.MaximumEntry,
141        managedLiveBefore = before, managedLiveAfter = after, marginalManagedBytes = before - after,
142        historyCollected = collected, currentSnapshotEdges = holder.Snapshot.EdgeCount,
143        holder.OperationMs, holder.AllocatedBytes, releaseReferenceMs, beforeGcMs, afterGcMs };
144}

146static long CollectLiveBytes()
147{
148    GC.Collect(GC.MaxGeneration, GCCollectionMode.Forced, blocking: true, compacting: true);
149    GC.WaitForPendingFinalizers();
150    return GC.GetTotalMemory(forceFullCollection: true);
151}

153static object MeasureCancellation(byte[] payload)
154{
155    RoadNetwork network = Decode(payload);
156    RoadSnapshot before = network.Snapshot;
157    RoadGridSpan[] spans = RoadBaselineFixtures.Spans(before);
158    using var cancellation = new CancellationTokenSource();
159    using var entered = new ManualResetEventSlim();
160    Task<RoadEditResult> task = Task.Run(() =>
161    {
162        entered.Set();
163        return network.PlanChangeProfile(spans, RoadProfileId.Highway, cancellation.Token);
164    });
165    if (!entered.Wait(TimeSpan.FromSeconds(5))) throw new TimeoutException("Worker never started.");
166    bool completedBeforeRequest = task.IsCompleted;
167    long start = Stopwatch.GetTimestamp();
168    cancellation.Cancel();
169    bool observed = false;
170    try { task.GetAwaiter().GetResult(); }
171    catch (OperationCanceledException) { observed = true; }
172    double tokenToWorkerExitMs = Stopwatch.GetElapsedTime(start).TotalMilliseconds;
173    if (!ReferenceEquals(before, network.Snapshot) || network.History.RetainedCount != 0)
174        throw new InvalidOperationException("Uncommitted cancellation changed the network.");
175    return new { completedBeforeRequest, cancellationObserved = observed, tokenToWorkerExitMs, stateUnchanged = true };
176}

178[MethodImpl(MethodImplOptions.NoInlining)]
179static CandidateHolder PrepareCandidate(byte[] payload)
180{
181    RoadNetwork network = Decode(payload);
182    RoadEditResult result = network.PlanChangeProfile(RoadBaselineFixtures.Spans(network.Snapshot), RoadProfileId.Highway);
183    return new CandidateHolder(network, result.Plan ?? throw new InvalidOperationException("Candidate rejected."));
184}

186[MethodImpl(MethodImplOptions.NoInlining)]
187static object MeasureCandidateRelease(byte[] payload)
188{
189    CandidateHolder holder = PrepareCandidate(payload);
190    var weak = new WeakReference(holder.Plan);
191    RoadSnapshot snapshot = holder.Network.Snapshot;
192    long before = CollectLiveBytes();
193    long started = Stopwatch.GetTimestamp();
194    holder.Release();
195    double releaseReferenceMs = Stopwatch.GetElapsedTime(started).TotalMilliseconds;
196    started = Stopwatch.GetTimestamp();
197    long after = CollectLiveBytes();
198    double collectionMs = Stopwatch.GetElapsedTime(started).TotalMilliseconds;
199    bool candidateCollected = !weak.IsAlive;
200    bool stateUnchanged = ReferenceEquals(snapshot, holder.Network.Snapshot) && holder.Network.History.RetainedCount == 0;
201    if (!candidateCollected || !stateUnchanged) throw new InvalidOperationException("Candidate release retained a root or changed state.");
202    GC.KeepAlive(holder.Network);
203    return new { marginalManagedBytes = before - after, releaseReferenceMs, collectionMs, candidateCollected, stateUnchanged };
204}

20    public static RoadGridSpan[] Spans(RoadSnapshot snapshot)
21    {
22        var spans = new Dictionary<RoadSpanKey, RoadGridSpan>();
23        foreach (RoadEdge edge in snapshot.Edges)
24        {
25            double travelled = 0;
26            for (int i = 1; i < edge.Points.Count; i++)
27            {
28                RoadPoint a = edge.Points[i - 1], b = edge.Points[i];
29                double dx = b.X - a.X, dy = b.Y - a.Y;
30                double length = Math.Sqrt(dx * dx + dy * dy);
31                int count = Math.Max(1, (int)Math.Ceiling(Math.Max(Math.Abs(dx), Math.Abs(dy)) / snapshot.Map.CellSizeMetres));
32                for (int j = 0; j < count; j++)
33                {
34                    double parameter = (travelled + length * (j + 0.5) / count) / edge.Length;
35                    RoadGridSpan span = RoadSpanQuery.Pick(snapshot, new(snapshot.Token, edge.Id, parameter))
36                        ?? throw new InvalidOperationException("Fixture has an unselectable span.");
37                    spans.TryAdd(span.Key, span);
38                }
39                travelled += length;
40            }
41        }
42        return spans.Values.ToArray();
43    }

45    public static RoadNetwork CreateMixed(int cell)
46    {
47        var network = new RoadNetwork(new MapDefinition(cell));
48        // Six-by-six complete square-eight lattice: 6*n*n + 2*n = 228 canonical edges.
49        for (int i = 0; i <= 6; i++)
50        {
51            Build(network, new(0, i * cell), new(6 * cell, i * cell));
52            Build(network, new(i * cell, 0), new(i * cell, 6 * cell));
53        }
54        for (int y = 0; y < 6; y++)
55            for (int x = 0; x < 6; x++)
56            {
57                Build(network, new(x * cell, y * cell), new((x + 1) * cell, (y + 1) * cell));
58                Build(network, new((x + 1) * cell, y * cell), new(x * cell, (y + 1) * cell));
59            }
60        // Twelve disconnected L chains add twelve edges, not twenty-four grid spans.
61        for (int row = 0; row < 4; row++)
62            for (int column = 0; column < 3; column++)
63            {
64                int x = (-12 + 3 * column) * cell;
65                int y = 3 * row * cell;
66                Build(network, new(x, y), new(x + cell, y));
67                Build(network, new(x + cell, y), new(x + cell, y + cell));
68            }
69        return network;
70    }

72    public static void Build(RoadNetwork network, RoadPoint from, RoadPoint to)
73    {
74        RoadBuildResult result = network.PlanBuild(new(network.Snapshot.Token, from, to, RoadProfileId.Street));
75        if (result.Plan is null || !network.TryCommit(result.Plan))
76            throw new InvalidOperationException($"Fixture build {from} -> {to}: {result.Reason}");
77    }

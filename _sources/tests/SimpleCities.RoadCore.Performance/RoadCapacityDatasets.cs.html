5    public static RoadNetwork CreateTarget(int cell)
6    {
7        var network = new RoadNetwork(new MapDefinition(cell));
8        const int size = 40;
9        RoadPoint Point(int x, int y) => new(-4000 + x * cell, -4000 + y * cell);
10        void Line(int ax, int ay, int bx, int by) => RoadBaselineFixtures.Build(network, Point(ax, ay), Point(bx, by));
11        for (int i = 0; i <= size; i++)
12        {
13            Line(0, i, size, i);
14            Line(i, 0, i, size);
15        }
16        // Whole diagonals cross the existing grid through the normal noding planner.
17        for (int offset = 1 - size; offset < size; offset++)
18        {
19            int x = Math.Max(0, -offset), y = Math.Max(0, offset), length = size - Math.Abs(offset);
20            Line(x, y, x + length, y + length);
21        }
22        for (int sum = 1; sum < 2 * size; sum++)
23            Line(Math.Max(0, sum - size), Math.Min(size, sum), Math.Min(size, sum), Math.Max(0, sum - size));
24        if (cell != 200) AddPolylines(network, cell, cell);
25        int expected = cell == 200 ? 9680 : 10000;
26        if (network.Snapshot.EdgeCount != expected) throw new InvalidOperationException($"Target {cell}: expected {expected} canonical edges.");
27        return network;
28    }

37    private static void AddPolylines(RoadNetwork network, int originX, int originY)

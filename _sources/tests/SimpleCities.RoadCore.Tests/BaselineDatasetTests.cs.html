1using SimpleCities.RoadCore.Performance;
2
3namespace SimpleCities.RoadCore.Tests;
4
5public sealed class BaselineDatasetTests

7    [Theory]
8    [InlineData(25)]
9    [InlineData(50)]
10    [InlineData(100)]
11    [InlineData(200)]
12    public void PolylineBaselineRetainsFortySegmentsPerIndependentChain(int cell)
13    {
14        RoadNetwork network = RoadBaselineFixtures.CreatePolylines(cell);
15        Assert.Equal(20, network.Snapshot.EdgeCount);
16        Assert.Equal(820, network.Snapshot.Edges.Sum(edge => edge.Points.Count));
17        Assert.All(network.Snapshot.Edges, edge => Assert.Equal(41, edge.Points.Count));
18        using var encoded = new MemoryStream();
19        RoadCodec.Write(encoded, network.Snapshot);
20        encoded.Position = 0;
21        Assert.Equal(20, RoadCodec.Read(encoded).Edges.Count);
22    }
23
24    [Theory]
25    [InlineData(25)]
26    [InlineData(50)]
27    [InlineData(100)]
28    [InlineData(200)]
29    public void MixedBaselineContainsDenseJunctionsAndPolylinesAndRoundTrips(int cell)
30    {
31        RoadNetwork network = RoadBaselineFixtures.CreateMixed(cell);
32        RoadSnapshot snapshot = network.Snapshot;
33        Assert.Equal(240, snapshot.EdgeCount);
34        Assert.Equal(36, snapshot.Nodes.Count(node => snapshot.Map.IsCellCenter(node.Position)));
35        Assert.Equal(12, snapshot.Edges.Count(edge => edge.Points.Count == 3));
36        using var encoded = new MemoryStream();
37        RoadCodec.Write(encoded, snapshot);
38        encoded.Position = 0;
39        var restored = new RoadNetwork();
40        Assert.True(restored.TryCommit(restored.PlanLoad(RoadCodec.Read(encoded))));
41        using var repeated = new MemoryStream();
42        RoadCodec.Write(repeated, restored.Snapshot);
43        Assert.Equal(encoded.ToArray(), repeated.ToArray());
44        Assert.Equal(0, restored.History.RetainedCount);
45    }

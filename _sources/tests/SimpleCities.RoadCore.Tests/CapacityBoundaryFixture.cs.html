1using System.Text;
2using System.Text.Json.Nodes;
3
4namespace SimpleCities.RoadCore.Tests;
5
6internal static class CapacityBoundaryFixture
7{
8    // Boundary fixtures use the real strict codec and load commit. Representative target
9    // datasets are separately built entirely through PlanBuild by RoadCapacityDatasets.
10    internal static void FillIsolatedRoads(RoadNetwork network)
11    {
12        Assert.Equal(25, network.Snapshot.Map.CellSizeMetres);
13        using var encoded = new MemoryStream();
14        RoadCodec.Write(encoded, network.Snapshot);
15        JsonObject payload = JsonNode.Parse(encoded.ToArray())!.AsObject();
16        JsonArray nodes = payload["nodes"]!.AsArray(), edges = payload["edges"]!.AsArray();
17        long nextNode = network.Snapshot.NextNodeId, nextEdge = network.Snapshot.NextEdgeId;
18        for (int i = 0; edges.Count < 16384; i++)
19        {
20            int x = -4000 + i % 160 * 50, y = -4000 + i / 160 * 50;
21            // Both seeded regression topologies and their proposed edits lie here.
22            if (x >= -200 && x <= 200 && y >= -200 && y <= 200) continue;
23            long start = nextNode++, end = nextNode++;
24            nodes.Add(new JsonObject { ["id"] = start, ["x"] = x, ["y"] = y });
25            nodes.Add(new JsonObject { ["id"] = end, ["x"] = x + 25, ["y"] = y });
26            edges.Add(new JsonObject { ["id"] = nextEdge++, ["startNodeId"] = start, ["endNodeId"] = end,
27                ["profile"] = "street", ["points"] = new JsonArray(
28                    new JsonObject { ["x"] = x, ["y"] = y }, new JsonObject { ["x"] = x + 25, ["y"] = y }) });
29        }
30        payload["nextNodeId"] = nextNode;
31        payload["nextEdgeId"] = nextEdge;
32        using var source = new MemoryStream(Encoding.UTF8.GetBytes(payload.ToJsonString()));
33        Assert.True(network.TryCommit(network.PlanLoad(RoadCodec.Read(source))));
34    }
35}

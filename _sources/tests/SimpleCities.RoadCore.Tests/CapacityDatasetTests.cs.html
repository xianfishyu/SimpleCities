1using SimpleCities.RoadCore.Performance;

5public sealed class CapacityDatasetTests

7    [Theory]
8    [InlineData(25, 10000, 3921, 320, 20320)]
9    [InlineData(50, 10000, 3921, 320, 20320)]
10    [InlineData(100, 10000, 3921, 320, 20320)]
11    [InlineData(200, 9680, 3281, 0, 19360)]
12    public void TargetNetworksPreserveExactCanonicalCountsThroughLoadAndEditing(int cell, int edges, int nodes, int polylines, int points)
13    {
14        RoadSnapshot snapshot = RoadCapacityDatasets.CreateTarget(cell).Snapshot;
15        Assert.Equal(edges, snapshot.EdgeCount);
16        Assert.Equal(nodes, snapshot.NodeCount);
17        Assert.Equal(1600, snapshot.Nodes.Count(n => snapshot.Map.IsCellCenter(n.Position)));
18        Assert.Equal(polylines, snapshot.Edges.Count(e => e.Points.Count > 2));
19        Assert.Equal(points, snapshot.Edges.Sum(e => e.Points.Count));
20        VerifyLoadAndEdit(snapshot);
21    }
22
23    [Fact]
24    public void FullTwoHundredMetreLatticeHasAnIndependentDensePolylineCompanion()
25    {
26        RoadSnapshot snapshot = RoadCapacityDatasets.CreateDensePolylines(200).Snapshot;
27        Assert.Equal(320, snapshot.EdgeCount);
28        Assert.Equal(640, snapshot.NodeCount);
29        Assert.Equal(960, snapshot.Edges.Sum(e => e.Points.Count));
30        Assert.All(snapshot.Edges, edge => Assert.Equal(3, edge.Points.Count));
31        Assert.DoesNotContain(snapshot.Nodes, node => snapshot.Map.IsCellCenter(node.Position));
32        VerifyLoadAndEdit(snapshot);
33    }

35    private static void VerifyLoadAndEdit(RoadSnapshot original)
36    {
37        using var encoded = new MemoryStream();
38        RoadCodec.Write(encoded, original);
39        encoded.Position = 0;
40        var network = new RoadNetwork();
41        Guid instance = network.Snapshot.Token.NetworkInstance;
42        Assert.True(network.TryCommit(network.PlanLoad(RoadCodec.Read(encoded))));
43        Assert.Equal(instance, network.Snapshot.Token.NetworkInstance);
44        Assert.NotEqual(original.Token.Lineage, network.Snapshot.Token.Lineage);
45        using var roundtrip = new MemoryStream();
46        RoadCodec.Write(roundtrip, network.Snapshot);
47        Assert.Equal(encoded.ToArray(), roundtrip.ToArray());
48        Assert.Equal(0, network.History.RetainedCount);
49        RoadSnapshot baseline = network.Snapshot;
50        RoadGridSpan Pick() => RoadSpanQuery.Pick(network.Snapshot, new(network.Snapshot.Token, network.Snapshot.Edges[0].Id, .5))!;
51        void Commit(RoadEditResult result)
52        {
53            Assert.Equal(RoadEditStatus.Ready, result.Status);
54            Assert.True(network.TryCommit(result.Plan!));
55        }
56        Commit(network.PlanChangeProfile(Pick(), RoadProfileId.Highway));
57        Commit(network.PlanUndo());
58        AssertContent(baseline, network.Snapshot);
59        Commit(network.PlanRedo());
60        Assert.Contains(network.Snapshot.Edges, edge => edge.Profile == RoadProfileId.Highway);
61        Commit(network.PlanUndo());
62        Commit(network.PlanRemove(Pick()));
63        Commit(network.PlanUndo());
64        AssertContent(baseline, network.Snapshot);
65        Commit(network.PlanRedo());
66        Commit(network.PlanUndo());
67        AssertContent(baseline, network.Snapshot);
68        Assert.True(network.Snapshot.NextNodeId >= baseline.NextNodeId);
69        Assert.True(network.Snapshot.NextEdgeId >= baseline.NextEdgeId);
70        RoadSnapshot before = network.Snapshot;
71        RoadEditHistory history = network.History;
72        RoadEdge occupied = before.Edges[0];
73        RoadBuildResult rejected = network.PlanBuild(new(before.Token, occupied.Points[0], occupied.Points[1], RoadProfileId.Street));
74        Assert.Equal(RoadBuildStatus.Rejected, rejected.Status);
75        Assert.NotEmpty(rejected.Conflicts);
76        Assert.Same(before, network.Snapshot);
77        Assert.Same(history, network.History);
78    }
79
80    private static void AssertContent(RoadSnapshot expected, RoadSnapshot actual)
81    {
82        Assert.Equal(expected.Nodes, actual.Nodes);
83        Assert.Equal(expected.EdgeCount, actual.EdgeCount);
84        for (int i = 0; i < expected.EdgeCount; i++)
85        {
86            RoadEdge a = expected.Edges[i], b = actual.Edges[i];
87            Assert.Equal((a.Id, a.Start, a.End, a.Profile), (b.Id, b.Start, b.End, b.Profile));
88            Assert.Equal(a.Points, b.Points);
89        }
90    }

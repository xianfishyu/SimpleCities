9    [InlineData(25)]
10    [InlineData(50)]
11    [InlineData(100)]
12    [InlineData(200)]
13    public void DiagonalCross_RoundTripsOneCellCenterAndItsBranchSources(int cellSize)
14    {
15        var network = new RoadNetwork(new MapDefinition(cellSize));
16        Build(network, new(0, 0), new(cellSize, cellSize));
17        Build(network, new(0, cellSize), new(cellSize, 0));
18        RoadSnapshot before = network.Snapshot;
19        var center = new RoadPoint(cellSize / 2.0, cellSize / 2.0);
20        RoadNode junction = Assert.Single(before.Nodes, node => node.Position == center);
21        Assert.Equal(5, before.NodeCount);
22        Assert.Equal(4, before.EdgeCount);
23        RoadJunctionReadModel original = RoadJunctionQuery.Read(before, junction.Id)!;
24        Assert.Equal(4, original.Incidences.Count);
25        byte[] saved = Save(network);
26        Assert.Equal(6, JsonNode.Parse(saved)!["schemaVersion"]!.GetValue<int>());
27
28        var restored = new RoadNetwork();
29        for (int read = 0; read < 2; read++)
30        {
31            using var source = new MemoryStream(saved);
32            Assert.True(restored.TryCommit(restored.PlanLoad(RoadCodec.Read(source))));
33            Assert.Equal(saved, Save(restored));
34            Assert.Equal(before.Nodes, restored.Snapshot.Nodes);
35            Assert.Equal(before.NextNodeId, restored.Snapshot.NextNodeId);
36            Assert.Equal(before.NextEdgeId, restored.Snapshot.NextEdgeId);
37            RoadJunctionReadModel current = RoadJunctionQuery.Read(restored.Snapshot, junction.Id)!;
38            Assert.Equal(restored.Snapshot.Token, current.Source);
39            Assert.Equal(original.Incidences, current.Incidences);
40            Assert.Equal(original.Turns, current.Turns);
41            foreach (RoadEdge edge in before.Edges)
42            {
43                double parameter = edge.Start == junction.Id ? 0 : 1;
44                Assert.Equal(center, restored.Snapshot.Resolve(new(restored.Snapshot.Token, edge.Id, parameter)));
45                Assert.Null(restored.Snapshot.Resolve(new(before.Token, edge.Id, parameter)));
46            }
47        }
48    }

51    [InlineData(25)]
52    [InlineData(50)]
53    [InlineData(100)]
54    [InlineData(200)]
55    public void ReloadedCellCenterTJunction_ReusesItsNodeWhenTheFourthDiagonalJoins(int cellSize)
56    {
57        var network = new RoadNetwork(new MapDefinition(cellSize));
58        var center = new RoadPoint(cellSize / 2.0, cellSize / 2.0);
59        Build(network, new(0, 0), new(cellSize, cellSize));
60        Build(network, center, new(0, cellSize), RoadProfileId.Dirt);
61        RoadNode junction = Assert.Single(network.Snapshot.Nodes, node => node.Position == center);
62        Assert.Equal(3, RoadJunctionQuery.Read(network.Snapshot, junction.Id)!.Incidences.Count);
63
64        RoadNetwork restored = Load(Save(network));
65        Build(restored, center, new(cellSize, 0), RoadProfileId.Highway);
66        Assert.Equal(junction, Assert.Single(restored.Snapshot.Nodes, node => node.Position == center));
67        RoadJunctionReadModel connected = RoadJunctionQuery.Read(restored.Snapshot, junction.Id)!;
68        Assert.Equal(4, connected.Incidences.Count);
69        Assert.Single(connected.Incidences, incidence => incidence.Profile == RoadProfileId.Dirt);
70        Assert.Single(connected.Incidences, incidence => incidence.Profile == RoadProfileId.Highway);
71        byte[] saved = Save(restored);
72        RoadNetwork again = Load(saved);
73        Assert.Equal(saved, Save(again));
74        Assert.Equal(connected.Incidences, RoadJunctionQuery.Read(again.Snapshot, junction.Id)!.Incidences);
75    }

130    [Theory]
131    [InlineData(25)]
132    [InlineData(50)]
133    [InlineData(100)]
134    [InlineData(200)]
135    public void OverlappingFromAReloadedCellCenter_LeavesItsSavedContentUntouched(int cellSize)
136    {
137        var network = new RoadNetwork(new MapDefinition(cellSize));
138        Build(network, new(0, 0), new(cellSize, cellSize));
139        Build(network, new(0, cellSize), new(cellSize, 0));
140        RoadNetwork restored = Load(Save(network));
141        RoadSnapshot before = restored.Snapshot;
142        byte[] saved = Save(restored);
143
144        RoadBuildResult overlap = restored.PlanBuild(new(before.Token,
145            new(cellSize / 2.0, cellSize / 2.0), new(-cellSize, -cellSize), RoadProfileId.Highway));
146
147        Assert.Equal(RoadBuildStatus.Rejected, overlap.Status);
148        Assert.Null(overlap.Plan);
149        Assert.NotEmpty(overlap.Conflicts);
150        Assert.Same(before, restored.Snapshot);
151        Assert.Equal(saved, Save(restored));
152        Assert.Equal(saved, Save(Load(saved)));
153    }

155    private static void AssertRejectedWithoutPublication(RoadNetwork network, byte[] payload, string? caseName = null)
156    {
157        RoadSnapshot before = network.Snapshot;
158        byte[] saved = Save(network);
159        using var source = new MemoryStream(payload);
160        Exception? error = Record.Exception(() => network.PlanLoad(RoadCodec.Read(source)));
161        Assert.True(error is InvalidDataException, $"{caseName ?? "old-schema"} must reject invalid content; actual: {error}");
162        Assert.Same(before, network.Snapshot);
163        Assert.Equal(saved, Save(network));
164    }
165
166    private static RoadNetwork Load(byte[] saved)
167    {
168        var network = new RoadNetwork();
169        using var source = new MemoryStream(saved);
170        Assert.True(network.TryCommit(network.PlanLoad(RoadCodec.Read(source))));
171        return network;
172    }
173
174    private static void Build(RoadNetwork network, RoadPoint start, RoadPoint end, RoadProfileId? profile = null)
175    {
176        RoadBuildResult result = network.PlanBuild(new(network.Snapshot.Token, start, end, profile ?? RoadProfileId.Street));
177        Assert.Equal(RoadBuildStatus.Ready, result.Status);
178        Assert.True(network.TryCommit(result.Plan!));
179    }
180
181    private static byte[] Save(RoadNetwork network)
182    {
183        using var destination = new MemoryStream();
184        RoadCodec.Write(destination, network.Snapshot);
185        return destination.ToArray();
186    }

174    [Theory]
175    [InlineData(25)]
176    [InlineData(50)]
177    [InlineData(100)]
178    [InlineData(200)]
179    public void CellCenterBoundaryDrag_PreservesItsDirectionAndStopsAtTheLastValidPoint(int cellSize)
180    {
181        var map = new MapDefinition(cellSize);
182        double half = cellSize / 2d;
183        foreach (int sign in new[] { -1, 1 })
184        {
185            var network = new RoadNetwork(map);
186            var center = new RoadPoint(sign * (4000 - half), sign * (4000 - half));
187            RoadPoint boundary = map.SnapDragEnd(center, new(sign * 5000, sign * 5000));
188            Assert.Equal(new RoadPoint(sign * 4000, sign * 4000), boundary);
189            Build(network, center, boundary);
190            Assert.Equal(boundary, map.SnapDragEnd(boundary, new(sign * 5000, sign * 5000)));
191        }
192        // 横竖草稿仍按整格保留格心奇偶，不因矩形裁剪产生混合奇偶的终点。
193        var originCenter = new RoadPoint(half, half);
194        Assert.Equal(new RoadPoint(4000 - half, half), map.SnapDragEnd(originCenter, new(5000, half)));
195        Assert.Equal(new RoadPoint(half, -4000 + half), map.SnapDragEnd(originCenter, new(half, -5000)));
196    }

198    [Fact]
199    public void CancelledOrStaleCellCenterPlan_DoesNotPublishSplitsOrConsumeIds()
200    {
201        var network = new RoadNetwork(new MapDefinition(25));
202        Build(network, new(0, 0), new(25, 25));
203        RoadSnapshot before = network.Snapshot;
204        var request = new RoadBuildRequest(before.Token, new(0, 25), new(25, 0), RoadProfileId.Street);
205        using var cancellation = new CancellationTokenSource();
206        cancellation.Cancel();
207        Assert.Throws<OperationCanceledException>(() => network.PlanBuild(request, cancellation.Token));
208        Assert.Same(before, network.Snapshot);
209        RoadBuildResult planned = network.PlanBuild(request);
210        Assert.Equal(RoadBuildStatus.Ready, planned.Status);
211        Assert.Same(before, network.Snapshot);
212        Build(network, new(-100, -100), new(-75, -75));
213        RoadSnapshot current = network.Snapshot;
214        Assert.Equal(before.NextNodeId + 2, current.NextNodeId);
215        Assert.Equal(before.NextEdgeId + 1, current.NextEdgeId);
216        Assert.False(network.TryCommit(planned.Plan!));
217        Assert.Equal(RoadBuildStatus.Rejected, network.PlanBuild(request).Status);
218        Assert.Same(current, network.Snapshot);
219        Assert.DoesNotContain(current.Nodes, node => node.Position == new RoadPoint(12.5, 12.5));
220    }

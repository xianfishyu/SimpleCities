7    [Fact]
8    public void LocalProfileEditKeepsDraftAndSnapshotLookupAllocationBounded()
9    {
10        RoadNetwork network = RoadBaselineFixtures.CreateMixed(25);
11        RoadSnapshot source = network.Snapshot;
12        RoadGridSpan span = RoadSpanQuery.Pick(source, new(source.Token, source.Edges[0].Id, .5))!;
13        _ = network.PlanChangeProfile(span, RoadProfileId.Highway);
14        long before = GC.GetAllocatedBytesForCurrentThread();
15        RoadPlan plan = network.PlanChangeProfile(span, RoadProfileId.Highway).Plan!;
16        long allocated = GC.GetAllocatedBytesForCurrentThread() - before;
17        output.WriteLine($"Local profile planning: {allocated} B");
18        Assert.True(allocated < 84 * 1024, $"Local profile planning allocated {allocated} B.");
19        Assert.Same(source, network.Snapshot);
20        Assert.Empty(plan.ChangeSet.Nodes);
21        Assert.Single(plan.ChangeSet.Edges);
22    }

24    [Theory]
25    [InlineData(false)]
26    [InlineData(true)]
27    public void BatchProfileContractionKeepsStableIdentityAndOldHistory(bool closed)
28    {
29        var network = new RoadNetwork();
30        RoadPoint[] points = closed
31            ? [new(0, 0), new(100, 0), new(100, 100), new(0, 100), new(0, 0)]
32            : Enumerable.Range(0, 13).Select(i => new RoadPoint(i * 100, (i % 2) * 100)).ToArray();
33        for (int i = 1; i < points.Length; i++)
34            Assert.True(network.TryCommit(network.PlanBuild(new(network.Snapshot.Token, points[i - 1], points[i],
35                i % 2 == 0 ? RoadProfileId.Dirt : RoadProfileId.Street)).Plan!));
36        RoadSnapshot original = network.Snapshot;
37        byte[] saved = Save(original);
38
39        Assert.True(network.TryCommit(network.PlanChangeProfile(RoadBaselineFixtures.Spans(original), RoadProfileId.Highway).Plan!));
40
41        RoadSnapshot changed = network.Snapshot;
42        RoadEdge edge = Assert.Single(changed.Edges);
43        Assert.Equal(new EdgeId(1), edge.Id);
44        Assert.Equal(new NodeId(1), edge.Start);
45        Assert.Equal(original.NextNodeId, changed.NextNodeId);
46        Assert.Equal(original.NextEdgeId, changed.NextEdgeId);
47        Assert.Equal(closed ? 1 : 2, changed.NodeCount);
48        Assert.Equal(closed ? new NodeId(1) : new NodeId(13), edge.End);
49        Assert.Equal(closed ? [points[0], points[3], points[2], points[1], points[0]] : points, edge.Points);
50        byte[] changedBytes = Save(changed);
51        Assert.True(network.TryCommit(network.PlanUndo().Plan!));
52        Assert.Equal(saved, Save(network.Snapshot));
53        Assert.True(network.TryCommit(network.PlanRedo().Plan!));
54        Assert.Equal(changedBytes, Save(network.Snapshot));
55        Assert.True(network.TryCommit(network.PlanRemove(RoadBaselineFixtures.Spans(network.Snapshot)).Plan!));
56        Assert.Empty(network.Snapshot.Nodes);
57        Assert.Empty(network.Snapshot.Edges);
58        Assert.True(network.TryCommit(network.PlanUndo().Plan!));
59        Assert.Equal(changedBytes, Save(network.Snapshot));
60    }
61
62    private static byte[] Save(RoadSnapshot snapshot)
63    {
64        using var bytes = new MemoryStream();
65        RoadCodec.Write(bytes, snapshot);
66        return bytes.ToArray();
67    }

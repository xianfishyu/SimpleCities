1using System.Text;
2using System.Text.Json;
3
4namespace SimpleCities.RoadCore.Tests;
5
6public sealed class EmptyMapRejectionTests
7{
8    // Authored independently of the writer: the schema example is the input oracle.
9    private const string Valid = """
10        {"formatFamily":"simple-cities-v4","payloadType":"road-network","schemaVersion":6,
11         "contentRevision":7,"nextNodeId":9,"nextEdgeId":12,"profileCatalogVersion":1,
12         "map":{"widthMetres":8000,"heightMetres":8000,"origin":"center","metresPerUnit":1,
13                "grid":"square-eight","cellSizeMetres":50},"nodes":[],"edges":[]}
14        """;
15
16    [Theory]
17    [InlineData("\"schemaVersion\":6", "\"schemaVersion\":1")]
18    [InlineData("simple-cities-v4", "simple-cities-v3")]
19    [InlineData("\"cellSizeMetres\":50", "\"cellSizeMetres\":75")]
20    [InlineData("\"cellSizeMetres\":50", "\"cellSizeMetres\":50.5")]
21    [InlineData("\"cellSizeMetres\":50", "\"cellSizeMetres\":50,\"cellSizeMetres\":100")]
22    [InlineData("\"schemaVersion\":6", "\"schemaVersion\":6,\"unknown\":0")]
23    [InlineData("\"nextNodeId\":9", "\"nextNodeId\":0")]
24    [InlineData("\"nextEdgeId\":12", "\"nextEdgeId\":-1")]
25    [InlineData("\"profileCatalogVersion\":1", "\"profileCatalogVersion\":2")]
26    [InlineData("\"widthMetres\":8000", "\"widthMetres\":8001")]
27    [InlineData("\"origin\":\"center\"", "\"origin\":\"corner\"")]
28    [InlineData("\"metresPerUnit\":1", "\"metresPerUnit\":100")]
29    [InlineData("\"nodes\":[]", "\"nodes\":[{}]")]
30    [InlineData("\"edges\":[]", "\"edges\":[{}]")]
31    public void InvalidPayload_DoesNotChangeActiveMap(string from, string to)
32    {
33        var network = new RoadNetwork();
34        RoadSnapshot before = network.Snapshot;
35        using var bytes = new MemoryStream(Encoding.UTF8.GetBytes(Valid.Replace(from, to)));
36        Assert.Throws<InvalidDataException>(() => network.PlanLoad(RoadCodec.Read(bytes)));
37        Assert.Equal(before.Token, network.Snapshot.Token);
38        Assert.Equal(100, network.Snapshot.Map.CellSizeMetres);
39        Assert.Equal(1, network.Snapshot.NextNodeId);
40    }
41
42    [Fact]
43    public void LoadedEmptyMap_RetainsPersistedWatermarksAndRejectsForeignPlan()
44    {
45        var first = new RoadNetwork();
46        var second = new RoadNetwork();
47        using var bytes = new MemoryStream(Encoding.UTF8.GetBytes(Valid));
48        RoadPlan plan = first.PlanLoad(RoadCodec.Read(bytes));
49        Assert.False(second.TryCommit(plan));
50        Assert.True(first.TryCommit(plan));
51        Assert.Equal(7, first.Snapshot.Token.ContentRevision);
52        Assert.Equal(9, first.Snapshot.NextNodeId);
53        Assert.Equal(12, first.Snapshot.NextEdgeId);
54        Assert.Equal(50, first.Snapshot.Map.CellSizeMetres);
55        Assert.Equal(100, second.Snapshot.Map.CellSizeMetres);
56    }
57
58    [Fact]
59    public void OversizedOrTruncatedPayload_IsRejected()
60    {
61        using var oversized = new MemoryStream(Encoding.UTF8.GetBytes(Valid + new string(' ', RoadCodec.MaximumPayloadBytes)));
62        Assert.Throws<InvalidDataException>(() => RoadCodec.Read(oversized));
63        using var truncated = new MemoryStream(Encoding.UTF8.GetBytes(Valid[..^1]));
64        Assert.ThrowsAny<JsonException>(() => RoadCodec.Read(truncated));
65        Assert.Throws<ArgumentOutOfRangeException>(() => new MapDefinition(75));
66    }
67}

75    [Fact]
76    public void LocationsMeasureTheWholePolylineArcLengthAndExpireAfterContinuation()
77    {
78        var network = new RoadNetwork();
79        Build(network, new(0, 0), new(300, 0));
80        RoadSnapshot straight = network.Snapshot;
81        var previous = new RoadLocation(straight.Token, Assert.Single(straight.Edges).Id, 0.5);
82        Assert.Equal(new RoadPoint(150, 0), straight.Resolve(previous));
83        Build(network, new(300, 0), new(300, 400));
84        RoadSnapshot turn = network.Snapshot;
85        EdgeId id = Assert.Single(turn.Edges).Id;
86
87        Assert.Null(turn.Resolve(previous));
88        Assert.Equal(new RoadPoint(0, 0), turn.Resolve(new(turn.Token, id, 0)));
89        Assert.Equal(new RoadPoint(175, 0), turn.Resolve(new(turn.Token, id, 0.25)));
90        Assert.Equal(new RoadPoint(300, 50), turn.Resolve(new(turn.Token, id, 0.5)));
91        Assert.Equal(new RoadPoint(300, 225), turn.Resolve(new(turn.Token, id, 0.75)));
92        Assert.Equal(new RoadPoint(300, 400), turn.Resolve(new(turn.Token, id, 1)));
93        var selected = new RoadLocation(turn.Token, id, 0.5);
94        Build(network, new(300, 400), new(500, 600));
95        Assert.Null(network.Snapshot.Resolve(selected));
96        Assert.Equal(new RoadPoint(300, 50), turn.Resolve(selected));
97    }

99    [Theory]
100    [InlineData(200, 0, true)] // Touch the interior of the first span.
101    [InlineData(200, -100, true)] // Cross the first span.
102    [InlineData(0, 0, true)] // Close onto the other open endpoint.
103    [InlineData(400, 400, true)] // Touch an existing internal turn.
104    [InlineData(200, 500, false)] // Overlap the last span in reverse.
105    [InlineData(4100, 200, false)] // Leave the map on a legal direction.
106    public void Continuation_PublishesLegalLoopsAndRejectsInvalidCoverage(double x, double y, bool allowed)
107    {
108        var network = new RoadNetwork();
109        Build(network, new(0, 0), new(400, 0));
110        Build(network, new(400, 0), new(400, 400));
111        Build(network, new(400, 400), new(200, 400));
112        Build(network, new(200, 400), new(200, 200));
113        RoadSnapshot before = network.Snapshot;
114        byte[] content = Save(network);
115
116        RoadBuildResult result = network.PlanBuild(new(before.Token, new(200, 200), new(x, y), RoadProfileId.Street));
117
118        Assert.Equal(allowed ? RoadBuildStatus.Ready : RoadBuildStatus.Rejected, result.Status);
119        Assert.Equal(before.Token, network.Snapshot.Token);
120        Assert.Equal(before.NextNodeId, network.Snapshot.NextNodeId);
121        Assert.Equal(before.NextEdgeId, network.Snapshot.NextEdgeId);
122        Assert.Equal(content, Save(network));
123        if (allowed)
124        {
125            Assert.True(network.TryCommit(result.Plan!));
126            Assert.Equal(before.Token.ChangeSequence + 1, network.Snapshot.Token.ChangeSequence);
127        }
128        else Assert.Null(result.Plan);
129    }

149    private static void Build(RoadNetwork network, RoadPoint start, RoadPoint end, RoadProfileId? profile = null)
150    {
151        RoadBuildResult result = network.PlanBuild(new(network.Snapshot.Token, start, end, profile ?? RoadProfileId.Street));
152        Assert.Equal(RoadBuildStatus.Ready, result.Status);
153        Assert.True(network.TryCommit(result.Plan!));
154    }

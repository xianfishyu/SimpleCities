8    [Fact]
9    public void UndoOfStillObservedContentAvoidsWholeSnapshotMaterialization()
10    {
11        RoadNetwork network = RoadBaselineFixtures.CreateMixed(25);
12        using var bytes = new MemoryStream();
13        RoadCodec.Write(bytes, network.Snapshot);
14        bytes.Position = 0;
15        Assert.True(network.TryCommit(network.PlanLoad(RoadCodec.Read(bytes))));
16        RoadSnapshot original = network.Snapshot;
17        RoadEdge edge = original.Edges[0];
18        RoadGridSpan span = RoadSpanQuery.Pick(original, new(original.Token, edge.Id, .5))!;
19        Assert.True(network.TryCommit(network.PlanChangeProfile(span, RoadProfileId.Highway).Plan!));
20        RoadSnapshot changed = network.Snapshot;
21        long start = GC.GetAllocatedBytesForCurrentThread();
22        RoadPlan undo = network.PlanUndo().Plan!;
23        long allocated = GC.GetAllocatedBytesForCurrentThread() - start;
24        output.WriteLine($"Observed-content undo allocation: {allocated}");
25        Assert.True(allocated < 16 * 1024, $"Undo rematerialized observed content: {allocated} bytes.");
26        Assert.Equal(changed.Token, undo.ChangeSet.Before.Token);
27        Assert.Equal(undo.Target.Token, undo.ChangeSet.After.Token);
28        Assert.Equal(original.Token.ContentRevision, undo.Target.Token.ContentRevision);
29        Assert.True(undo.Target.Token.ChangeSequence > changed.Token.ChangeSequence);
30        Assert.Equal(original.Resolve(new(original.Token, edge.Id, .5)), undo.Target.Resolve(new(undo.Target.Token, edge.Id, .5)));
31        Assert.Null(undo.Target.Resolve(new(original.Token, edge.Id, .5)));
32        Assert.True(network.TryCommit(undo));
33        Assert.Equal(0, network.History.UndoCount);
34        Assert.Equal(1, network.History.RedoCount);
35        RoadPlan redo = network.PlanRedo().Plan!;
36        Assert.Equal(network.Snapshot.Token, redo.ChangeSet.Before.Token);
37        Assert.Equal(redo.Target.Token, redo.ChangeSet.After.Token);
38        Assert.True(network.TryCommit(redo));
39        Assert.Equal(changed.Edges.Select(e => e.Profile), network.Snapshot.Edges.Select(e => e.Profile));
40        GC.KeepAlive(original);
41        GC.KeepAlive(changed);
42    }

44    [Fact]
45    public void UndoOfLocalEditDoesNotRescanRetainedLargeHistory()
46    {
47        RoadNetwork network = RoadBaselineFixtures.CreateMixed(25);
48        using var bytes = new MemoryStream();
49        RoadCodec.Write(bytes, network.Snapshot);
50        bytes.Position = 0;
51        Assert.True(network.TryCommit(network.PlanLoad(RoadCodec.Read(bytes))));
52        for (int i = 0; i < 63; i++)
53            Assert.True(network.TryCommit(network.PlanChangeProfile(RoadBaselineFixtures.Spans(network.Snapshot),
54                i % 2 == 0 ? RoadProfileId.Highway : RoadProfileId.Dirt).Plan!));
55        RoadSnapshot observed = network.Snapshot;
56        RoadGridSpan span = RoadSpanQuery.Pick(observed, new(observed.Token, observed.Edges[0].Id, .5))!;
57        Assert.True(network.TryCommit(network.PlanChangeProfile(span, RoadProfileId.Street).Plan!));
58        long estimate = network.History.EstimatedBytes;
59        _ = network.PlanUndo();
60
61        long before = GC.GetAllocatedBytesForCurrentThread();
62        RoadPlan undo = network.PlanUndo().Plan!;
63        long allocated = GC.GetAllocatedBytesForCurrentThread() - before;
64
65        output.WriteLine($"Local undo with 63 retained full-map changes: {allocated}");
66        Assert.True(allocated <= 1024, $"Local undo rescanned retained history: {allocated} bytes.");
67        Assert.True(network.TryCommit(undo));
68        Assert.Equal(63, network.History.UndoCount);
69        Assert.Equal(1, network.History.RedoCount);
70        Assert.Equal(estimate, network.History.EstimatedBytes);
71        Assert.True(network.TryCommit(network.PlanRedo().Plan!));
72        Assert.Equal(64, network.History.UndoCount);
73        Assert.Equal(estimate, network.History.EstimatedBytes);
74        GC.KeepAlive(observed);
75    }

77    [Fact]
78    public void HistoryDoesNotRootPreviousSnapshotAndCanReplayAfterItIsCollected()
79    {
80        RoadNetwork network = CreateSingleRoad();
81        WeakReference previous = EditAndReleaseSnapshot(network);
82        GC.Collect(); GC.WaitForPendingFinalizers(); GC.Collect();
83        Assert.False(previous.IsAlive);
84        Assert.True(network.TryCommit(network.PlanUndo().Plan!));
85        Assert.All(network.Snapshot.Edges, edge => Assert.Equal(RoadProfileId.Street, edge.Profile));
86    }
87
88    [MethodImpl(MethodImplOptions.NoInlining)]
89    private static RoadNetwork CreateSingleRoad()
90    {
91        var network = new RoadNetwork();
92        Assert.True(network.TryCommit(network.PlanBuild(new(network.Snapshot.Token, new(0, 0), new(200, 0), RoadProfileId.Street)).Plan!));
93        return network;
94    }
95
96    [MethodImpl(MethodImplOptions.NoInlining)]
97    private static WeakReference EditAndReleaseSnapshot(RoadNetwork network)
98    {
99        RoadSnapshot previous = network.Snapshot;
100        RoadGridSpan span = RoadSpanQuery.Pick(previous, new(previous.Token, previous.Edges[0].Id, .25))!;
101        Assert.True(network.TryCommit(network.PlanChangeProfile(span, RoadProfileId.Highway).Plan!));
102        return new(previous);
103    }

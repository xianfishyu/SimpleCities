6    [InlineData("nextNodeId", long.MaxValue - 2)]
7    [InlineData("nextEdgeId", long.MaxValue - 1)]
8    [InlineData("contentRevision", long.MaxValue - 1)]
9    public void BuildAtIdentityLimit_IsRejectedAndCurrentContentStillRoundTrips(string field, long value)
10    {
11        const string empty = """
12            {"formatFamily":"simple-cities-v4","payloadType":"road-network","schemaVersion":6,
13            "contentRevision":1,"nextNodeId":1,"nextEdgeId":1,"profileCatalogVersion":1,
14            "map":{"widthMetres":8000,"heightMetres":8000,"origin":"center","metresPerUnit":1,"grid":"square-eight","cellSizeMetres":100},
15            "nodes":[],"edges":[]}
16            """;
17        using var source = new MemoryStream(System.Text.Encoding.UTF8.GetBytes(empty.Replace($"\"{field}\":1", $"\"{field}\":{value}")));
18        var network = new RoadNetwork();
19        Assert.True(network.TryCommit(network.PlanLoad(RoadCodec.Read(source))));
20        RoadSnapshot before = network.Snapshot;
21        RoadBuildResult result = network.PlanBuild(new(before.Token, new(0, 0), new(100, 0), RoadProfileId.Street));
22        Assert.Equal(RoadBuildStatus.Rejected, result.Status);
23        Assert.Null(result.Plan);
24        Assert.Equal(before.Token, network.Snapshot.Token);
25        Assert.Equal(before.NextNodeId, network.Snapshot.NextNodeId);
26        Assert.Equal(before.NextEdgeId, network.Snapshot.NextEdgeId);
27        using var saved = new MemoryStream();
28        RoadCodec.Write(saved, network.Snapshot);
29        saved.Position = 0;
30        var restored = new RoadNetwork();
31        Assert.True(restored.TryCommit(restored.PlanLoad(RoadCodec.Read(saved))));
32        Assert.Equal(before.Token.ContentRevision, restored.Snapshot.Token.ContentRevision);
33        Assert.Equal(before.NextNodeId, restored.Snapshot.NextNodeId);
34        Assert.Equal(before.NextEdgeId, restored.Snapshot.NextEdgeId);
35    }

37    [Fact]
38    public void RejectedAndNoChangeRequests_DoNotConsumeIdsOrPublish()
39    {
40        var network = new RoadNetwork();
41        RoadSnapshot before = network.Snapshot;
42        Assert.Equal(RoadBuildStatus.NoChange, network.PlanBuild(new(before.Token, new(0, 0), new(0, 0), RoadProfileId.Dirt)).Status);
43        foreach (RoadPoint end in new[] { new RoadPoint(100, 200), new RoadPoint(25, 25), new RoadPoint(4100, 0), new RoadPoint(double.NaN, 0) })
44            Assert.Equal(RoadBuildStatus.Rejected, network.PlanBuild(new(before.Token, new(0, 0), end, RoadProfileId.Dirt)).Status);
45        Assert.Equal(RoadBuildStatus.Rejected, network.PlanBuild(new(default, new(0, 0), new(100, 0), RoadProfileId.Dirt)).Status);
46        Assert.Equal(RoadBuildStatus.Rejected, network.PlanBuild(new(before.Token, new(0, 0), new(100, 0), default)).Status);
47        Assert.Equal(before.Token, network.Snapshot.Token);
48        Assert.Equal(1, network.Snapshot.NextNodeId);
49        Assert.Equal(1, network.Snapshot.NextEdgeId);
50        RoadPlan first = network.PlanBuild(new(before.Token, new(0, 0), new(100, 0), RoadProfileId.Dirt)).Plan!;
51        RoadPlan concurrent = network.PlanBuild(new(before.Token, new(0, 0), new(200, 0), RoadProfileId.Dirt)).Plan!;
52        Assert.True(network.TryCommit(first));
53        Assert.False(network.TryCommit(concurrent));
54        Assert.Equal(RoadBuildStatus.Ready, network.PlanBuild(new(network.Snapshot.Token, new(0, 200), new(100, 200), RoadProfileId.Dirt)).Status);
55        Assert.Equal(1, network.Snapshot.EdgeCount);
56        Assert.Equal(1, network.Snapshot.Token.ChangeSequence);
57    }

75    public void RoadRoundTrip_RetainsEntitiesAndResolvesOnlyCurrentLocations(string profile, double width)
76    {
77        var network = new RoadNetwork(new MapDefinition(25));
78        var result = network.PlanBuild(new(network.Snapshot.Token, new(-100, 25), new(200, 325), new(profile)));
79        Assert.True(network.TryCommit(result.Plan!));
80        RoadSurfaceData surface = RoadPresentation.Prepare(network.Snapshot.Nodes, network.Snapshot.Edges)!;
81        Assert.InRange(surface.Pieces[0].Corners[0].DistanceTo(surface.Pieces[0].Corners[3]), width - 1e-9, width + 1e-9);
82        var location = new RoadLocation(network.Snapshot.Token, new EdgeId(1), 0.5);
83        Assert.Equal(new RoadPoint(50, 175), network.Snapshot.Resolve(location));
84        using var bytes = new MemoryStream();
85        RoadCodec.Write(bytes, network.Snapshot);
86        bytes.Position = 0;
87        var target = new RoadNetwork();
88        Assert.True(target.TryCommit(target.PlanLoad(RoadCodec.Read(bytes))));
89        Assert.Equal(2, target.Snapshot.NodeCount);
90        RoadEdge edge = Assert.Single(target.Snapshot.Edges);
91        Assert.Equal(profile, edge.Profile.Value);
92        Assert.Equal(new RoadPoint(-100, 25), target.Snapshot.Nodes[0].Position);
93        Assert.Equal(new RoadPoint(200, 325), target.Snapshot.Nodes[1].Position);
94        Assert.Equal(3, target.Snapshot.NextNodeId);
95        Assert.Equal(2, target.Snapshot.NextEdgeId);
96        Assert.Null(target.Snapshot.Resolve(location));
97        using var savedAgain = new MemoryStream();
98        RoadCodec.Write(savedAgain, target.Snapshot);
99        Assert.Equal(bytes.ToArray(), savedAgain.ToArray());
100    }

103    [InlineData("\"endNodeId\":2", "\"endNodeId\":9")]
104    [InlineData("\"profile\":\"street\"", "\"profile\":\"unknown\"")]

108    [InlineData("\"nextNodeId\":3", "\"nextNodeId\":2")]
109    [InlineData("\"startNodeId\":1", "\"startNodeId\":2")]
110    public void MalformedRoadPayload_IsRejectedBeforePublishing(string original, string replacement)
111    {

114            "contentRevision":2,"nextNodeId":3,"nextEdgeId":2,"profileCatalogVersion":1,
115            "map":{"widthMetres":8000,"heightMetres":8000,"origin":"center","metresPerUnit":1,"grid":"square-eight","cellSizeMetres":100},
116            "nodes":[{"id":1,"x":0,"y":0},{"id":2,"x":300,"y":0}],
117            "edges":[{"id":1,"startNodeId":1,"endNodeId":2,"profile":"street","points":[{"x":0,"y":0},{"x":300,"y":0}]}]}
118            """;
119        var network = new RoadNetwork();
120        RoadStateToken before = network.Snapshot.Token;
121        using var source = new MemoryStream(System.Text.Encoding.UTF8.GetBytes(valid.Replace(original, replacement)));
122        Assert.Throws<InvalidDataException>(() => network.PlanLoad(RoadCodec.Read(source)));
123        Assert.Equal(before, network.Snapshot.Token);
124        Assert.Equal(0, network.Snapshot.EdgeCount);
125    }

127    [Theory]
128    [InlineData(300, 0)]
129    [InlineData(300, 300)]
130    [InlineData(0, 300)]
131    [InlineData(-300, 300)]
132    [InlineData(-300, 0)]
133    [InlineData(-300, -300)]
134    [InlineData(0, -300)]
135    [InlineData(300, -300)]
136    public void BuildAcrossSeveralCells_PublishesOneRoadOnce(double x, double y)
137    {
138        var network = new RoadNetwork();
139        RoadSnapshot before = network.Snapshot;
140        var request = new RoadBuildRequest(before.Token, new RoadPoint(0, 0), new RoadPoint(x, y), RoadProfileId.Street);
141        RoadBuildResult result = network.PlanBuild(request);
142
143        Assert.Equal(RoadBuildStatus.Ready, result.Status);
144        Assert.Equal(before.Token, network.Snapshot.Token);
145        Assert.Equal(1, network.Snapshot.NextNodeId);
146        Assert.True(network.TryCommit(result.Plan!));
147        RoadSnapshot after = network.Snapshot;
148        Assert.Equal(2, after.NodeCount);
149        RoadEdge edge = Assert.Single(after.Edges);
150        Assert.Equal(new RoadPoint(0, 0), after.Nodes[0].Position);
151        Assert.Equal(new RoadPoint(x, y), after.Nodes[1].Position);
152        Assert.Equal(RoadProfileId.Street, edge.Profile);
153        Assert.Equal(new NodeId(1), edge.Start);
154        Assert.Equal(new NodeId(2), edge.End);
155        Assert.Equal(new EdgeId(1), edge.Id);
156        Assert.Equal(3, after.NextNodeId);
157        Assert.Equal(2, after.NextEdgeId);
158        Assert.Equal(2, after.Token.ContentRevision);
159        Assert.Equal(1, after.Token.ChangeSequence);
160        Assert.False(network.TryCommit(result.Plan!));
161        Assert.Equal(before.Token, result.Plan!.SourceToken);
162        Assert.Equal(2, result.Plan.AddedNodeCount);
163        Assert.Equal(1, result.Plan.AddedEdgeCount);
164    }

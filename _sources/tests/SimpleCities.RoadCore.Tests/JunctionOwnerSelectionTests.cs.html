5    [Fact]
6    public void EightWayPreparationHasBoundedAllocationWithoutPerPatchSorting()
7    {
8        var (nodes, edges) = EightWay();
9        for (int i = 0; i < 3; i++) RoadPresentation.Prepare(nodes, edges);
10        long before = GC.GetAllocatedBytesForCurrentThread();
11        RoadSurfaceData? surface = RoadPresentation.Prepare(nodes, edges);
12        long allocated = GC.GetAllocatedBytesForCurrentThread() - before;
13        output.WriteLine($"Eight-way surface allocated bytes: {allocated}");
14        Assert.NotNull(surface);
15        Assert.Contains(surface.Pieces, piece => piece.JunctionNode == nodes[0].Id);
16        // The small eight-way hull must not allocate per-corner objects or a
17        // multi-key sorting/deduplication pipeline on every preparation.
18        Assert.InRange(allocated, 0, 16 * 1024);
19    }

21    [Theory]
22    [InlineData(0, 0, false)]
23    [InlineData(50, 0, false)]
24    [InlineData(-3975, -3975, true)]
25    [InlineData(3950, 3950, true)]
26    public void EightWayOwnersAndGeometryRemainStableWhenTranslatedAndReordered(int x, int y, bool reverse)
27    {
28        var (nodes, edges) = EightWay();
29        RoadSurfacePiece[] expected = RoadPresentation.Prepare(nodes, edges)!.Pieces.Where(p => p.JunctionNode == nodes[0].Id).ToArray();
30        AssertOrderedOwners(nodes[0], edges, expected);
31        var movedNodes = nodes.Select(n => n with { Position = Translate(n.Position) }).ToArray();
32        var movedEdges = edges.Select(e => new RoadEdge(e.Id, e.Start, e.End, e.Profile, e.Points.Select(Translate))).ToArray();
33        var actual = RoadPresentation.Prepare(reverse ? movedNodes.Reverse().ToArray() : movedNodes,
34            reverse ? movedEdges.Reverse().ToArray() : movedEdges)!.Pieces.Where(p => p.JunctionNode == nodes[0].Id).ToArray();
35        Assert.Equal(expected.Length, actual.Length);
36        for (int i = 0; i < expected.Length; i++)
37        {
38            Assert.Equal(expected[i].Edge.Id, actual[i].Edge.Id);
39            Assert.Equal(expected[i].StartParameter, actual[i].StartParameter);
40            Assert.Equal(expected[i].EndParameter, actual[i].EndParameter);
41            Assert.Equal(expected[i].Corners.Select(Translate), actual[i].Corners);
42        }
43        RoadPoint Translate(RoadPoint point) => new(point.X + x, point.Y + y);
44    }

46    [Fact]
47    public void SelfLoopKeepsBothEndRoleOwners()
48    {
49        RoadNode center = new(new(1), default);
50        RoadNode branch = new(new(2), new(-25, 0));
51        RoadEdge loop = new(new(9), center.Id, center.Id, RoadProfileId.Highway,
52            [default, new(25, 0), new(25, 25), new(0, 25), default]);
53        RoadEdge arm = new(new(2), center.Id, branch.Id, RoadProfileId.Dirt, [center.Position, branch.Position]);
54        var patches = RoadPresentation.Prepare([center, branch], [loop, arm])!.Pieces.Where(p => p.JunctionNode == center.Id).ToArray();
55        AssertOrderedOwners(center, [loop, arm], patches);
56        Assert.Contains(patches, p => p.Edge == loop && p.StartParameter == 0);
57        Assert.Contains(patches, p => p.Edge == loop && p.StartParameter == 1);
58    }

60    [Fact]
61    public void EqualRayScoresKeepEdgeIdAndEndRoleTieBreaks()
62    {
63        // Synthetic presentation inputs deliberately share a ray to exercise exact
64        // owner ties; they do not claim these overlaps pass normal network admission.
65        RoadNode center = new(new(1), default);
66        RoadNode east = new(new(2), new(25, 0));
67        RoadNode north = new(new(3), new(0, 25));
68        RoadEdge high = new(new(20), center.Id, east.Id, RoadProfileId.Street, [center.Position, east.Position]);
69        RoadEdge low = new(new(3), center.Id, center.Id, RoadProfileId.Street,
70            [center.Position, east.Position, new(25, -25), east.Position, center.Position]);
71        RoadEdge arm = new(new(30), center.Id, north.Id, RoadProfileId.Street, [center.Position, north.Position]);
72        var patches = RoadPresentation.Prepare([center, east, north], [high, low, arm])!.Pieces.Where(p => p.JunctionNode == center.Id).ToArray();
73        AssertOrderedOwners(center, [high, low, arm], patches);
74        Assert.Contains(patches, p => p.Edge == low);
75        Assert.DoesNotContain(patches, p => p.Edge == high || (p.Edge == low && p.StartParameter == 1));
76    }

78    [Theory]
79    [InlineData(2)]
80    [InlineData(3)]
81    public void ManyRepeatedMouthsKeepStableGeometryAndOwnersWhenReordered(int copies)
82    {
83        // Synthetic overlapping incidences exercise deduplication with more
84        // corners than a legal eight-direction node, without claiming admission.
85        var (nodes, edges) = EightWay();
86        var repeated = Enumerable.Range(0, copies).SelectMany(copy => edges.Select(edge =>
87            new RoadEdge(new(edge.Id.Value + copy * 100), edge.Start, edge.End, edge.Profile, edge.Points)))
88            .ToArray();
89        var expected = RoadPresentation.Prepare(nodes, repeated)!.Pieces.Where(p => p.JunctionNode == nodes[0].Id).ToArray();
90        var actual = RoadPresentation.Prepare(nodes.Reverse().ToArray(), repeated.Reverse().ToArray())!
91            .Pieces.Where(p => p.JunctionNode == nodes[0].Id).ToArray();
92        AssertOrderedOwners(nodes[0], repeated, actual);
93        Assert.Equal(expected.Length, actual.Length);
94        for (int i = 0; i < expected.Length; i++)
95        {
96            Assert.Equal(expected[i].Edge.Id, actual[i].Edge.Id);
97            Assert.Equal(expected[i].StartParameter, actual[i].StartParameter);
98            Assert.Equal(expected[i].Corners, actual[i].Corners);
99        }
100    }

102    private static void AssertOrderedOwners(RoadNode center, IReadOnlyList<RoadEdge> edges, IReadOnlyList<RoadSurfacePiece> patches)
103    {
104        var rays = new List<(RoadEdge Edge, RoadEndRole Role, RoadPoint Outward)>();
105        foreach (RoadEdge edge in edges)
106        {
107            if (edge.Start == center.Id) Add(edge, RoadEndRole.Start, edge.Points[1]);
108            if (edge.End == center.Id) Add(edge, RoadEndRole.End, edge.Points[^2]);
109        }
110        Assert.NotEmpty(patches);
111        foreach (RoadSurfacePiece patch in patches)
112        {
113            RoadPoint middle = new((patch.Corners[1].X + patch.Corners[2].X) / 2,
114                (patch.Corners[1].Y + patch.Corners[2].Y) / 2);
115            var expected = rays.OrderByDescending(r => middle.X * r.Outward.X + middle.Y * r.Outward.Y)
116                .ThenBy(r => r.Edge.Id.Value).ThenBy(r => r.Role).First();
117            Assert.Same(expected.Edge, patch.Edge);
118            Assert.Equal(expected.Role == RoadEndRole.Start ? 0 : 1, patch.StartParameter);
119        }
120        void Add(RoadEdge edge, RoadEndRole role, RoadPoint next)
121        {
122            double length = center.Position.DistanceTo(next);
123            rays.Add((edge, role, new((next.X - center.Position.X) / length, (next.Y - center.Position.Y) / length)));
124        }
125    }

127    private static (RoadNode[] Nodes, RoadEdge[] Edges) EightWay()
128    {
129        RoadPoint[] tips = [new(25, 0), new(25, 25), new(0, 25), new(-25, 25), new(-25, 0), new(-25, -25), new(0, -25), new(25, -25)];
130        RoadNode[] nodes = [new(new(1), default), .. tips.Select((p, i) => new RoadNode(new(i + 2), p))];
131        RoadProfileId[] profiles = [RoadProfileId.Street, RoadProfileId.Dirt, RoadProfileId.Highway, RoadProfileId.Arterial];
132        RoadEdge[] edges = tips.Select((point, i) => i % 2 == 0
133            ? new RoadEdge(new(8 - i), nodes[0].Id, nodes[i + 1].Id, profiles[i % 4], [nodes[0].Position, point])
134            : new RoadEdge(new(8 - i), nodes[i + 1].Id, nodes[0].Id, profiles[i % 4], [point, nodes[0].Position])).ToArray();
135        return (nodes, edges);
136    }

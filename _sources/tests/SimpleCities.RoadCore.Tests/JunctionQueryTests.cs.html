1namespace SimpleCities.RoadCore.Tests;
2
3public sealed class JunctionQueryTests
4{
5    [Fact]
6    public void CrossReadsStableEndRolesDirectionsAndGeometricTurns()
7    {
8        var network = new RoadNetwork();
9        Build(network, new(-300, 0), new(300, 0));
10        Build(network, new(0, -300), new(0, 300));
11        RoadSnapshot snapshot = network.Snapshot;
12        RoadNode center = Assert.Single(snapshot.Nodes, node => node.Position == new RoadPoint(0, 0));
13        RoadJunctionReadModel junction = RoadJunctionQuery.Read(snapshot, center.Id)!;
14        Assert.Equal(snapshot.Token, junction.Source);
15        Assert.Equal(4, junction.Incidences.Count);
16        Assert.Equal(16, junction.Turns.Count);
17        Assert.Equal(new[] { new RoadVector(1, 0), new RoadVector(0, 1), new RoadVector(-1, 0), new RoadVector(0, -1) },
18            junction.Incidences.Select(incidence => incidence.Outward));
19        Assert.Equal(4, junction.Incidences.Select(incidence => incidence.Key).Distinct().Count());
20        Assert.Equal(4, junction.Turns.Count(turn => turn.From == turn.To && turn.SignedAngleRadians == Math.PI));
21        Assert.Equal(4, junction.Turns.Count(turn => turn.SignedAngleRadians == 0));
22        Assert.Equal(junction.Incidences, RoadJunctionQuery.Read(snapshot, center.Id)!.Incidences);
23    }
24
25    private static void Build(RoadNetwork network, RoadPoint start, RoadPoint end)
26    {
27        RoadBuildResult result = network.PlanBuild(new(network.Snapshot.Token, start, end, RoadProfileId.Street));
28        Assert.Equal(RoadBuildStatus.Ready, result.Status);
29        Assert.True(network.TryCommit(result.Plan!));
30    }

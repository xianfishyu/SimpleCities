9    [InlineData("pure")]
10    [InlineData("branch")]
11    [InlineData("parallel")]
12    [InlineData("cell-center")]
13    public void LoopRoundTrip_PreservesSeamIdentitiesEndRolesAndLocations(string shape)
14    {
15        RoadNetwork network;
16        if (shape == "cell-center")
17        {
18            network = new RoadNetwork(new MapDefinition(25));
19            LoopRoadTests.Build(network, new(0, 0), new(25, 25));
20            LoopRoadTests.Build(network, new(0, 0), new(25, 0));
21            LoopRoadTests.Build(network, new(25, 0), new(12.5, 12.5));
22        }
23        else
24        {
25            network = LoopRoadTests.Square();
26            if (shape == "branch") LoopRoadTests.Build(network, new(100, 100), new(200, 100));
27            if (shape == "parallel") LoopRoadTests.Build(network, new(0, 0), new(100, 100));
28        }
29        RoadSnapshot original = network.Snapshot;
30        byte[] bytes = Save(network);
31        Assert.Equal(6, JsonNode.Parse(bytes)!["schemaVersion"]!.GetValue<int>());
32        var restored = new RoadNetwork();
33        for (int i = 0; i < 2; i++)
34        {
35            using var stream = new MemoryStream(bytes);
36            Assert.True(restored.TryCommit(restored.PlanLoad(RoadCodec.Read(stream))));
37            RoadSnapshot loaded = restored.Snapshot;
38            Assert.Equal(bytes, Save(restored));
39            Assert.Equal(original.Nodes, loaded.Nodes);
40            Assert.Equal(original.NextNodeId, loaded.NextNodeId);
41            Assert.Equal(original.NextEdgeId, loaded.NextEdgeId);
42            foreach (RoadEdge edge in original.Edges)
43            {
44                RoadEdge copy = Assert.Single(loaded.Edges, item => item.Id == edge.Id);
45                Assert.Equal(edge.Start, copy.Start);
46                Assert.Equal(edge.End, copy.End);
47                Assert.Equal(edge.Profile, copy.Profile);
48                Assert.Equal(edge.Points, copy.Points);
49                foreach (double parameter in new[] { 0, 0.25, 0.5, 0.75, 1 })
50                    Assert.Equal(original.Resolve(new(original.Token, edge.Id, parameter)), loaded.Resolve(new(loaded.Token, edge.Id, parameter)));
51                Assert.Null(loaded.Resolve(new(original.Token, edge.Id, 0.5)));
52            }
53            foreach (RoadNode node in original.Nodes)
54            {
55                RoadJunctionReadModel before = RoadJunctionQuery.Read(original, node.Id)!;
56                RoadJunctionReadModel after = RoadJunctionQuery.Read(loaded, node.Id)!;
57                Assert.Equal(before.Incidences, after.Incidences);
58                Assert.Equal(before.Turns, after.Turns);
59                Assert.Equal(loaded.Token, after.Source);
60            }
61        }
62    }

64    [Theory]
65    [InlineData("reverse")]
66    [InlineData("missing-close")]
67    [InlineData("wrong-seam-position")]
68    [InlineData("redundant-degree-two-node")]
69    [InlineData("overlapping-return")]
70    public void NoncanonicalLoopPayload_IsRejectedBeforePublication(string corruption)
71    {
72        var network = LoopRoadTests.Square();
73        RoadSnapshot before = network.Snapshot;
74        byte[] bytes = Save(network);
75        JsonNode payload = JsonNode.Parse(bytes)!;
76        JsonNode edge = payload["edges"]![0]!;
77        var points = (JsonArray)edge["points"]!;
78        switch (corruption)
79        {
80            case "reverse":
81                edge["points"] = new JsonArray(points.Reverse().Select(point => point!.DeepClone()).ToArray());
82                break;
83            case "missing-close":
84                points.RemoveAt(points.Count - 1);
85                break;
86            case "wrong-seam-position":
87                payload["nodes"]![0]!["x"] = -100;
88                break;
89            case "redundant-degree-two-node":
90                payload["nodes"] = JsonNode.Parse("""[{"id":1,"x":0,"y":0},{"id":2,"x":100,"y":100}]""");
91                payload["edges"] = JsonNode.Parse("""[{"id":1,"startNodeId":1,"endNodeId":2,"profile":"street","points":[{"x":0,"y":0},{"x":0,"y":100},{"x":100,"y":100}]},{"id":2,"startNodeId":1,"endNodeId":2,"profile":"street","points":[{"x":0,"y":0},{"x":100,"y":0},{"x":100,"y":100}]}]""");
92                payload["nextEdgeId"] = 3;
93                break;
94            case "overlapping-return":
95                edge["points"] = JsonNode.Parse("""[{"x":0,"y":0},{"x":100,"y":0},{"x":0,"y":0}]""");
96                break;
97        }
98        using var stream = new MemoryStream(Encoding.UTF8.GetBytes(payload.ToJsonString()));
99        Assert.Throws<InvalidDataException>(() => network.PlanLoad(RoadCodec.Read(stream)));
100        Assert.Same(before, network.Snapshot);
101        Assert.Equal(bytes, Save(network));
102    }

104    [Fact]
105    public void LoopAtExhaustedNodeWatermark_RejectsANewBranchWithoutPartialChanges()
106    {
107        var network = LoopRoadTests.Square();
108        JsonNode payload = JsonNode.Parse(Save(network))!;
109        payload["nextNodeId"] = long.MaxValue - 1;
110        using (var stream = new MemoryStream(Encoding.UTF8.GetBytes(payload.ToJsonString())))
111            Assert.True(network.TryCommit(network.PlanLoad(RoadCodec.Read(stream))));
112        RoadSnapshot before = network.Snapshot;
113        byte[] saved = Save(network);
114        RoadBuildResult result = network.PlanBuild(new(before.Token, new(100, 100), new(200, 100), RoadProfileId.Street));
115        Assert.Equal(RoadBuildStatus.Rejected, result.Status);
116        Assert.Contains("耗尽", result.Reason);
117        Assert.Null(result.Plan);
118        Assert.Same(before, network.Snapshot);
119        Assert.Equal(saved, Save(network));
120        using var readable = new MemoryStream(saved);
121        Assert.Single(RoadCodec.Read(readable).Edges);
122    }
123
124    private static byte[] Save(RoadNetwork network)
125    {
126        using var stream = new MemoryStream();
127        RoadCodec.Write(stream, network.Snapshot);
128        return stream.ToArray();
129    }

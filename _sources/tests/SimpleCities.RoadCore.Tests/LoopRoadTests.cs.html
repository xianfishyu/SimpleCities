28    public void PureLoopSeam_JoinsBothEndRolesWithoutEndpointCaps()
29    {
30        var network = new RoadNetwork();
31        Build(network, new(0, 0), new(100, 0));
32        Build(network, new(100, 0), new(100, 100));
33        Build(network, new(100, 100), new(0, 100));
34        Build(network, new(0, 100), new(0, 0));
35        RoadSnapshot snapshot = network.Snapshot;
36        RoadEdge loop = Assert.Single(snapshot.Edges);
37        RoadNode seam = Assert.Single(snapshot.Nodes);
38        RoadJunctionReadModel read = RoadJunctionQuery.Read(snapshot, seam.Id)!;
39        Assert.Equal(2, read.Incidences.Count);
40        Assert.Equal(4, read.Turns.Count);
41        Assert.Contains(read.Incidences, item => item.Key == new RoadIncidenceKey(loop.Id, RoadEndRole.Start) && item.Outward == new RoadVector(0, 1));
42        Assert.Contains(read.Incidences, item => item.Key == new RoadIncidenceKey(loop.Id, RoadEndRole.End) && item.Outward == new RoadVector(1, 0));
43        RoadSurfaceData surface = RoadPresentation.Prepare(snapshot.Nodes, snapshot.Edges)!;
44        RoadSurfacePiece[] joins = surface.Pieces.Where(piece => piece.Start == seam.Position && piece.End == seam.Position).ToArray();
45        Assert.Contains(joins, piece => piece.StartParameter == 0 && piece.EndParameter == 0);
46        Assert.Contains(joins, piece => piece.StartParameter == 1 && piece.EndParameter == 1);
47        Assert.All(joins, piece => Assert.Null(piece.JunctionNode));
48        RoadSurfacePiece first = Assert.Single(surface.Pieces, piece => piece.StartParameter == 0 && piece.EndParameter > 0);
49        Assert.Equal(0, first.Corners.Min(point => point.Y));
50        RoadSurfacePiece last = Assert.Single(surface.Pieces, piece => piece.StartParameter < 1 && piece.EndParameter == 1);
51        Assert.Equal(0, last.Corners.Min(point => point.X));
52    }

87    {
88        var network = Square();
89        RoadNode oldSeam = Assert.Single(network.Snapshot.Nodes);
90        Build(network, new(100, 100), new(200, 100));
91        RoadSnapshot snapshot = network.Snapshot;
92        Assert.Equal(2, snapshot.NodeCount);
93        Assert.Equal(2, snapshot.EdgeCount);
94        Assert.DoesNotContain(snapshot.Nodes, node => node.Id == oldSeam.Id);
95        RoadNode junction = Assert.Single(snapshot.Nodes, node => node.Position == new RoadPoint(100, 100));
96        RoadEdge loop = Assert.Single(snapshot.Edges, edge => edge.Start == edge.End);
97        Assert.Equal(junction.Id, loop.Start);
98        RoadJunctionReadModel read = RoadJunctionQuery.Read(snapshot, junction.Id)!;
99        Assert.Equal(3, read.Incidences.Count);
100        Assert.Equal(9, read.Turns.Count);
101        Assert.Equal(2, read.Incidences.Count(item => item.Key.Edge == loop.Id));
102        Assert.Contains(read.Incidences, item => item.Key.Role == RoadEndRole.Start && item.Key.Edge == loop.Id && item.Outward == new RoadVector(-1, 0));
103        Assert.Contains(read.Incidences, item => item.Key.Role == RoadEndRole.End && item.Key.Edge == loop.Id && item.Outward == new RoadVector(0, -1));
104        RoadSurfaceData surface = RoadPresentation.Prepare(snapshot.Nodes, snapshot.Edges)!;
105        RoadSurfacePiece[] junctionPieces = surface.Pieces.Where(piece => piece.JunctionNode == junction.Id).ToArray();
106        Assert.Contains(junctionPieces, piece => piece.Edge.Id == loop.Id && piece.StartParameter == 0 && piece.EndParameter == 0);
107        Assert.Contains(junctionPieces, piece => piece.Edge.Id == loop.Id && piece.StartParameter == 1 && piece.EndParameter == 1);
108        Assert.All(junctionPieces, piece => Assert.Equal(junction.Position, snapshot.Resolve(new(snapshot.Token, piece.Edge.Id, piece.StartParameter))));
109    }

188        Assert.Same(before, network.Snapshot);

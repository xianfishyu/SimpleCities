15    [Fact]
16    public void TurningRoad_RoundTripsEveryTurnAndCanContinueAfterReload()
17    {
18        var network = new RoadNetwork();
19        Build(network, new(0, 0), new(200, 0), RoadProfileId.Street);
20        Build(network, new(200, 0), new(200, 300), RoadProfileId.Street);
21        byte[] saved = Save(network);
22        var restored = Load(saved);
23
24        Assert.Equal(2, restored.Snapshot.NodeCount);
25        RoadEdge edge = Assert.Single(restored.Snapshot.Edges);
26        Assert.Equal(new[] { new RoadPoint(0, 0), new RoadPoint(200, 0), new RoadPoint(200, 300) }, edge.Points);
27        Assert.Equal(saved, Save(restored));
28        Build(restored, new(200, 300), new(300, 400), RoadProfileId.Street);
29        Assert.Equal(4, Assert.Single(restored.Snapshot.Edges).Points.Count);
30        Assert.Equal(new RoadPoint(300, 400), restored.Snapshot.Edges[0].Points[^1]);
31    }

33    [Fact]
34    public void ReverseContinuationAndProfileBoundary_RetainStableContentAfterReload()
35    {
36        var network = new RoadNetwork();
37        Build(network, new(0, 0), new(200, 0), RoadProfileId.Street);
38        Build(network, new(0, 0), new(-100, 100), RoadProfileId.Street);
39        Build(network, new(-100, 100), new(-100, 300), RoadProfileId.Dirt);
40        RoadSnapshot before = network.Snapshot;
41        byte[] saved = Save(network);
42        RoadNetwork restored = Load(saved);
43
44        Assert.Equal(3, restored.Snapshot.NodeCount);
45        Assert.Equal(2, restored.Snapshot.EdgeCount);
46        RoadEdge street = Assert.Single(restored.Snapshot.Edges, edge => edge.Profile == RoadProfileId.Street);
47        RoadEdge dirt = Assert.Single(restored.Snapshot.Edges, edge => edge.Profile == RoadProfileId.Dirt);
48        Assert.Equal(new[] { new RoadPoint(200, 0), new RoadPoint(0, 0), new RoadPoint(-100, 100) }, street.Points);
49        Assert.Equal(new[] { new RoadPoint(-100, 100), new RoadPoint(-100, 300) }, dirt.Points);
50        Assert.Equal(before.NextNodeId, restored.Snapshot.NextNodeId);
51        Assert.Equal(before.NextEdgeId, restored.Snapshot.NextEdgeId);
52        Assert.Equal(before.Nodes, restored.Snapshot.Nodes);
53        Assert.Equal(before.Edges.Select(edge => edge.Id), restored.Snapshot.Edges.Select(edge => edge.Id));
54        Assert.Equal(saved, Save(restored));
55
56        Build(restored, new(-100, 300), new(-300, 300), RoadProfileId.Dirt);
57        Assert.Equal(3, restored.Snapshot.NodeCount);
58        Assert.Equal(2, restored.Snapshot.EdgeCount);
59    }

68    [InlineData("\"x\":200,\"y\":0", "\"x\":200,\"y\":50")]

73    public void InvalidPolylineContent_IsRejectedWithoutChangingPublishedRoad(string original, string replacement)
74    {
75        var network = new RoadNetwork();
76        Build(network, new(0, 0), new(-200, 0), RoadProfileId.Highway);
77        RoadSnapshot before = network.Snapshot;
78        using var source = new MemoryStream(Encoding.UTF8.GetBytes(TurningRoad.Replace(original, replacement)));
79
80        Assert.Throws<InvalidDataException>(() => network.PlanLoad(RoadCodec.Read(source)));
81        Assert.Same(before, network.Snapshot);
82        Assert.Equal(RoadProfileId.Highway, Assert.Single(network.Snapshot.Edges).Profile);
83    }

85    [Fact]
86    public void NonSeekablePayload_IsReadInChunksAndCannotExceedByteBudget()
87    {
88        using var valid = new ChunkedReadStream(Encoding.UTF8.GetBytes(TurningRoad));
89        PreparedRoadState prepared = RoadCodec.Read(valid);
90        Assert.Equal(3, Assert.Single(prepared.Edges).Points.Count);
91
92        byte[] oversized = Encoding.UTF8.GetBytes(TurningRoad + new string(' ', RoadCodec.MaximumPayloadBytes));
93        using var invalid = new ChunkedReadStream(oversized);
94        Assert.Throws<InvalidDataException>(() => RoadCodec.Read(invalid));
95        Assert.Equal(RoadCodec.MaximumPayloadBytes + 1, invalid.BytesRead);
96    }

98    private static void Build(RoadNetwork network, RoadPoint start, RoadPoint end, RoadProfileId profile)
99    {
100        RoadBuildResult result = network.PlanBuild(new(network.Snapshot.Token, start, end, profile));
101        Assert.Equal(RoadBuildStatus.Ready, result.Status);
102        Assert.True(network.TryCommit(result.Plan!));
103    }
104
105    private static byte[] Save(RoadNetwork network)
106    {
107        using var destination = new MemoryStream();
108        RoadCodec.Write(destination, network.Snapshot);
109        return destination.ToArray();
110    }
111
112    private static RoadNetwork Load(byte[] bytes)
113    {
114        using var source = new MemoryStream(bytes);
115        var result = new RoadNetwork();
116        Assert.True(result.TryCommit(result.PlanLoad(RoadCodec.Read(source))));
117        return result;
118    }

120    private sealed class ChunkedReadStream(byte[] bytes) : Stream
121    {
122        public int BytesRead { get; private set; }
123        public override bool CanRead => true;
124        public override bool CanSeek => false;
125        public override bool CanWrite => false;
126        public override long Length => throw new NotSupportedException();
127        public override long Position { get => throw new NotSupportedException(); set => throw new NotSupportedException(); }
128        public override int Read(byte[] buffer, int offset, int count)
129        {
130            int length = Math.Min(Math.Min(count, 37), bytes.Length - BytesRead);
131            bytes.AsSpan(BytesRead, length).CopyTo(buffer.AsSpan(offset, length));
132            BytesRead += length;
133            return length;
134        }
135        public override void Flush() => throw new NotSupportedException();
136        public override long Seek(long offset, SeekOrigin origin) => throw new NotSupportedException();
137        public override void SetLength(long value) => throw new NotSupportedException();
138        public override void Write(byte[] buffer, int offset, int count) => throw new NotSupportedException();
139    }

18    [Fact]
19    public void SurfacePieces_RetainWholeEdgeArcLengthSourcesAcrossTheTurn()
20    {
21        var network = new RoadNetwork();
22        Build(network, new(0, 0), new(300, 0), RoadProfileId.Street);
23        Build(network, new(300, 0), new(300, 400), RoadProfileId.Street);
24        RoadSurfaceData surface = RoadPresentation.Prepare(network.Snapshot.Nodes, network.Snapshot.Edges)!;
25        RoadSurfacePiece horizontal = Assert.Single(surface.Pieces, piece => piece.Start == new RoadPoint(0, 0));
26        RoadSurfacePiece vertical = Assert.Single(surface.Pieces, piece => piece.End == new RoadPoint(300, 400));
27
28        Assert.Equal(0, horizontal.StartParameter);
29        Assert.Equal(3.0 / 7, horizontal.EndParameter);
30        Assert.Equal(horizontal.EndParameter, vertical.StartParameter);
31        Assert.Equal(1, vertical.EndParameter);
32        RoadSurfacePiece[] joins = surface.Pieces.Where(piece => piece.Start == piece.End).ToArray();
33        Assert.NotEmpty(joins);
34        Assert.All(joins, piece =>
35        {
36            Assert.Equal(new RoadPoint(300, 0), piece.Start);
37            Assert.Equal(3.0 / 7, piece.StartParameter);
38            Assert.Equal(piece.StartParameter, piece.EndParameter);
39        });

60    [Fact]
61    public void DifferentWidthConnection_HasNoGapAndRetainsBothProfileSources()
62    {
63        var network = new RoadNetwork();
64        Build(network, new(0, 0), new(300, 0), RoadProfileId.Street);
65        Build(network, new(300, 0), new(300, 400), RoadProfileId.Arterial);
66        RoadSurfaceData surface = RoadPresentation.Prepare(network.Snapshot.Nodes, network.Snapshot.Edges)!;
67
68        Assert.Contains(surface.Pieces, piece => Contains(piece.Corners, new(301, -0.5)));
69        Assert.Contains(surface.Pieces, piece => piece.Edge.Profile == RoadProfileId.Street && Contains(piece.Corners, new(150, 0)));
70        Assert.Contains(surface.Pieces, piece => piece.Edge.Profile == RoadProfileId.Arterial && Contains(piece.Corners, new(300, 200)));
71        RoadSurfacePiece[] joins = surface.Pieces.Where(piece => piece.Start == piece.End).ToArray();
72        Assert.Contains(joins, piece => piece.Edge.Profile == RoadProfileId.Street && piece.StartParameter == 1);
73        Assert.Contains(joins, piece => piece.Edge.Profile == RoadProfileId.Arterial && piece.StartParameter == 0);
74        Assert.All(joins, piece => Assert.All(piece.Corners,
75            corner => Assert.InRange(corner.DistanceTo(new(300, 0)), 0, 3)));
76    }

78    private static void Build(RoadNetwork network, RoadPoint start, RoadPoint end, RoadProfileId profile)
79    {
80        RoadBuildResult result = network.PlanBuild(new(network.Snapshot.Token, start, end, profile));
81        Assert.Equal(RoadBuildStatus.Ready, result.Status);
82        Assert.True(network.TryCommit(result.Plan!));
83    }

85    private static bool Contains(IReadOnlyList<RoadPoint> polygon, RoadPoint point)
86    {
87        bool inside = false;
88        for (int i = 0, j = polygon.Count - 1; i < polygon.Count; j = i++)
89        {
90            RoadPoint a = polygon[i], b = polygon[j];
91            if ((a.Y > point.Y) != (b.Y > point.Y) &&
92                point.X < (b.X - a.X) * (point.Y - a.Y) / (b.Y - a.Y) + a.X)
93                inside = !inside;
94        }
95        return inside;
96    }

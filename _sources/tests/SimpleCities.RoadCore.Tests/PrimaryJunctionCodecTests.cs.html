8    [Fact]
9    public void MainGridCross_RoundTripsCanonicalConnectivityWithoutDuplicatingJunctions()
10    {
11        var network = new RoadNetwork();
12        Build(network, new(-200, 0), new(200, 0));
13        Build(network, new(0, -200), new(0, 200));
14        byte[] saved = Save(network);
15        Assert.Equal(6, JsonNode.Parse(saved)!["schemaVersion"]!.GetValue<int>());
16
17        RoadSnapshot before = network.Snapshot;
18        Assert.Equal(5, before.NodeCount);
19        Assert.Equal(4, before.EdgeCount);
20        RoadNode junction = Assert.Single(before.Nodes, node => node.Position == new RoadPoint(0, 0));
21        Assert.Equal(4, before.Edges.Count(edge => edge.Start == junction.Id || edge.End == junction.Id));
22        RoadJunctionReadModel junctionBefore = RoadJunctionQuery.Read(before, junction.Id)!;
23
24        var restored = new RoadNetwork();
25        for (int read = 0; read < 2; read++)
26        {
27            using var source = new MemoryStream(saved);
28            Assert.True(restored.TryCommit(restored.PlanLoad(RoadCodec.Read(source))));
29            Assert.Equal(saved, Save(restored));
30            Assert.Equal(before.Nodes, restored.Snapshot.Nodes);
31            Assert.Equal(before.NextNodeId, restored.Snapshot.NextNodeId);
32            Assert.Equal(before.NextEdgeId, restored.Snapshot.NextEdgeId);
33            RoadJunctionReadModel junctionAfter = RoadJunctionQuery.Read(restored.Snapshot, junction.Id)!;
34            Assert.Equal(restored.Snapshot.Token, junctionAfter.Source);
35            Assert.Equal(junctionBefore.Node, junctionAfter.Node);
36            Assert.Equal(junctionBefore.Incidences, junctionAfter.Incidences);
37            Assert.Equal(junctionBefore.Turns, junctionAfter.Turns);
38            foreach (RoadEdge edge in before.Edges)
39            {
40                RoadEdge reloaded = Assert.Single(restored.Snapshot.Edges, item => item.Id == edge.Id);
41                Assert.Equal(edge.Start, reloaded.Start);
42                Assert.Equal(edge.End, reloaded.End);
43                Assert.Equal(edge.Profile, reloaded.Profile);
44                Assert.Equal(edge.Points, reloaded.Points);
45                Assert.Equal(before.Resolve(new(before.Token, edge.Id, 0.25)),
46                    restored.Snapshot.Resolve(new(restored.Snapshot.Token, edge.Id, 0.25)));
47                Assert.Null(restored.Snapshot.Resolve(new(before.Token, edge.Id, 0.25)));
48            }
49        }
50    }

52    [Fact]
53    public void ReloadedTJunction_ReusesItsNodeForDiagonalBranches()
54    {
55        var network = new RoadNetwork();
56        Build(network, new(-200, 0), new(200, 0));
57        Build(network, new(0, 0), new(0, 200), RoadProfileId.Dirt);
58        RoadNode junction = Assert.Single(network.Snapshot.Nodes, node => node.Position == new RoadPoint(0, 0));
59        Assert.Equal(3, RoadJunctionQuery.Read(network.Snapshot, junction.Id)!.Incidences.Count);
60
61        var restored = new RoadNetwork();
62        using (var source = new MemoryStream(Save(network)))
63            Assert.True(restored.TryCommit(restored.PlanLoad(RoadCodec.Read(source))));
64        Build(restored, new(0, 0), new(200, 200), RoadProfileId.Highway);
65
66        RoadNode reused = Assert.Single(restored.Snapshot.Nodes, node => node.Position == new RoadPoint(0, 0));
67        Assert.Equal(junction.Id, reused.Id);
68        Assert.Equal(5, restored.Snapshot.NodeCount);
69        Assert.Equal(4, restored.Snapshot.EdgeCount);
70        RoadJunctionReadModel read = RoadJunctionQuery.Read(restored.Snapshot, junction.Id)!;
71        Assert.Equal(4, read.Incidences.Count);
72        Assert.Single(read.Incidences, incidence => incidence.Profile == RoadProfileId.Dirt);
73        Assert.Single(read.Incidences, incidence => incidence.Profile == RoadProfileId.Highway);
74        byte[] saved = Save(restored);
75        using (var source = new MemoryStream(saved))
76            Assert.True(restored.TryCommit(restored.PlanLoad(RoadCodec.Read(source))));
77        Assert.Equal(saved, Save(restored));
78        Assert.Equal(read.Incidences, RoadJunctionQuery.Read(restored.Snapshot, junction.Id)!.Incidences);
79        Assert.Equal(read.Turns, RoadJunctionQuery.Read(restored.Snapshot, junction.Id)!.Turns);
80    }

100    [MemberData(nameof(InvalidTopologies))]
101    public void NoncanonicalJunctionPayload_IsRejectedWithoutChangingActiveRoad(string caseName, string nodes, string edges)
102    {
103        var network = new RoadNetwork();
104        Build(network, new(-300, -300), new(-100, -300), RoadProfileId.Highway);
105        RoadSnapshot before = network.Snapshot;
106        byte[] saved = Save(network);
107        string payload = $$"""
108            {"formatFamily":"simple-cities-v4","payloadType":"road-network","schemaVersion":6,
109            "contentRevision":3,"nextNodeId":10,"nextEdgeId":10,"profileCatalogVersion":1,
110            "map":{"widthMetres":8000,"heightMetres":8000,"origin":"center","metresPerUnit":1,"grid":"square-eight","cellSizeMetres":100},
111            "nodes":{{nodes}},"edges":{{edges}}}
112            """;
113        using var source = new MemoryStream(Encoding.UTF8.GetBytes(payload));
114
115        Exception? error = Record.Exception(() => network.PlanLoad(RoadCodec.Read(source)));
116        Assert.True(error is InvalidDataException, $"{caseName} must be rejected as invalid road content; actual: {error}");
117        Assert.Same(before, network.Snapshot);
118        Assert.Equal(saved, Save(network));
119    }

140    private static void Build(RoadNetwork network, RoadPoint start, RoadPoint end, RoadProfileId? profile = null)
141    {
142        RoadBuildResult result = network.PlanBuild(new(network.Snapshot.Token, start, end, profile ?? RoadProfileId.Street));
143        Assert.Equal(RoadBuildStatus.Ready, result.Status);
144        Assert.True(network.TryCommit(result.Plan!));
145    }

147    private static byte[] Save(RoadNetwork network)

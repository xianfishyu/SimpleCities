18        RoadSurfaceData surface = RoadPresentation.Prepare(network.Snapshot.Nodes, network.Snapshot.Edges)!;

20        Assert.NotEmpty(patches);
21        Assert.Equal(4, patches.Select(piece => piece.Edge.Id).Distinct().Count());
22        foreach (RoadSurfacePiece piece in patches)
23        {
24            // The actual rendering boundary widens its binary32 vertices before cross products.
25            RoadPoint[] corners = piece.Corners.Select(point => new RoadPoint((float)point.X, (float)point.Y)).ToArray();
26            RoadPoint a = corners[0], b = corners[1], c = corners[2];
27            double area = (b.X - a.X) * (c.Y - a.Y) - (b.Y - a.Y) * (c.X - a.X);
28            Assert.True(area > 0, $"Display triangle collapsed at {piece.Start}: {string.Join(", ", corners)}");
29        }

60    public void UnequalWidthTJunction_HasNoCentralGapAndStableOwnedPatches(int cellSize, bool diagonal)
61    {
62        var network = new RoadNetwork(new MapDefinition(cellSize));
63        Build(network, new(-cellSize, 0), new(cellSize, 0), RoadProfileId.Highway);
64        Build(network, new(diagonal ? -cellSize : 0, -cellSize), new(0, 0), RoadProfileId.Dirt);
65        RoadSnapshot snapshot = network.Snapshot;
66        RoadNode node = Assert.Single(snapshot.Nodes, node => node.Position == new RoadPoint(0, 0));
67        RoadSurfaceData surface = RoadPresentation.Prepare(snapshot.Nodes, snapshot.Edges)!;
68        RoadSurfacePiece[] patches = surface.Pieces.Where(piece => piece.JunctionNode == node.Id).ToArray();
69        Assert.NotEmpty(patches);
70
71        // Literal interior points exercise both sides of the join and its center.
72        foreach (RoadPoint point in new RoadPoint[] { new(0, 0), new(-0.5, -0.25), new(0.5, -0.25), new(-0.5, 0.25), new(0.5, 0.25) })
73            Assert.Contains(patches, piece => Contains(piece.Corners, point));
74        Assert.All(patches, piece =>
75        {
76            Assert.Equal(node.Position, piece.Start);
77            Assert.Equal(node.Position, piece.End);
78            Assert.Equal(piece.Edge.Start == node.Id ? 0 : 1, piece.StartParameter);
79            Assert.Equal(piece.StartParameter, piece.EndParameter);
80            Assert.All(piece.Corners, point => Assert.InRange(point.DistanceTo(node.Position), 0, 5.75));
81        });
82
83        RoadSurfacePiece[] reordered = RoadPresentation.Prepare(snapshot.Nodes.Reverse().ToArray(), snapshot.Edges.Reverse().ToArray())!
84            .Pieces.Where(piece => piece.JunctionNode == node.Id).ToArray();
85        Assert.Equal(patches.Length, reordered.Length);
86        for (int i = 0; i < patches.Length; i++)
87        {
88            Assert.Equal(patches[i].Edge.Id, reordered[i].Edge.Id);
89            Assert.Equal(patches[i].Corners, reordered[i].Corners);
90            Assert.Equal(patches[i].StartParameter, reordered[i].StartParameter);
91        }
92    }

121    private static void Build(RoadNetwork network, RoadPoint start, RoadPoint end, RoadProfileId profile)
122    {
123        RoadBuildResult result = network.PlanBuild(new(network.Snapshot.Token, start, end, profile));
124        Assert.Equal(RoadBuildStatus.Ready, result.Status);
125        Assert.True(network.TryCommit(result.Plan!));

128    private static bool Contains(IReadOnlyList<RoadPoint> polygon, RoadPoint point)
129    {
130        bool inside = false;
131        for (int i = 0, j = polygon.Count - 1; i < polygon.Count; j = i++)
132        {
133            RoadPoint a = polygon[i], b = polygon[j];
134            double cross = (point.X - a.X) * (b.Y - a.Y) - (point.Y - a.Y) * (b.X - a.X);
135            if (Math.Abs(cross) < 1e-9 && point.X >= Math.Min(a.X, b.X) && point.X <= Math.Max(a.X, b.X) &&
136                point.Y >= Math.Min(a.Y, b.Y) && point.Y <= Math.Max(a.Y, b.Y)) return true;
137            if ((a.Y > point.Y) != (b.Y > point.Y) &&
138                point.X < (b.X - a.X) * (point.Y - a.Y) / (b.Y - a.Y) + a.X)
139                inside = !inside;
140        }
141        return inside;
142    }

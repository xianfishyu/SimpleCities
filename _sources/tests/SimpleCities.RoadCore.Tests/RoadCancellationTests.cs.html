1namespace SimpleCities.RoadCore.Tests;
2
3public sealed class RoadCancellationTests
4{
5    [Fact]
6    public void CancelledPlanningDoesNotChangeContentAndDoesNotPrepareAPresentation()
7    {
8        var network = new RoadNetwork();
9        RoadSnapshot before = network.Snapshot;
10        var request = new RoadBuildRequest(before.Token, new(0, 0), new(300, 0), RoadProfileId.Street);
11        using var cancellation = new CancellationTokenSource();
12        cancellation.Cancel();
13        Assert.Throws<OperationCanceledException>(() => network.PlanBuild(request, cancellation.Token));
14        Assert.Equal(before.Token, network.Snapshot.Token);
15        Assert.Equal(1, network.Snapshot.NextNodeId);
16        RoadPlan plan = network.PlanBuild(request).Plan!;
17        Assert.Throws<OperationCanceledException>(() => RoadPresentation.Prepare(plan.Target.Nodes, plan.Target.Edges, cancellation.Token));
18        Assert.Equal(0, network.Snapshot.EdgeCount);
19    }
20}

10    public void ChainPointBudgetIsIndependentOfEdgeAndNodeCounts(int count, bool accepted)
11    {
12        var points = new List<RoadPoint> { new(-4000, -4000) };
13        for (int row = 0; points.Count < count; row++)
14        {
15            int direction = row % 2 == 0 ? 1 : -1;
16            int x = direction > 0 ? -4000 : 4000, y = -4000 + row * 25;
17            if (row > 0) points.Add(new(x, y));
18            for (int column = 0; column < 320 && points.Count < count; column++)
19            {
20                points.Add(new(x + direction * 12.5, y + 12.5));
21                if (points.Count < count) points.Add(new(x + direction * 25, y));

23            }
24        }

80    private sealed class PaddedStream(byte[] prefix, long length) : Stream
81    {
82        public long TotalRead { get; private set; }
83        public override int Read(byte[] buffer, int offset, int count)
84        {
85            int amount = (int)Math.Min(count, length - TotalRead);
86            buffer.AsSpan(offset, amount).Fill((byte)' ');
87            if (TotalRead < prefix.Length)
88                prefix.AsSpan((int)TotalRead, Math.Min(amount, prefix.Length - (int)TotalRead)).CopyTo(buffer.AsSpan(offset));
89            TotalRead += amount;
90            return amount;
91        }
92        public override bool CanRead => true;
93        public override bool CanSeek => false;
94        public override bool CanWrite => false;
95        public override long Length => throw new NotSupportedException();
96        public override long Position { get => throw new NotSupportedException(); set => throw new NotSupportedException(); }
97        public override void Flush() => throw new NotSupportedException();
98        public override long Seek(long offset, SeekOrigin origin) => throw new NotSupportedException();
99        public override void SetLength(long value) => throw new NotSupportedException();
100        public override void Write(byte[] buffer, int offset, int count) => throw new NotSupportedException();
101    }

183    private static byte[] Payload(IReadOnlyList<RoadPoint[]> chains, int extraNodes = 0, bool alternateProfiles = false)
184    {
185        var nodes = new List<object>();
186        var edges = new List<object>();
187        var identities = new Dictionary<RoadPoint, int>();
188        int Node(RoadPoint point)
189        {
190            if (identities.TryGetValue(point, out int found)) return found;
191            int id = nodes.Count + 1;
192            nodes.Add(new { id, x = point.X, y = point.Y });
193            identities.Add(point, id);
194            return id;
195        }
196        for (int i = 0; i < chains.Count; i++)
197        {
198            RoadPoint[] chain = chains[i];
199            int start = Node(chain[0]), end = Node(chain[^1]);
200            IEnumerable<RoadPoint> ordered = start < end ? chain : chain.Reverse();
201            edges.Add(new { id = i + 1, startNodeId = Math.Min(start, end), endNodeId = Math.Max(start, end),
202                profile = alternateProfiles && i % 2 == 1 ? "dirt" : "street",
203                points = ordered.Select(p => new { x = p.X, y = p.Y }) });
204        }
205        for (int i = 0; i < extraNodes; i++) nodes.Add(new { id = nodes.Count + 1, x = 4000d, y = 4000d });
206        return JsonSerializer.SerializeToUtf8Bytes(new { formatFamily = "simple-cities-v4", payloadType = "road-network",
207            schemaVersion = 6, contentRevision = 1, nextNodeId = nodes.Count + 1, nextEdgeId = edges.Count + 1, profileCatalogVersion = 1,
208            map = new { widthMetres = 8000, heightMetres = 8000, origin = "center", metresPerUnit = 1, grid = "square-eight", cellSizeMetres = 25 },
209            nodes, edges });
210    }

1namespace SimpleCities.RoadCore.Tests;
2
3public sealed class RoadDisplayWidthTests
4{
5    [Theory]
6    [InlineData("dirt", 2)]
7    [InlineData("street", 3)]
8    [InlineData("arterial", 6)]
9    [InlineData("highway", 8)]
10    public void TwentyFiveMetreGridUsesProportionalSymbolWidthsWithoutChangingRoadGeometry(string profile, double expectedWidth)
11    {
12        var network = new RoadNetwork(new MapDefinition(25));
13        var build = network.PlanBuild(new(network.Snapshot.Token, new(0, 0), new(100, 0), new(profile)));
14        Assert.Equal(RoadBuildStatus.Ready, build.Status);
15        Assert.True(network.TryCommit(build.Plan!));
16        RoadSnapshot snapshot = network.Snapshot;
17        using var before = new MemoryStream();
18        RoadCodec.Write(before, snapshot);
19
20        RoadSurfacePiece ribbon = Assert.Single(RoadPresentation.Prepare(snapshot.Nodes, snapshot.Edges)!.Pieces);
21        Assert.Equal(expectedWidth, ribbon.Corners[0].DistanceTo(ribbon.Corners[3]), 8);
22        Assert.Equal(new RoadPoint(0, 0), ribbon.Start);
23        Assert.Equal(new RoadPoint(100, 0), ribbon.End);
24        Assert.Equal(100, ribbon.Edge.Length);
25        Assert.All(ribbon.Corners, corner => Assert.InRange(Math.Abs(corner.Y), 0, 4));
26        Assert.Same(snapshot, network.Snapshot);
27        using var after = new MemoryStream();
28        RoadCodec.Write(after, network.Snapshot);
29        Assert.Equal(before.ToArray(), after.ToArray());
30    }
31}

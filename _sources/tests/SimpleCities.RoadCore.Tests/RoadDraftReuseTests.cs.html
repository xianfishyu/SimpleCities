7    [Fact]
8    public void DistantBuildKeepsExistingEntitiesWithoutCopyingTheirPointChains()
9    {
10        RoadNetwork network = RoadBaselineFixtures.CreateMixed(25);
11        RoadSnapshot source = network.Snapshot;
12        var request = new RoadBuildRequest(source.Token, new(3000, 3000), new(3025, 3000), RoadProfileId.Street);
13        _ = network.PlanBuild(request);
14        long before = GC.GetAllocatedBytesForCurrentThread();
15
16        RoadPlan plan = network.PlanBuild(request).Plan!;
17
18        long allocated = GC.GetAllocatedBytesForCurrentThread() - before;
19        output.WriteLine($"Distant build: {allocated} B");
20        Assert.True(allocated < 300 * 1024, $"Distant build allocated {allocated} B.");
21        Assert.Same(source, network.Snapshot);
22        Assert.Equal(1, plan.AddedEdgeCount);
23        Assert.Equal(2, plan.AddedNodeCount);
24        Assert.Single(plan.ChangeSet.Edges);
25        foreach (RoadEdge edge in source.Edges) Assert.Same(edge, plan.Target.FindEdge(edge.Id));
26        Assert.True(network.TryCommit(plan));
27        Assert.True(network.TryCommit(network.PlanUndo().Plan!));
28        Assert.Equal(source.Edges, network.Snapshot.Edges);
29    }

31    [Theory]
32    [InlineData(0, 0, -100, 0)]
33    [InlineData(200, 200, 300, 200)]
34    [InlineData(100, 0, 100, -100)]
35    public void LoopBranchAtSeamCornerOrInteriorPreservesOldChainAndCorrectIncidences(
36        int x, int y, int toX, int toY)
37    {
38        var network = new RoadNetwork();
39        RoadBaselineFixtures.Build(network, new(0, 0), new(200, 0));
40        RoadBaselineFixtures.Build(network, new(200, 0), new(200, 200));
41        RoadBaselineFixtures.Build(network, new(200, 200), new(0, 200));
42        RoadBaselineFixtures.Build(network, new(0, 200), new(0, 0));
43        RoadSnapshot old = network.Snapshot;
44        RoadEdge original = Assert.Single(old.Edges);
45        RoadPoint[] originalPoints = original.Points.ToArray();
46
47        RoadBaselineFixtures.Build(network, new(x, y), new(toX, toY));
48
49        RoadSnapshot current = network.Snapshot;
50        Assert.Equal(2, current.EdgeCount);
51        RoadNode branch = Assert.Single(current.Nodes, n => n.Position == new RoadPoint(x, y));
52        RoadJunctionReadModel junction = RoadJunctionQuery.Read(current, branch.Id)!;
53        Assert.Equal(3, junction.Incidences.Count);
54        RoadEdge loop = Assert.Single(current.Edges, e => e.Start == e.End);
55        Assert.Equal(branch.Id, loop.Start);
56        Assert.Equal(2, junction.Incidences.Count(i => i.Key.Edge == loop.Id));
57        Assert.Equal(originalPoints, original.Points);
58        Assert.Equal(1, old.EdgeCount);
59        if (x == 0 && y == 0) Assert.Same(original, loop);
60        Assert.True(network.TryCommit(network.PlanUndo().Plan!));
61        Assert.Equal(originalPoints, Assert.Single(network.Snapshot.Edges).Points);
62        Assert.True(network.TryCommit(network.PlanRedo().Plan!));
63        using var bytes = new MemoryStream();
64        RoadCodec.Write(bytes, network.Snapshot);
65        bytes.Position = 0;
66        PreparedRoadState restored = RoadCodec.Read(bytes);
67        Assert.Equal(2, restored.Edges.Count);
68        Assert.Equal(originalPoints, original.Points);
69    }

5    [Fact]
6    public void RepeatedLengthAndPointReadsDoNotAllocate()
7    {
8        RoadEdge edge = Create([new(0, 0), new(300, 0), new(300, 400)]);
9        for (int i = 0; i < 4; i++) { _ = edge.Length; _ = edge.PointAt(0.5); }
10        double total = 0;
11        long before = GC.GetAllocatedBytesForCurrentThread();
12        for (int i = 0; i < 4096; i++) total += edge.Length + edge.PointAt(0.5).Y;
13        long allocated = GC.GetAllocatedBytesForCurrentThread() - before;
14        output.WriteLine($"Length/PointAt allocated bytes: {allocated}");
15        Assert.Equal(4096 * 750d, total);
16        Assert.Equal(0, allocated);
17    }

19    [Fact]
20    public void LengthPreservesOriginalAccumulationBitsIncludingInvalidModelInputs()
21    {
22        RoadPoint[][] paths = [[], [new(0, 0)], [new(0, 0), new(0, 0)],
23            [new(-4000, -4000), new(-3975, -3975), new(-3950, -3975), new(-3950, -3950)],
24            [new(0, 0), new(100, 0), new(100, 100), new(0, 100), new(0, 0)],
25            [new(double.NaN, 0), new(1, 1)], [new(0, 0), new(double.PositiveInfinity, 0)],
26            Enumerable.Range(0, 821).Select(i => new RoadPoint(i * 0.125, i % 3 * 0.25)).ToArray()];
27        foreach (RoadPoint[] points in paths)
28        {
29            // This is the prior public getter, retained as a numeric oracle.
30            double expected = points.Zip(points.Skip(1), (a, b) => a.DistanceTo(b)).Sum();
31            Assert.Equal(BitConverter.DoubleToInt64Bits(expected), BitConverter.DoubleToInt64Bits(Create(points).Length));
32        }
33    }

35    [Fact]
36    public void OwnedPointsAndRecordCopyKeepLengthAndLocationsStable()
37    {
38        RoadPoint[] points = [new(0, 0), new(300, 0), new(300, 400)];
39        RoadEdge edge = Create(points);
40        points[1] = new(999, 999);
41        Assert.Throws<NotSupportedException>(() => { ((IList<RoadPoint>)edge.Points)[1] = new(888, 888); });
42        RoadEdge clone = edge with { };
43        Assert.Equal(edge, clone);
44        Assert.Same(edge.Points, clone.Points);
45        foreach (RoadEdge current in new[] { edge, clone })
46        {
47            Assert.Equal(700, current.Length);
48            Assert.Equal(new RoadPoint(0, 0), current.PointAt(0));
49            Assert.Equal(new RoadPoint(175, 0), current.PointAt(0.25));
50            Assert.Equal(new RoadPoint(300, 50), current.PointAt(0.5));
51            Assert.Equal(new RoadPoint(300, 400), current.PointAt(1));
52        }
53    }

55    private static RoadEdge Create(RoadPoint[] points) => new(new(1), new(1), new(2), RoadProfileId.Street, points);

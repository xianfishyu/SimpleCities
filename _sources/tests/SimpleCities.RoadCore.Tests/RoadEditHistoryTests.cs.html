79        AssertRestoredContent(open, network.Snapshot);
80        Assert.True(network.TryCommit(network.PlanRedo().Plan!));
81        AssertRestoredContent(loop, network.Snapshot);
82        RoadGridSpan wrap = Pick(network.Snapshot, Assert.Single(network.Snapshot.Edges), 0.01);
83        Assert.Equal(2, wrap.Ranges.Count);
84

90        Assert.True(network.TryCommit(network.PlanUndo().Plan!));
91
92        AssertRestoredContent(loop, network.Snapshot);
93        RoadNode seam = Assert.Single(network.Snapshot.Nodes);
94        Assert.Equal(new RoadPoint(50, 50), seam.Position);
95        Assert.Equal(2, RoadJunctionQuery.Read(network.Snapshot, seam.Id)!.Incidences.Count);

99        Assert.Equal(Save(edited), Save(network.Snapshot));

128        Assert.Equal(0, network.History.UndoCount);

131        Assert.Equal(4, RoadJunctionQuery.Read(network.Snapshot, junction.Id)!.Incidences.Count);
132        Assert.True(network.TryCommit(network.PlanRedo().Plan!));

162    public void EditingAfterUndoDiscardsRedoAndAllocatesFreshRevisionAndIdentities()
163    {
164        var network = new RoadNetwork();
165        Build(network, new(0, 0), new(100, 0));
166        Build(network, new(0, 100), new(100, 100));
167        RoadSnapshot abandoned = network.Snapshot;
168        Assert.True(network.TryCommit(network.PlanUndo().Plan!));
169        RoadPlan staleRedo = network.PlanRedo().Plan!;
170
171        RoadBuildResult fork = network.PlanBuild(new(network.Snapshot.Token, new(0, 200), new(100, 200), RoadProfileId.Street));
172        Assert.Equal(RoadBuildStatus.Ready, fork.Status);
173        Assert.Equal(1, network.History.RedoCount);
174        Assert.True(network.TryCommit(fork.Plan!));
175
176        Assert.Equal(abandoned.Token.ContentRevision + 1, network.Snapshot.Token.ContentRevision);
177        RoadEdge branch = Assert.Single(network.Snapshot.Edges, edge => edge.Points[0].Y == 200);
178        Assert.True(branch.Id.Value >= abandoned.NextEdgeId);
179        Assert.True(branch.Start.Value >= abandoned.NextNodeId);
180        Assert.True(branch.End.Value >= abandoned.NextNodeId);
181        Assert.Equal(2, network.History.RetainedCount);
182        Assert.Equal(2, network.History.UndoCount);
183        Assert.Equal(0, network.History.RedoCount);
184        Assert.False(network.TryCommit(staleRedo));
185        Assert.Equal(RoadEditStatus.NoChange, network.PlanRedo().Status);
186        Assert.True(network.TryCommit(network.PlanUndo().Plan!));
187        Assert.Single(network.Snapshot.Edges);
188        Assert.True(network.TryCommit(network.PlanRedo().Plan!));
189        Assert.Contains(network.Snapshot.Edges, edge => edge.Id == branch.Id);
190        Assert.DoesNotContain(network.Snapshot.Edges, edge => edge.Points[0].Y == 100);
191    }

246    {
247        var network = new RoadNetwork();
248        for (int i = 0; i < count; i++) Build(network, new(0, -3200 + i * 100), new(100, -3200 + i * 100));
249        RoadSnapshot built = network.Snapshot;
250        RoadEditHistory completeHistory = network.History;
251        long bytes = completeHistory.EstimatedBytes;
252        Assert.Equal(64, completeHistory.RetainedCount);
253        Assert.True(bytes > 0);
254
255        for (int i = 0; i < 64; i++)
256        {
257            Assert.True(network.TryCommit(network.PlanUndo().Plan!));
258            Assert.Equal(63 - i, network.History.UndoCount);
259            Assert.Equal(i + 1, network.History.RedoCount);
260            Assert.Equal(64, network.History.RetainedCount);
261            Assert.Equal(bytes, network.History.EstimatedBytes);
262        }
263        Assert.Equal(count - 64, network.Snapshot.EdgeCount);
264        if (count == 65)
265            Assert.Equal(new[] { new RoadPoint(0, -3200), new RoadPoint(100, -3200) }, Assert.Single(network.Snapshot.Edges).Points);
266        RoadSnapshot oldest = network.Snapshot;
267        Assert.Equal(RoadEditStatus.NoChange, network.PlanUndo().Status);
268        Assert.Same(oldest, network.Snapshot);
269        Assert.Equal(64, completeHistory.UndoCount);
270
271        for (int i = 0; i < 64; i++) Assert.True(network.TryCommit(network.PlanRedo().Plan!));
272        Assert.Equal(Save(built), Save(network.Snapshot));
273        Assert.Equal(64, network.History.UndoCount);
274        Assert.Equal(0, network.History.RedoCount);
275        Assert.Equal(RoadEditStatus.NoChange, network.PlanRedo().Status);
276    }

280    {
281        var network = new RoadNetwork();

283        RoadBuildResult build = network.PlanBuild(new(empty.Token, new(0, 0), new(100, 0), RoadProfileId.Street));

287        Assert.Equal(0, network.History.RedoCount);

289
290        RoadEditResult undo = network.PlanUndo();
291        Assert.Equal(RoadEditStatus.Ready, undo.Status);
292        Assert.Same(built, network.Snapshot);
293        Assert.Equal(1, network.History.UndoCount);
294        Assert.True(network.TryCommit(undo.Plan!));

297        Assert.Equal(built.Token.ChangeSequence + 1, network.Snapshot.Token.ChangeSequence);

299        Assert.Equal(built.NextEdgeId, network.Snapshot.NextEdgeId);
300        Assert.Equal(0, network.History.UndoCount);
301        Assert.Equal(1, network.History.RedoCount);
302        Assert.Null(network.Snapshot.Resolve(location));
303
304        Assert.True(network.TryCommit(network.PlanRedo().Plan!));

327    private static byte[] Save(RoadSnapshot snapshot)
328    {
329        using var stream = new MemoryStream();
330        RoadCodec.Write(stream, snapshot);
331        return stream.ToArray();
332    }

334    private static void AssertRestoredContent(RoadSnapshot expected, RoadSnapshot actual)
335    {
336        Assert.Equal(expected.Map, actual.Map);
337        Assert.Equal(expected.Token.ContentRevision, actual.Token.ContentRevision);
338        Assert.Equal(expected.Nodes, actual.Nodes);
339        Assert.Equal(expected.EdgeCount, actual.EdgeCount);
340        foreach (RoadEdge edge in expected.Edges)
341        {
342            RoadEdge restored = Assert.Single(actual.Edges, candidate => candidate.Id == edge.Id);
343            Assert.Equal(edge.Start, restored.Start);
344            Assert.Equal(edge.End, restored.End);
345            Assert.Equal(edge.Profile, restored.Profile);
346            Assert.Equal(edge.Points, restored.Points);
347        }
348        Assert.True(actual.NextNodeId >= expected.NextNodeId);
349        Assert.True(actual.NextEdgeId >= expected.NextEdgeId);
350        AssertRoundTrip(actual);
351    }
352
353    private static void AssertRoundTrip(RoadSnapshot snapshot)
354    {
355        byte[] bytes = Save(snapshot);
356        using var stream = new MemoryStream(bytes);
357        var restored = new RoadNetwork();
358        Assert.True(restored.TryCommit(restored.PlanLoad(RoadCodec.Read(stream))));
359        Assert.Equal(bytes, Save(restored.Snapshot));
360        Assert.Equal(0, restored.History.RetainedCount);
361    }

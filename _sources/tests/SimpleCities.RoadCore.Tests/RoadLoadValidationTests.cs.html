1using System.Text;
2
3namespace SimpleCities.RoadCore.Tests;
4
5public sealed class RoadLoadValidationTests
6{
7    // 独立编写的规范载荷；非法输入直接变更原始 JSON，以保留重复字段。
8    private const string Valid = """
9        {"formatFamily":"simple-cities-v4","payloadType":"road-network","schemaVersion":6,
10        "contentRevision":2,"nextNodeId":3,"nextEdgeId":2,"profileCatalogVersion":1,
11        "map":{"widthMetres":8000,"heightMetres":8000,"origin":"center","metresPerUnit":1,"grid":"square-eight","cellSizeMetres":100},
12        "nodes":[{"id":1,"x":0,"y":0},{"id":2,"x":100,"y":0}],
13        "edges":[{"id":1,"startNodeId":1,"endNodeId":2,"profile":"street","points":[{"x":0,"y":0},{"x":100,"y":0}]}]}
14        """;
15
16    [Theory]
17    [InlineData("\"x\":100,\"y\":0", "\"x\":4100,\"y\":0")]
18    [InlineData("\"x\":100,\"y\":0", "\"x\":100,\"y\":200")]
19    [InlineData("\"nextEdgeId\":2", "\"nextEdgeId\":1")]
20    [InlineData("\"profile\":\"street\"", "\"profile\":\"street\",\"profile\":\"street\"")]
21    [InlineData("\"profile\":\"street\"", "\"profile\":\"street\",\"geometryKind\":\"bezier\"")]
22    public void InvalidLoadPreservesActiveRoadContentAndBothHistoryDirections(string original, string replacement)
23    {
24        var network = new RoadNetwork(new MapDefinition(25));
25        Build(network, new(-200, 0), new(-100, 0));
26        Build(network, new(-200, 100), new(-100, 100));
27        Assert.True(network.TryCommit(network.PlanUndo().Plan!));
28        RoadSnapshot before = network.Snapshot;
29        RoadEditHistory history = network.History;
30        byte[] saved = Save(before);
31        Assert.Equal(1, history.UndoCount);
32        Assert.Equal(1, history.RedoCount);
33
34        using (var validSource = new MemoryStream(Encoding.UTF8.GetBytes(Valid)))
35            Assert.Single(RoadCodec.Read(validSource).Edges);
36        Assert.Contains(original, Valid);
37        using var source = new MemoryStream(Encoding.UTF8.GetBytes(Valid.Replace(original, replacement)));
38
39        Assert.Throws<InvalidDataException>(() => network.PlanLoad(RoadCodec.Read(source)));
40
41        Assert.Same(before, network.Snapshot);
42        Assert.Same(history, network.History);
43        Assert.Equal(saved, Save(network.Snapshot));
44        Assert.Equal(1, network.History.UndoCount);
45        Assert.Equal(1, network.History.RedoCount);
46    }
47
48    private static void Build(RoadNetwork network, RoadPoint start, RoadPoint end)
49    {
50        RoadBuildResult result = network.PlanBuild(new(network.Snapshot.Token, start, end, RoadProfileId.Street));
51        Assert.Equal(RoadBuildStatus.Ready, result.Status);
52        Assert.True(network.TryCommit(result.Plan!));
53    }
54
55    private static byte[] Save(RoadSnapshot snapshot)
56    {
57        using var stream = new MemoryStream();
58        RoadCodec.Write(stream, snapshot);
59        return stream.ToArray();
60    }
61}

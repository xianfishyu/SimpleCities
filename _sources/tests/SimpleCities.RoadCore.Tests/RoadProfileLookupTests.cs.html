1namespace SimpleCities.RoadCore.Tests;
2
3public sealed class RoadProfileLookupTests(Xunit.Abstractions.ITestOutputHelper output)
4{
5    [Fact]
6    public void RepeatedCatalogLookupsDoNotAllocate()
7    {
8        IReadOnlyList<RoadProfile> catalog = RoadProfiles.All;
9        for (int i = 0; i < catalog.Count; i++) RoadProfiles.Get(catalog[i].Id);
10        double widths = 0;
11        long before = GC.GetAllocatedBytesForCurrentThread();
12        for (int repeat = 0; repeat < 4096; repeat++)
13            for (int i = 0; i < catalog.Count; i++)
14                widths += RoadProfiles.Get(catalog[i].Id).DisplayWidth;
15        long allocated = GC.GetAllocatedBytesForCurrentThread() - before;
16        output.WriteLine($"Catalog lookup allocated bytes: {allocated}");
17        Assert.Equal(4096 * 19d, widths);
18        Assert.Equal(0, allocated);
19    }
20
21    [Fact]
22    public void EveryCatalogIdReturnsTheOriginalProfileAndDisplayAttributes()
23    {
24        Assert.Equal(new[] { RoadProfileId.Dirt, RoadProfileId.Street, RoadProfileId.Arterial, RoadProfileId.Highway },
25            RoadProfiles.All.Select(p => p.Id));
26        foreach (RoadProfile profile in RoadProfiles.All)
27        {
28            // Reconstruct the value rather than relying on a caller's cached ID instance.
29            RoadProfile found = RoadProfiles.Get(new RoadProfileId(profile.Id.Value));
30            Assert.Same(profile, found);
31            Assert.Equal(profile.NominalWidth * 0.25, found.DisplayWidth);
32            Assert.Equal((profile.Red, profile.Green, profile.Blue), (found.Red, found.Green, found.Blue));
33        }
34    }
35
36    [Fact]
37    public void DefaultIdKeepsTheExistingNoMatchFailure()
38    {
39        InvalidOperationException expected = Assert.Throws<InvalidOperationException>(
40            () => RoadProfiles.All.Single(profile => profile.Id == default));
41        InvalidOperationException actual = Assert.Throws<InvalidOperationException>(() => RoadProfiles.Get(default));
42        Assert.Equal(expected.Message, actual.Message);
43    }

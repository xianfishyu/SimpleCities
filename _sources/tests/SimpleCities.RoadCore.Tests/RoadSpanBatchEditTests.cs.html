12    public void AdjacentDiagonalSpans_DeduplicateRegardlessOfOrderAndDoNotAllocateInteriorBoundaries(int cell)
13    {
14        var network = new RoadNetwork(new MapDefinition(cell));
15        Build(network, new(0, 0), new(10 * cell, 10 * cell));
16        RoadSnapshot before = network.Snapshot;
17        RoadEdge edge = Assert.Single(before.Edges);
18        RoadGridSpan[] spans = Enumerable.Range(0, 8).Select(i => Pick(before, edge, (i + 0.5) / 10)).ToArray();
19
20        RoadEditResult forward = network.PlanChangeProfile(spans, RoadProfileId.Highway);
21        RoadEditResult returning = network.PlanChangeProfile(spans.Reverse().Concat(spans).ToArray(), RoadProfileId.Highway);
22
23        Assert.Equal(RoadEditStatus.Ready, forward.Status);
24        Assert.Equal(RoadEditStatus.Ready, returning.Status);
25        Assert.Equal(Save(forward.Plan!.Target), Save(returning.Plan!.Target));
26        Assert.Equal(before.NextNodeId + 1, forward.Plan.Target.NextNodeId);
27        Assert.Equal(3, forward.Plan.Target.NodeCount);
28        Assert.Equal(2, forward.Plan.Target.EdgeCount);
29        Assert.Equal(8 * cell * Math.Sqrt(2), Assert.Single(forward.Plan.Target.Edges, road => road.Profile == RoadProfileId.Highway).Length, 8);
30        Assert.Equal(Save(forward.Plan.Target), ApplyChange(before, forward.Plan.ChangeSet));
31        Assert.True(network.TryCommit(forward.Plan));
32        AssertRoundTrip(network.Snapshot);
33    }

35    [Fact]
36    public void MixedProfilesAcrossAJunction_ChangeOnlyDifferentIntervalsAndAllSameOrEmptyIsSilent()
37    {
38        var network = new RoadNetwork();
39        Build(network, new(-200, 0), new(200, 0));
40        Build(network, new(0, -200), new(0, 200), RoadProfileId.Dirt);
41        RoadSnapshot before = network.Snapshot;
42        RoadGridSpan[] changed = before.Edges.Where(edge => edge.Profile == RoadProfileId.Street)
43            .Select(edge => Pick(before, edge, 0.75)).ToArray();
44        RoadEdge[] unchanged = before.Edges.Where(edge => edge.Profile == RoadProfileId.Dirt).ToArray();
45        RoadGridSpan[] selection = changed.Concat(unchanged.Select(edge => Pick(before, edge, 0.25))).ToArray();
46
47        RoadEditResult result = network.PlanChangeProfile(selection, RoadProfileId.Dirt);
48
49        Assert.Equal(RoadEditStatus.Ready, result.Status);
50        Assert.Same(before, network.Snapshot);
51        Assert.True(network.TryCommit(result.Plan!));
52        Assert.Equal(600, network.Snapshot.Edges.Where(edge => edge.Profile == RoadProfileId.Dirt).Sum(edge => edge.Length));
53        Assert.Equal(200, network.Snapshot.Edges.Where(edge => edge.Profile == RoadProfileId.Street).Sum(edge => edge.Length));
54        foreach (RoadEdge edge in unchanged)
55        {
56            RoadEdge current = network.Snapshot.FindEdge(edge.Id)!;
57            Assert.Equal(edge.Points, current.Points);
58            Assert.Equal(edge.Profile, current.Profile);
59            Assert.DoesNotContain(result.Plan!.ChangeSet.Edges, change => change.Id == edge.Id);
60        }
61        RoadSnapshot after = network.Snapshot;
62        byte[] bytes = Save(after);
63        RoadGridSpan[] same = after.Edges.Where(edge => edge.Profile == RoadProfileId.Dirt)
64            .Select(edge => Pick(after, edge, 0.25)).ToArray();
65        foreach (RoadEditResult noChange in new[] { network.PlanChangeProfile(same, RoadProfileId.Dirt), network.PlanRemove(Array.Empty<RoadGridSpan>()) })
66        {
67            Assert.Equal(RoadEditStatus.NoChange, noChange.Status);
68            Assert.Null(noChange.Plan);
69            Assert.Empty(noChange.Reason);
70        }
71        Assert.Same(after, network.Snapshot);
72        Assert.Equal(bytes, Save(network.Snapshot));
73        AssertRoundTrip(after);
74    }

265    private static byte[] Save(RoadSnapshot snapshot)
266    {
267        using var stream = new MemoryStream();
268        RoadCodec.Write(stream, snapshot);
269        return stream.ToArray();
270    }

272    // A history consumer can reconstruct domain content through public metadata and codec alone.
273    private static byte[] ApplyChange(RoadSnapshot before, RoadChangeSet change, bool reverse = false)
274    {
275        JsonObject document = JsonNode.Parse(Save(before))!.AsObject();
276        RoadChangeMetadata target = reverse ? change.Before : change.After;
277        document["contentRevision"] = target.Token.ContentRevision;
278        document["nextNodeId"] = target.NextNodeId;
279        document["nextEdgeId"] = target.NextEdgeId;
280        document["map"]!["cellSizeMetres"] = target.Map.CellSizeMetres;
281        var nodes = before.Nodes.ToDictionary(node => node.Id);
282        foreach (RoadNodeChange node in change.Nodes)
283        {
284            nodes.Remove(node.Id);
285            RoadNode? state = reverse ? node.Before : node.After;
286            if (state is not null) nodes.Add(node.Id, state);
287        }
288        var edges = before.Edges.ToDictionary(edge => edge.Id);
289        foreach (RoadEdgeChange edge in change.Edges)
290        {
291            edges.Remove(edge.Id);
292            RoadEdge? state = reverse ? edge.Before : edge.After;
293            if (state is not null) edges.Add(edge.Id, state);
294        }
295        document["nodes"] = new JsonArray(nodes.Values.OrderBy(node => node.Id.Value).Select(node => (JsonNode)new JsonObject
296        {
297            ["id"] = node.Id.Value, ["x"] = node.Position.X, ["y"] = node.Position.Y,
298        }).ToArray());
299        document["edges"] = new JsonArray(edges.Values.OrderBy(edge => edge.Id.Value).Select(edge => (JsonNode)new JsonObject
300        {
301            ["id"] = edge.Id.Value, ["startNodeId"] = edge.Start.Value, ["endNodeId"] = edge.End.Value,
302            ["profile"] = edge.Profile.Value,
303            ["points"] = new JsonArray(edge.Points.Select(point => (JsonNode)new JsonObject { ["x"] = point.X, ["y"] = point.Y }).ToArray()),
304        }).ToArray());
305        using var stream = new MemoryStream(System.Text.Encoding.UTF8.GetBytes(document.ToJsonString()));
306        var restored = new RoadNetwork();
307        Assert.True(restored.TryCommit(restored.PlanLoad(RoadCodec.Read(stream))));
308        return Save(restored.Snapshot);
309    }

311    private static void AssertRoundTrip(RoadSnapshot snapshot)
312    {
313        byte[] bytes = Save(snapshot);
314        using var stream = new MemoryStream(bytes);
315        var restored = new RoadNetwork();
316        Assert.True(restored.TryCommit(restored.PlanLoad(RoadCodec.Read(stream))));
317        Assert.Equal(bytes, Save(restored.Snapshot));

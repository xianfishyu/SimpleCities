6    public void SpanFromDifferentUncommittedTargetWithSameToken_IsRejectedBeforeAnyEdit()
7    {
8        var network = new RoadNetwork();
9        RoadSnapshot empty = network.Snapshot;
10        RoadBuildResult candidate = network.PlanBuild(new(empty.Token, new(0, 0), new(200, 0), RoadProfileId.Street));
11        RoadBuildResult committed = network.PlanBuild(new(empty.Token, new(0, 0), new(1000, 0), RoadProfileId.Street));
12        RoadSnapshot other = candidate.Plan!.Target;
13        RoadGridSpan span = RoadSpanQuery.Pick(other, new(other.Token, Assert.Single(other.Edges).Id, 0.75))!;
14        Assert.True(network.TryCommit(committed.Plan!));
15        RoadSnapshot before = network.Snapshot;
16        Assert.Equal(other.Token, before.Token);
17
18        Assert.Equal(RoadEditStatus.Rejected, network.PlanRemove(span).Status);
19        Assert.Equal(RoadEditStatus.Rejected, network.PlanChangeProfile(span, RoadProfileId.Highway).Status);
20        Assert.Equal(RoadEditStatus.Rejected, network.PlanChangeProfile(span, RoadProfileId.Street).Status);
21        Assert.Same(before, network.Snapshot);
22    }

129        }
130        RoadNode center = Assert.Single(network.Snapshot.Nodes, node => node.Position == new RoadPoint(50, 50));
131        Assert.Equal(changeProfile ? 4 : 3, RoadJunctionQuery.Read(network.Snapshot, center.Id)!.Incidences.Count);
132        if (changeProfile)
133        {
134            RoadEdge changed = Assert.Single(network.Snapshot.Edges, edge => edge.Profile == RoadProfileId.Dirt);
135            Assert.Equal(selected.Points, changed.Points);
136        }
137        else
138        {
139            Assert.DoesNotContain(network.Snapshot.Nodes, node => node.Position == new RoadPoint(0, 0));
140            Assert.DoesNotContain(network.Snapshot.Edges, edge => edge.Id == selected.Id);

142        AssertRoundTrip(network);

174    {
175        var network = new RoadNetwork();
176        Build(network, new(50, 50), new(0, 0));
177        Build(network, new(0, 0), new(100, 0));
178        Build(network, new(100, 0), new(50, 50));

180        Assert.Equal(2, wrap.Ranges.Count);
181        RoadEditResult result = changeProfile
182            ? network.PlanChangeProfile(wrap, RoadProfileId.Highway)
183            : network.PlanRemove(wrap);
184        Assert.Equal(RoadEditStatus.Ready, result.Status);
185        Assert.True(network.TryCommit(result.Plan!));
186        Assert.DoesNotContain(network.Snapshot.Nodes, node => node.Position == new RoadPoint(50, 50));
187        RoadEdge straight = Assert.Single(network.Snapshot.Edges, edge => edge.Profile == RoadProfileId.Street);
188        Assert.Equal(100, straight.Length);
189        Assert.Equal(2, straight.Points.Count);
190        if (changeProfile)
191        {
192            RoadEdge changed = Assert.Single(network.Snapshot.Edges, edge => edge.Profile == RoadProfileId.Highway);
193            Assert.Contains(new RoadPoint(50, 50), changed.Points);
194            Assert.Equal(100 * Math.Sqrt(2), changed.Length, 8);
195            RoadEditResult back = network.PlanChangeProfile(Pick(network, changed), RoadProfileId.Street);
196            Assert.Equal(RoadEditStatus.Ready, back.Status);
197            Assert.True(network.TryCommit(back.Plan!));
198            RoadEdge loop = Assert.Single(network.Snapshot.Edges);
199            Assert.Equal(loop.Start, loop.End);
200            Assert.Equal(1, network.Snapshot.NodeCount);
201        }
202        else Assert.Single(network.Snapshot.Edges);
203        AssertRoundTrip(network);
204    }

206    [Theory]
207    [InlineData(false)]
208    [InlineData(true)]
209    public void ForeignStaleCancelledAndSupersededEdits_LeaveTheCurrentSnapshotUntouched(bool changeProfile)
210    {
211        var network = new RoadNetwork();
212        Build(network, new(0, 0), new(1000, 0));
213        RoadGridSpan span = Pick(network, Assert.Single(network.Snapshot.Edges), 0.45);
214        RoadEditResult Plan(RoadNetwork target, CancellationToken token = default) => changeProfile
215            ? target.PlanChangeProfile(span, RoadProfileId.Highway, token)
216            : target.PlanRemove(span, token);
217        var foreign = new RoadNetwork();
218        RoadSnapshot foreignBefore = foreign.Snapshot;
219        Assert.Equal(RoadEditStatus.Rejected, Plan(foreign).Status);
220        Assert.Same(foreignBefore, foreign.Snapshot);
221        RoadSnapshot before = network.Snapshot;
222        Assert.Throws<OperationCanceledException>(() => Plan(network, new CancellationToken(true)));
223        Assert.Same(before, network.Snapshot);
224        RoadEditResult planned = Plan(network);
225        Assert.Equal(RoadEditStatus.Ready, planned.Status);
226        Build(network, new(0, 100), new(100, 100));
227        RoadSnapshot current = network.Snapshot;
228        Assert.Equal(RoadEditStatus.Rejected, Plan(network).Status);
229        Assert.False(network.TryCommit(planned.Plan!));
230        Assert.Same(current, network.Snapshot);
231    }

233    private static void AssertRoundTrip(RoadNetwork network)
234    {
235        byte[] bytes = Save(network.Snapshot);
236        using var stream = new MemoryStream(bytes);
237        var restored = new RoadNetwork();
238        Assert.True(restored.TryCommit(restored.PlanLoad(RoadCodec.Read(stream))));
239        Assert.Equal(bytes, Save(restored.Snapshot));

242    private static void Build(RoadNetwork network, RoadPoint start, RoadPoint end, RoadProfileId? profile = null)
243    {

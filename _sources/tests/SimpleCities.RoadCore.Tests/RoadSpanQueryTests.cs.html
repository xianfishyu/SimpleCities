10    public void GridIntervalsFollowDirectionAndCellSize(int cell)
11    {
12        foreach (RoadPoint end in new RoadPoint[] { new(-10 * cell, 0), new(0, -10 * cell), new(10 * cell, 10 * cell) })
13        {
14            var network = new RoadNetwork(new MapDefinition(cell));
15            Build(network, new(0, 0), end);
16            RoadSnapshot snapshot = network.Snapshot;
17            RoadEdge edge = Assert.Single(snapshot.Edges);
18            RoadGridSpan span = RoadSpanQuery.Pick(snapshot, new(snapshot.Token, edge.Id, 0.45))!;
19            Assert.Equal(new RoadPoint(end.X * 0.4, end.Y * 0.4), span.Points[0]);
20            Assert.Equal(new RoadPoint(end.X * 0.5, end.Y * 0.5), span.Points[^1]);
21            Assert.Equal(0.4, span.Ranges[0].StartParameter, 12);
22            Assert.Equal(0.5, span.Ranges[0].EndParameter, 12);
23        }
24    }

41    [Fact]
42    public void TrueCellCenterJunction_TruncatesEachHalfAndRejectsTheAmbiguousCenter()
43    {
44        var network = new RoadNetwork();
45        Build(network, new(0, 0), new(100, 100));
46        Build(network, new(0, 100), new(100, 0));
47        RoadSnapshot snapshot = network.Snapshot;
48        RoadNode center = Assert.Single(snapshot.Nodes, node => node.Position == new RoadPoint(50, 50));
49        var keys = new HashSet<RoadSpanKey>();
50        foreach (RoadEdge edge in snapshot.Edges)
51        {
52            double endpoint = edge.Start == center.Id ? 0 : 1;
53            Assert.Null(RoadSpanQuery.Pick(snapshot, new(snapshot.Token, edge.Id, endpoint)));
54            RoadGridSpan half = RoadSpanQuery.Pick(snapshot, new(snapshot.Token, edge.Id, 0.5))!;
55            Assert.Equal(2, half.Points.Count);
56            Assert.Contains(center.Position, half.Points);
57            Assert.Equal(new[] { new RoadSpanRange(0, 1) }, half.Ranges);
58            Assert.True(keys.Add(half.Key));
59            Assert.NotNull(RoadSpanQuery.Pick(snapshot, new(snapshot.Token, edge.Id, 1 - endpoint)));
60        }
61    }

79    [Fact]
80    public void SyntheticCellCenterLoopSeam_IsTransparentAndHasOneStableWrapKey()
81    {
82        var network = new RoadNetwork();
83        Build(network, new(50, 50), new(0, 0));
84        Build(network, new(0, 0), new(100, 0));
85        Build(network, new(100, 0), new(50, 50));
86        RoadSnapshot snapshot = network.Snapshot;
87        RoadEdge edge = Assert.Single(snapshot.Edges);
88        Assert.Equal(new RoadPoint(50, 50), edge.Points[0]);
89        RoadGridSpan first = RoadSpanQuery.Pick(snapshot, new(snapshot.Token, edge.Id, 0.01))!;
90        RoadGridSpan last = RoadSpanQuery.Pick(snapshot, new(snapshot.Token, edge.Id, 0.99))!;
91        Assert.Equal(first.Key, last.Key);
92        Assert.Equal(first.Key, RoadSpanQuery.Pick(snapshot, new(snapshot.Token, edge.Id, 0))!.Key);
93        Assert.Equal(first.Key, RoadSpanQuery.Pick(snapshot, new(snapshot.Token, edge.Id, 1))!.Key);
94        Assert.Equal(2, first.Ranges.Count);
95        Assert.Equal(1, first.Ranges[0].EndParameter);
96        Assert.Equal(0, first.Ranges[1].StartParameter);
97        Assert.Equal(new RoadPoint(50, 50), first.Points[1]);
98        Assert.True(new HashSet<RoadPoint> { new(0, 0), new(100, 0) }.SetEquals(
99            first.Points.Where(point => point != new RoadPoint(50, 50))));
100    }

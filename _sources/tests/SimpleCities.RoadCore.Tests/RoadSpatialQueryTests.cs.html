5    [Fact]
6    public void SingleBucketMissDoesNotAllocateCandidateSizedDeduplicationStorage()
7    {
8        var fragments = Enumerable.Range(0, 2048).Select(i => new SpatialFragment<int>(i, new(1, 1, 2, 2)));
9        var index = new SpatialQueryIndex<int>(default, fragments);
10        var point = new SpatialBounds(90, 90, 90, 90);
11        for (int i = 0; i < 3; i++) index.QueryBounds(point);
12        long before = GC.GetAllocatedBytesForCurrentThread();
13        var result = index.QueryBounds(point);
14        long allocated = GC.GetAllocatedBytesForCurrentThread() - before;
15        Assert.Equal(SpatialQueryStatus.Ready, result.Status);
16        Assert.Empty(result.Results);
17        Assert.Equal(1, result.Metrics.BucketsVisited);
18        Assert.Equal(2048, result.Metrics.FragmentCandidates);
19        Assert.InRange(allocated, 0, 1024);
20    }

39        Assert.Equal(1, rejected.Metrics.BucketsVisited);

43    public void SingleBucketPointMatchesSegmentReferenceAcrossBoundariesAndBudgets()
44    {
45        var fragments = new List<SpatialFragment<int>>();
46        for (int x = -2; x <= 2; x++)
47            for (int y = -2; y <= 2; y++)
48                fragments.Add(new((x + y) % 3, new(x * 100 - 1, y * 100 - 1, x * 100 + 1, y * 100 + 1)));
49        var index = new SpatialQueryIndex<int>(new(Guid.NewGuid(), Guid.NewGuid(), 1, 1), fragments);
50        foreach (double x in new[] { -200, -100, -0.01, 0, 0.01, 99.9, 100, 200 })
51            foreach (double y in new[] { -200, -100, -0.01, 0, 0.01, 99.9, 100, 200 })
52                foreach (int candidates in new[] { 1, 64 })
53                {
54                    var budget = new SpatialQueryBudget(1, candidates, 1);
55                    var fast = index.QueryBounds(new(x, y, x, y), budget);
56                    var reference = index.QuerySegment(new(x, y), new(x, y), budget);
57                    Assert.Equal(reference.Status, fast.Status);
58                    Assert.Equal(reference.Source, fast.Source);
59                    Assert.Equal(reference.Metrics, fast.Metrics);
60                    Assert.Equal(reference.Reason, fast.Reason);
61                    Assert.Equal(reference.Results, fast.Results);
62                }
63    }

139    {
140        var network = new RoadNetwork();

142        var index = new RoadSpatialQueryIndex(network.Snapshot);
143        foreach (SpatialQueryBudget limits in new[]
144        {

146            new SpatialQueryBudget(4096, 1, 16384),
147            new SpatialQueryBudget(4096, 4096, 1),

149        {

151            Assert.Equal(SpatialQueryStatus.BudgetExceeded, rejected.Status);

158        {

160            index.QueryNearby(new(50, 0), double.PositiveInfinity),
161            index.QueryNearby(new(50, 0), -1),
162            index.QueryNearby(new(50, 0), 1, new SpatialQueryBudget(0, 10, 10)),
163            index.QueryBounds(new(10, 0, -10, 0)),
164            index.QuerySegment(new(0, 0), new(double.NaN, 0)),
165        })
166        {
167            Assert.Equal(SpatialQueryStatus.InvalidParameters, invalid.Status);
168            Assert.Empty(invalid.Results);
169            Assert.NotEmpty(invalid.Reason);

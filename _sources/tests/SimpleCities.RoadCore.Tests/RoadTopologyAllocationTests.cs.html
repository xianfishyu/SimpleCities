17    [Fact]
18    public void RepeatedLocalBuildDoesNotRematerializeFullValidationObjects()
19    {
20        RoadNetwork network = RoadBaselineFixtures.CreateMixed(25);
21        RoadSnapshot source = network.Snapshot;
22        var request = new RoadBuildRequest(source.Token, new(3000, 3000), new(3025, 3000), RoadProfileId.Street);
23        _ = network.PlanBuild(request);
24        long before = GC.GetAllocatedBytesForCurrentThread();
25        RoadPlan plan = network.PlanBuild(request).Plan!;
26        long allocated = GC.GetAllocatedBytesForCurrentThread() - before;
27        output.WriteLine($"Repeated local build validation: {allocated} B");
28        Assert.True(allocated < 128 * 1024, $"Full validation allocated {allocated} B.");
29        Assert.Same(source, network.Snapshot);
30        Assert.Equal(source.EdgeCount + 1, plan.Target.EdgeCount);
31        Assert.Single(plan.ChangeSet.Edges);
32    }
33
34    [Fact]
35    public void RejectedIntersectionsCannotContaminateLaterSmallOrLargeValidation()
36    {
37        RoadNetwork network = RoadBaselineFixtures.CreateMixed(25);
38        RoadSnapshot before = network.Snapshot;
39        RoadEditHistory history = network.History;
40        for (int i = 0; i < 24; i++)
41        {
42            RejectCrossing();
43            var small = new RoadNetwork();
44            RoadBaselineFixtures.Build(small, new(0, 0), new(100, 0));
45            Assert.Single(small.Snapshot.Edges);
46            RoadPlan plan = network.PlanBuild(new(before.Token, new(3000, 3000), new(3025, 3000), RoadProfileId.Street)).Plan!;
47            Assert.Equal(before.EdgeCount + 1, plan.Target.EdgeCount);
48            Assert.Same(before, network.Snapshot);
49            Assert.Same(history, network.History);
50        }
51    }
52
53    [Fact]
54    public async Task ConcurrentValidAndInvalidPlanningKeepsScratchOwnershipSeparate()
55    {
56        var source = new RoadNetwork();
57        RoadBaselineFixtures.Build(source, new(0, 0), new(200, 0));
58        RoadSnapshot before = source.Snapshot;
59        await Task.WhenAll(Enumerable.Range(0, 8).Select(index => Task.Run(() =>
60        {
61            for (int repeat = 0; repeat < 32; repeat++)
62            {
63                RejectCrossing();
64                var start = new RoadPoint(0, 200 + index * 100);
65                var end = new RoadPoint(200, start.Y);
66                RoadPlan plan = source.PlanBuild(new(before.Token, start, end, RoadProfileId.Street)).Plan!;
67                Assert.Equal(2, plan.Target.EdgeCount);
68                RoadEdge added = Assert.Single(plan.ChangeSet.Edges).After!;
69                Assert.Equal(new[] { start, end }, added.Points);
70                Assert.Same(before, source.Snapshot);
71                Assert.Throws<OperationCanceledException>(() => source.PlanBuild(
72                    new(before.Token, start, end, RoadProfileId.Street), new(true)));
73            }
74        })));
75    }
76
77    private static void RejectCrossing()
78    {
79        using var input = new MemoryStream(Encoding.UTF8.GetBytes(CrossingWithoutJunction));
80        InvalidDataException error = Assert.Throws<InvalidDataException>(() => RoadCodec.Read(input));
81        Assert.Equal("道路交叉或接入位置缺少共同结构节点", error.Message);
82    }

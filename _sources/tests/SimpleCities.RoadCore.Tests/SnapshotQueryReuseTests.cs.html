1using System.Runtime.CompilerServices;
2using SimpleCities.RoadCore.Performance;

4namespace SimpleCities.RoadCore.Tests;

6public sealed class SnapshotQueryReuseTests(Xunit.Abstractions.ITestOutputHelper output)

8    [Theory]
9    [InlineData(false)]
10    [InlineData(true)]
11    public void FallbackUndoReusesUnchangedGeometryWithoutRetainingOldSnapshot(bool wholeMap)
12    {
13        (RoadNetwork network, WeakReference old) = CreateEdited(wholeMap);
14        GC.Collect(); GC.WaitForPendingFinalizers(); GC.Collect();
15        Assert.False(old.IsAlive);
16        RoadSnapshot current = network.Snapshot;
17        RoadEditHistory history = network.History;
18        _ = network.PlanUndo();
19
20        long before = GC.GetAllocatedBytesForCurrentThread();
21        RoadPlan undo = network.PlanUndo().Plan!;
22        long allocated = GC.GetAllocatedBytesForCurrentThread() - before;
23
24        output.WriteLine($"Fallback undo wholeMap={wholeMap}: {allocated} B");
25        Assert.True(allocated < 100 * 1024, $"Fallback undo allocated {allocated} B.");
26        Assert.Same(current, network.Snapshot);
27        Assert.Same(history, network.History);
28        AssertMatchesFreshLoad(undo.Target);
29        AssertMatchesFreshLoad(current);
30        Assert.True(network.TryCommit(undo));
31        Assert.True(network.TryCommit(network.PlanRedo().Plan!));
32        AssertMatchesFreshLoad(network.Snapshot);
33    }

35    [Theory]
36    [InlineData(25)]
37    [InlineData(50)]
38    [InlineData(100)]
39    [InlineData(200)]
40    public void EditsKeepQueryResultsEquivalentToFreshLoadAndOldSnapshotsIndependent(int cell)
41    {
42        var network = new RoadNetwork(new MapDefinition(cell));
43        var observed = new List<RoadSnapshot>();
44        Build(new(0, 0), new(4 * cell, 0));
45        Build(new(4 * cell, 0), new(4 * cell, 4 * cell));
46        Build(new(4 * cell, 4 * cell), new(0, 4 * cell));
47        Build(new(0, 4 * cell), new(0, 0));
48        Assert.Equal(2, RoadJunctionQuery.Read(network.Snapshot, network.Snapshot.Nodes[0].Id)!.Incidences.Count);
49        Build(new(4 * cell, 4 * cell), new(6 * cell, 4 * cell));
50        Publish(network.PlanChangeProfile(Span(), RoadProfileId.Highway).Plan!);
51        Publish(network.PlanRemove(Span()).Plan!);
52        Publish(network.PlanUndo().Plan!);
53        Publish(network.PlanUndo().Plan!);
54        Publish(network.PlanRedo().Plan!);
55        Build(new(0, 6 * cell), new(6 * cell, 6 * cell));
56        Assert.Equal(0, network.History.RedoCount);
57        Publish(network.PlanRemove(RoadBaselineFixtures.Spans(network.Snapshot)).Plan!);
58        Assert.Empty(network.Snapshot.Nodes);
59        Assert.Empty(network.Snapshot.Edges);
60        Publish(network.PlanUndo().Plan!);
61        foreach (RoadSnapshot old in observed) AssertMatchesFreshLoad(old);
62
63        RoadGridSpan Span() => RoadSpanQuery.Pick(network.Snapshot,
64            new(network.Snapshot.Token, network.Snapshot.Edges[0].Id, .25))!;
65        void Build(RoadPoint from, RoadPoint to) => Publish(network.PlanBuild(
66            new(network.Snapshot.Token, from, to, RoadProfileId.Street)).Plan!);
67        void Publish(RoadPlan plan)
68        {
69            RoadSnapshot before = network.Snapshot;
70            RoadEditHistory history = network.History;
71            AssertMatchesFreshLoad(plan.Target);
72            Assert.Same(before, network.Snapshot);
73            Assert.Same(history, network.History);
74            Assert.True(network.TryCommit(plan));
75            foreach (RoadNode removed in before.Nodes.Where(node => !plan.Target.Nodes.Contains(node)))
76                Assert.Null(RoadJunctionQuery.Read(plan.Target, removed.Id));
77            foreach (RoadEdge removed in before.Edges.Where(edge => plan.Target.FindEdge(edge.Id) is null))
78                Assert.Null(plan.Target.Resolve(new(plan.Target.Token, removed.Id, .5)));
79            observed.Add(before);
80        }
81    }

83    [MethodImpl(MethodImplOptions.NoInlining)]
84    private static (RoadNetwork, WeakReference) CreateEdited(bool wholeMap)
85    {
86        RoadNetwork network = RoadBaselineFixtures.CreateMixed(25);
87        RoadSnapshot before = network.Snapshot;
88        RoadGridSpan[] spans = wholeMap ? RoadBaselineFixtures.Spans(before)
89            : [RoadSpanQuery.Pick(before, new(before.Token, before.Edges[0].Id, .5))!];
90        Assert.True(network.TryCommit(network.PlanChangeProfile(spans, RoadProfileId.Highway).Plan!));
91        return (network, new(before));
92    }
93
94    private static void AssertMatchesFreshLoad(RoadSnapshot actual)
95    {
96        using var bytes = new MemoryStream();
97        RoadCodec.Write(bytes, actual);
98        bytes.Position = 0;
99        RoadSnapshot expected = new RoadNetwork().PlanLoad(RoadCodec.Read(bytes)).Target;
100        foreach (RoadEdge edge in actual.Edges)
101        {
102            foreach (double parameter in new[] { 0, .1, .25, .5, .75, .9, 1 })
103            {
104                Assert.Equal(expected.Resolve(new(expected.Token, edge.Id, parameter)),
105                    actual.Resolve(new(actual.Token, edge.Id, parameter)));
106                RoadGridSpan? a = RoadSpanQuery.Pick(actual, new(actual.Token, edge.Id, parameter));
107                RoadGridSpan? b = RoadSpanQuery.Pick(expected, new(expected.Token, edge.Id, parameter));
108                Assert.Equal(b?.Ranges, a?.Ranges);
109                Assert.Equal(b?.Points, a?.Points);
110                if (a is not null) Assert.Equal(actual.Token, a.Source);
111            }
112            Assert.Null(actual.Resolve(new(expected.Token, edge.Id, .5)));
113        }
114        foreach (RoadNode node in actual.Nodes)
115        {
116            RoadJunctionReadModel a = RoadJunctionQuery.Read(actual, node.Id)!;
117            RoadJunctionReadModel b = RoadJunctionQuery.Read(expected, node.Id)!;
118            Assert.Equal(actual.Token, a.Source);
119            Assert.Equal(b.Node, a.Node);
120            Assert.Equal(b.Incidences, a.Incidences);
121            Assert.Equal(b.Turns, a.Turns);
122        }
123    }

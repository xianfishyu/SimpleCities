1using Godot;
2using System;
3using System.Threading;
4
5/// <summary>Debug-only failures at display preflight/publish and a controllable retry worker boundary.</summary>
6public partial class RoadDisplayFailureProbe : RefCounted
7{
8    private readonly ManualResetEventSlim _release = new(false);
9    private RoadMapScene? _scene;
10    private RoadMapView? _view;
11    private Action? _preflight;
12    private Action? _publish;
13    private Action? _draw;
14    private Action? _retry;
15    private Action<SimpleCities.RoadCore.RoadStateToken, int>? _partialDraw;
16    private int _faultCalls;
17    private int _retryCalls;
18    private int _completed;
19    private int _workerThread;
20    private int _mainThread;
21    private bool _disposed;
22
23    public int FaultCalls => Volatile.Read(ref _faultCalls);
24    public int RetryCalls => Volatile.Read(ref _retryCalls);
25    public bool Started => RetryCalls != 0;
26    public bool Completed => Volatile.Read(ref _completed) != 0;
27    public bool UsedBackgroundThread => Started && Volatile.Read(ref _workerThread) != _mainThread;
28
29    public void Install(RoadMapScene scene, string stage, bool persistent = false)
30    {
31        if (stage is not "preflight" and not "publish" and not "draw")
32            throw new ArgumentOutOfRangeException(nameof(stage));
33        _scene = scene;
34        RoadMapView view = scene.GetNode<RoadMapView>("View");
35        _view = view;
36        string baselineToken = view.Presented?.Token.ToString() ?? "";
37        Action fail = () =>
38        {
39            // Only fail a new candidate's draw; the current display and its fallback remain usable.
40            if (stage == "draw" && view.Presented?.Token.ToString() == baselineToken) return;
41            int call = Interlocked.Increment(ref _faultCalls);
42            if (persistent || call == 1)
43                throw new InvalidOperationException($"Injected V4 display {stage} failure.");
44        };
45        if (stage == "preflight") view.BeforeDisplayPreflight = _preflight = fail;
46        else if (stage == "publish") view.BeforeDisplayPublish = _publish = fail;
47        else view.BeforeDisplayDraw = _draw = fail;
48    }
49
50    public void InstallRetryGate(RoadMapScene scene, bool failAfterRelease = false)
51    {
52        _scene = scene;
53        _mainThread = System.Environment.CurrentManagedThreadId;
54        scene.BeforeDisplayRetryWork = _retry = () =>
55        {
56            Volatile.Write(ref _workerThread, System.Environment.CurrentManagedThreadId);
57            Interlocked.Increment(ref _retryCalls);
58            _release.Wait();
59            Volatile.Write(ref _completed, 1);
60            if (failAfterRelease)
61                throw new InvalidOperationException("Injected V4 display retry worker failure.");
62        };
63    }
64
65    public void Release() => _release.Set();
66
67    public void InstallPartialDraw(RoadMapScene scene)
68    {
69        _scene = scene;
70        _view = scene.GetNode<RoadMapView>("View");
71        var baseline = _view.Presented!.Token;
72        RoadDisplay.AfterMeshDraw = _partialDraw = (source, ordinal) =>
73        {
74            if (source != baseline && ordinal == 1 && Interlocked.Increment(ref _faultCalls) == 1)
75                throw new InvalidOperationException("Injected V4 failure after the second mesh draw.");
76        };
77    }
78
79    public void RemoveFault()
80    {
81        if (ReferenceEquals(RoadDisplay.AfterMeshDraw, _partialDraw)) RoadDisplay.AfterMeshDraw = null;
82        _partialDraw = null;
83        if (_view is not null && GodotObject.IsInstanceValid(_view))
84        {
85            if (ReferenceEquals(_view.BeforeDisplayPreflight, _preflight)) _view.BeforeDisplayPreflight = null;
86            if (ReferenceEquals(_view.BeforeDisplayPublish, _publish)) _view.BeforeDisplayPublish = null;
87            if (ReferenceEquals(_view.BeforeDisplayDraw, _draw)) _view.BeforeDisplayDraw = null;
88        }
89        _preflight = null;
90        _publish = null;
91        _draw = null;
92    }
93
94    public void Cleanup()
95    {
96        if (_disposed) return;
97        Release();
98        if (Started && !Completed)
99            throw new InvalidOperationException("Await the released retry worker before disposing its gate.");
100        RemoveFault();
101        if (_scene is not null && GodotObject.IsInstanceValid(_scene) && ReferenceEquals(_scene.BeforeDisplayRetryWork, _retry))
102            _scene.BeforeDisplayRetryWork = null;
103        _retry = null;
104        _release.Dispose();
105        _disposed = true;
106    }
107}

13    public Godot.Collections.Dictionary MeasurePointerAllocation(RoadMapScene scene, int mode)
14    {
15        const int count = 1000;
16        if (!scene.SetToolMode(mode)) throw new InvalidOperationException("Pointer fixture cannot select tool.");
17        using var motion = new InputEventMouseMotion();
18        void Move(int i)
19        {
20            motion.Position = scene.GetCanvasTransform() * new Vector2(i % 2 == 0 ? 35 : 40, 0);
21            scene._Input(motion);
22            scene._UnhandledInput(motion);
23        }
24        for (int i = 0; i < 20; i++) Move(i);
25        string source = scene.StateToken;
26        long before = GC.GetAllocatedBytesForCurrentThread();
27        for (int i = 0; i < count; i++) Move(i);
28        long allocated = GC.GetAllocatedBytesForCurrentThread() - before;
29        return new()
30        {
31            ["mode"] = mode, ["count"] = count, ["allocatedBytes"] = allocated,
32            ["unchanged"] = scene.StateToken == source,
33            ["currentQuerySource"] = scene.GetQueryState()["sourceToken"].AsString() == source,
34            ["hasHover"] = scene.GetSelectionState()["hasHover"],
35            ["status"] = scene.GetNode<Label>("HUD/Panel/Margin/Controls/Status").Text,
36        };
37    }

39    public Godot.Collections.Dictionary MeasureGeometryAllocation(string fixturePath)
40    {
41        using var bytes = System.IO.File.OpenRead(ProjectSettings.GlobalizePath(fixturePath));
42        RoadSnapshot snapshot = new RoadNetwork().PlanLoad(RoadCodec.Read(bytes)).Target;
43        RoadSurfaceData surface = RoadPresentation.Prepare(snapshot.Nodes, snapshot.Edges)
44            ?? throw new InvalidOperationException("Missing allocation fixture surface.");
45        // Warm the same public preparation path, then measure only its managed work.
46        RoadDisplay.PrepareGeometry(snapshot, surface);
47        var measured = Task.Run(() =>
48        {
49            long before = GC.GetAllocatedBytesForCurrentThread();
50            var geometry = RoadDisplay.PrepareGeometry(snapshot, surface);
51            return (Geometry: geometry, Bytes: GC.GetAllocatedBytesForCurrentThread() - before);
52        }).GetAwaiter().GetResult();
53        using var display = measured.Geometry.Upload();
54        var allVertices = new List<Vector3>();
55        var allColors = new List<Color>();
56        var allIndices = new List<int>();
57        foreach (ArrayMesh mesh in display.Meshes)
58        {
59            using var arrays = mesh.SurfaceGetArrays(0);
60            int offset = allVertices.Count;
61            allVertices.AddRange(arrays[(int)Mesh.ArrayType.Vertex].AsVector3Array());
62            allColors.AddRange(arrays[(int)Mesh.ArrayType.Color].AsColorArray());
63            allIndices.AddRange(arrays[(int)Mesh.ArrayType.Index].AsInt32Array().Select(index => index + offset));
64        }
65        Vector3[] vertices = allVertices.ToArray();
66        Color[] colors = allColors.ToArray();
67        int[] indices = allIndices.ToArray();
68        using var hash = System.Security.Cryptography.IncrementalHash.CreateHash(System.Security.Cryptography.HashAlgorithmName.SHA256);
69        hash.AppendData(System.Runtime.InteropServices.MemoryMarshal.AsBytes(vertices.AsSpan()));
70        hash.AppendData(System.Runtime.InteropServices.MemoryMarshal.AsBytes(colors.AsSpan()));
71        hash.AppendData(System.Runtime.InteropServices.MemoryMarshal.AsBytes(indices.AsSpan()));
72        return new()
73        {
74            ["allocatedBytes"] = measured.Bytes,
75            ["edges"] = snapshot.EdgeCount,
76            ["pieces"] = surface.Pieces.Count,
77            ["vertices"] = vertices.Length,
78            ["indices"] = indices.Length,
79            ["meshBytes"] = vertices.Length * 28L + indices.Length * 4L,
80            ["sha256"] = Convert.ToHexString(hash.GetHashAndReset()),
81        };
82    }

109            using var display = geometry.Upload();

111            return ReferenceEquals(display.Snapshot, snapshot) && display.Meshes.Single().SurfaceGetArrayLen(0) == 4 &&
112                hit.Location.Source == snapshot.Token && hit.Location.Edge == ribbon.Edge.Id;

114        foreach (Vector2 direction in new Vector2[]
115        {
116            new(1, 0), new(1, 1), new(0, 1), new(-1, 1),
117            new(-1, 0), new(-1, -1), new(0, -1), new(1, -1),

121            {
122                var target = new RoadNetwork();
123                Vector2 start = new(-300, -300), end = start + direction * 600;
124                Build(target, new(start.X, start.Y), new(end.X, end.Y), RoadProfileId.Highway);
125                RoadSnapshot source = target.Snapshot;
126                RoadEdge edge = source.Edges.Single();
127                using var display = RoadDisplay.Prepare(source, RoadPresentation.Prepare(source.Nodes, source.Edges));
128                Vector2 normal = new Vector2(-direction.Y, direction.X).Normalized();
129                // Exercise real picking across multiple spatial buckets in every direction,
130                // including both sides of the 8-unit ribbon and its endpoint caps.
131                for (int sample = 0; sample <= 32; sample++)
132                {
133                    float fraction = sample / 32f;
134                    Vector2 center = start.Lerp(end, fraction);
135                    foreach (float offset in new[] { -3.5f, 0, 3.5f })
136                    {
137                        var hits = display.QueryHit(center + normal * offset);
138                        if (hits.Status != SpatialQueryStatus.Ready || hits.Results.Count != 1 ||
139                            hits.Results[0].Location.Source != source.Token || hits.Results[0].Location.Edge != edge.Id ||
140                            Math.Abs(hits.Results[0].Location.Parameter - fraction) > 0.00001) return false;
141                    }
142                    foreach (float offset in new[] { -4.5f, 4.5f })
143                    {
144                        var miss = display.QueryHit(center + normal * offset);
145                        if (miss.Status != SpatialQueryStatus.Ready || miss.Results.Count != 0) return false;
146                    }
147                }
148                return true;
149            });
150        }

291        void Check(string name, Func<bool> assertion)
292        {
293            string error = "";
294            bool passed;
295            try { passed = assertion(); }
296            catch (Exception exception) { passed = false; error = exception.Message; }
297            results.Add(new Godot.Collections.Dictionary { ["name"] = name, ["passed"] = passed, ["error"] = error });

302        RoadPoint[] corners, NodeId? node = null) =>

310    private static void Build(RoadNetwork network, RoadPoint start, RoadPoint end, RoadProfileId profile)
311    {
312        RoadBuildResult result = network.PlanBuild(new(network.Snapshot.Token, start, end, profile));
313        if (result.Status != RoadBuildStatus.Ready || result.Plan is null || !network.TryCommit(result.Plan))
314            throw new InvalidOperationException($"Preflight fixture could not build: {result.Reason}");
315    }

317    private static RoadSaveParticipant.Prepared ReadLoad(IStreamingLoadReader reader, RoadSnapshot snapshot)
318    {
319        using var bytes = new System.IO.MemoryStream();
320        RoadCodec.Write(bytes, snapshot);
321        bytes.Position = 0;
322        return (RoadSaveParticipant.Prepared)Task.Run(() => reader.PrepareLoad(bytes)).GetAwaiter().GetResult();
323    }

1using Godot;
2using System;
3using System.Collections.Generic;
4using System.Diagnostics;
5using System.Linq;
6using System.Text.Json;
7
8/// <summary>Debug-only bounded frame sampler. No dictionary/token polling or per-frame managed containers.</summary>
9public partial class RoadFrameAllocationProbe : Node
10{
11    private readonly record struct Sample(double AtMs, double FrameMs, long MainBytes, int Gen0, int Gen1, int Gen2, bool Focused);
12    private sealed record Window(string Name, int Start, int Count, double MeanMainBytes, int Gen0, int Gen1, int Gen2,
13        double P95Ms, double P99Ms, double MaxMs, int OverBudget, int OverBudgetWithGc, int FocusedFrames);
14    private readonly Sample[] _samples = new Sample[262144];
15    private readonly List<Window> _windows = new(8);
16    private bool _capturing;
17    private bool _overflow;
18    private bool _wrongThread;
19    private int _count, _start, _thread, _lastGen0, _lastGen1, _lastGen2;
20    private long _lastTicks, _lastBytes, _origin;
21    private string _name = "";
22
23    public override void _Ready() => RenderingServer.FramePostDraw += Capture;
24    public override void _ExitTree() => RenderingServer.FramePostDraw -= Capture;
25
26    public void BeginWindow(string name)
27    {
28        if (_capturing) throw new InvalidOperationException("A probe window is already active.");
29        _name = name;
30        _start = _count;
31        _thread = System.Environment.CurrentManagedThreadId;
32        _origin = _lastTicks = Stopwatch.GetTimestamp();
33        _lastGen0 = GC.CollectionCount(0);
34        _lastGen1 = GC.CollectionCount(1);
35        _lastGen2 = GC.CollectionCount(2);
36        _lastBytes = GC.GetAllocatedBytesForCurrentThread();
37        _capturing = true;
38    }
39
40    private void Capture()
41    {
42        if (!_capturing) return;
43        if (_count == _samples.Length) { _overflow = true; _capturing = false; return; }
44        long now = Stopwatch.GetTimestamp();
45        long allocated = GC.GetAllocatedBytesForCurrentThread();
46        int gen0 = GC.CollectionCount(0), gen1 = GC.CollectionCount(1), gen2 = GC.CollectionCount(2);
47        _wrongThread |= System.Environment.CurrentManagedThreadId != _thread;
48        _samples[_count++] = new Sample(Stopwatch.GetElapsedTime(_origin, now).TotalMilliseconds,
49            Stopwatch.GetElapsedTime(_lastTicks, now).TotalMilliseconds, allocated - _lastBytes,
50            gen0 - _lastGen0, gen1 - _lastGen1, gen2 - _lastGen2, DisplayServer.WindowIsFocused());
51        _lastTicks = now;
52        _lastBytes = allocated;
53        _lastGen0 = gen0;
54        _lastGen1 = gen1;
55        _lastGen2 = gen2;
56    }
57
58    public double EndWindow()
59    {
60        _capturing = false;
61        int count = _count - _start;
62        if (count == 0) throw new InvalidOperationException("No rendered frames were sampled.");
63        long bytes = 0;
64        int gen0 = 0, gen1 = 0, gen2 = 0, over = 0, overWithGc = 0, focused = 0;
65        var times = new double[count];
66        for (int i = 0; i < count; i++)
67        {
68            Sample sample = _samples[_start + i];
69            times[i] = sample.FrameMs;
70            bytes += sample.MainBytes;
71            gen0 += sample.Gen0; gen1 += sample.Gen1; gen2 += sample.Gen2;
72            if (sample.Focused) focused++;
73            if (sample.FrameMs > 1000d / 144)
74            {
75                over++;
76                if (sample.Gen0 + sample.Gen1 + sample.Gen2 > 0) overWithGc++;
77            }
78        }
79        Array.Sort(times);
80        double mean = (double)bytes / count;
81        _windows.Add(new Window(_name, _start, count, mean, gen0, gen1, gen2,
82            times[(int)Math.Ceiling(count * .95) - 1], times[(int)Math.Ceiling(count * .99) - 1], times[^1], over, overWithGc, focused));
83        return mean;
84    }
85
86    public bool Save(string path)
87    {
88        if (_capturing) throw new InvalidOperationException("Stop sampling before writing its report.");
89        bool valid = !_overflow && !_wrongThread && _windows.TrueForAll(window => window.FocusedFrames == window.Count);
90        System.IO.File.WriteAllText(ProjectSettings.GlobalizePath(path), JsonSerializer.Serialize(new
91        {
92            valid, overflow = _overflow, wrongThread = _wrongThread, windows = _windows,
93            samples = _samples.Take(_count).ToArray(),
94            method = "Main-thread allocated bytes and GC collection-count deltas between actual post-draw samples. Fixed preallocated sample buffer. GC coincidence is not proof for every long frame.",
95        }));
96        return valid;
97    }
98}

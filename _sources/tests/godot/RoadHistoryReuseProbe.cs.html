12    public bool CaptureCachedMesh(RoadMapScene scene)
13    {
14        _cachedMeshes = scene.GetNode<RoadMapView>("View").ReusableDisplay?.Meshes;
15        return _cachedMeshes is { Count: > 0 };
16    }
17    public bool CachedMeshReleased => _cachedMeshes is { Count: > 0 } && _cachedMeshes.All(mesh => !GodotObject.IsInstanceValid(mesh));
18    public bool CacheIsEmpty(RoadMapScene scene) => scene.GetNode<RoadMapView>("View").ReusableDisplay is null;

20    public Godot.Collections.Dictionary Run()
21    {
22        var rows = new Godot.Collections.Array<Godot.Collections.Dictionary>();
23        var network = new RoadNetwork();
24        network.TryCommit(network.PlanBuild(new(network.Snapshot.Token, new(0, 0), new(300, 0), RoadProfileId.Street)).Plan!);
25        RoadSnapshot original = network.Snapshot;
26        using var oldDisplay = Display(original);
27        ArrayMesh oldMesh = oldDisplay.Meshes.Single();
28        var originalHit = oldDisplay.QueryHit(new(50, 0)).Results.Single().Location;
29        RoadGridSpan span = RoadSpanQuery.Pick(original, new(original.Token, original.Edges[0].Id, .5))!;
30        network.TryCommit(network.PlanChangeProfile(span, RoadProfileId.Highway).Plan!);
31        using var changedDisplay = Display(network.Snapshot);
32        RoadPlan undo = network.PlanUndo().Plan!;
33        var prepared = Task.Run(() => oldDisplay.TryPrepareReuse(undo.Target, CancellationToken.None)).GetAwaiter().GetResult();
34        Check("observed_content_prepares_on_worker", prepared is not null && network.Snapshot.Token == undo.ChangeSet.Before.Token);
35        using var restored = prepared!.Upload();
36        Check("reuse_keeps_exact_mesh", ReferenceEquals(restored.Meshes.Single(), oldMesh));
37        network.TryCommit(undo);
38        var hit = restored.QueryHit(new(50, 0));
39        var picked = restored.QuerySpan(new(50, 0));
40        var trace = restored.QueryTraceSpans(new(25, 0), new(275, 0));
41        Check("all_queries_bind_new_publication", restored.Snapshot.Token != original.Token &&
42            hit.Source == undo.Target.Token && hit.Results.Single().Location.Source == undo.Target.Token &&
43            picked.Results.Count == 1 && picked.Results.All(item => item.Source == undo.Target.Token) &&
44            trace.Results.Count == 3 && trace.Results.All(item => item.Source == undo.Target.Token));
45        try { restored.SurfaceCenter(originalHit); Check("old_location_rejected", false); }
46        catch (InvalidOperationException) { Check("old_location_rejected", true); }
47        Check("old_reader_remains_bound_to_old_token", oldDisplay.QueryHit(new(50, 0)).Results.Single().Location.Source == original.Token);
48        try { prepared.Upload(); Check("reused_upload_is_single_use", false); }
49        catch (InvalidOperationException) { Check("reused_upload_is_single_use", true); }
50        var redo = network.PlanRedo().Plan!;
51        Check("different_content_cannot_reuse", oldDisplay.TryPrepareReuse(redo.Target, CancellationToken.None) is null);
52        using var redone = changedDisplay.TryPrepareReuse(redo.Target, CancellationToken.None)!.Upload();
53        Check("redo_reuses_other_mesh", ReferenceEquals(redone.Meshes, changedDisplay.Meshes));
54        using (var cancellation = new CancellationTokenSource())
55        {
56            cancellation.Cancel();
57            try { restored.TryPrepareReuse(undo.Target, cancellation.Token); Check("cancel_before_reuse", false); }
58            catch (OperationCanceledException) { Check("cancel_before_reuse", true); }
59        }
60        using (var bytes = new System.IO.MemoryStream())
61        {
62            RoadCodec.Write(bytes, original);
63            bytes.Position = 0;
64            RoadPlan load = network.PlanLoad(RoadCodec.Read(bytes));
65            Check("load_lineage_cannot_reuse", restored.TryPrepareReuse(load.Target, CancellationToken.None) is null);
66            bytes.Position = 0;
67            var foreign = new RoadNetwork().PlanLoad(RoadCodec.Read(bytes));
68            Check("foreign_network_cannot_reuse", restored.TryPrepareReuse(foreign.Target, CancellationToken.None) is null);
69        }
70        oldDisplay.Dispose();
71        oldDisplay.Dispose();
72        Check("cache_eviction_preserves_current_mesh", GodotObject.IsInstanceValid(oldMesh) && restored.Meshes.Single().GetSurfaceCount() == 1);
73        Check("disposed_cache_cannot_prepare", oldDisplay.TryPrepareReuse(undo.Target, CancellationToken.None) is null);
74        // Abandoning a prepared candidate acquires no native resource, including after its cache is evicted.
75        var abandoned = restored.TryPrepareReuse(undo.Target, CancellationToken.None)!;
76        restored.Dispose();
77        Check("last_owner_releases_mesh", !GodotObject.IsInstanceValid(oldMesh));
78        try { abandoned.Upload(); Check("evicted_candidate_cannot_upload", false); }
79        catch (ObjectDisposedException) { Check("evicted_candidate_cannot_upload", true); }
80        return new() { ["rows"] = rows, ["passed"] = rows.All(row => row["passed"].AsBool()) };
81
82        void Check(string name, bool passed) => rows.Add(new() { ["name"] = name, ["passed"] = passed });
83    }

85    public Godot.Collections.Dictionary Measure(string fixturePath)
86    {
87        using var bytes = System.IO.File.OpenRead(ProjectSettings.GlobalizePath(fixturePath));
88        var network = new RoadNetwork();
89        network.TryCommit(network.PlanLoad(RoadCodec.Read(bytes)));
90        RoadSnapshot original = network.Snapshot;
91        using var display = Display(original);
92        RoadEdge edge = original.Edges[0];
93        var span = RoadSpanQuery.Pick(original, new(original.Token, edge.Id, .5))!;
94        network.TryCommit(network.PlanChangeProfile(span, edge.Profile == RoadProfileId.Highway ? RoadProfileId.Street : RoadProfileId.Highway).Plan!);
95        var measured = Task.Run(() =>
96        {
97            long start = GC.GetAllocatedBytesForCurrentThread();
98            RoadPlan undo = network.PlanUndo().Plan!;
99            var geometry = display.TryPrepareReuse(undo.Target, CancellationToken.None)!;
100            return (Plan: undo, Geometry: geometry, Bytes: GC.GetAllocatedBytesForCurrentThread() - start);
101        }).GetAwaiter().GetResult();
102        long before = GC.GetAllocatedBytesForCurrentThread();
103        using var restored = measured.Geometry.Upload();
104        long uploadBytes = GC.GetAllocatedBytesForCurrentThread() - before;
105        return new()
106        {
107            ["edges"] = original.EdgeCount, ["workerBytes"] = measured.Bytes, ["uploadBytes"] = uploadBytes,
108            ["sameMesh"] = ReferenceEquals(display.Meshes, restored.Meshes),
109            ["newToken"] = restored.Snapshot.Token != original.Token,
110            ["committed"] = network.TryCommit(measured.Plan),
111        };
112    }

114    private static RoadDisplay Display(RoadSnapshot snapshot) =>
115        RoadDisplay.Prepare(snapshot, RoadPresentation.Prepare(snapshot.Nodes, snapshot.Edges));

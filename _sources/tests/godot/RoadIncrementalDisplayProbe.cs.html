21    private static RoadNetwork SparseNetwork(string fixturePath)
22    {
23        var document = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(ProjectSettings.GlobalizePath(fixturePath)))!.AsObject();
24        foreach (var edge in document["edges"]!.AsArray()) edge!["id"] = checked(edge["id"]!.GetValue<long>() * 64);
25        document["nextEdgeId"] = checked(document["nextEdgeId"]!.GetValue<long>() * 64);
26        using var bytes = new System.IO.MemoryStream(System.Text.Encoding.UTF8.GetBytes(document.ToJsonString()));
27        var network = new RoadNetwork();
28        if (!network.TryCommit(network.PlanLoad(RoadCodec.Read(bytes)))) throw new InvalidOperationException("Sparse fixture rejected.");
29        return network;
30    }

32    public Godot.Collections.Dictionary RunPaging(string fixturePath)
33    {
34        var rows = new Godot.Collections.Array<Godot.Collections.Dictionary>();
35        RoadDisplay? current = null;
36        try
37        {
38            RoadNetwork sparse = SparseNetwork(fixturePath);
39            using (var sparseDisplay = RoadDisplay.PrepareFull(sparse.Snapshot).Upload())
40            using (var expected = Full(sparse.Snapshot))
41            {
42                Check("sparse_archive_draw_calls_are_bounded", sparseDisplay.Meshes.Count < 512);
43                Check("sparse_archive_gpu_and_pick_match_full", Fingerprint(sparseDisplay) == Fingerprint(expected) && SameQueries(sparseDisplay, expected, sparse.Snapshot.Map.CellSizeMetres));
44            }
45            var network = new RoadNetwork(new MapDefinition(25));
46            for (int i = 0; i < 128; i++)
47                if (!network.TryCommit(Build(i))) throw new InvalidOperationException("Paging fixture rejected.");
48            current = RoadDisplay.PrepareFull(network.Snapshot).Upload();
49            Check("dense_initial_owner_pages", current.Meshes.Count == 2);
50            Step(Build(128), "split");
51            Check("insertion_splits_one_full_page", current.Meshes.Count == 3);
52            Step(network.PlanRemove(Pick(1)).Plan!, "first_owner_rekey");
53            Step(network.PlanRemove(Pick(65)).Plan!, "redistribute");
54            RoadSnapshot beforeBatch = network.Snapshot;
55            RoadGridSpan[] removed = beforeBatch.Edges.Take(50).Select(edge =>
56                RoadSpanQuery.Pick(beforeBatch, new(beforeBatch.Token, edge.Id, .5))!).ToArray();
57            Step(network.PlanRemove(removed).Plan!, "adjacent_page_batch_delete");
58            for (int i = 0; i < 80; i++)
59            {
60                Step(Build(200 + i), "append_" + i);
61                if (i % 2 == 0) Step(network.PlanRemove(Pick(network.Snapshot.Edges[0].Id.Value)).Plan!, "trim_" + i);
62            }
63            Check("churn_keeps_pages_proportional_to_live_owners", current.Meshes.Count <= Math.Max(1, (network.Snapshot.EdgeCount + 15) / 16));
64            return new() { ["passed"] = rows.All(row => row["passed"].AsBool()), ["rows"] = rows };
65
66            RoadPlan Build(int index)
67            {
68                var start = new RoadPoint(-2000 + 100 * (index % 16), -1000 + 100 * (index / 16));
69                return network.PlanBuild(new(network.Snapshot.Token, start, new(start.X + 25, start.Y), RoadProfileId.Street)).Plan!;
70            }
71            RoadGridSpan Pick(long id) => RoadSpanQuery.Pick(network.Snapshot, new(network.Snapshot.Token, new(id), .5))!;
72            void Step(RoadPlan plan, string name)
73            {
74                RoadDisplay candidate = RoadDisplay.PrepareChange(current!, plan, CancellationToken.None).Upload();
75                using var full = Full(plan.Target);
76                Check(name + "_flat_geometry_matches", Fingerprint(candidate) == Fingerprint(full));
77                Check(name + "_commits", network.TryCommit(plan));
78                current!.Dispose();
79                current = candidate;
80            }
81        }
82        catch (Exception exception) { return new() { ["passed"] = false, ["error"] = exception.ToString(), ["rows"] = rows }; }
83        finally { current?.Dispose(); }
84
85        void Check(string name, bool passed) => rows.Add(new() { ["name"] = name, ["passed"] = passed });
86    }

95    private Godot.Collections.Dictionary RunCore(string fixturePath)
96    {
97        var rows = new Godot.Collections.Array<Godot.Collections.Dictionary>();
98        using var input = System.IO.File.OpenRead(ProjectSettings.GlobalizePath(fixturePath));
99        var network = new RoadNetwork();
100        Check("fixture_loaded", network.TryCommit(network.PlanLoad(RoadCodec.Read(input))));
101        using var original = RoadDisplay.PrepareFull(network.Snapshot).Upload();
102        using (var reference = Full(network.Snapshot))
103            Check("full_entry_matches_reference_gpu_and_pick", Fingerprint(original) == Fingerprint(reference) && SameQueries(original, reference, network.Snapshot.Map.CellSizeMetres));
104        string originalHash = Fingerprint(original);
105        RoadSnapshot source = network.Snapshot;
106        RoadGridSpan Pick(RoadSnapshot snapshot, int index) => RoadSpanQuery.Pick(snapshot,
107            new(snapshot.Token, snapshot.Edges[index].Id, .5)) ?? throw new InvalidOperationException("Fixture span missing.");
108        RoadPlan plan = network.PlanChangeProfile(Pick(source, 0), RoadProfileId.Highway).Plan!;
109        var measured = Task.Run(() =>
110        {
111            _ = RoadDisplay.PrepareChange(original, plan, CancellationToken.None);
112            long before = GC.GetAllocatedBytesForCurrentThread();
113            var prepared = RoadDisplay.PrepareChange(original, plan, CancellationToken.None);
114            return (Prepared: prepared, Bytes: GC.GetAllocatedBytesForCurrentThread() - before);
115        }).GetAwaiter().GetResult();
116        long uploadedBytes = measured.Prepared.UploadBytes;
117        int pages = measured.Prepared.ChangedPageCount;
118        using var changed = measured.Prepared.Upload();
119        using var full = Full(plan.Target);
120        Check("local_change_matches_full_gpu_arrays", Fingerprint(changed) == Fingerprint(full));
121        Check("local_change_keeps_old_gpu_arrays", originalHash == Fingerprint(original));
122        Check("local_change_shares_untouched_meshes", changed.Meshes.Intersect(original.Meshes).Any() && !changed.Meshes.SequenceEqual(original.Meshes));
123        Check("local_preparation_is_bounded", measured.Bytes < 1024 * 1024 && uploadedBytes < 1024 * 1024 && pages < original.Meshes.Count,
124            new() { ["workerBytes"] = measured.Bytes, ["uploadBytes"] = uploadedBytes, ["changedPages"] = pages, ["originalMeshes"] = original.Meshes.Count });
125        Check("local_change_matches_full_picking", SameQueries(changed, full, source.Map.CellSizeMetres));
126        Check("uncommitted_prepare_keeps_network_and_history", ReferenceEquals(network.Snapshot, source) && network.History.RetainedCount == 0);
127        try { measured.Prepared.Upload(); Check("incremental_upload_is_single_use", false); }
128        catch (InvalidOperationException) { Check("incremental_upload_is_single_use", true); }
129        using (var cancellation = new CancellationTokenSource())
130        {
131            cancellation.Cancel();
132            try { RoadDisplay.PrepareChange(original, plan, cancellation.Token); Check("cancel_before_incremental_prepare", false); }
133            catch (OperationCanceledException) { Check("cancel_before_incremental_prepare", true); }
134        }
135        try { RoadDisplay.PrepareChange(changed, plan, CancellationToken.None); Check("wrong_source_display_rejected", false); }
136        catch (InvalidOperationException) { Check("wrong_source_display_rejected", true); }
137
138        // The injected failure is after the second new mesh exists. All newly created resources
139        // and every temporary lease must be released while original pages stay usable.
140        RoadPlan batch = network.PlanChangeProfile(new[] { Pick(source, 0), Pick(source, source.EdgeCount / 3), Pick(source, source.EdgeCount * 2 / 3) }, RoadProfileId.Highway).Plan!;
141        var failed = RoadDisplay.PrepareChange(original, batch, CancellationToken.None);
142        var created = new List<ArrayMesh>();
143        RoadDisplay.AfterMeshUpload = (mesh, ordinal) =>
144        {
145            created.Add(mesh);
146            if (ordinal == 1) throw new InvalidOperationException("Injected second mesh upload failure.");
147        };
148        try { using var unexpected = failed.Upload(); Check("second_mesh_failure_injected", false); }
149        catch (InvalidOperationException exception) { Check("second_mesh_failure_injected", exception.Message == "Injected second mesh upload failure."); }
150        finally { RoadDisplay.AfterMeshUpload = null; }
151        Check("failed_upload_releases_all_new_meshes", created.Count == 2 && created.All(mesh => !GodotObject.IsInstanceValid(mesh)));
152        Check("failed_upload_preserves_all_source_meshes", original.Meshes.All(GodotObject.IsInstanceValid) && Fingerprint(original) == originalHash);
153
154        Check("first_change_committed", network.TryCommit(plan));
155        using var removed = Advance(changed, network.PlanRemove(Pick(network.Snapshot, 0)).Plan!, "delete");
156        using var undone = Advance(removed, network.PlanUndo().Plan!, "undo");
157        using var redone = Advance(undone, network.PlanRedo().Plan!, "redo");
158        using var reset = Advance(redone, network.PlanUndo().Plan!, "restore");
159        int cell = network.Snapshot.Map.CellSizeMetres;
160        if (cell != 200)
161        {
162            RoadPlan build = network.PlanBuild(new(network.Snapshot.Token, new(-4000, 4000), new(-4000 + 4 * cell, 4000), RoadProfileId.Street)).Plan!;
163            using var built = Advance(reset, build, "build");
164        }
165        CheckLoop();
166        return new() { ["passed"] = rows.All(row => row["passed"].AsBool()), ["rows"] = rows };
167
168        void CheckLoop()
169        {
170            var loop = new RoadNetwork();
171            RoadDisplay current = RoadDisplay.PrepareFull(loop.Snapshot).Upload();
172            int step = 0;
173            try
174            {
175                foreach (var stroke in new (RoadPoint Start, RoadPoint End)[]
176                {
177                    (new(50, 50), new(0, 0)), (new(0, 0), new(100, 0)),
178                    (new(100, 0), new(50, 50)), (new(50, 50), new(150, 150)),
179                }) Step(loop.PlanBuild(new(loop.Snapshot.Token, stroke.Start, stroke.End, RoadProfileId.Street)).Plan!);
180                Step(loop.PlanUndo().Plan!);
181                Step(loop.PlanRedo().Plan!);
182                RoadSnapshot snapshot = loop.Snapshot;
183                RoadEdge edge = snapshot.Edges.First();
184                Step(loop.PlanChangeProfile(RoadSpanQuery.Pick(snapshot, new(snapshot.Token, edge.Id, .01))!, RoadProfileId.Highway).Plan!);
185            }
186            finally { current.Dispose(); }
187
188            void Step(RoadPlan next)
189            {
190                RoadDisplay candidate = RoadDisplay.PrepareChange(current, next, CancellationToken.None).Upload();
191                using var expected = Full(next.Target);
192                Check($"loop_step_{step}_matches_full", Fingerprint(candidate) == Fingerprint(expected) && SameQueries(candidate, expected, 100, Vector2.Zero));
193                Check($"loop_step_{step}_commits", loop.TryCommit(next));
194                step++;
195                current.Dispose();
196                current = candidate;
197            }
198        }
199
200        RoadDisplay Advance(RoadDisplay basis, RoadPlan next, string name)
201        {
202            string before = Fingerprint(basis);
203            var geometry = Task.Run(() => RoadDisplay.PrepareChange(basis, next, CancellationToken.None)).GetAwaiter().GetResult();
204            RoadDisplay display = geometry.Upload();
205            using var expected = Full(next.Target);
206            Check(name + "_gpu_and_query_match_full", Fingerprint(display) == Fingerprint(expected) && SameQueries(display, expected, next.Target.Map.CellSizeMetres));
207            Check(name + "_source_remains_independent", Fingerprint(basis) == before);
208            Check(name + "_committed", network.TryCommit(next));
209            return display;
210        }
211        void Check(string name, bool passed, Godot.Collections.Dictionary? detail = null) => rows.Add(new()
212        { ["name"] = name, ["passed"] = passed, ["detail"] = detail ?? new() });
213    }

218    private static string Fingerprint(RoadDisplay display)
219    {
220        using var verticesHash = IncrementalHash.CreateHash(HashAlgorithmName.SHA256);
221        using var colorsHash = IncrementalHash.CreateHash(HashAlgorithmName.SHA256);
222        using var indicesHash = IncrementalHash.CreateHash(HashAlgorithmName.SHA256);
223        int vertexCount = 0, indexCount = 0;
224        foreach (ArrayMesh mesh in display.Meshes)
225        {
226            if (mesh.GetSurfaceCount() != 1) throw new InvalidOperationException("Unexpected surface count.");
227            using var arrays = mesh.SurfaceGetArrays(0);
228            Vector3[] vertices = arrays[(int)Mesh.ArrayType.Vertex].AsVector3Array();
229            Color[] colors = arrays[(int)Mesh.ArrayType.Color].AsColorArray();
230            int[] indices = arrays[(int)Mesh.ArrayType.Index].AsInt32Array();
231            if (vertices.Length != colors.Length || indices.Length % 3 != 0 || indices.Any(index => index < 0 || index >= vertices.Length))
232                throw new InvalidOperationException("Invalid uploaded geometry.");
233            for (int i = 0; i < indices.Length; i++) indices[i] = checked(indices[i] + vertexCount);
234            verticesHash.AppendData(MemoryMarshal.AsBytes(vertices.AsSpan()));
235            colorsHash.AppendData(MemoryMarshal.AsBytes(colors.AsSpan()));
236            indicesHash.AppendData(MemoryMarshal.AsBytes(indices.AsSpan()));
237            vertexCount += vertices.Length;
238            indexCount += indices.Length;
239        }
240        // Mesh boundaries are an implementation choice. Compare the flattened painter
241        // sequence with global indices, so splitting/merging pages cannot hide reordering.
242        return $"{vertexCount}/{indexCount}:" + Convert.ToHexString(verticesHash.GetHashAndReset()) +
243            Convert.ToHexString(colorsHash.GetHashAndReset()) + Convert.ToHexString(indicesHash.GetHashAndReset());
244    }

246    private static bool SameQueries(RoadDisplay left, RoadDisplay right, int cell, Vector2? origin = null)
247    {
248        Vector2 start = origin ?? new Vector2(-4000, -4000);
249        foreach (SpatialQueryBudget budget in new[] { SpatialQueryBudget.Default, new(1, 1, 1), new(0, 1, 1) })
250        {
251            for (int x = -2; x <= 12; x++)
252                for (int y = -2; y <= 12; y++)
253                {
254                    Vector2 point = start + new Vector2(x * cell / 4f, y * cell / 4f);
255                    var a = left.QueryHit(point, budget);
256                    var b = right.QueryHit(point, budget);
257                    if (a.Source != b.Source || a.Status != b.Status || a.Metrics != b.Metrics || a.Reason != b.Reason || !a.Results.SequenceEqual(b.Results)) return false;
258                }
259            foreach ((Vector2 from, Vector2 to) in new[]
260            {
261                (start, start + new Vector2(6 * cell, 0)),
262                (start + new Vector2(6 * cell, 0), start),
263                (start, start + new Vector2(6 * cell, 6 * cell)),
264            })
265            {
266                var a = left.QueryTraceSpans(from, to, budget);
267                var b = right.QueryTraceSpans(from, to, budget);
268                if (a.Source != b.Source || a.Status != b.Status || a.Metrics != b.Metrics || a.Reason != b.Reason ||
269                    !a.Results.Select(span => span.Key).SequenceEqual(b.Results.Select(span => span.Key))) return false;
270            }
271        }
272        return true;
273    }

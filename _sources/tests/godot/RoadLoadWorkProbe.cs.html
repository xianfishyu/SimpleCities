1using Godot;
2using System;
3using System.Collections.Generic;
4using System.IO;
5using System.Threading;

7/// <summary>Debug-only load worker gate and external notification failure boundary.</summary>
8public partial class RoadLoadWorkProbe : RefCounted
9{
10    private static RoadLoadWorkProbe? _active;
11    private readonly ManualResetEventSlim _release = new(false);
12    private SaveManager? _manager;
13    private RoadMapScene? _scene;
14    private string _mode = "";
15    private int _started;
16    private int _completed;
17    private int _workerThread;
18    private int _mainThread;
19    private FileStream? _saveLock;
20    private SaveManager? _warningManager;
21
22    public bool Started => Volatile.Read(ref _started) != 0;
23    public bool Completed => Volatile.Read(ref _completed) != 0;
24    public bool UsedBackgroundThread => Started && Volatile.Read(ref _workerThread) != _mainThread;
25    public bool NotificationObserved { get; private set; }
26    public bool ObservedCleanState { get; private set; }
27
28    public void Install(SaveManager manager, RoadMapScene scene, string mode)
29    {
30        if (_active is not null || manager.IsOperationBusy || mode is not ("normal" or "fail-preflight" or "observe-notifications"))
31            throw new InvalidOperationException("Load probe must be installed while idle with a known mode.");
32        _manager = manager;
33        _scene = scene;
34        _mode = mode;
35        _mainThread = System.Environment.CurrentManagedThreadId;
36        Volatile.Write(ref _active, this);
37    }
38
39    internal static void AtWorkerEntry(SaveManager manager)
40    {
41        RoadLoadWorkProbe? probe = Volatile.Read(ref _active);
42        if (probe is null || !ReferenceEquals(probe._manager, manager)) return;
43        Volatile.Write(ref probe._workerThread, System.Environment.CurrentManagedThreadId);
44        Volatile.Write(ref probe._started, 1);
45        probe._release.Wait();
46        Volatile.Write(ref probe._completed, 1);
47    }
48
49    internal static void AfterPreflight(SaveManager manager, IReadOnlyList<INonThrowingLoadCommitPlan> plans)
50    {
51        RoadLoadWorkProbe? probe = Volatile.Read(ref _active);
52        if (probe is null || !ReferenceEquals(probe._manager, manager)) return;
53        if (probe._mode == "fail-preflight")
54            throw new InvalidOperationException("Injected V4 load failure after all participant preflights.");
55        if (probe._mode == "observe-notifications")
56        {
57            if (plans is not List<INonThrowingLoadCommitPlan> mutable)
58                throw new InvalidOperationException("Load probe requires the pre-aggregate plan list.");
59            mutable.Insert(0, new Observer(probe));
60        }
61    }
62
63    public void Release() => _release.Set();
64
65    public void LockSaveManifest(string slotID)
66    {
67        if (_saveLock is not null || !slotID.StartsWith("manual-", StringComparison.Ordinal) ||
68            !Guid.TryParseExact(slotID[7..], "N", out _))
69            throw new ArgumentException("Expected a fresh manual test slot.", nameof(slotID));
70        string path = ProjectSettings.GlobalizePath($"user://saves-v4/{slotID}/manifest.json");
71        _saveLock = new FileStream(path, FileMode.Open, System.IO.FileAccess.Read, FileShare.None);
72    }
73
74    public void ReleaseSaveLock()
75    {
76        _saveLock?.Dispose();
77        _saveLock = null;
78    }
79
80    public void ArmLoadWarnings(SaveManager manager)
81    {
82        _warningManager = manager;
83        manager.ArmNextSlotTargetLoadCompleteCommitFailure();
84        manager.ArmNextAggregateLoadPostCommitFailure();
85    }
86
87    public int GetSlotCleanupFailureCount() => _warningManager?.GetSlotTargetLoadCompleteCommitFailureCount() ?? 0;
88    public int GetPostCommitFailureCount() => _warningManager?.GetAggregateLoadPostCommitFailureCount() ?? 0;
89    public bool IsSlotCleanupFailureArmed() => _warningManager?.IsSlotTargetLoadCompleteCommitFailureArmed() ?? false;
90    public bool IsPostCommitFailureArmed() => _warningManager?.IsAggregateLoadPostCommitFailureArmed() ?? false;
91
92    public void StopObserving()
93    {
94        if (_manager?.IsOperationBusy == true) throw new InvalidOperationException("Wait for load completion before disarming.");
95        if (ReferenceEquals(_active, this)) Volatile.Write(ref _active, null);
96    }
97
98    public void DisarmLoadWarnings()
99    {
100        _warningManager?.DisarmSlotTargetLoadCompleteCommitFailure();
101        _warningManager?.DisarmAggregateLoadPostCommitFailure();
102        _warningManager = null;
103    }
104
105    public void Cleanup()
106    {
107        if (!Completed || _manager?.IsOperationBusy == true)
108            throw new InvalidOperationException("Await the released load before cleaning its probe.");
109        if (ReferenceEquals(_active, this)) Volatile.Write(ref _active, null);
110        DisarmLoadWarnings();
111        _manager = null;
112        _scene = null;
113        _release.Dispose();
114    }
115
116    private sealed class Observer(RoadLoadWorkProbe probe) : INonThrowingLoadCommitPlan
117    {
118        public string ParticipantID => "v4-load-observer-probe";
119        public bool IsGenerationCurrent => true;
120        public void CommitReferences() { }
121        public IReadOnlyList<string> PublishNotifications()
122        {
123            RoadMapScene scene = probe._scene ?? throw new InvalidOperationException("Missing probe scene.");
124            var selection = scene.GetSelectionState();
125            var preview = scene.GetBuildPreview();
126            probe.NotificationObserved = true;
127            probe.ObservedCleanState = !scene.HasBuildPreview && scene.IsPresentationCurrent &&
128                selection["selectedCount"].AsInt32() == 0 && !selection["hasHover"].AsBool() &&
129                !selection["selecting"].AsBool() && selection["strokes"].AsGodotArray().Count == 0 &&
130                preview["phase"].AsString() == "None" && preview["segments"].AsGodotArray().Count == 0 &&
131                scene.GetHistoryState()["retainedCount"].AsInt32() == 0;
132            throw new InvalidOperationException("Injected V4 external load observer failure.");
133        }
134        public void CompleteCommit() { }
135        public void Dispose() { }
136    }

1using Godot;
2using System;
3using System.Threading;
4using System.Reflection;
5
6/// <summary>Debug-only controllable worker boundary; releases late work even after cancellation.</summary>
7public partial class RoadOperationWorkProbe : RefCounted
8{
9    private readonly ManualResetEventSlim _release = new(false);
10    private int _started;
11    private int _completed;
12    private int _workerThread;
13    private int _mainThread;
14    private bool _disposed;
15
16    public bool Started => Volatile.Read(ref _started) != 0;
17    public bool Completed => Volatile.Read(ref _completed) != 0;
18    public bool UsedBackgroundThread => Started && Volatile.Read(ref _workerThread) != _mainThread;
19
20    public void Install(RoadMapScene scene)
21    {
22        _mainThread = System.Environment.CurrentManagedThreadId;
23        scene.BeforeBuildWork = Wait;
24    }
25
26    public void InstallPreview(RoadMapScene scene)
27    {
28        _mainThread = System.Environment.CurrentManagedThreadId;
29        scene.BeforePreviewWork = Wait;
30    }
31
32    public void InstallPreviewFailure(RoadMapScene scene)
33    {
34        _mainThread = System.Environment.CurrentManagedThreadId;
35        scene.BeforePreviewWork = () =>
36        {
37            Wait();
38            throw new InvalidOperationException("Injected V4 preview worker failure.");
39        };
40    }
41
42    public void RemovePreview(RoadMapScene scene)
43    {
44        scene.BeforePreviewWork = null;
45        Release();
46    }
47
48    private void Wait()
49    {
50        Volatile.Write(ref _workerThread, System.Environment.CurrentManagedThreadId);
51        Volatile.Write(ref _started, 1);
52        _release.Wait();
53        Volatile.Write(ref _completed, 1);
54    }
55
56    public void Release() => _release.Set();
57
58    // Hold the main-thread continuation while the real worker exits, then dispatch Esc.
59    // This deterministic race is correctness evidence, not a latency measurement.
60    public bool CancelAfterWorkerExit(RoadMapScene scene)
61    {
62        FieldInfo workerExit = typeof(RoadMapScene).GetField("_workerExitedAt", BindingFlags.Instance | BindingFlags.NonPublic)!;
63        Release();
64        if (!SpinWait.SpinUntil(() => (long)workerExit.GetValue(scene)! != 0, TimeSpan.FromSeconds(5))) return false;
65        using var input = new InputEventKey { Keycode = Key.Escape, Pressed = true };
66        Input.ParseInputEvent(input);
67        Input.FlushBufferedEvents();
68        return true;
69    }
70
71    public void Remove(RoadMapScene scene)
72    {
73        scene.BeforeBuildWork = null;
74        Release();
75    }
76
77    public void Cleanup()
78    {
79        if (_disposed) return;
80        if (!Completed)
81            throw new InvalidOperationException("Release and await the worker before disposing its gate.");
82        _release.Dispose();
83        _disposed = true;
84    }
85}

70    private const string DeleteRecoverGateTimeoutMessage =
71        "Timed out waiting to release the Delete Recover test gate.";
72    private const string DeletePostCommitGateTimeoutMessage =
73        "Timed out waiting to release the Delete post-commit test gate.";
74    private static readonly TimeSpan DeleteRecoverGateTimeout = TimeSpan.FromSeconds(15);
75
76    private readonly ManualResetEventSlim _deleteRecoverGateRelease = new(initialState: true);
77    private int _deleteRecoverGateArmed;
78    private int _deleteRecoverGateEntered;
79    private int _deleteRecoverGateTriggerCount;
80    private int _deleteRecoverGateCancelRequestCount;
81    private string _deleteRecoverGateCancelOperationToken = string.Empty;
82    private readonly ManualResetEventSlim _deletePostCommitGateRelease = new(initialState: true);
83    private int _deletePostCommitGateArmed;
84    private int _deletePostCommitGateEntered;
85    private int _deletePostCommitGateTriggerCount;
86    private int _deletePostCommitGateCancelRequestCount;
87    private string _deletePostCommitGateCancelOperationToken = string.Empty;
88
89    partial void ProbeObserveCancelOperation(ref string operationToken)
90    {
91        if (Volatile.Read(ref _deleteRecoverGateEntered) != 0)
92        {
93            _deleteRecoverGateCancelOperationToken = operationToken;
94            Interlocked.Increment(ref _deleteRecoverGateCancelRequestCount);
95        }
96
97        if (Volatile.Read(ref _deletePostCommitGateEntered) != 0)
98        {
99            _deletePostCommitGateCancelOperationToken = operationToken;
100            Interlocked.Increment(ref _deletePostCommitGateCancelRequestCount);
101        }
102    }
103
104    partial void ProbeWaitAtDeleteRecover(ref SaveSlotStore store)
105    {
106        if (Volatile.Read(ref _deletePostCommitGateArmed) != 0)
107            store = new SaveSlotStore(store.SaveBaseDirectory, WaitAtDeletePostCommit);
108
109        if (Interlocked.Exchange(ref _deleteRecoverGateArmed, 0) == 0)
110            return;
111
112        Interlocked.Increment(ref _deleteRecoverGateTriggerCount);
113        Volatile.Write(ref _deleteRecoverGateEntered, 1);
114        try
115        {
116            if (!_deleteRecoverGateRelease.Wait(DeleteRecoverGateTimeout))
117                throw new TimeoutException(DeleteRecoverGateTimeoutMessage);
118        }
119        finally
120        {
121            Volatile.Write(ref _deleteRecoverGateEntered, 0);
122        }
123    }
124
125    private void WaitAtDeletePostCommit(SavePublicationPhase phase)
126    {
127        if (phase != SavePublicationPhase.DeletionTombstoned ||
128            Interlocked.Exchange(ref _deletePostCommitGateArmed, 0) == 0)
129        {
130            return;
131        }
132
133        Interlocked.Increment(ref _deletePostCommitGateTriggerCount);
134        Volatile.Write(ref _deletePostCommitGateEntered, 1);
135        try
136        {
137            if (!_deletePostCommitGateRelease.Wait(DeleteRecoverGateTimeout))
138                throw new TimeoutException(DeletePostCommitGateTimeoutMessage);
139        }
140        finally
141        {
142            Volatile.Write(ref _deletePostCommitGateEntered, 0);
143        }
144    }

146    internal void ArmNextDeleteRecoverGate()
147    {
148        if (IsOperationBusy)
149        {
150            throw new InvalidOperationException(
151                "SaveManager must be idle before arming its Delete Recover test gate.");
152        }
153        if (Interlocked.CompareExchange(ref _deleteRecoverGateArmed, 1, 0) != 0)
154        {
155            throw new InvalidOperationException(
156                "Delete Recover test gate is already armed.");
157        }
158
159        Interlocked.Exchange(ref _deleteRecoverGateTriggerCount, 0);
160        Interlocked.Exchange(ref _deleteRecoverGateCancelRequestCount, 0);
161        _deleteRecoverGateCancelOperationToken = string.Empty;
162        Volatile.Write(ref _deleteRecoverGateEntered, 0);
163        _deleteRecoverGateRelease.Reset();
164    }

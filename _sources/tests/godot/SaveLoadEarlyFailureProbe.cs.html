5/// <summary>通用StartLoad早期边界故障；仅编译进Debug。</summary>
6public partial class SaveLoadEarlyFailureProbe : RefCounted
7{
8    private SaveManager? _manager;
9    public void Arm(SaveManager manager, string stage)
10    {
11        if (stage is not ("participant-capture" or "prepare-phase" or "worker-entry") || manager.IsOperationBusy)
12            throw new InvalidOperationException("Arm a known early load stage while idle.");
13        _manager = manager;
14        manager.QaLoadFailureStage = stage;
15        manager.QaLoadFailureCount = 0;
16    }
17    public int TriggerCount => _manager?.QaLoadFailureCount ?? 0;
18    public bool IsArmed => _manager?.QaLoadFailureStage.Length > 0;
19    public void Disarm() { if (_manager is not null) _manager.QaLoadFailureStage = ""; _manager = null; }
20}
21
22public partial class SaveManager
23{
24    internal string QaLoadFailureStage = "";
25    internal int QaLoadFailureCount;
26    internal void ThrowQaLoadFailure(string stage)
27    {
28        if (QaLoadFailureStage != stage) return;
29        QaLoadFailureStage = "";
30        QaLoadFailureCount++;
31        throw new InvalidOperationException("Injected generic load early failure: " + stage);
32    }
33    partial void ProbeAggregateLoadPostParticipantCaptureFailure(IReadOnlyList<CapturedLoadParticipant> loadParticipants) =>
34        ThrowQaLoadFailure("participant-capture");
35    partial void ProbeAggregateLoadPostPreparePhaseFailure(SaveOperationPhase phase) =>
36        ThrowQaLoadFailure("prepare-phase");
37}

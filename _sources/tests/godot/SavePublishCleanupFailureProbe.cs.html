33public partial class SaveManager
34{
35    internal const string PublishCleanupFailureMessage =
36        "Injected Save publish cleanup failure.";
37
38    private int _publishCleanupFailureArmed;
39    private int _publishCleanupFailureCount;
40
41    partial void ProbeConfigurePublishCleanupFailure(ref SaveSlotStore store)
42    {
43        if (Volatile.Read(ref _publishCleanupFailureArmed) == 0)
44            return;
45
46        store = new SaveSlotStore(store.SaveBaseDirectory, phase =>
47        {
48            if (phase != SavePublicationPhase.CanonicalPublished ||
49                Interlocked.Exchange(ref _publishCleanupFailureArmed, 0) == 0)
50            {
51                return;
52            }
53
54            Interlocked.Increment(ref _publishCleanupFailureCount);
55            throw new IOException(PublishCleanupFailureMessage);
56        });
57    }
58
59    internal void ArmNextPublishCleanupFailure()
60    {
61        if (IsOperationBusy)
62        {
63            throw new InvalidOperationException(
64                "SaveManager must be idle before arming its publish cleanup failure probe.");
65        }
66        if (Interlocked.CompareExchange(ref _publishCleanupFailureArmed, 1, 0) != 0)
67        {
68            throw new InvalidOperationException(
69                "Save publish cleanup failure probe is already armed.");
70        }
71    }
72
73    internal void DisarmPublishCleanupFailure() =>
74        Interlocked.Exchange(ref _publishCleanupFailureArmed, 0);
75
76    internal bool IsPublishCleanupFailureArmed() =>
77        Volatile.Read(ref _publishCleanupFailureArmed) != 0;
78
79    internal int GetPublishCleanupFailureCount() =>
80        Volatile.Read(ref _publishCleanupFailureCount);
81}

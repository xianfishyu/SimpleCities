229    partial void ProbeGetPublishCaptureGate(ref Task? gate)
230    {
231        if (Interlocked.Exchange(ref _publishCaptureGateArmed, 0) == 0)
232            return;
233
234        Interlocked.Increment(ref _publishCaptureGateTriggerCount);
235        Volatile.Write(ref _publishCaptureGateEntered, 1);
236        TaskCompletionSource release = Volatile.Read(ref _publishCaptureGateRelease);
237        gate = WaitAtPublishCaptureAsync(release.Task);
238    }

240    private async Task WaitAtPublishCaptureAsync(Task release)
241    {
242        try
243        {
244            try
245            {
246                await release.WaitAsync(PublishOperationGateTimeout);
247            }
248            catch (TimeoutException)
249            {
250                throw new TimeoutException(PublishCaptureGateTimeoutMessage);
251            }
252        }
253        finally
254        {
255            Volatile.Write(ref _publishCaptureGateEntered, 0);
256        }
257    }

259    partial void ProbeWaitAtPublishPrepare(ref SaveSlotStore store)
260    {
261        if (Volatile.Read(ref _publishStagedGateArmed) != 0)
262            store = new SaveSlotStore(store.SaveBaseDirectory, WaitAtPublishStaged);
263        else if (Volatile.Read(ref _publishPreCommitBoundaryGateArmed) != 0)
264            store = new SaveSlotStore(store.SaveBaseDirectory, WaitAtPublishPreCommitBoundary);
265        else if (Volatile.Read(ref _publishPostCommitGateArmed) != 0)
266            store = new SaveSlotStore(store.SaveBaseDirectory, WaitAtPublishPostCommit);
267
268        if (Interlocked.Exchange(ref _publishPrepareGateArmed, 0) == 0)
269            return;
270
271        Interlocked.Increment(ref _publishPrepareGateTriggerCount);
272        Volatile.Write(ref _publishPrepareGateEntered, 1);
273        try
274        {
275            if (!_publishPrepareGateRelease.Wait(PublishOperationGateTimeout))
276                throw new TimeoutException(PublishPrepareGateTimeoutMessage);
277        }
278        finally
279        {
280            Volatile.Write(ref _publishPrepareGateEntered, 0);
281        }
282    }

284    internal void ArmNextPublishCaptureGate()
285    {
286        if (IsOperationBusy)
287        {
288            throw new InvalidOperationException(
289                "SaveManager must be idle before arming its publish Capture test gate.");
290        }
291        if (Interlocked.CompareExchange(ref _publishCaptureGateArmed, 1, 0) != 0)
292        {
293            throw new InvalidOperationException(
294                "Save publish Capture test gate is already armed.");
295        }
296
297        Interlocked.Exchange(ref _publishCaptureGateTriggerCount, 0);
298        Interlocked.Exchange(ref _publishCaptureGateCancelRequestCount, 0);
299        _publishCaptureGateCancelOperationToken = string.Empty;
300        Volatile.Write(ref _publishCaptureGateEntered, 0);
301        Interlocked.Exchange(
302            ref _publishCaptureGateRelease,
303            new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously));
304    }

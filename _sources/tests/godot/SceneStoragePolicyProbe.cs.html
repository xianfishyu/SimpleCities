1using Godot;
2using System;
3using System.IO;
4using System.Linq;

6/// <summary>跨保存根删除授权回归的测试装配，仅编译进Debug。</summary>
7public partial class SceneStoragePolicyProbe : RefCounted
8{
9    private readonly string _root = Path.Combine(
10        Path.GetTempPath(), $"simple-cities-root-auth-{Guid.NewGuid():N}");
11    private SaveSlotSummary? _captured;
12    public string SlotID => "manual-shared";
13    public string RootA => Path.Combine(_root, "a");
14    public string RootB => Path.Combine(_root, "b");
15
16    public void CreateMatchingSlots(RoadMapScene map)
17    {
18        new SaveSlotStore(RootA).Save(SlotID, "Root authorization fixture", [map.QaRoadParticipant]);
19        string source = Path.Combine(RootA, SlotID);
20        string destination = Path.Combine(RootB, SlotID);
21        Directory.CreateDirectory(destination);
22        foreach (string file in Directory.GetFiles(source))
23            File.Copy(file, Path.Combine(destination, Path.GetFileName(file)));
24    }
25
26    public bool BindRoot(SaveManager manager, RoadMapScene map, string root) =>
27        map.BindQaStorage(manager, new SceneStoragePolicy(root, [map.QaRoadParticipant.SaveFileName]));
28
29    public string CaptureAndAuthorize(SaveManager manager)
30    {
31        _captured = manager.ListSlots().Single(slot => slot.SlotID == SlotID);
32        return manager.ArmDeletion(_captured);
33    }
34
35    public string RearmCapturedSummary(SaveManager manager) =>
36        manager.ArmDeletion(_captured ?? throw new InvalidOperationException("No slot summary was captured."));
37
38    public bool BothSlotsExist() => Directory.Exists(Path.Combine(RootA, SlotID)) &&
39        Directory.Exists(Path.Combine(RootB, SlotID));
40
41    public void Cleanup(SaveManager manager, RoadMapScene map)
42    {
43        if (!map.BindQaStorage(manager, RoadSaveParticipant.Storage))
44            throw new InvalidOperationException("Could not restore the V4 scene storage.");
45        string directory = Path.GetFullPath(_root);
46        string prefix = Path.Combine(Path.GetFullPath(Path.GetTempPath()), "simple-cities-root-auth-");
47        if (!directory.StartsWith(prefix, StringComparison.OrdinalIgnoreCase))
48            throw new InvalidOperationException("Unexpected temporary storage root.");
49        if (Directory.Exists(directory))
50            Directory.Delete(directory, recursive: true);
51    }
52}
53
54public partial class RoadMapScene
55{
56    internal RoadSaveParticipant QaRoadParticipant => _roads ??
57        throw new InvalidOperationException("The V4 scene is not ready.");
58    internal bool BindQaStorage(SaveManager manager, SceneStoragePolicy policy)
59    {
60        manager.UnregisterSceneLoad(this);
61        return manager.RegisterSceneLoad(new SceneLoadParticipants(QaRoadParticipant, this, _view, policy));
62    }
63}

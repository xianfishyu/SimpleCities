24func run() -> void:
25	var map: Node = await instantiate_scene(MAP_SCENE)
26	if map != null:
27		var autosave_controller: Node = map.get_node("AutosaveController")
28		autosave_controller.set("AutosaveEnabled", false)

42	await verify_sub_milliscale_survives_ready_normalization()

77func verify_zoom_settings_are_editor_visible(camera: Camera2D, scene_name: String) -> void:
78	var editor_property_names: Array[String] = []
79	var editor_property_hints: Dictionary = {}
80	for property: Dictionary in camera.get_property_list():
81		if int(property.usage) & PROPERTY_USAGE_EDITOR:
82			var property_name := str(property.name)
83			editor_property_names.append(property_name)
84			editor_property_hints[property_name] = str(property.hint_string)
85
86	for property_name in [
87		"smoothing",
88		"referenceFps",
89		"panSpeed",
90		"zoomInfluence",
91		"accelerationTime",
92		"decelerationTime",
93	]:
94		require(editor_property_names.has(property_name),
95			"%s does not expose %s in the Godot Inspector" % [scene_name, property_name])
96
97	require(str(editor_property_hints.get("minScale", "")).begins_with("0.000001,"),
98		"%s minScale Inspector range truncates sub-milliscale values" % scene_name)

101func verify_wheel_event_reaches_camera(camera: Camera2D, scene_name: String) -> void:
102	camera.zoom = Vector2.ONE
103	camera.set("defaultScale", 1.0)
104
105	Input.parse_input_event(wheel_event(MOUSE_BUTTON_WHEEL_UP))
106	await process_frame

108
109	var target_zoom := float(camera.get("defaultScale"))
110	require(is_equal_approx(target_zoom, 1.125), "%s wheel-up did not update MainCamera's zoom target" % scene_name)
111	require(camera.zoom.x > 1.0 and camera.zoom.x < target_zoom,
112		"%s wheel-up did not begin smooth zoom convergence within two frames" % scene_name)

115func verify_zoom_is_frame_rate_independent(camera: Camera2D, scene_name: String) -> void:
116	var low_rate_zoom := zoom_after_one_second(camera, LOW_FRAME_RATE)
117	var high_rate_zoom := zoom_after_one_second(camera, HIGH_FRAME_RATE)
118	var difference := absf(high_rate_zoom - low_rate_zoom)
119
120	print("Camera zoom one-second convergence (%s): %d FPS=%.6f, %d FPS=%.6f, difference=%.6f" % [
121		scene_name,
122		LOW_FRAME_RATE,
123		low_rate_zoom,
124		HIGH_FRAME_RATE,
125		high_rate_zoom,
126		difference,
127	])
128	require(difference <= MAX_ONE_SECOND_ZOOM_DIFFERENCE,
129		"%s zoom smoothing depends on frame rate; delta=%.6f exceeds %.6f" % [
130			scene_name,
131			difference,
132			MAX_ONE_SECOND_ZOOM_DIFFERENCE,
133		])

155func verify_keyboard_pan_follows_zoom_influence() -> void:
156	const INFLUENCE := 0.75
157	var zoomed_out_screen_distance := await screen_pan_after_one_second(0.125, INFLUENCE)
158	var zoomed_in_screen_distance := await screen_pan_after_one_second(4.0, INFLUENCE)
159	var zoomed_out_normalized := zoomed_out_screen_distance / pow(0.125, 1.0 - INFLUENCE)
160	var zoomed_in_normalized := zoomed_in_screen_distance / pow(4.0, 1.0 - INFLUENCE)
161	var difference := absf(zoomed_out_normalized - zoomed_in_normalized)
162
163	print("Camera pan zoom influence: zoom 0.125=%.6f, zoom 4.0=%.6f, normalized difference=%.6f" % [
164		zoomed_out_screen_distance,
165		zoomed_in_screen_distance,
166		difference,
167	])
168	require(zoomed_in_screen_distance > zoomed_out_screen_distance,
169		"zoomInfluence did not produce the expected hybrid pan-speed curve")
170	require(difference <= MAX_NORMALIZED_PAN_DIFFERENCE_ACROSS_ZOOM,
171		"keyboard pan does not follow zoomInfluence; delta=%.6f exceeds %.6f" % [
172			difference,
173			MAX_NORMALIZED_PAN_DIFFERENCE_ACROSS_ZOOM,
174		])

206func verify_middle_drag_stops_keyboard_pan() -> void:
207	var scene: Node = await instantiate_scene(ISOLATED_CAMERA_SCENE)
208	if scene == null:
209		return
210
211	var camera: Camera2D = scene.get_node("Camera2D")
212	Input.action_press("KeyBoard_MoveRight")
213	camera._UnhandledInput(InputEventKey.new())
214	var middle_press := wheel_event(MOUSE_BUTTON_MIDDLE)
215	camera._UnhandledInput(middle_press)
216	var position_before_process := camera.position
217	camera._Process(1.0 / 60.0)
218	require(camera.position.is_equal_approx(position_before_process),
219		"middle drag allowed stale keyboard input to move the camera")
220
221	var middle_release := wheel_event(MOUSE_BUTTON_MIDDLE)
222	middle_release.pressed = false
223	Input.action_release("KeyBoard_MoveRight")
224	camera._UnhandledInput(middle_release)
225	camera._Process(1.0 / 60.0)
226	require(camera.position.is_equal_approx(position_before_process),
227		"middle drag left stale keyboard velocity after release")
228	await free_scene(scene)

231func verify_zoom_bounds_apply_to_restore_and_wheel_input() -> void:
232	var scene: Node = await instantiate_scene(ISOLATED_CAMERA_SCENE)
233	if scene == null:
234		return

241		"restore allowed zoom above maxScale")
242	camera.call("RestoreState", "{\"PositionX\":0,\"PositionY\":0,\"Zoom\":0.001}")
243	require(is_equal_approx(float(camera.get("defaultScale")), 0.009),
244		"restore allowed zoom below minScale")
245
246	camera.set("defaultScale", 1.99)
247	camera._UnhandledInput(wheel_event(MOUSE_BUTTON_WHEEL_UP))
248	require(is_equal_approx(float(camera.get("defaultScale")), 2.0),
249		"wheel input exceeded maxScale")
250	await free_scene(scene)

294func verify_wheel_zoom_preserves_mouse_anchor() -> void:
295	var scene: Node = await instantiate_scene(ISOLATED_CAMERA_SCENE)
296	if scene == null:
297		return
298
299	var camera: Camera2D = scene.get_node("Camera2D")
300	camera.position = Vector2(100.0, 200.0)
301	camera.zoom = Vector2.ONE
302	camera.set("defaultScale", 1.0)
303	var viewport_rect := camera.get_viewport().get_visible_rect()
304	var viewport_center := viewport_rect.position + viewport_rect.size * 0.5
305	var anchor_position := viewport_rect.position + viewport_rect.size * Vector2(0.75, 0.35)
306	var world_before := camera.position + (anchor_position - viewport_center) / camera.zoom
307	var event := wheel_event(MOUSE_BUTTON_WHEEL_UP)
308	event.position = anchor_position
309	camera._UnhandledInput(event)
310	camera._Process(1.0 / 60.0)
311	var world_after := camera.position + (anchor_position - viewport_center) / camera.zoom
312
313	require(world_before.distance_to(world_after) <= 0.001,
314		"wheel zoom moved the world point under the mouse anchor")
315	await free_scene(scene)

318func pan_after_one_second(frame_rate: int) -> float:
319	var scene: Node = await instantiate_scene(ISOLATED_CAMERA_SCENE)
320	if scene == null:
321		return 0.0
322
323	var camera: Camera2D = scene.get_node("Camera2D")
324	camera.position = Vector2.ZERO
325	camera.set("defaultScale", 1.0)
326	Input.action_press("KeyBoard_MoveRight")
327	camera._UnhandledInput(InputEventKey.new())
328	var delta := 1.0 / float(frame_rate)
329	for frame_index in range(frame_rate):
330		camera._Process(delta)
331	Input.action_release("KeyBoard_MoveRight")
332	camera._UnhandledInput(InputEventKey.new())
333	var position_x := camera.position.x
334	await free_scene(scene)
335	return position_x

338func screen_pan_after_one_second(zoom_scale: float, zoom_influence: float = 0.75) -> float:
339	var scene: Node = await instantiate_scene(ISOLATED_CAMERA_SCENE)
340	if scene == null:
341		return 0.0
342
343	var camera: Camera2D = scene.get_node("Camera2D")
344	camera.position = Vector2.ZERO
345	camera.zoom = Vector2.ONE * zoom_scale
346	camera.set("defaultScale", zoom_scale)
347	camera.set("zoomInfluence", zoom_influence)
348	Input.action_press("KeyBoard_MoveRight")
349	camera._UnhandledInput(InputEventKey.new())
350	for frame_index in range(60):
351		camera._Process(1.0 / 60.0)
352	Input.action_release("KeyBoard_MoveRight")
353	camera._UnhandledInput(InputEventKey.new())
354	var screen_distance := camera.position.x * zoom_scale
355	await free_scene(scene)
356	return screen_distance

359func screen_pan_sample_after_one_frame(
360	start_zoom: float,
361	target_zoom: float,
362	acceleration_time: float,
363) -> Vector2:
364	var scene: Node = await instantiate_scene(ISOLATED_CAMERA_SCENE)
365	if scene == null:
366		return Vector2.ZERO
367
368	var camera: Camera2D = scene.get_node("Camera2D")
369	camera.position = Vector2.ZERO
370	camera.zoom = Vector2.ONE * start_zoom
371	camera.set("defaultScale", target_zoom)
372	camera.set("accelerationTime", acceleration_time)
373	Input.action_press("KeyBoard_MoveRight")
374	camera._UnhandledInput(InputEventKey.new())
375	camera._Process(1.0 / 60.0)
376	Input.action_release("KeyBoard_MoveRight")
377	camera._UnhandledInput(InputEventKey.new())
378	var screen_distance := camera.position.x * camera.zoom.x
379	var rendered_zoom := camera.zoom.x
380	await free_scene(scene)
381	return Vector2(screen_distance, rendered_zoom)

393func wheel_event(button: MouseButton) -> InputEventMouseButton:
394	var event := InputEventMouseButton.new()
395	event.button_index = button

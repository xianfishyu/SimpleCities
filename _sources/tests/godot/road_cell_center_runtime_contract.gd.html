95func focus(map: Node, world: Vector2, zoom: float) -> void:
96	var camera: Camera2D = map.get_node("Camera2D")
97	camera.zoom = Vector2(zoom, zoom)
98	# Keep all raw mouse gestures clear of the left-hand control panel.
99	camera.position = world - (Vector2(1000, 450) - root.get_visible_rect().size / 2.0) / zoom

101func junction_at(state: Dictionary, position: Vector2) -> Dictionary:
102	var found: Array = state.get("junctions", []).filter(func(item): return item.get("position") == position)
103	return found[0] if found.size() == 1 else {}

105func junction_matches(map: Node, center: Vector2, nodes: int, branches: int) -> bool:
106	var state: Dictionary = map.GetRoadState()
107	var junction: Dictionary = junction_at(state, center)
108	var incidences: Array = junction.get("incidences", [])
109	var turns: Array = junction.get("turns", [])
110	if state.get("nodeCount", 0) != nodes or state.get("edges", []).size() != branches or state.get("junctions", []).size() != 1 or incidences.size() != branches or turns.size() != branches * branches:
111		return false
112	var keys: Dictionary = {}
113	var directions: Array = []
114	var previous_bearing: float = -1.0
115	for incidence in incidences:
116		var edge_id: int = incidence.get("edgeId", 0)
117		var role: String = incidence.get("role", "")
118		var outward: Vector2 = incidence.get("outward", Vector2.ZERO)
119		var bearing: float = incidence.get("bearingRadians", -1.0)
120		var key: String = "%d:%s" % [edge_id, role]
121		if edge_id <= 0 or role not in ["Start", "End"] or keys.has(key) or directions.has(outward) or bearing < previous_bearing or not is_equal_approx(absf(outward.x), sqrt(0.5)) or not is_equal_approx(absf(outward.y), sqrt(0.5)):
122			return false
123		keys[key] = true
124		directions.append(outward)
125		previous_bearing = bearing
126	var turn_keys: Dictionary = {}
127	for turn in turns:
128		var from_key: String = "%d:%s" % [turn.get("fromEdgeId", 0), turn.get("fromRole", "")]
129		var to_key: String = "%d:%s" % [turn.get("toEdgeId", 0), turn.get("toRole", "")]
130		var pair: String = from_key + ">" + to_key
131		if not keys.has(from_key) or not keys.has(to_key) or turn_keys.has(pair): return false
132		turn_keys[pair] = true
133	var hit: Dictionary = map.PickRoad(center)
134	var owners: Array = incidences.filter(func(item): return item.get("edgeId") == hit.get("edgeId"))
135	var node_id: int = junction.get("nodeId", 0)
136	return map.IsPresentationCurrent and state.get("sourceToken") == map.StateToken and junction.get("sourceToken") == map.StateToken and node_id > 0 and map.GetJunctionState(node_id) == junction and hit.get("sourceToken") == map.StateToken and hit.get("junctionNodeId", 0) == node_id and hit.get("surfaceCenter") == center and owners.size() == 1

138func rejected_drag(map: Node, start: Vector2, end: Vector2, reason: String, conflict: bool) -> bool:
139	var before: Dictionary = map.GetRoadState()
140	var token: String = map.StateToken
141	press(map, start)
142	motion(map, end)
143	var ready: bool = await wait_preview(map)
144	var preview: Dictionary = map.GetBuildPreview()
145	var preview_reason: String = preview.get("reason", "")
146	var segments: Array = preview.get("segments", [])
147	var red: Array = segments.filter(func(segment): return segment.get("conflict", false))
148	var rejected: bool = ready and preview.get("phase") == "Rejected" and preview.get("sourceToken") == token and preview_reason.contains(reason) and (not conflict or red.size() > 0) and map.StateToken == token
149	release(map, end)
150	var idle: bool = await wait_idle(map)
151	var status: String = map.get_node(CONTROLS + "Status").text
152	return rejected and idle and status.contains(reason) and map.StateToken == token and map.GetRoadState() == before and map.IsPresentationCurrent

154func lineage(token: String) -> String:
155	return token.get_slice("Lineage = ", 1).get_slice(",", 0).strip_edges()

157func build(map: Node, start: Vector2, end: Vector2) -> bool:
158	var token: String = map.StateToken
159	press(map, start)
160	motion(map, end)
161	release(map, end)
162	var idle: bool = await wait_idle(map)
163	return idle and map.StateToken != token and map.IsPresentationCurrent

171func button(position: Vector2, pressed: bool) -> void:
172	var event := InputEventMouseButton.new()
173	event.position = position
174	event.button_index = MOUSE_BUTTON_LEFT
175	event.pressed = pressed
176	Input.parse_input_event(event)
177
178func motion(map: Node, point: Vector2) -> void:
179	var event := InputEventMouseMotion.new()
180	event.position = map.get_canvas_transform() * point
181	event.button_mask = MOUSE_BUTTON_MASK_LEFT
182	Input.parse_input_event(event)

184func wait_idle(map: Node) -> bool:
185	var deadline: int = Time.get_ticks_msec() + 5000
186	while Time.get_ticks_msec() < deadline:
187		await process_frame
188		if not map.IsBuildBusy: return true
189	return false

191func wait_preview(map: Node) -> bool:
192	var deadline: int = Time.get_ticks_msec() + 5000
193	while Time.get_ticks_msec() < deadline:
194		await process_frame
195		if map.GetBuildPreview().get("phase") != "Pending": return true
196	return false

116func road_pixels(map: Node) -> PackedByteArray:
117	# Include all three possible road rows while excluding the changing status panel.
118	var center: Vector2 = map.get_canvas_transform() * Vector2(250, 0)
119	var rect := Rect2i(Vector2i(center) - Vector2i(120, 260), Vector2i(240, 520))
120	var image: Image = root.get_texture().get_image()
121	if not Rect2i(Vector2i.ZERO, image.get_size()).encloses(rect): return PackedByteArray()
122	return image.get_region(rect).get_data()

135func retry_failures_and_success(map: Node, manager: Node, slot: String, road: Dictionary, history: Dictionary) -> void:
136	var payload_path: String = "user://saves-v4/" + slot + "/road_network_v4.json"
137	var saved_bytes: PackedByteArray = FileAccess.get_file_as_bytes(payload_path) if not slot.is_empty() else PackedByteArray()
138	var retry: Button = map.get_node(DISPLAY_CONTROLS + "DisplayRecovery/Retry")

156	var save_button: Button = map.get_node(DISPLAY_CONTROLS + "Save")
157	check("retry_keeps_camera_and_save_available", not save_button.disabled and await pan_camera(map) and await SAVE.save(manager, slot) and FileAccess.get_file_as_bytes(payload_path) == saved_bytes and map.GetRoadState() == road and map.GetHistoryState() == history)

160	check("retry_worker_failure_preserves_committed_state_and_allows_retry", gate.Completed and presentation(map).get("phase") == "Failed" and recovery_visible(map) and map.GetRoadState() == road and map.GetHistoryState() == history)

169	gate = DISPLAY_PROBE.new()
170	gate.InstallRetryGate(map, false)
171	await click_control(retry)
172	await wait_worker(gate)
173	check("successful_retry_still_uses_single_worker", gate.Started and gate.UsedBackgroundThread and gate.RetryCalls == 1 and not map.RetryRoadDisplay())
174	gate.Release()

176	await RenderingServer.frame_post_draw

179	check("retry_success_does_not_keep_stale_cancel_hint", not status.text.contains("Esc") and not status.text.contains("取消"), status.text)
180	check("retry_does_not_recommit_or_change_codec_bytes", map.GetRoadState() == road and map.GetHistoryState() == history and await SAVE.save(manager, slot) and not saved_bytes.is_empty() and FileAccess.get_file_as_bytes(payload_path) == saved_bytes)
181	check("recovered_road_can_be_picked_and_current_retry_is_noop", map.PickRoad(Vector2(250, 200)).get("sourceToken") == map.StateToken and map.SetToolMode(1) and not map.RetryRoadDisplay() and map.GetRoadState() == road and map.GetHistoryState() == history)

183	check("recovered_history_controls_are_enabled", not map.get_node(DISPLAY_CONTROLS + "History/Undo").disabled)

216func late_retry_is_discarded(map: Node, manager: Node, slot: String, target: Dictionary, replacement: String, late_failure: bool) -> void:
217	var label: String = replacement + ("_late_failure" if late_failure else "_late_success")
218	check(label + "_fixture_map_created", map.CreateMap(100) and map.SetToolMode(0))
219	focus(map, Vector2(250, 0), 1.2)
220	await process_frame
221	check(label + "_fixture_baseline_built", await build(map, Vector2.ZERO, Vector2(500, 0)))
222	var fault = DISPLAY_PROBE.new()
223	fault.Install(map, "publish", false)
224	await request_build(map, Vector2(0, 200), Vector2(500, 200))
225	check(label + "_fixture_display_failed", fault.FaultCalls == 1 and presentation(map).get("phase") == "Failed")
226	fault.Cleanup()
227	var gate = DISPLAY_PROBE.new()
228	gate.InstallRetryGate(map, late_failure)
229	await click_control(map.get_node(DISPLAY_CONTROLS + "DisplayRecovery/Retry"))
230	await wait_worker(gate)
231	check(label + "_worker_is_pending", gate.Started and map.IsDisplayRetryBusy and presentation(map).get("phase") == "Preparing")
232	var replaced: bool = map.CreateMap(50) if replacement == "new_map" else await SAVE.operation_succeeded(manager, map.LoadSlot(slot))
233	await process_frame
234	await RenderingServer.frame_post_draw
235	var road: Dictionary = map.GetRoadState()
236	var history: Dictionary = map.GetHistoryState()
237	var token: String = map.StateToken
238	check(label + "_replacement_publishes_while_old_worker_waits", replaced and map.IsPresentationCurrent and presentation(map).get("phase") == "Current" and history_counts(map, 0, 0) and (map.CellSizeMetres == 50 and road.is_empty() if replacement == "new_map" else content_matches(map, target)))
239	gate.Release()
240	await wait_worker_completed(gate)
241	await wait_retry_finished(map)
242	for _frame in range(6):
243		await process_frame
244	var after: Dictionary = presentation(map)
245	check(label + "_cannot_publish_or_report_error_into_replacement", gate.Completed and map.StateToken == token and map.GetRoadState() == road and map.GetHistoryState() == history and after.get("phase") == "Current" and after.get("error", "missing").is_empty() and after.get("presentedToken") == token and after.get("drawnToken") == token and not map.IsDisplayRetryBusy and not map.get_node(DISPLAY_CONTROLS + "DisplayRecovery").visible, after)
246	await cleanup_gate(gate)

276func wait_worker(probe: RefCounted) -> bool:
277	var deadline: int = Time.get_ticks_msec() + 5000
278	while not probe.Started and Time.get_ticks_msec() < deadline:
279		await process_frame
280	return probe.Started

282func wait_worker_completed(probe: RefCounted) -> bool:
283	var deadline: int = Time.get_ticks_msec() + 5000
284	while not probe.Completed and Time.get_ticks_msec() < deadline:
285		await process_frame
286	return probe.Completed

288func cleanup_gate(probe: RefCounted) -> void:
289	probe.Release()
290	if probe.Started:
291		await wait_worker_completed(probe)
292	if not probe.Started or probe.Completed:
293		probe.Cleanup()
294	else:
295		check("display_retry_worker_cleanup_timeout", false)

297func wait_retry_finished(map: Node) -> bool:
298	var deadline: int = Time.get_ticks_msec() + 5000
299	while Time.get_ticks_msec() < deadline:
300		await process_frame
301		if not map.IsDisplayRetryBusy and presentation(map).get("phase") not in ["Preparing", "AwaitingDraw"]: return true
302	return false

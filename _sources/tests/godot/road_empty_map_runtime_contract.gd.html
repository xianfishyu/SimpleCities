84func rejects_invalid_payload(map: Node, manager: Node, slot: String) -> bool:
85	var payload_path := "user://saves-v4/" + slot + "/road_network_v4.json"
86	var original := FileAccess.get_file_as_string(payload_path)
87	var passed := true
88	for replacement in [["\"schemaVersion\":6", "\"schemaVersion\":1"], ["\"cellSizeMetres\":50", "\"cellSizeMetres\":75"]]:
89		var before: String = map.StateToken
90		var changed := original.replace(replacement[0], replacement[1])
91		passed = publish_fixture(slot, changed) and passed
92		var result := await SAVE.wait_for_operation(manager, map.LoadSlot(slot))
93		await SAVE.wait_for_idle(manager)
94		await process_frame
95		var rejected: bool = not result.get("committed", true) and result.get("resultKind", -1) == 2
96		var intact: bool = map.StateToken == before and map.CellSizeMetres == 50 and map.IsPresentationCurrent
97		print("V4_INVALID_LOAD_RESULT ", JSON.stringify({"field": replacement[0], "rejected": rejected, "intact": intact, "error": result.get("error", "")}))
98		passed = passed and rejected and intact
99	passed = publish_fixture(slot, original) and passed
100	return passed

102func publish_fixture(slot: String, text: String) -> bool:
103	# Only the fresh slot created in this test is edited. Refresh integrity metadata
104	# so malformed content reaches the actual V4 reader instead of failing the hash gate.
105	var directory := "user://saves-v4/" + slot + "/"
106	var file := FileAccess.open(directory + "road_network_v4.json", FileAccess.WRITE)
107	if file == null:
108		return false
109	file.store_string(text)
110	file.close()
111	var manifest: Dictionary = JSON.parse_string(FileAccess.get_file_as_string(directory + "manifest.json"))
112	manifest["schemaVersion"] = int(manifest["schemaVersion"])
113	manifest["files"][0]["encodedLength"] = text.to_utf8_buffer().size()
114	manifest["files"][0]["sha256"] = text.sha256_text()
115	var output := FileAccess.open(directory + "manifest.json", FileAccess.WRITE)
116	if output == null:
117		return false
118	output.store_string(JSON.stringify(manifest))
119	output.close()
120	return true

8func run() -> void:

22	var horizontal: Dictionary = map.PickRoad(Vector2(200, 0))
23	var vertical: Dictionary = map.PickRoad(Vector2(400, 150))
24	var joined: bool = not map.PickRoad(Vector2(402, -2)).is_empty()
25	var positions: bool = horizontal.get("profile", "") == "street" and vertical.get("profile", "") == "street" and absf(horizontal.get("parameter", -1.0) - 2.0 / 7.0) < 0.00001 and absf(vertical.get("parameter", -1.0) - 5.5 / 7.0) < 0.00001
26	passed = passed and merged and joined and positions
27	passed = await build(map, Vector2(-300, -300), Vector2.ZERO) and passed
28	var reverse: Dictionary = map.GetRoadState()
29	var reverse_ok: bool = reverse.get("nodeCount", 0) == 2 and reverse["edges"][0]["points"].size() == 4
30	map.get_node("HUD/Panel/Margin/Controls/Profile").select(3)
31	passed = await build(map, Vector2(-300, -300), Vector2(-300, -600)) and passed
32	var before: Dictionary = map.GetRoadState()
33	var type_boundary: bool = before.get("nodeCount", 0) == 3 and before["edges"].size() == 2 and map.PickRoad(Vector2(-300, -450)).get("profile", "") == "highway"
34	passed = passed and reverse_ok and type_boundary
35	var saved := await SAVE.save_as(manager, "V4 turning road QA")
36	var slot: String = manager.CurrentSlotID
37	map.CreateMap(25)
38	var loaded := await SAVE.operation_succeeded(manager, map.LoadSlot(slot))
39	await process_frame
40	await RenderingServer.frame_post_draw

42	var round_trip: bool = loaded and map.CellSizeMetres == 100 and after.get("edges") == before.get("edges") and after.get("sourceToken") != before.get("sourceToken") and map.IsPresentationCurrent
43	passed = passed and saved and round_trip

66func build(map: Node, start: Vector2, end: Vector2) -> bool:
67	var before: String = map.StateToken
68	var transform: Transform2D = map.get_canvas_transform()
69	button(transform * start, true)
70	var motion := InputEventMouseMotion.new()
71	motion.position = transform * end
72	motion.button_mask = MOUSE_BUTTON_MASK_LEFT
73	Input.parse_input_event(motion)
74	button(transform * end, false)
75	var deadline := Time.get_ticks_msec() + 5000
76	while Time.get_ticks_msec() < deadline:
77		await process_frame
78		if not map.IsBuildBusy:
79			return map.StateToken != before and map.IsPresentationCurrent
80	return false

59func batch_history(map: Node, built: Dictionary) -> void:

61	var profile: OptionButton = map.get_node("HUD/Panel/Margin/Controls/Profile")

68	release(map, Vector2(350, 0))
69	await settle(map)
70	var changed: Dictionary = map.GetRoadState()
71	check("batch_change_adds_one_entry", history_counts(map, 2, 0) and [150, 250, 350].all(func(x): return map.PickRoad(Vector2(x, 0)).get("profile") == "highway") and [50, 450].all(func(x): return map.PickRoad(Vector2(x, 0)).get("profile") == "street"))
72	shortcut(KEY_Z)

74	check("one_undo_restores_entire_profile_batch", content_matches(map, built) and history_counts(map, 1, 1))

76	await settle(map)
77	check("one_redo_restores_entire_profile_batch", content_matches(map, changed) and history_counts(map, 2, 0))
78	# Undo/redo restores content under a fresh sequence. No-change must preserve this
79	# newly published snapshot, including its token, rather than the pre-undo token.

81	var history_before: Dictionary = map.GetHistoryState()
82	var phase_before: String = map.BuildPhase

86	await process_frame
87	release(map, Vector2(350, 0))

92	await process_frame

94	await process_frame
95	check("held_delete_crosses_profile_edges_without_history_write", map.GetSelectionState().get("selectedCount") == 3 and map.GetRoadState() == changed and map.GetHistoryState() == history_before)

98	var deleted: Dictionary = map.GetRoadState()
99	check("batch_delete_adds_one_entry_and_preserves_remainder", history_counts(map, 3, 0) and [50, 150, 250].all(func(x): return map.PickRoad(Vector2(x, 0)).is_empty()) and map.PickRoad(Vector2(350, 0)).get("profile") == "highway" and map.PickRoad(Vector2(450, 0)).get("profile") == "street")

102	check("one_undo_restores_entire_delete_batch", content_matches(map, changed) and history_counts(map, 2, 1))
103	shortcut(KEY_Z, true)
104	await settle(map)
105	check("one_redo_restores_entire_delete_batch", content_matches(map, deleted) and history_counts(map, 3, 0))
106	shortcut(KEY_Z)
107	await settle(map)
108	check("new_build_after_undo_clears_redo", map.SetToolMode(0) and await build(map, Vector2(0, 200), Vector2(500, 200)) and history_counts(map, 3, 0) and not map.RedoRoadEdit() and map.PickRoad(Vector2(250, 200)).get("profile") == "highway")

110func cancelled_history(map: Node) -> void:
111	var before: Dictionary = map.GetRoadState()
112	var history_before: Dictionary = map.GetHistoryState()
113	var probe = WORK_PROBE.new()
114	probe.Install(map)
115	var accepted: bool = map.UndoRoadEdit()
116	var deadline: int = Time.get_ticks_msec() + 5000
117	while Time.get_ticks_msec() < deadline and (not probe.Started or map.BuildElapsedMilliseconds <= 350):
118		await process_frame
119	var status: Label = map.get_node("HUD/Panel/Margin/Controls/Status")
120	var undo: Button = map.get_node("HUD/Panel/Margin/Controls/History/Undo")
121	var redo: Button = map.get_node("HUD/Panel/Margin/Controls/History/Redo")
122	check("pending_undo_keeps_history_and_shows_escape_hint", accepted and probe.Started and probe.UsedBackgroundThread and map.IsBuildBusy and map.BuildPhase == "Preparing" and status.text.contains("等待") and status.text.contains("Esc") and map.GetHistoryState() == history_before and map.GetRoadState() == before)
123	check("pending_undo_rejects_second_history_operation", undo.disabled and redo.disabled and not map.UndoRoadEdit() and not map.RedoRoadEdit() and map.GetHistoryState() == history_before)
124	escape()
125	await process_frame
126	check("escape_cancels_pending_undo_without_moving_stacks", map.BuildPhase == "Cancelling" and map.IsBuildBusy and map.GetHistoryState() == history_before and map.GetRoadState() == before)
127	probe.Release()
128	var completed: bool = await settle(map)
129	probe.Remove(map)
130	check("cancelled_undo_rejects_late_result_and_keeps_all_history", completed and probe.Completed and map.BuildPhase == "Cancelled" and map.GetHistoryState() == history_before and map.GetRoadState() == before and not undo.disabled)
131	if probe.Completed: probe.Cleanup()
132	shortcut(KEY_Z)
133	await settle(map)
134	var committed: Dictionary = map.GetHistoryState()
135	escape()
136	await process_frame
137	check("escape_after_completed_undo_does_not_undo_again", history_counts(map, 2, 1) and map.GetHistoryState() == committed and map.PickRoad(Vector2(250, 200)).is_empty() and map.PickRoad(Vector2(250, 0)).get("profile") == "highway" and not status.text.contains("Esc"))

139func loop_history_and_load(map: Node) -> void:
140	check("new_map_resets_existing_history", map.SetToolMode(0) and map.CreateMap(100) and history_counts(map, 0, 0))
141	var profile: OptionButton = map.get_node("HUD/Panel/Margin/Controls/Profile")
142	profile.select(1)
143	focus(map, Vector2(300, 200), 0.6)
144	await process_frame
145	var built: bool = await build(map, Vector2.ZERO, Vector2(400, 0))
146	built = await build(map, Vector2(400, 0), Vector2(400, 400)) and built
147	built = await build(map, Vector2(400, 400), Vector2(0, 400)) and built
148	built = await build(map, Vector2(0, 400), Vector2.ZERO) and built
149	var pure: Dictionary = map.GetRoadState()
150	var edges: Array = pure.get("edges", [])
151	check("loop_gestures_have_four_entries", built and history_counts(map, 4, 0) and pure.get("nodeCount") == 1 and edges.size() == 1 and edges[0].get("startNodeId") == edges[0].get("endNodeId"))
152	check("branch_moves_loop_seam_as_one_edit", await build(map, Vector2(400, 400), Vector2(600, 400)) and history_counts(map, 5, 0) and map.GetRoadState().get("nodeCount") == 2 and map.GetRoadState().get("edges", []).size() == 2)
153	var branched: Dictionary = map.GetRoadState()
154	shortcut(KEY_Z)
155	await settle(map)
156	check("undo_restores_loop_seam_and_visible_owner", content_matches(map, pure) and history_counts(map, 4, 1) and map.PickRoad(Vector2(500, 400)).is_empty() and map.PickRoad(Vector2(200, 0)).get("sourceToken") == map.StateToken)
157	shortcut(KEY_Y)
158	await settle(map)
159	check("redo_restores_loop_branch_and_visible_owner", content_matches(map, branched) and history_counts(map, 5, 0) and map.PickRoad(Vector2(500, 400)).get("sourceToken") == map.StateToken)
160	shortcut(KEY_Z)
161	await settle(map)
162	var history_before: Dictionary = map.GetHistoryState()
163	var manager: Node = root.get_node("SaveManager")
164	var saved: bool = await SAVE.save_as(manager, "V4 history restored loop QA")
165	check("save_preserves_undone_loop_and_session_history", saved and content_matches(map, pure) and history_counts(map, 4, 1) and map.GetHistoryState() == history_before)
166	if not saved: return
167	var slot: String = manager.CurrentSlotID
168	var payload_path: String = "user://saves-v4/" + slot + "/road_network_v4.json"
169	var bytes: PackedByteArray = FileAccess.get_file_as_bytes(payload_path)
170	shortcut(KEY_Y)
171	await settle(map)
172	check("redo_after_save_restores_branch", content_matches(map, branched) and history_counts(map, 5, 0))
173	var loaded: bool = await SAVE.operation_succeeded(manager, map.LoadSlot(slot))
174	await process_frame
175	check("load_restores_saved_loop_with_empty_history", loaded and content_matches(map, pure) and history_counts(map, 0, 0) and map.GetHistoryState().get("estimatedRetainedBytes", -1) == 0 and map.StateToken != history_before.get("sourceToken") and not map.UndoRoadEdit() and not map.RedoRoadEdit())
176	var resaved: bool = await SAVE.save(manager, slot)
177	check("history_is_not_serialized_into_codec", resaved and not bytes.is_empty() and FileAccess.get_file_as_bytes(payload_path) == bytes)
178	var deleted: bool = await SAVE.delete_slot(manager, slot)
179	check("history_save_fixture_deleted", deleted and not DirAccess.dir_exists_absolute(ProjectSettings.globalize_path("user://saves-v4/" + slot)))

181func history_counts(map: Node, undo_count: int, redo_count: int) -> bool:
182	var state: Dictionary = map.GetHistoryState()
183	return state.get("undoCount", -1) == undo_count and state.get("redoCount", -1) == redo_count and state.get("retainedCount", -1) == undo_count + redo_count
184
185func history_transition_matches(before: Dictionary, after: Dictionary) -> bool:
186	return after.get("sourceToken") != before.get("sourceToken") and after.get("changeSequence", 0) == before.get("changeSequence", 0) + 1 and after.get("nextNodeId", 0) >= before.get("nextNodeId", 0) and after.get("nextEdgeId", 0) >= before.get("nextEdgeId", 0)
187
188func content_matches(map: Node, expected: Dictionary) -> bool:
189	var actual: Dictionary = map.GetRoadState()
190	return map.IsPresentationCurrent and actual.get("edges", []) == expected.get("edges", []) and actual.get("nodeCount", 0) == expected.get("nodeCount", 0)

198func shortcut(key: Key, shift: bool = false) -> void:
199	var event := InputEventKey.new()
200	event.keycode = key
201	event.ctrl_pressed = true
202	event.shift_pressed = shift
203	event.pressed = true
204	Input.parse_input_event(event)
205	var released := InputEventKey.new()
206	released.keycode = key
207	released.ctrl_pressed = true
208	released.shift_pressed = shift
209	Input.parse_input_event(released)

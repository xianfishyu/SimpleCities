115func mouse_button(position: Vector2, pressed: bool) -> void:
116	var event := InputEventMouseButton.new()
117	event.position = position
118	event.button_index = MOUSE_BUTTON_LEFT
119	event.pressed = pressed
120	Input.parse_input_event(event)
121
122func mouse_motion(position: Vector2) -> void:
123	var event := InputEventMouseMotion.new()
124	event.position = position
125	event.button_mask = MOUSE_BUTTON_MASK_LEFT
126	Input.parse_input_event(event)

128func wait_build(map: Node) -> bool:
129	return await wait_idle(map) and map.RoadCount == 1
130
131func wait_idle(map: Node) -> bool:
132	var deadline := Time.get_ticks_msec() + 5000
133	while Time.get_ticks_msec() < deadline:
134		await process_frame
135		if not map.IsBuildBusy:
136			return true
137	return false

96func closed_edge(edge: Dictionary) -> bool:
97	var points: Array = edge.get("points", [])
98	return edge.get("startNodeId", 0) > 0 and edge.get("startNodeId") == edge.get("endNodeId") and points.size() >= 4 and points[0] == points[-1]

100func self_loop_roles(read: Dictionary, edge_id: int) -> bool:
101	var incidences: Array = read.get("incidences", []).filter(func(item): return item.get("edgeId", 0) == edge_id)
102	return edge_id > 0 and incidences.size() == 2 and incidences[0].get("role") != incidences[1].get("role") and incidences[0].get("outward") != incidences[1].get("outward")

104func roles_match(map: Node, read: Dictionary, count: int) -> bool:
105	var incidences: Array = read.get("incidences", [])
106	var turns: Array = read.get("turns", [])
107	if read.get("nodeId", 0) <= 0 or read.get("sourceToken", "") != map.StateToken or not map.IsPresentationCurrent or incidences.size() != count or turns.size() != count * count:
108		return false
109	var keys: Dictionary = {}
110	var directions: Array = []
111	for incidence in incidences:
112		var role: String = incidence.get("role", "")
113		var key: String = "%d:%s" % [incidence.get("edgeId", 0), role]
114		var outward: Vector2 = incidence.get("outward", Vector2.ZERO)
115		if keys.has(key) or role not in ["Start", "End"] or directions.has(outward) or not is_equal_approx(outward.length(), 1.0): return false
116		keys[key] = true
117		directions.append(outward)
118	var pairs: Dictionary = {}
119	for turn in turns:
120		var from_key: String = "%d:%s" % [turn.get("fromEdgeId", 0), turn.get("fromRole", "")]
121		var to_key: String = "%d:%s" % [turn.get("toEdgeId", 0), turn.get("toRole", "")]
122		var pair: String = from_key + ">" + to_key
123		if not keys.has(from_key) or not keys.has(to_key) or pairs.has(pair): return false
124		pairs[pair] = true
125	return true

127func junction_surface_matches(map: Node, read: Dictionary) -> bool:
128	var hit: Dictionary = map.PickRoad(read.get("position", Vector2.INF))
129	var parameter: float = hit.get("parameter", -1.0)
130	var owners: Array = read.get("incidences", []).filter(func(item): return item.get("edgeId") == hit.get("edgeId") and ((parameter == 0.0 and item.get("role") == "Start") or (parameter == 1.0 and item.get("role") == "End")))
131	return hit.get("sourceToken") == map.StateToken and hit.get("junctionNodeId", 0) == read.get("nodeId") and hit.get("surfaceCenter") == read.get("position") and owners.size() == 1

133func round_trip(map: Node, manager: Node, shape: String) -> bool:
134	var before: Dictionary = map.GetRoadState()
135	var reads: Dictionary = {}
136	for edge in before.get("edges", []):
137		for node_id in [edge.get("startNodeId", 0), edge.get("endNodeId", 0)]:
138			reads[node_id] = map.GetJunctionState(node_id)
139	var saved: bool = await SAVE.save_as(manager, "V4 loop %s QA" % shape)
140	if not saved: return false
141	var slot: String = manager.CurrentSlotID
142	var payload_path: String = "user://saves-v4/" + slot + "/road_network_v4.json"
143	var bytes: PackedByteArray = FileAccess.get_file_as_bytes(payload_path)
144	var created: bool = map.CreateMap(25)
145	var empty_token: String = map.StateToken
146	var loaded: bool = await SAVE.operation_succeeded(manager, map.LoadSlot(slot))
147	await process_frame
148	await RenderingServer.frame_post_draw
149	var after: Dictionary = map.GetRoadState()
150	var matched: bool = created and loaded and map.CellSizeMetres == 100 and map.IsPresentationCurrent and after.get("edges") == before.get("edges") and after.get("nodeCount") == before.get("nodeCount") and lineage(map.StateToken) != lineage(empty_token) and lineage(map.StateToken) != lineage(before.get("sourceToken", ""))
151	for node_id in reads:
152		var previous: Dictionary = reads[node_id]
153		var current: Dictionary = map.GetJunctionState(node_id)
154		matched = matched and current.get("position") == previous.get("position") and current.get("incidences") == previous.get("incidences") and current.get("turns") == previous.get("turns") and current.get("sourceToken") == map.StateToken
155	var saved_again: bool = await SAVE.save(manager, slot)
156	matched = matched and saved_again and FileAccess.get_file_as_bytes(payload_path) == bytes
157	var deleted: bool = await SAVE.delete_slot(manager, slot)
158	deleted = deleted and not DirAccess.dir_exists_absolute(ProjectSettings.globalize_path("user://saves-v4/" + slot))
159	return matched and deleted

169func square(map: Node) -> bool:
170	var passed: bool = await build(map, Vector2.ZERO, Vector2(400, 0))
171	passed = await build(map, Vector2(400, 0), Vector2(400, 400)) and passed
172	passed = await build(map, Vector2(400, 400), Vector2(0, 400)) and passed
173	passed = await build(map, Vector2(0, 400), Vector2.ZERO) and passed
174	return passed

176func build(map: Node, start: Vector2, end: Vector2) -> bool:
177	var before: String = map.StateToken
178	var transform: Transform2D = map.get_canvas_transform()
179	button(transform * start, true)
180	var motion := InputEventMouseMotion.new()
181	motion.position = transform * end
182	motion.button_mask = MOUSE_BUTTON_MASK_LEFT
183	Input.parse_input_event(motion)
184	button(transform * end, false)
185	return await wait_idle(map) and map.StateToken != before and map.IsPresentationCurrent

187func wait_idle(map: Node) -> bool:
188	var deadline: int = Time.get_ticks_msec() + 5000
189	while Time.get_ticks_msec() < deadline:
190		await process_frame
191		if not map.IsBuildBusy: return true
192	return false

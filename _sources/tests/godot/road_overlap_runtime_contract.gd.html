83func press(map: Node, point: Vector2) -> void:
84	button(map.get_canvas_transform() * point, true)
85
86func release(map: Node, point: Vector2) -> void:
87	button(map.get_canvas_transform() * point, false)
88
89func button(position: Vector2, pressed: bool) -> void:
90	var event := InputEventMouseButton.new()
91	event.position = position
92	event.button_index = MOUSE_BUTTON_LEFT
93	event.pressed = pressed
94	Input.parse_input_event(event)
95
96func motion(map: Node, point: Vector2) -> void:
97	var event := InputEventMouseMotion.new()
98	event.position = map.get_canvas_transform() * point
99	event.button_mask = MOUSE_BUTTON_MASK_LEFT
100	Input.parse_input_event(event)

102func wait_idle(map: Node) -> bool:
103	var deadline := Time.get_ticks_msec() + 5000
104	while Time.get_ticks_msec() < deadline:
105		await process_frame
106		if not map.IsBuildBusy: return true
107	return false
108
109func wait_preview(map: Node) -> bool:
110	var deadline := Time.get_ticks_msec() + 5000
111	while Time.get_ticks_msec() < deadline:
112		await process_frame
113		if map.GetBuildPreview().get("phase") != "Pending": return true
114	return false

116func preview_late_results(map: Node) -> bool:
117	map.CreateMap(100)
118	var probe = WORK.new()
119	probe.InstallPreview(map)
120	press(map, Vector2.ZERO)
121	motion(map, Vector2(300, 0))
122	motion(map, Vector2(-300, 0))
123	var deadline := Time.get_ticks_msec() + 5000
124	while not probe.Started and Time.get_ticks_msec() < deadline: await process_frame
125	var pending: Dictionary = map.GetBuildPreview()
126	var neutral: bool = probe.Started and pending.get("phase") == "Pending" and pending.get("segments", []).all(func(segment): return not segment["conflict"] and segment["color"].is_equal_approx(Color("9aa5ad")))
127	probe.Release()
128	var ready := await wait_preview(map)
129	var latest: Dictionary = map.GetBuildPreview()
130	var coalesced: bool = ready and latest.get("phase") == "Ready" and latest["segments"][0]["end"] == Vector2(-300, 0) and map.RoadCount == 0
131	probe.RemovePreview(map)
132	if probe.Completed: probe.Cleanup()
133	var escaped := InputEventKey.new()
134	escaped.keycode = KEY_ESCAPE
135	escaped.pressed = true
136	Input.parse_input_event(escaped)
137	await process_frame
138	var stale_probe = WORK.new()
139	stale_probe.InstallPreview(map)
140	press(map, Vector2.ZERO)
141	motion(map, Vector2(300, 0))
142	deadline = Time.get_ticks_msec() + 5000
143	while not stale_probe.Started and Time.get_ticks_msec() < deadline: await process_frame
144	var created: bool = map.CreateMap(50)
145	var token: String = map.StateToken
146	stale_probe.Release()
147	deadline = Time.get_ticks_msec() + 5000
148	while not stale_probe.Completed and Time.get_ticks_msec() < deadline: await process_frame
149	for frame in range(3): await process_frame
150	var stale_rejected: bool = created and stale_probe.Completed and map.StateToken == token and map.CellSizeMetres == 50 and map.RoadCount == 0 and map.GetBuildPreview().get("phase") == "None" and map.GetBuildPreview()["segments"].is_empty()
151	stale_probe.RemovePreview(map)
152	if stale_probe.Completed: stale_probe.Cleanup()
153	print("V4_PREVIEW_LIFECYCLE ", JSON.stringify({"pending_neutral":neutral,"latest_only":coalesced,"stale_rejected":stale_rejected}))
154	return neutral and coalesced and stale_rejected

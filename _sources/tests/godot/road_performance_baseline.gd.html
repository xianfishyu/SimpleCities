70func load_fixture(kind: String, expected: int) -> bool:
71	var manager: Node = root.get_node("SaveManager")
72	if _slot.is_empty():
73		if not check("create_fixture_slot", await SAVE.save_as(manager, "V4 baseline temporary %s" % _cell)):
74			return false
75		_slot = manager.CurrentSlotID
76	var payload := FileAccess.get_file_as_bytes(_output.path_join("datasets/%s-%d.json" % [kind, _cell]))
77	if not check("read_fixture_%s" % kind, not payload.is_empty()): return false
78	var base := "user://saves-v4/" + _slot
79	var file := FileAccess.open(base.path_join("road_network_v4.json"), FileAccess.WRITE)
80	if not check("write_fixture_%s" % kind, file != null): return false
81	file.store_buffer(payload)
82	file.close()
83	var manifest: Dictionary = JSON.parse_string(FileAccess.get_file_as_string(base.path_join("manifest.json")))
84	manifest.schemaVersion = int(manifest.schemaVersion)
85	var payload_hash := HashingContext.new()
86	payload_hash.start(HashingContext.HASH_SHA256)
87	payload_hash.update(payload)
88	manifest.files[0].encodedLength = payload.size()
89	manifest.files[0].sha256 = payload_hash.finish().hex_encode()
90	file = FileAccess.open(base.path_join("manifest.json"), FileAccess.WRITE)
91	if not check("write_manifest_%s" % kind, file != null): return false
92	file.store_string(JSON.stringify(manifest))
93	file.close()
94	var loaded := await SAVE.operation_succeeded(manager, _map.LoadSlot(_slot))
95	await RenderingServer.frame_post_draw
96	return check("normal_load_%s" % kind, loaded and _map.RoadCount == expected and _map.CellSizeMetres == _cell and _map.IsPresentationCurrent,
97		{"loaded": loaded, "roadCount": _map.RoadCount, "status": _map.get_node("HUD/Panel/Margin/Controls/Status").text})

108func on_draw() -> void:
109	if not _sampling: return
110	var now := Time.get_ticks_usec()
111	if _last_draw != 0:
112		_frame_ms.append((now - _last_draw) / 1000.0)
113		_frame_us.append(now - _origin)
114		if _frame_timeline:
115			_process_us.append(_process_at - _origin)
116			_pre_draw_us.append(_pre_draw_at - _origin)
117			_drawn_frames.append(Engine.get_frames_drawn())
118		if DisplayServer.window_is_focused(): _focused_frames += 1
119	_last_draw = now

164func end_window(name: String) -> void:
165	_sampling = false
166	var wall_end_ms := Time.get_unix_time_from_system() * 1000.0
167	var duration := (Time.get_ticks_usec() - _origin) / 1000000.0
168	var sorted := _frame_ms.duplicate()
169	sorted.sort()
170	var over := 0
171	for ms in _frame_ms:
172		if ms > 1000.0 / 144.0: over += 1
173	var row := {"phase": name, "startedAtUs": _origin, "focusedFrames": _focused_frames, "seconds": duration, "frames": _frame_ms.size(), "p95Ms": percentile(sorted, 0.95),
174		"wallStartUnixMs": _window_wall_start_ms, "wallEndUnixMs": wall_end_ms,
175		"p99Ms": percentile(sorted, 0.99), "maxMs": percentile(sorted, 1.0), "overBudget": over,
176		"frameMs": _frame_ms, "frameTimeUs": _frame_us}
177	if _frame_timeline:
178		row.frameTimeline = {"processFrameUs": _process_us, "preDrawUs": _pre_draw_us,
179			"drawnFrame": _drawn_frames, "syncCalls": _sync_calls}
180	_windows.append(row)
181	print("BASELINE_WINDOW cell=%d phase=%s frames=%d p95=%.4f p99=%.4f max=%.4f over=%d" %
182		[_cell, name, row.frames, row.p95Ms, row.p99Ms, row.maxMs, over])

184func percentile(sorted: PackedFloat64Array, fraction: float) -> float:
185	if sorted.is_empty(): return -1
186	return sorted[clampi(int(ceil(sorted.size() * fraction)) - 1, 0, sorted.size() - 1)]

192func motion_window(name: String, mode: int) -> void:

194	await process_frame
195	begin_window(name)
196	var tick := _origin
197	var until := _origin + int(_seconds * 1000000)
198	var index := 0
199	var visible_hover_observed := false
200	while Time.get_ticks_usec() < until:
201		var now := Time.get_ticks_usec()
202		if now >= tick and mode != 0:
203			tick = now + 16667
204			index += 1
205			if mode == 1 or mode == 3:
206				motion(Vector2(1000, 800), Vector2(0.6 if index % 120 < 60 else -0.6, 0), MOUSE_BUTTON_MASK_MIDDLE)
207			if mode >= 2:
208				motion(screen(Vector2(2.48 if index % 2 == 0 else 2.52, 3.0) * _cell))
209				if index == 10: visible_hover_observed = _map.GetSelectionState().hasHover
210		await process_frame
211	end_window(name)
212	if mode >= 2: check(name + "_actually_highlights", visible_hover_observed)

239func gesture(name: String, from: Vector2, to: Vector2, expected_selection: int = 0) -> bool:

257func record_operation(name: String, source: String, injected: int) -> bool:
258	var until := Time.get_ticks_msec() + 10000
259	while Time.get_ticks_msec() < until:
260		await process_frame
261		if not _map.IsBuildBusy and _map.StateToken != source and _map.BuildPhase == "Drawn": break
262	var metrics: Dictionary = _map.GetOperationMetrics()
263	metrics.operation = name
264	metrics.sampleWindow = _active_window if _sampling else "setup"
265	metrics.inputInjectionAtUs = injected
266	metrics.inputInjectionUs = injected - _origin
267	metrics.injectionToObservedCompletionMs = (Time.get_ticks_usec() - injected) / 1000.0
268	metrics.renderFrame = Engine.get_frames_drawn()
269	var stages := float(metrics.domainMs) + float(metrics.surfaceMs) + float(metrics.get("geometryMs", 0)) + float(metrics.preflightMs) + float(metrics.referenceCommitMs) + float(metrics.presentationPublishMs)
270	var valid: bool = not _map.IsBuildBusy and metrics.phase == "Drawn" and source != _map.StateToken and _map.IsPresentationCurrent and metrics.presentedToken == _map.StateToken and metrics.inputToDrawMs >= stages
271	_operations.append(metrics)
272	return check(name + "_drawn", valid, metrics if not valid else {})

274func cancellation_window() -> void:
275	set_tool_mode(0)
276	if not await gesture("cancelFixture", Vector2(0, -2), Vector2(4, -2)): return
277	begin_window("escapeCancellation")
278	for attempt in range(20):
279		var source: String = _map.StateToken
280		keyboard(KEY_Z, true)
281		var phase_before: String = _map.BuildPhase
282		keyboard(KEY_ESCAPE)
283		await process_frame
284		var until := Time.get_ticks_msec() + 10000
285		while _map.IsBuildBusy and Time.get_ticks_msec() < until: await process_frame
286		var metrics: Dictionary = _map.GetOperationMetrics()
287		var accepted: bool = metrics.phase == "Cancelled"
288		metrics.accepted = accepted
289		metrics.phaseBeforeEscape = phase_before
290		metrics.stateUnchanged = source == _map.StateToken
291		_cancellations.append(metrics)
292		if accepted:
293			check("cancel_keeps_state", not _map.IsBuildBusy and metrics.stateUnchanged and metrics.phase == "Cancelled" and metrics.cancelToToolReadyMs >= 0 and metrics.cancelToWorkerExitMs >= 0)
294		else:
295			check("commit_race_not_cancelled", metrics.phase == "Drawn")
296			if not await history("redoAfterCancelRace", KEY_Y): break
297	end_window("escapeCancellation")

319func keyboard(key: Key, control: bool = false) -> void:
320	var started := probe_call_start()
321	var event := InputEventKey.new()
322	event.keycode = key
323	event.ctrl_pressed = control
324	event.pressed = true
325	Input.parse_input_event(event)
326	var released := InputEventKey.new()
327	released.keycode = key
328	released.ctrl_pressed = control
329	Input.parse_input_event(released)
330	Input.flush_buffered_events()
331	if started != 0: probe_call_end("keyboard:%d" % key, started)

333func check(name: String, passed: bool, detail: Dictionary = {}) -> bool:

339func finish() -> void:

343		_slot = ""
344	var report := {"passed": _passed, "cell": _cell, "engine": Engine.get_version_info(),
345		"processId": OS.get_process_id(), "executable": OS.get_executable_path(), "cpu": OS.get_processor_name(),
346		"adapter": RenderingServer.get_video_adapter_name(), "vendor": RenderingServer.get_video_adapter_vendor(),
347		"driver": RenderingServer.get_current_rendering_driver_name(), "method": RenderingServer.get_current_rendering_method(),
348		"resolution": DisplayServer.window_get_size(), "vsync": DisplayServer.window_get_vsync_mode(),
349		"maxFps": Engine.max_fps, "debugBuild": OS.is_debug_build(), "warmupSeconds": _warmup, "requestedWindowSeconds": _seconds,
350		"frameTimelineEnabled": _frame_timeline,
351		"checks": _checks, "windows": _windows, "operations": _operations, "cancellations": _cancellations}
352	var file := FileAccess.open(_output.path_join("render-%d.json" % _cell), FileAccess.WRITE)
353	if file != null:
354		file.store_string(JSON.stringify(report))
355		file.close()
356	else: _passed = false

359	await process_frame
360	quit(0 if _passed else 1)

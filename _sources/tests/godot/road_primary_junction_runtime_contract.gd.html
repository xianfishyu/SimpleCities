83func center_junction(state: Dictionary) -> Dictionary:
84	var found: Array = state.get("junctions", []).filter(func(item): return item.get("position") == Vector2.ZERO)
85	return found[0] if found.size() == 1 else {}

87func junction_matches(map: Node, nodes: int, edges: int, directions: Array) -> bool:
88	var state: Dictionary = map.GetRoadState()
89	var junction: Dictionary = center_junction(state)
90	var incidences: Array = junction.get("incidences", [])
91	var turns: Array = junction.get("turns", [])
92	if state.get("nodeCount", 0) != nodes or state.get("edges", []).size() != edges or state.get("junctions", []).size() != 1 or incidences.size() != directions.size() or turns.size() != directions.size() * directions.size():
93		return false
94	var previous_bearing := -1.0
95	for index in range(incidences.size()):
96		var incidence: Dictionary = incidences[index]
97		var outward: Vector2 = incidence.get("outward", Vector2.ZERO)
98		var bearing: float = incidence.get("bearingRadians", -1.0)
99		if not outward.is_equal_approx(directions[index]) or bearing < previous_bearing or not incidence.get("role", "") in ["Start", "End"]:
100			return false
101		previous_bearing = bearing
102	return junction_surface_matches(map, junction)

104func junction_surface_matches(map: Node, junction: Dictionary) -> bool:
105	var hit: Dictionary = map.PickRoad(Vector2.ZERO)
106	var owners: Array = junction.get("incidences", []).filter(func(item): return item.get("edgeId") == hit.get("edgeId"))
107	return map.IsPresentationCurrent and junction.get("sourceToken", "") == map.StateToken and hit.get("sourceToken", "") == map.StateToken and junction.get("nodeId", 0) > 0 and hit.get("junctionNodeId", 0) == junction.get("nodeId") and hit.get("surfaceCenter") == Vector2.ZERO and owners.size() == 1

109func turn_matches(junction: Dictionary, from_direction: Vector2, to_direction: Vector2, angle: float) -> bool:
110	var incidences: Array = junction.get("incidences", [])
111	var from_items: Array = incidences.filter(func(item): return item.get("outward") == from_direction)
112	var to_items: Array = incidences.filter(func(item): return item.get("outward") == to_direction)
113	if from_items.size() != 1 or to_items.size() != 1: return false
114	var turns: Array = junction.get("turns", []).filter(func(item): return item.get("fromEdgeId") == from_items[0]["edgeId"] and item.get("fromRole") == from_items[0]["role"] and item.get("toEdgeId") == to_items[0]["edgeId"] and item.get("toRole") == to_items[0]["role"])
115	return turns.size() == 1 and absf(turns[0].get("signedAngleRadians", INF) - angle) < 0.00001

117func lineage(token: String) -> String:
118	return token.get_slice("Lineage = ", 1).get_slice(",", 0).strip_edges()

120func build(map: Node, start: Vector2, end: Vector2) -> bool:
121	var before: String = map.StateToken
122	gesture(map, start, end)
123	return await wait_idle(map) and map.StateToken != before and map.IsPresentationCurrent

125func gesture(map: Node, start: Vector2, end: Vector2) -> void:
126	var transform: Transform2D = map.get_canvas_transform()
127	button(transform * start, true)
128	var motion := InputEventMouseMotion.new()
129	motion.position = transform * end
130	motion.button_mask = MOUSE_BUTTON_MASK_LEFT
131	Input.parse_input_event(motion)
132	button(transform * end, false)

134func wait_idle(map: Node) -> bool:
135	var deadline := Time.get_ticks_msec() + 5000
136	while Time.get_ticks_msec() < deadline:
137		await process_frame
138		if not map.IsBuildBusy: return true
139	return false

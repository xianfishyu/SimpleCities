28static func operation_succeeded(save_manager: Node, operation_token: String) -> bool:
29	var result := await wait_for_operation(save_manager, operation_token)
30	if not await wait_for_idle(save_manager):
31		print("SAVE_OPERATION_FAILURE idle-timeout token=%s" % operation_token)
32		return false
33	var result_kind := int(result.get("resultKind", -1))
34	var succeeded := result_kind == RESULT_SUCCEEDED or result_kind == RESULT_SUCCEEDED_WITH_WARNINGS

39static func wait_for_idle(

42	var deadline_msec := Time.get_ticks_msec() + int(timeout_seconds * 1000.0)

44	while Time.get_ticks_msec() < deadline_msec:
45		if not is_instance_valid(save_manager):
46			return false
47		if not bool(save_manager.get("IsOperationBusy")):
48			idle_frames += 1

50				return true

85static func refresh_manifest_payload(slot_id: String) -> bool:
86	var payload_path := slot_path(slot_id, PAYLOAD_FILE_NAME)
87	var payload_file := FileAccess.open(payload_path, FileAccess.READ)
88	if payload_file == null:
89		return false
90	var payload_length := payload_file.get_length()
91	var hash_context := HashingContext.new()
92	if hash_context.start(HashingContext.HASH_SHA256) != OK:
93		payload_file.close()
94		return false
95	while payload_file.get_position() < payload_length:
96		var remaining := payload_length - payload_file.get_position()
97		var chunk := payload_file.get_buffer(mini(64 * 1024, remaining))
98		if chunk.is_empty() or hash_context.update(chunk) != OK:
99			payload_file.close()
100			return false
101	payload_file.close()
102	var payload_sha256 := hash_context.finish().hex_encode()
103
104	var manifest_path := slot_path(slot_id, "manifest.json")
105	var manifest: Variant = JSON.parse_string(FileAccess.get_file_as_string(manifest_path))
106	if not manifest is Dictionary:
107		return false
108	manifest["schemaVersion"] = int(manifest.get("schemaVersion", -1))
109	var files: Array = manifest.get("files", [])
110	if files.size() != 1 or not files[0] is Dictionary:
111		return false
112
113	var file_entry: Dictionary = files[0]
114	file_entry["name"] = PAYLOAD_FILE_NAME
115	file_entry["encodedLength"] = payload_length
116	file_entry["sha256"] = payload_sha256
117	files[0] = file_entry
118	manifest["files"] = files
119
120	var manifest_file := FileAccess.open(manifest_path, FileAccess.WRITE)
121	if manifest_file == null:
122		return false
123	manifest_file.store_string(JSON.stringify(manifest))
124	manifest_file.close()
125	return true

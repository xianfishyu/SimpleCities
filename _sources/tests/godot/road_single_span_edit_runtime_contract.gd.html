138func selected_stroke(map: Node) -> Dictionary:
139	var strokes: Array = map.GetSelectionState().get("strokes", []).filter(func(stroke): return stroke.get("selected", false))
140	return strokes[0] if not strokes.is_empty() else {}

142func point_set_matches(stroke: Dictionary, expected: Array) -> bool:
143	var points: Array = Array(stroke.get("points", PackedVector2Array()))
144	return points.size() == expected.size() and points.all(func(point): return expected.has(point))

146func roundtrip(map: Node, label: String, check_no_change: bool = false) -> void:
147	var manager: Node = root.get_node("SaveManager")
148	var before: Dictionary = map.GetRoadState()
149	var saved: bool = await SAVE.save_as(manager, "V4 single span QA " + label)
150	check(label + "_saved", saved)
151	if not saved: return
152	var slot: String = manager.CurrentSlotID
153	var payload_path: String = "user://saves-v4/" + slot + "/road_network_v4.json"
154	var bytes: PackedByteArray = FileAccess.get_file_as_bytes(payload_path)
155	check(label + "_payload_present", not bytes.is_empty())
156	if check_no_change:
157		# The selected target remains Highway; codec bytes include content revision and ID allocators.
158		var previous_phase: String = map.BuildPhase
159		var previous_elapsed: float = map.BuildElapsedMilliseconds
160		var previous_drawn: float = map.BuildDrawnElapsedMilliseconds
161		motion(map, Vector2(250, 0), false)
162		await process_frame
163		check("same_profile_has_no_hover_highlight", map.GetSelectionState().get("strokes", []).is_empty())
164		press(map, Vector2(250, 0))
165		await process_frame
166		check("same_profile_has_no_pending_highlight", map.GetSelectionState().get("strokes", []).is_empty())
167		release(map, Vector2(250, 0))
168		await settle(map)
169		check("same_profile_does_not_start_operation", not map.IsBuildBusy and map.BuildPhase == previous_phase and map.BuildElapsedMilliseconds == previous_elapsed and map.BuildDrawnElapsedMilliseconds == previous_drawn)
170		check("same_profile_keeps_snapshot_token_and_ids", map.GetRoadState() == before)
171		var no_change_saved: bool = await SAVE.save(manager, slot)
172		check("same_profile_preserves_codec_bytes", no_change_saved and FileAccess.get_file_as_bytes(payload_path) == bytes)
173	var loaded: bool = await SAVE.operation_succeeded(manager, map.LoadSlot(slot))
174	await process_frame
175	check(label + "_reload", loaded and map.IsPresentationCurrent and map.StateToken != before.get("sourceToken") and map.GetRoadState().get("edges") == before.get("edges"))
176	var resaved: bool = await SAVE.save(manager, slot)
177	check(label + "_resave", resaved and FileAccess.get_file_as_bytes(payload_path) == bytes)
178	var deleted: bool = await SAVE.delete_slot(manager, slot)
179	check(label + "_fixture_deleted", deleted and not DirAccess.dir_exists_absolute(ProjectSettings.globalize_path("user://saves-v4/" + slot)))

181func focus(map: Node, world: Vector2, zoom: float) -> void:
182	var camera: Camera2D = map.get_node("Camera2D")
183	camera.zoom = Vector2(zoom, zoom)
184	camera.position = world - (Vector2(1000, 450) - root.get_visible_rect().size / 2.0) / zoom

186func build(map: Node, start: Vector2, end: Vector2) -> bool:
187	var token: String = map.StateToken
188	press(map, start)
189	motion(map, end)
190	release(map, end)
191	return await settle(map) and map.StateToken != token

193func settle(map: Node) -> bool:
194	var deadline: int = Time.get_ticks_msec() + 5000
195	while Time.get_ticks_msec() < deadline:
196		await process_frame
197		if not map.IsBuildBusy: return map.IsPresentationCurrent
198	return false

200func press(map: Node, point: Vector2) -> void:
201	button(map.get_canvas_transform() * point, true)
202
203func release(map: Node, point: Vector2) -> void:
204	button(map.get_canvas_transform() * point, false)
205
206func button(position: Vector2, pressed: bool) -> void:
207	var event := InputEventMouseButton.new()
208	event.position = position
209	event.button_index = MOUSE_BUTTON_LEFT
210	event.pressed = pressed
211	Input.parse_input_event(event)
212
213func motion(map: Node, point: Vector2, held: bool = true) -> void:
214	var event := InputEventMouseMotion.new()
215	event.position = map.get_canvas_transform() * point
216	event.button_mask = MOUSE_BUTTON_MASK_LEFT if held else 0
217	Input.parse_input_event(event)
218
219func escape() -> void:
220	var event := InputEventKey.new()
221	event.keycode = KEY_ESCAPE
222	event.pressed = true
223	Input.parse_input_event(event)
224	var released := InputEventKey.new()
225	released.keycode = KEY_ESCAPE
226	Input.parse_input_event(released)

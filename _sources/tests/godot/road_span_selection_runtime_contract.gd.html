158func check(name: String, passed: bool, details: Variant = null) -> void:
159	_rows.append({"name": name, "passed": passed, "details": details})
160	_passed = passed and _passed

162func count(map: Node) -> int:
163	return map.GetSelectionState().get("selectedCount", 0)
164
165func selected_strokes(map: Node) -> Array:
166	return map.GetSelectionState().get("strokes", []).filter(func(stroke): return stroke.get("selected", false))
167
168func first_stroke(map: Node, selected: bool) -> Dictionary:
169	var strokes: Array = map.GetSelectionState().get("strokes", []).filter(func(stroke): return stroke.get("selected", false) == selected)
170	return strokes[0] if not strokes.is_empty() else {}
171
172func exact_points(stroke: Dictionary, expected: Array) -> bool:
173	return Array(stroke.get("points", PackedVector2Array())) == expected
174
175func point_set_matches(stroke: Dictionary, expected: Array) -> bool:
176	var points: Array = Array(stroke.get("points", PackedVector2Array()))
177	return points.size() == expected.size() and points.all(func(point): return expected.has(point))
178
179func exact_range(stroke: Dictionary, start: float, end: float) -> bool:
180	var ranges: Array = stroke.get("ranges", [])
181	return ranges.size() == 1 and is_equal_approx(ranges[0].get("startParameter", -1.0), start) and is_equal_approx(ranges[0].get("endParameter", -1.0), end)

183func focus(map: Node, world: Vector2, zoom: float) -> void:
184	var camera: Camera2D = map.get_node("Camera2D")
185	camera.zoom = Vector2(zoom, zoom)
186	camera.position = world - (Vector2(1000, 450) - root.get_visible_rect().size / 2.0) / zoom

188func build(map: Node, start: Vector2, end: Vector2) -> bool:
189	var token: String = map.StateToken
190	press(map, start)
191	motion(map, end)
192	release(map, end)
193	var deadline: int = Time.get_ticks_msec() + 5000
194	while Time.get_ticks_msec() < deadline:
195		await process_frame
196		if not map.IsBuildBusy: return map.StateToken != token and map.IsPresentationCurrent
197	return false

199func press(map: Node, point: Vector2) -> void:
200	button(map.get_canvas_transform() * point, true)
201
202func release(map: Node, point: Vector2) -> void:
203	button(map.get_canvas_transform() * point, false)
204
205func button(position: Vector2, pressed: bool) -> void:
206	var event := InputEventMouseButton.new()
207	event.position = position
208	event.button_index = MOUSE_BUTTON_LEFT
209	event.pressed = pressed
210	Input.parse_input_event(event)
211
212func motion(map: Node, point: Vector2, held: bool = true) -> void:
213	var event := InputEventMouseMotion.new()
214	event.position = map.get_canvas_transform() * point
215	event.button_mask = MOUSE_BUTTON_MASK_LEFT if held else 0
216	Input.parse_input_event(event)
217
218func escape() -> void:
219	var event := InputEventKey.new()
220	event.keycode = KEY_ESCAPE
221	event.pressed = true
222	Input.parse_input_event(event)
223	var released := InputEventKey.new()
224	released.keycode = KEY_ESCAPE
225	Input.parse_input_event(released)

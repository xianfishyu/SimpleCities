1extends "res://tests/godot/road_target_performance.gd"

3# Exercise the exact production QA gesture sequence without collecting frame or
4# latency samples. This remains a functional contract, never FPS acceptance.
5var _observed: Array[String] = []
6var _opened: Array[String] = []
7var _closed: Array[String] = []
8var _cases: Array = []
9
10func _initialize() -> void:
11	_output = "res://.scratch/v4-target-build-20261005"
12	for argument in OS.get_cmdline_user_args():
13		if argument.begins_with("--output="): _output = argument.trim_prefix("--output=")
14	_cycles = 2
15	run.call_deferred()
16
17func run() -> void:
18	_map = load("res://Scenes/RoadMap.tscn").instantiate()
19	root.add_child(_map)
20	current_scene = _map
21	await process_frame
22	for row in [[25, "target"], [50, "target"], [100, "target"], [200, "target"], [200, "polylines"]]:
23		_cell = row[0]
24		_kind = row[1]
25		if not await load_fixture(_kind, expected_edges()): break
26		var original: String = _map.GetRoadContentFingerprint()
27		_observed.clear()
28		_opened.clear()
29		_closed.clear()
30		await build_window()
31		var expected: Array[String] = []
32		if _cell == 200 and _kind == "target": expected.append("createBuildSpace")
33		for cycle in range(_cycles):
34			expected.append_array(["build", "undoBuild", "redoBuild", "undoBuildReset"])
35		if _cell == 200 and _kind == "target": expected.append("restoreBuildSpace")
36		check("exact_build_and_history_sequence", _observed == expected, {"observed": _observed.duplicate()})
37		check("build_window_executed_once", _opened == ["buildAndHistory"] and _closed == _opened)
38		check("build_history_restores_geometry_and_ids", original == _map.GetRoadContentFingerprint() and _map.RoadCount == expected_edges())
39		check("no_performance_sampling", not _sampling and _windows.is_empty() and _frame_ms.is_empty() and _operations.is_empty())
40		_cases.append({"cell": _cell, "kind": _kind, "operations": _observed.duplicate(), "canonicalEdges": _map.RoadCount})
41		if not _passed: break
42	await finish()
43
44func begin_window(name: String) -> void:
45	_opened.append(name)
46
47func end_window(name: String) -> void:
48	_closed.append(name)
49
50func record_operation(name: String, source: String, _injected: int) -> bool:
51	var deadline := Time.get_ticks_msec() + 10000
52	while Time.get_ticks_msec() < deadline:
53		await process_frame
54		if not _map.IsBuildBusy and _map.StateToken != source and _map.BuildPhase == "Drawn": break
55	_observed.append(name)
56	return check(name + "_drawn", not _map.IsBuildBusy and _map.StateToken != source and _map.BuildPhase == "Drawn" and _map.IsPresentationCurrent)
57
58func finish() -> void:
59	check("all_five_cases_covered", _cases.size() == 5)
60	if not _slot.is_empty():
61		check("temporary_slot_deleted", await SAVE.delete_slot(root.get_node("SaveManager"), _slot))
62		_slot = ""
63	var report := {"passed": _passed, "checks": _checks, "cases": _cases, "performanceSampled": false}
64	var file := FileAccess.open(_output.path_join("build-functional.json"), FileAccess.WRITE)
65	if file == null:
66		_passed = false
67	else:
68		file.store_string(JSON.stringify(report))
69		file.close()
70	print("V4_TARGET_BUILD_FUNCTIONAL ", JSON.stringify({"passed": _passed, "checks": _checks.size(), "cases": _cases.size(), "performanceSampled": false}))
71	_map.queue_free()
72	await process_frame
73	quit(0 if _passed else 1)

4# Target-scale replay through normal admission and real input. Dataset files
5# must exist under --output/datasets; all timings include the unchanged grid.
6var _kind := "target"
7var _require_release := false
8var _settle_seconds := 0.0
9var _collector_token := ""
10var _collector_wait_seconds := 30.0

12func _initialize() -> void:
13	for argument in OS.get_cmdline_user_args():
14		if argument.begins_with("--dataset="): _kind = argument.trim_prefix("--dataset=")
15		if argument == "--require-release": _require_release = true
16		if argument.begins_with("--settle="): _settle_seconds = float(argument.trim_prefix("--settle="))
17		if argument.begins_with("--collector-token="): _collector_token = argument.trim_prefix("--collector-token=")
18		if argument.begins_with("--collector-wait-seconds="): _collector_wait_seconds = float(argument.trim_prefix("--collector-wait-seconds="))
19	super._initialize()

21func expected_edges() -> int:
22	return 320 if _kind == "polylines" else (9680 if _cell == 200 else 10000)

24func run() -> void:
25	if _require_release and not check("release_export_identity", not OS.is_debug_build() and OS.has_feature("road_performance")):
26		quit(1)
27		return
28	if not check("valid_measurement_arguments", _cell in [25, 50, 100, 200] and _seconds > 0 and _warmup >= 0 and _cycles > 0 and is_finite(_settle_seconds) and _settle_seconds >= 0 and _settle_seconds <= 60 and (_kind == "target" or (_kind == "polylines" and _cell == 200))):
29		quit(1)
30		return
31	if not _collector_token.is_empty() and not check("valid_collector_arguments", _collector_token.length() == 32 and _collector_token.is_valid_hex_number() and is_finite(_collector_wait_seconds) and _collector_wait_seconds > 0 and _collector_wait_seconds <= 60):
32		quit(1)
33		return
34	Engine.max_fps = 0
35	DisplayServer.window_set_vsync_mode(DisplayServer.VSYNC_DISABLED)
36	if OS.has_feature("road_performance"):
37		# Export templates may disable CLI script overrides. The QA feature selects
38		# this MainLoop in project settings and loads the map with its normal autoloads.
39		await process_frame
40		_map = current_scene
41		if not check("release_main_scene_loaded", is_instance_valid(_map) and _map.scene_file_path == "res://Scenes/CityMap.tscn" and _map.UseProductControls):
42			quit(1)
43			return
44	else:
45		_map = load("res://Scenes/CityMap.tscn").instantiate()
46		root.add_child(_map)
47		current_scene = _map
48	RenderingServer.frame_post_draw.connect(on_draw)
49	await process_frame
50	await process_frame
51	root.grab_focus()
52	if await load_fixture(_kind, expected_edges()):
53		focus(Vector2(-4000, -4000) + Vector2(3, 3) * _cell, 65.0 / _cell)
54		await idle_for(_warmup)
55		await motion_window("idle", 0)
56		await motion_window("camera", 1)
57		await motion_window("visibleHover", 2)
58		await motion_window("cameraAndHover", 3)
59		await operation_window()
60		await build_window()
61		await cancellation_window()
62		if _settle_seconds > 0:
63			var source: String = _map.StateToken
64			begin_window("settle")
65			await idle_for(_settle_seconds)
66			end_window("settle")
67			check("settle_keeps_state", _map.StateToken == source and not _map.IsBuildBusy and _map.IsPresentationCurrent)
68	await finish()

70func finish() -> void:
71	if _require_release:
72		check("release_has_rendered_windows", _windows.size() >= 7)
73		check("release_has_build_window", _windows.any(func(window): return window.phase == "buildAndHistory" and window.frames > 0))
74		for window in _windows:
75			check(window.phase + "_focused_frames", window.frames > 0 and window.focusedFrames == window.frames)
76			# A frame P95 is one gate; all long frames remain in the report even when it passes.
77			check(window.phase + "_p95_budget", window.p95Ms >= 0 and window.p95Ms <= 1000.0 / 144.0)
78	if not _collector_token.is_empty():
79		await wait_for_collector()
80	await super.finish()

82func wait_for_collector() -> void:
83	# Measurement has ended. Keep the CLR alive until its external collector has
84	# stopped and drained the stream; these frames never enter a sampled window.
85	_sampling = false
86	var source: String = _map.StateToken
87	var temporary := _output.path_join("collector-ready.tmp")
88	var ready := _output.path_join("collector-ready.json")
89	var file := FileAccess.open(temporary, FileAccess.WRITE)
90	if not check("collector_ready_opened", file != null): return
91	file.store_string(JSON.stringify({"token": _collector_token, "processId": OS.get_process_id(),
92		"measurementEndUnixMs": _windows[-1].wallEndUnixMs if not _windows.is_empty() else 0,
93		"readyUnixMs": Time.get_unix_time_from_system() * 1000.0, "passed": _passed}))
94	file.close()
95	if not check("collector_ready_published", DirAccess.rename_absolute(ProjectSettings.globalize_path(temporary), ProjectSettings.globalize_path(ready)) == OK): return
96	var acknowledged := false
97	var status := "timeout"
98	var acknowledgement := _output.path_join("collector-done.json")
99	var deadline := Time.get_ticks_msec() + int(_collector_wait_seconds * 1000.0)
100	while Time.get_ticks_msec() < deadline:
101		if FileAccess.file_exists(acknowledgement):
102			var ack: Variant = JSON.parse_string(FileAccess.get_file_as_string(acknowledgement))
103			if ack is Dictionary and ack.get("token") == _collector_token and ack.get("processId") == OS.get_process_id():
104				status = str(ack.get("status", "invalid"))
105				acknowledged = status == "stopped"
106				break
107		await process_frame
108	check("collector_acknowledged", acknowledged, {"status": status})
109	check("collector_wait_keeps_state", _map.StateToken == source and not _map.IsBuildBusy and _map.IsPresentationCurrent)

128func motion_window(name: String, mode: int) -> void:
129	set_tool_mode(1 if mode >= 2 else 0)
130	await process_frame
131	var camera: Camera2D = _map.get_node("Camera2D")
132	var camera_start := camera.position
133	var camera_travel := 0.0
134	if mode == 1 or mode == 3: middle_button(true)
135	begin_window(name)
136	var tick := _origin
137	var until := _origin + int(_seconds * 1000000)
138	var index := 0
139	var visible := false
140	while Time.get_ticks_usec() < until:
141		var now := Time.get_ticks_usec()
142		if now >= tick and mode != 0:
143			tick = now + 16667
144			index += 1
145			if mode == 1 or mode == 3:
146				motion(Vector2(1000, 450), Vector2(0.6 if index % 120 < 60 else -0.6, 0), MOUSE_BUTTON_MASK_MIDDLE)
147				camera_travel = maxf(camera_travel, camera.position.distance_to(camera_start))
148			if mode >= 2:
149				var hover := Vector2(0.25 if index % 2 == 0 else 0.3, 0) if _kind == "polylines" else Vector2(2.25 if index % 2 == 0 else 2.3, 3)
150				motion(screen(Vector2(-4000, -4000) + hover * _cell))
151				if index == 10: visible = _map.GetSelectionState().hasHover
152		await process_frame
153	end_window(name)
154	if mode == 1 or mode == 3:
155		middle_button(false)
156		check(name + "_actually_moves_camera", camera_travel > 0.0, {"maximumWorldTravel": camera_travel})
157	if mode >= 2: check(name + "_actually_highlights", visible)

159func operation_window() -> void:
160	var corner := Vector2.ONE * (-4000.0 / _cell)
161	var from := corner + Vector2(0.25, 0)
162	var span_count := 1 if _kind == "polylines" else 3
163	var to := corner + Vector2(span_count - 0.25, 0)
164	focus((corner + Vector2(2, 0)) * _cell, 65.0 / _cell)
165	await process_frame
166	# One full change/undo warmup precedes the retained repeated edit window.
167	set_tool_mode(3)
168	select_profile(3)
169	if not await gesture("warmupChange", from, to, span_count): return
170	if not await history("warmupUndo", KEY_Z): return
171	begin_window("editsAndHistory")
172	for cycle in range(_cycles):
173		set_tool_mode(3)
174		select_profile(3)
175		if not await gesture("change%d" % span_count, from, to, span_count): break
176		if not await history("undoChange", KEY_Z): break
177		if not await history("redoChange", KEY_Y): break
178		if not await history("undoChangeReset", KEY_Z): break
179		set_tool_mode(2)
180		if not await gesture("delete%d" % span_count, from, to, span_count): break
181		if not await history("undoDelete", KEY_Z): break
182		if not await history("redoDelete", KEY_Y): break
183		if not await history("undoDeleteReset", KEY_Z): break
184	end_window("editsAndHistory")
185	check("cycles_restore_fixture", _map.RoadCount == expected_edges())

187func build_window() -> void:
188	var full_grid := _cell == 200 and _kind == "target"
189	var corner := Vector2.ONE * (-4000.0 / _cell)
190	var from := Vector2(corner.x + 1, -corner.y - 1)
191	var to := from + Vector2(4, 0)
192	if full_grid:
193		from = corner
194		to = corner + Vector2(3, 0)
195		focus((corner + Vector2(2, 0)) * _cell, 65.0 / _cell)
196		set_tool_mode(2)
197		if not await gesture("createBuildSpace", from + Vector2(0.25, 0), to - Vector2(0.25, 0), 3): return
198	focus((from + to) * _cell / 2, 65.0 / _cell)
199	set_tool_mode(0)
200	select_profile(1)
201	await process_frame
202	begin_window("buildAndHistory")
203	for cycle in range(_cycles):
204		# Keep injected coordinates inside the map while still snapping to boundary nodes.
205		var inset := Vector2(0.02, 0.02) if full_grid else Vector2.ZERO
206		if not await gesture("build", from + inset, to + inset): break
207		if not await history("undoBuild", KEY_Z): break
208		if not await history("redoBuild", KEY_Y): break
209		if not await history("undoBuildReset", KEY_Z): break
210	end_window("buildAndHistory")
211	if full_grid: await history("restoreBuildSpace", KEY_Z)
212	check("build_cycles_restore_fixture", _map.RoadCount == expected_edges())

214func cancellation_window() -> void:
215	begin_window("escapeCancellation")
216	for attempt in range(20):
217		var source: String = _map.StateToken
218		keyboard(KEY_Y, true)
219		var phase_before: String = _map.BuildPhase
220		keyboard(KEY_ESCAPE)
221		await process_frame
222		var until := Time.get_ticks_msec() + 10000
223		while _map.IsBuildBusy and Time.get_ticks_msec() < until: await process_frame
224		var metrics: Dictionary = _map.GetOperationMetrics()
225		var accepted: bool = metrics.phase == "Cancelled"
226		metrics.accepted = accepted
227		metrics.phaseBeforeEscape = phase_before
228		metrics.stateUnchanged = source == _map.StateToken
229		_cancellations.append(metrics)
230		if accepted:
231			check("cancel_keeps_state", not _map.IsBuildBusy and metrics.stateUnchanged and metrics.cancelToToolReadyMs >= 0 and metrics.cancelToWorkerExitMs >= 0)
232		else:
233			check("commit_race_not_cancelled", metrics.phase == "Drawn")
234			if not await history("undoAfterCancelRace", KEY_Z): break
235	end_window("escapeCancellation")
